    ->     u.id AS user_id,
    ->     u.email,
    ->     u.role,
    ->     us.remember_multiplier,
    ->     us.forgot_multiplier,
    ->     d.id AS deck_id,
    ->     d.title AS deck_title,
    ->     f.id AS card_id,
    ->     f.front,
    ->     f.back,
    ->     f.next_review_at
    -> FROM users u
    -> LEFT JOIN user_settings us ON u.id = us.user_id
    -> LEFT JOIN decks d ON u.id = d.user_id AND d.deleted_at IS NULL
    -> LEFT JOIN flashcards f ON d.id = f.deck_id AND f.deleted_at IS NULL
    -> WHERE u.deleted_at IS NULL;
