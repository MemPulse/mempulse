<?php
// 1. Initialize user session tracking
session_start();

// 2. Route Protection: Redirect to login view if user credentials are uninitialized
if (!isset($_SESSION['logged_in']) || $_SESSION['logged_in'] !== true || !isset($_SESSION['user_id'])) {
    header("Location: ../../authentication/login.php");
    exit;
}

// 3. Connect to database module and load business services
require_once __DIR__ . '/../../config/db.php';
require_once __DIR__ . '/../api/FlashcardService.php';

$userId = (int) $_SESSION['user_id'];

ArchiveService::purgeExpired($conn, $userId);

// 4. Validate and sanitize input deck ID parameter
$deckId = isset($_GET['deck_id']) && ctype_digit((string) $_GET['deck_id']) ? (int) $_GET['deck_id'] : 0;

// 5. Fetch target deck details and verify user ownership
$deck = FlashcardService::getDeckDetails($conn, $deckId, $userId);

// Redirect to deck list if the requested deck is missing or unauthorized
if (!$deck) {
    header("Location: decks.php");
    exit;
}

// 6. Fetch all active flashcards belonging to the deck
$flashcards = FlashcardService::getDeckFlashcards($conn, $deckId);
$archivedFlashcards = FlashcardService::getDeckFlashcards($conn, $deckId, '', true);

// 7. CSRF token initialization for card mutation requests (Add / Edit / Delete)
if (empty($_SESSION['csrf_token'])) {
    $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
}

$pageTitle   = $deck['title'] . " - Flashcards";
$currentPage = "decks";
?>

            const nextReviewAt = modalCardDue ? modalCardDue.value.trim() : '';
            await handleCardSave(cardId, front, back, nextReviewAt, modalError, modalSaveBtn, false);
        });
    }

    // 3. Delete Card Execution
    if (deleteCardConfirm) {
        deleteCardConfirm.addEventListener('click', async () => {
            if (!targetDeleteCardId) return;
            deleteCardConfirm.disabled = true;
            try {
                await sendCardAction({ action: 'delete_permanent', card_id: targetDeleteCardId });
                if (tabulatorTable) {
                    const row = tabulatorTable.getRow(Number(targetDeleteCardId));
                    if (row) row.delete();
                    archivedCards--;
                }
                updateCardCountLabel();
                if (deleteCardDialog) deleteCardDialog.close();
                Archive.toast('Card deleted permanently', { type: 'error' });
            } catch (err) {
                if (deleteCardError) {
                    deleteCardError.textContent = err.message;
                    deleteCardError.classList.remove('hidden');
                }
            } finally {
                deleteCardConfirm.disabled = false;
            }
        });
    }

    // ---------------------------------------------------------------------------
    // Archive: archive / restore / view switching
    // ---------------------------------------------------------------------------
    function removeRow(id) {
        if (!tabulatorTable) return;
        const row = tabulatorTable.getRow(Number(id));
        if (row) row.delete();
    }

    async function archiveCard(card) {
        try {
            await sendCardAction({ action: 'archive', card_id: card.id });
            removeRow(card.id);
            offscreenList.unshift({ ...card, archived_at: new Date().toISOString(), days_left: Archive.RETENTION_DAYS });
            totalCards--;
            archivedCards++;
            updateCardCountLabel();
            Archive.toast('Card archived', { actionLabel: 'Undo', onAction: () => restoreCard(card.id) });
        } catch (err) {
            Archive.toast(err.message, { type: 'error' });
        }
    }

    async function restoreCard(cardId) {
        try {
            await sendCardAction({ action: 'restore', card_id: cardId });
            const id = Number(cardId);

            if (viewArchived) {
                const row = tabulatorTable.getRow(id);
                const data = row ? row.getData() : null;
                if (row) row.delete();
                if (data) offscreenList.unshift({ ...data, archived_at: null, days_left: null });
                Archive.toast('Card restored');
            } else {
                const idx = offscreenList.findIndex((c) => Number(c.id) === id);
                if (idx !== -1) {
                    const [data] = offscreenList.splice(idx, 1);
                    await tabulatorTable.addRow({ ...data, archived_at: null, days_left: null }, true);
                }
            }

            totalCards++;
            archivedCards--;
            updateCardCountLabel();
        } catch (err) {
            Archive.toast(err.message, { type: 'error' });
        }
    }

    function setViewMode(archived) {
        viewArchived = archived;
        closeActionsMenu();

        if (archiveBanner) archiveBanner.classList.toggle('hidden', !archived);
        if (openAddCardBtn) openAddCardBtn.classList.toggle('hidden', archived);
        if (startReviewBtn) startReviewBtn.classList.toggle('hidden', archived);
        if (toggleArchiveIcon) toggleArchiveIcon.textContent = archived ? 'arrow_back' : 'inventory_2';
        if (toggleArchiveLabel) toggleArchiveLabel.textContent = archived ? 'Back to cards' : 'Archive';
        if (toggleArchiveBtn) {
            const label = archived ? 'Back to cards' : 'View archive';
            toggleArchiveBtn.title = label;
            toggleArchiveBtn.setAttribute('aria-label', label);
            toggleArchiveBtn.setAttribute('aria-pressed', String(archived));
        }
        if (emptyTitle) emptyTitle.textContent = archived ? 'Archive is empty' : 'No flashcards yet';
        if (emptyText) {
            emptyText.textContent = archived
                ? `Archived cards appear here for ${Archive.RETENTION_DAYS} days before they are permanently deleted.`
                : 'Click "Create Card" to add your first card.';
        }

        if (tabulatorTable) {
            tabulatorTable.setColumns(
                allColumns.filter((c) => c.field !== (archived ? 'next_review_at' : 'days_left'))
            );

            const current = tabulatorTable.getData().concat(pending); // full list of the view we're leaving
            const next = offscreenList;
            offscreenList = current;
            selectedSidebarCardId = null;
            selectedTabulatorRow = null;

            loadFirstPage(next).then(() => {
                tabulatorTable.redraw(true);
                const first = tabulatorTable.getRows()[0];
                if (first && isDesktopView()) {
                    first.select();
                    populateSidebar(first.getData(), first);
                }
            });
        }
        updateCardCountLabel();
    }

    if (toggleArchiveBtn) toggleArchiveBtn.addEventListener('click', () => setViewMode(!viewArchived));

    // ---------------------------------------------------------------------------
    // Row overflow menu (Duplicate / Reset progress)
    // ---------------------------------------------------------------------------
    function openActionsMenu(btn, cardId) {
        if (!actionsMenu) return;
        menuCardId = cardId;
        const r = btn.getBoundingClientRect();
        actionsMenu.classList.remove('hidden');
        const w = actionsMenu.offsetWidth;
        actionsMenu.style.top = `${r.bottom + 4}px`;
        actionsMenu.style.left = `${Math.max(8, r.right - w)}px`;
    }

    function closeActionsMenu() {
        if (actionsMenu) actionsMenu.classList.add('hidden');
        menuCardId = null;
    }

    document.addEventListener('click', (e) => {
        if (!e.target.closest('#card-actions-menu') && !e.target.closest('[data-more-card]')) closeActionsMenu();
    });
    document.addEventListener('keydown', (e) => { if (e.key === 'Escape') closeActionsMenu(); });
    window.addEventListener('resize', closeActionsMenu);
    if (tabulatorTable) {
        tabulatorTable.on('scrollVertical', closeActionsMenu);
        tabulatorTable.on('scrollHorizontal', closeActionsMenu);
    }

    if (actionsMenu) {
        actionsMenu.addEventListener('click', async (e) => {
            const item = e.target.closest('[data-menu-action]');
            if (!item || !menuCardId) return;
            const id = menuCardId;
            const act = item.dataset.menuAction;
            closeActionsMenu();

            try {
                if (act === 'duplicate') {
                    const res = await sendCardAction({ action: 'duplicate', card_id: id });
                    if (tabulatorTable) await tabulatorTable.addRow(res.card, true);
                    totalCards++;
                    updateCardCountLabel();
                    Archive.toast('Card duplicated');
                } else if (act === 'reset') {
                    const res = await sendCardAction({ action: 'reset_progress', card_id: id });
                    const row = tabulatorTable ? tabulatorTable.getRow(Number(id)) : null;
                    if (row) await row.update(res.card);
                    if (isDesktopView() && selectedSidebarCardId === Number(id)) {
                        populateSidebar(row ? row.getData() : res.card, row || null);
                    }
                    Archive.toast('Progress reset. The card is due now.');
                }
            } catch (err) {
                Archive.toast(err.message, { type: 'error' });
            }
        });
    }
});

/**
 * MemPulse shared archive UI helpers (toast + retention label).
 * Used by the deck page (cards) and reusable by the decks page (decks).
 * Reads retentionDays from window.MemPulseDeckData or window.MemPulseArchiveConfig.
 */
(function () {
    const cfg = window.MemPulseDeckData || window.MemPulseArchiveConfig || {};
    const RETENTION_DAYS = Number(cfg.retentionDays) || 30;

    function retentionLabel(daysLeft) {
        if (daysLeft === null || daysLeft === undefined) return '';
        if (daysLeft <= 0) return 'Today';
        return daysLeft + (daysLeft === 1 ? ' day' : ' days');
    }

    function toast(message, opts = {}) {
        const { actionLabel, onAction, type = 'info', duration = 6000 } = opts;

        let root = document.getElementById('mp-toast-root');
        if (!root) {
            root = document.createElement('div');
            root.id = 'mp-toast-root';
            root.className = 'fixed bottom-4 left-1/2 -translate-x-1/2 z-[100] flex flex-col items-center gap-2 px-4';
            document.body.appendChild(root);
        }

        const el = document.createElement('div');
        el.setAttribute('role', 'status');
        el.className = 'flex items-center gap-3 pl-4 pr-2 py-2 rounded-xl shadow-xl text-sm font-medium '
            + (type === 'error' ? 'bg-rose-700 text-white' : 'bg-inverse-surface text-inverse-on-surface');

        const text = document.createElement('span');
        text.textContent = message;
        el.appendChild(text);

        const remove = () => el.remove();

        if (actionLabel && typeof onAction === 'function') {
            const btn = document.createElement('button');
            btn.type = 'button';
            btn.textContent = actionLabel;
            btn.className = 'px-2.5 py-1 rounded-lg font-semibold text-primary-fixed-dim hover:bg-white/10';
            btn.addEventListener('click', () => { remove(); onAction(); });
            el.appendChild(btn);
        }

        root.appendChild(el);
        setTimeout(remove, duration);
    }

    window.MemPulseArchive = { RETENTION_DAYS, retentionLabel, toast };
})();

<?php
/**
 * Flashcard Actions API Endpoint
 * 
 * Secure HTTP POST JSON endpoint handling asynchronous flashcard mutations 
 * including addition, editing, and soft-deletion.
 */

session_start();
header('Content-Type: application/json; charset=utf-8');

/**
 * Send a structured JSON HTTP response and terminate request execution.
 * 
 * @param int $status HTTP status code.
 * @param array $body Response payload data.
 * @return void
 */
function respond(int $status, array $body): void
{
    http_response_code($status);
    echo json_encode($body);
    exit;
}

// 1. Request Method Verification
if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
    respond(405, ['error' => 'Method not allowed.']);
}

// 2. Authentication Check
if (!isset($_SESSION['logged_in']) || $_SESSION['logged_in'] !== true || !isset($_SESSION['user_id'])) {
    respond(401, ['error' => 'Please log in again.']);
}

// 3. Input Data Parsing
$input = json_decode(file_get_contents('php://input'), true);
if (!is_array($input)) {
    respond(400, ['error' => 'Invalid request structure.']);
}

// 4. Anti-CSRF Token Verification
$token = $input['csrf_token'] ?? '';
if (empty($_SESSION['csrf_token']) || !is_string($token) || !hash_equals($_SESSION['csrf_token'], $token)) {
    respond(403, ['error' => 'Security check failed. Refresh the page and try again.']);
}

require_once __DIR__ . '/../../config/db.php';

$user_id = (int) $_SESSION['user_id'];
$action  = $input['action'] ?? '';
$deck_id = filter_var($input['deck_id'] ?? null, FILTER_VALIDATE_INT);

if (!$deck_id || $deck_id < 1) {
    respond(400, ['error' => 'Invalid deck selection.']);
}

// 5. Verify User Ownership of Target Deck
$checkStmt = mysqli_prepare($conn, "SELECT id FROM decks WHERE id = ? AND user_id = ? AND deleted_at IS NULL");
mysqli_stmt_bind_param($checkStmt, "ii", $deck_id, $user_id);
mysqli_stmt_execute($checkStmt);
mysqli_stmt_store_result($checkStmt);
$ownsDeck = mysqli_stmt_num_rows($checkStmt) === 1;
mysqli_stmt_close($checkStmt);

if (!$ownsDeck) {
    respond(404, ['error' => 'Deck not found or access denied.']);
}

// ---------------------------------------------------------------------------
// Action Processor
// ---------------------------------------------------------------------------
try {
    // Action: Add Flashcard
    if ($action === 'add') {
        $front = trim((string) ($input['front'] ?? ''));
        $back  = trim((string) ($input['back'] ?? ''));

        if ($front === '' || $back === '') {
            respond(422, ['error' => 'Both the front and back of the card are required.']);
        }
        if (mb_strlen($front, 'UTF-8') > 5000 || mb_strlen($back, 'UTF-8') > 5000) {
            respond(422, ['error' => 'Each side must be 5000 characters or fewer.']);
        }

        $stmt = mysqli_prepare($conn, "INSERT INTO flashcards (deck_id, front, back) VALUES (?, ?, ?)");
        mysqli_stmt_bind_param($stmt, "iss", $deck_id, $front, $back);
        mysqli_stmt_execute($stmt);
        $newCardId = mysqli_insert_id($conn);
        mysqli_stmt_close($stmt);

        // Fetch newly created card record with defaults
        $fetchStmt = mysqli_prepare($conn, "SELECT id, deck_id, front, back, interval_days, next_review_at, review_count, lapses, created_at, updated_at FROM flashcards WHERE id = ?");
        mysqli_stmt_bind_param($fetchStmt, "i", $newCardId);
        mysqli_stmt_execute($fetchStmt);
        $card = mysqli_fetch_assoc(mysqli_stmt_get_result($fetchStmt));
        mysqli_stmt_close($fetchStmt);

        respond(200, [
            'ok'   => true,
            'card' => $card
        ]);
    }

    // Action: Update Flashcard
    if ($action === 'update') {
        $card_id        = filter_var($input['card_id'] ?? null, FILTER_VALIDATE_INT);
        $front          = trim((string) ($input['front'] ?? ''));
        $back           = trim((string) ($input['back'] ?? ''));
        $next_review_at = trim((string) ($input['next_review_at'] ?? ''));

        if (!$card_id || $card_id < 1) {
            respond(400, ['error' => 'Invalid card specified.']);
        }
        if ($front === '' || $back === '') {
            respond(422, ['error' => 'Both the front and back of the card are required.']);
        }
        if (mb_strlen($front, 'UTF-8') > 5000 || mb_strlen($back, 'UTF-8') > 5000) {
            respond(422, ['error' => 'Each side must be 5000 characters or fewer.']);
        }

        // Validate and sanitize custom next_review_at if provided
        $formattedNextReview = null;
        if ($next_review_at !== '') {
            $dt = DateTime::createFromFormat('Y-m-d\TH:i', $next_review_at) 
               ?: DateTime::createFromFormat('Y-m-d H:i:s', $next_review_at);
            if ($dt !== false) {
                $formattedNextReview = $dt->format('Y-m-d H:i:s');
            }
        }

        if ($formattedNextReview !== null) {
            $stmt = mysqli_prepare($conn, "UPDATE flashcards SET front = ?, back = ?, next_review_at = ? WHERE id = ? AND deck_id = ? AND deleted_at IS NULL");
            mysqli_stmt_bind_param($stmt, "sssii", $front, $back, $formattedNextReview, $card_id, $deck_id);
        } else {
            $stmt = mysqli_prepare($conn, "UPDATE flashcards SET front = ?, back = ? WHERE id = ? AND deck_id = ? AND deleted_at IS NULL");
            mysqli_stmt_bind_param($stmt, "ssii", $front, $back, $card_id, $deck_id);
        }

        mysqli_stmt_execute($stmt);
        mysqli_stmt_close($stmt);

        // Fetch refreshed card record
        $fetchStmt = mysqli_prepare($conn, "SELECT id, deck_id, front, back, interval_days, next_review_at, review_count, lapses, created_at, updated_at FROM flashcards WHERE id = ?");
        mysqli_stmt_bind_param($fetchStmt, "i", $card_id);
        mysqli_stmt_execute($fetchStmt);
        $card = mysqli_fetch_assoc(mysqli_stmt_get_result($fetchStmt));
        mysqli_stmt_close($fetchStmt);

        respond(200, [
            'ok'   => true,
            'card' => $card
        ]);
    }

        // Actions: Archive / Restore / Permanent delete
    if (in_array($action, ['archive', 'restore', 'delete_permanent'], true)) {
        $card_id = filter_var($input['card_id'] ?? null, FILTER_VALIDATE_INT);
        if (!$card_id || $card_id < 1) {
            respond(400, ['error' => 'Invalid card specified.']);
        }

        if ($action === 'archive') {
            $ok = ArchiveService::archive($conn, 'flashcards', $card_id, $user_id);
        } elseif ($action === 'restore') {
            $ok = ArchiveService::restore($conn, 'flashcards', $card_id, $user_id);
        } else {
            $ok = ArchiveService::deletePermanently($conn, 'flashcards', $card_id, $user_id);
        }

        if (!$ok) {
            respond(404, ['error' => 'Card not found.']);
        }
        respond(200, ['ok' => true, 'card_id' => $card_id]);
    }

    // Action: Duplicate (copies front/back, fresh progress)
    if ($action === 'duplicate') {
        $card_id = filter_var($input['card_id'] ?? null, FILTER_VALIDATE_INT);
        if (!$card_id || $card_id < 1) {
            respond(400, ['error' => 'Invalid card specified.']);
        }

        $stmt = mysqli_prepare($conn, "INSERT INTO flashcards (deck_id, front, back)
                                       SELECT deck_id, front, back FROM flashcards
                                        WHERE id = ? AND deck_id = ? AND deleted_at IS NULL");
        mysqli_stmt_bind_param($stmt, "ii", $card_id, $deck_id);
        mysqli_stmt_execute($stmt);
        $newCardId = (int) mysqli_insert_id($conn);
        mysqli_stmt_close($stmt);

        if ($newCardId < 1) {
            respond(404, ['error' => 'Card not found.']);
        }
        respond(200, ['ok' => true, 'card' => FlashcardService::getFlashcardById($conn, $newCardId, $user_id)]);
    }

    // Action: Reset progress (interval back to the user's initial interval, due now)
    if ($action === 'reset_progress') {
        $card_id = filter_var($input['card_id'] ?? null, FILTER_VALIDATE_INT);
        if (!$card_id || $card_id < 1) {
            respond(400, ['error' => 'Invalid card specified.']);
        }

        $stmt = mysqli_prepare($conn, "UPDATE flashcards
                                          SET interval_days = COALESCE((SELECT initial_interval_days FROM user_settings WHERE user_id = ?), 1),
                                              next_review_at = NOW()
                                        WHERE id = ? AND deck_id = ? AND deleted_at IS NULL");
        mysqli_stmt_bind_param($stmt, "iii", $user_id, $card_id, $deck_id);
        mysqli_stmt_execute($stmt);
        mysqli_stmt_close($stmt);

        $card = FlashcardService::getFlashcardById($conn, $card_id, $user_id);
        if (!$card) {
            respond(404, ['error' => 'Card not found.']);
        }
        respond(200, ['ok' => true, 'card' => $card]);
    }

    respond(400, ['error' => 'Unknown action.']);

} catch (Throwable $e) {
    error_log('flashcard-actions.php: ' . $e->getMessage());
    respond(500, ['error' => 'Something went wrong. Please try again.']);
}

<?php
/**
 * FlashcardService
 * 
 * Centralized service layer for fetching deck details, managing flashcard 
 * operations, and enforcing data integrity and user authorization.
 */
require_once __DIR__ . '/ArchiveService.php';

class FlashcardService
{
    /**
     * Retrieve single deck details and verify user ownership.
     * 
     * @param mysqli $conn Active database connection instance.
     * @param int $deckId Target deck ID.
     * @param int $userId Authenticated user ID.
     * @return array|null Returns deck array if authorized and active, null otherwise.
     */
    public static function getDeckDetails(mysqli $conn, int $deckId, int $userId): ?array
    {
        $sql = "SELECT d.id, 
                       d.title, 
                       d.created_at,
                       (SELECT COUNT(*) 
                          FROM flashcards f 
                         WHERE f.deck_id = d.id AND f.deleted_at IS NULL) AS card_count,
                       (SELECT COUNT(*) 
                          FROM flashcards f 
                         WHERE f.deck_id = d.id AND f.deleted_at IS NULL AND f.next_review_at <= NOW()) AS due_count,
                       (SELECT COUNT(*)
                          FROM flashcards f
                         WHERE f.deck_id = d.id AND f.deleted_at IS NOT NULL) AS archived_count
                  FROM decks d
                 WHERE d.id = ? AND d.user_id = ? AND d.deleted_at IS NULL";

        $stmt = mysqli_prepare($conn, $sql);
        if (!$stmt) {
            return null;
        }

        mysqli_stmt_bind_param($stmt, "ii", $deckId, $userId);
        mysqli_stmt_execute($stmt);
        $result = mysqli_stmt_get_result($stmt);
        $deck = mysqli_fetch_assoc($result);
        mysqli_stmt_close($stmt);

        return $deck ?: null;
    }

    /**
     * Fetch active flashcards belonging to a specific deck with optional search filtering.
     * 
     * @param mysqli $conn Active database connection instance.
     * @param int $deckId Target deck ID.
     * @param string $search Search query to match against front or back content.
     * @return array Array of flashcard records.
     */
        public static function getDeckFlashcards(mysqli $conn, int $deckId, string $search = '', bool $archived = false): array
    {
        $search      = trim($search);
        $deletedCond = $archived ? 'IS NOT NULL' : 'IS NULL';
        $orderBy     = $archived ? 'f.deleted_at DESC, f.id DESC' : 'f.id DESC';
        $retention   = ArchiveService::RETENTION_DAYS;

        $sql = "SELECT f.id, f.deck_id, f.front, f.back, f.interval_days, f.next_review_at,
                    f.review_count, f.lapses, f.created_at, f.updated_at, f.deleted_at,
                    GREATEST(0, ? - TIMESTAMPDIFF(DAY, f.deleted_at, NOW())) AS days_left,
                    (SELECT MAX(l.reviewed_at)
                        FROM review_logs l WHERE l.card_id = f.id) AS last_reviewed_at,
                    (SELECT ROUND(100 * SUM(l.rating = 'remembered') / COUNT(*))
                        FROM review_logs l WHERE l.card_id = f.id) AS accuracy,
                    (SELECT ROUND(AVG(l.response_time_ms))
                        FROM review_logs l WHERE l.card_id = f.id) AS avg_response_ms
                FROM flashcards f
                WHERE f.deck_id = ? AND f.deleted_at {$deletedCond}";

        if ($search !== '') {
            $sql .= " AND (f.front LIKE ? OR f.back LIKE ?)";
        }
        $sql .= " ORDER BY {$orderBy}";

        $stmt = mysqli_prepare($conn, $sql);
        if (!$stmt) {
            return [];
        }

        if ($search !== '') {
            $like = '%' . $search . '%';
            mysqli_stmt_bind_param($stmt, "iiss", $retention, $deckId, $like, $like);
        } else {
            mysqli_stmt_bind_param($stmt, "ii", $retention, $deckId);
        }

        mysqli_stmt_execute($stmt);
        $result = mysqli_stmt_get_result($stmt);

        $cards = [];
        while ($row = mysqli_fetch_assoc($result)) {
            $cards[] = [
                'id'               => (int) $row['id'],
                'deck_id'          => (int) $row['deck_id'],
                'front'            => $row['front'],
                'back'             => $row['back'],
                'interval_days'    => (int) $row['interval_days'],
                'next_review_at'   => $row['next_review_at'],
                'review_count'     => (int) $row['review_count'],
                'lapses'           => (int) $row['lapses'],
                'created_at'       => $row['created_at'],
                'updated_at'       => $row['updated_at'],
                'archived_at'      => $row['deleted_at'],
                'days_left'        => $row['days_left'] !== null ? (int) $row['days_left'] : null,
                'last_reviewed_at' => $row['last_reviewed_at'],
                'accuracy'         => $row['accuracy'] !== null ? (int) $row['accuracy'] : null,
                'avg_response_ms'  => $row['avg_response_ms'] !== null ? (int) $row['avg_response_ms'] : null,
            ];
        }
        mysqli_stmt_close($stmt);

        return $cards;
    }

    /**
     * Fetch a single flashcard record and verify deck ownership.
     * 
     * @param mysqli $conn Active database connection instance.
     * @param int $cardId Target flashcard ID.
     * @param int $userId Authenticated user ID.
     * @return array|null Returns flashcard array if found and authorized, null otherwise.
     */
    public static function getFlashcardById(mysqli $conn, int $cardId, int $userId): ?array
    {
        $sql = "SELECT f.id, f.deck_id, f.front, f.back, f.next_review_at, f.interval_days, f.review_count, f.lapses, f.created_at, f.updated_at
                  FROM flashcards f
                  JOIN decks d ON f.deck_id = d.id
                 WHERE f.id = ? AND d.user_id = ? AND f.deleted_at IS NULL AND d.deleted_at IS NULL";

        $stmt = mysqli_prepare($conn, $sql);
        if (!$stmt) {
            return null;
        }

        mysqli_stmt_bind_param($stmt, "ii", $cardId, $userId);
        mysqli_stmt_execute($stmt);
        $result = mysqli_stmt_get_result($stmt);
        $card = mysqli_fetch_assoc($result);
        mysqli_stmt_close($stmt);

        return $card ?: null;
    }

    /**
     * Safely escape HTML string values for view rendering.
     * 
     * @param mixed $value Raw string content.
     * @return string Escaped HTML string.
     */
    public static function h($value): string
    {
        return htmlspecialchars((string) $value, ENT_QUOTES, 'UTF-8');
    }
