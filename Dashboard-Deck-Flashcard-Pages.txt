<?php
/**
 * DashboardService
 * 
 * Centralized service layer for handling database reads, user profiling,
 * spaced repetition metric analytics, and streak tracking.
 */
class DashboardService
{
    /**
     * Retrieve authenticated user profile details.
     * 
     * @param mysqli $conn Active database connection instance.
     * @param int $userId Authenticated user ID.
     * @return array Profile metadata containing display name.
     */
    public static function getUserProfile(mysqli $conn, int $userId): array
    {
        $sql = "SELECT first_name, last_name FROM users WHERE id = ? AND deleted_at IS NULL";
        $stmt = mysqli_prepare($conn, $sql);
        
        if (!$stmt) {
            return ['displayName' => 'Active Learner'];
        }

        mysqli_stmt_bind_param($stmt, "i", $userId);
        mysqli_stmt_execute($stmt);
        mysqli_stmt_bind_result($stmt, $firstName, $lastName);
        mysqli_stmt_fetch($stmt);
        mysqli_stmt_close($stmt);

        $fullName = trim(($firstName ?? '') . ' ' . ($lastName ?? ''));
        
        return [
            'displayName' => !empty($fullName) ? $fullName : 'Active Learner'
        ];
    }

    /**
     * Fetch core flashcard and study session metrics.
     * 
     * @param mysqli $conn Active database connection instance.
     * @param int $userId Authenticated user ID.
     * @return array Calculated user metrics.
     */
    public static function getMetrics(mysqli $conn, int $userId): array
    {
        // 1. Cards Due Today
        $due = self::fetchCount($conn, 
            "SELECT COUNT(*) FROM flashcards f 
             JOIN decks d ON f.deck_id = d.id 
             WHERE d.user_id = ? AND f.deleted_at IS NULL AND f.next_review_at <= NOW()", 
            $userId
        );

        // 2. Mastered Cards (Interval >= 21 days)
        $mastered = self::fetchCount($conn, 
            "SELECT COUNT(*) FROM flashcards f 
             JOIN decks d ON f.deck_id = d.id 
             WHERE d.user_id = ? AND f.deleted_at IS NULL AND f.interval_days >= 21", 
            $userId
        );

        // 3. Tough Cards / Leeches (Lapses >= 8)
        $tough = self::fetchCount($conn, 
            "SELECT COUNT(*) FROM flashcards f 
             JOIN decks d ON f.deck_id = d.id 
             WHERE d.user_id = ? AND f.deleted_at IS NULL AND f.lapses >= 8", 
            $userId
        );

        // 4. Cards Reviewed Today
        $reviewedToday = self::fetchCount($conn, 
            "SELECT COALESCE(SUM(total_reviewed), 0) FROM review_sessions 
             WHERE user_id = ? AND ended_at IS NOT NULL AND DATE(started_at) = CURDATE()", 
            $userId
        );

        // 5. Retention Rate Percentage
        $retentionRate = 0.0;
        $retentionSql = "SELECT COALESCE(SUM(remembered_count), 0), COALESCE(SUM(total_reviewed), 0) 
                         FROM review_sessions 
                         WHERE user_id = ? AND ended_at IS NOT NULL";
                         
        if ($stmt = mysqli_prepare($conn, $retentionSql)) {
            mysqli_stmt_bind_param($stmt, "i", $userId);
            mysqli_stmt_execute($stmt);
            mysqli_stmt_bind_result($stmt, $remembered, $reviewed);
            if (mysqli_stmt_fetch($stmt) && $reviewed > 0) {
                $retentionRate = round(($remembered / $reviewed) * 100, 1);
            }
            mysqli_stmt_close($stmt);
        }

        return [
            'cardsDue' => $due,
            'masteredCards' => $mastered,
            'toughCards' => $tough,
            'cardsReviewedToday' => $reviewedToday,
            'retentionRate' => $retentionRate
        ];
    }

    /**
     * Compute activity history, streaks, and heatmap data.
     * 
     * @param mysqli $conn Active database connection instance.
     * @param int $userId Authenticated user ID.
     * @return array Streak counters and calendar heatmap data.
     */
    public static function getStreakAndHeatmapData(mysqli $conn, int $userId): array
    {
        $sessionSql = "SELECT DATE(started_at) as session_date, 
                              SUM(remembered_count) as total_remembered, 
                              SUM(total_reviewed) as total_reviewed
                       FROM review_sessions 
                       WHERE user_id = ? AND ended_at IS NOT NULL
                       GROUP BY DATE(started_at)
                       HAVING SUM(total_reviewed) > 0
                       ORDER BY session_date DESC";

        $stmt = mysqli_prepare($conn, $sessionSql);
        if (!$stmt) {
            return self::emptyStreakState();
        }

        mysqli_stmt_bind_param($stmt, "i", $userId);
        mysqli_stmt_execute($stmt);
        $result = mysqli_stmt_get_result($stmt);

        $activityHistory = [];
        while ($row = mysqli_fetch_assoc($result)) {
            $activityHistory[$row['session_date']] = [
                'remembered' => (int) $row['total_remembered'],
                'reviewed' => (int) $row['total_reviewed']
            ];
        }
        mysqli_stmt_close($stmt);

        if (empty($activityHistory)) {
            return self::emptyStreakState();
        }

        // Streak Analytics Processing
        $todayStr = date('Y-m-d');
        $yesterdayStr = date('Y-m-d', strtotime('-1 day'));
        
        $currentStreak = 0;
        $bestStreak = 0;
        $streakAccuracy = 0.0;
        
        $dates = array_keys($activityHistory);
        $latestDate = $dates[0];

        // Active Streak calculation
        if ($latestDate === $todayStr || $latestDate === $yesterdayStr) {
            $checkDate = new DateTime($latestDate);
            $streakRemembered = 0;
            $streakReviewed = 0;

            while (true) {
                $dateKey = $checkDate->format('Y-m-d');
                if (isset($activityHistory[$dateKey])) {
                    $currentStreak++;
                    $streakRemembered += $activityHistory[$dateKey]['remembered'];
                    $streakReviewed += $activityHistory[$dateKey]['reviewed'];
                    $checkDate->modify('-1 day');
                } else {
                    break;
                }
            }

            if ($streakReviewed > 0) {
                $streakAccuracy = round(($streakRemembered / $streakReviewed) * 100, 1);
            }
        }

        // Best Historical Streak calculation
        $tempStreak = 0;
        $expectedDate = null;

        foreach (array_reverse($dates) as $dateStr) {
            $currentDate = new DateTime($dateStr);
            if ($expectedDate === null || $currentDate == $expectedDate) {
                $tempStreak++;
            } else {
                if ($tempStreak > $bestStreak) {
                    $bestStreak = $tempStreak;
                }
                $tempStreak = 1;
            }
            $expectedDate = clone $currentDate;
            $expectedDate->modify('+1 day');
        }

        if ($tempStreak > $bestStreak) {
            $bestStreak = $tempStreak;
        }

        // Heatmap bounds calculation
        $heatmapEnd = new DateTime('today');
        $heatmapStart = (clone $heatmapEnd)->modify('-364 days');
        $heatmapData = [];
        $studyDays = 0;

        foreach ($activityHistory as $dateStr => $metrics) {
            $activityDate = DateTime::createFromFormat('!Y-m-d', $dateStr);
            if ($activityDate && $activityDate >= $heatmapStart && $activityDate <= $heatmapEnd && $metrics['reviewed'] > 0) {
                $heatmapData[$dateStr] = $metrics['reviewed'];
                $studyDays++;
            }
        }

        $calendarStart = clone $heatmapStart;
        $calendarStart->modify('monday this week');

        return [
            'currentStreak' => $currentStreak,
            'bestStreak' => $bestStreak,
            'streakAccuracy' => $streakAccuracy,
            'studyDays' => $studyDays,
            'heatmapData' => $heatmapData,
            'heatmapStart' => $heatmapStart,
            'heatmapEnd' => $heatmapEnd,
            'calendarStart' => $calendarStart
        ];
    }

    /**
     * Retrieve recent completed study sessions for the activity feed.
     * 
     * @param mysqli $conn Active database connection instance.
     * @param int $userId Authenticated user ID.
     * @param int $limit Max sessions to return.
     * @return array Session history items.
     */
    public static function getRecentSessions(mysqli $conn, int $userId, int $limit = 4): array
    {
        $sql = "SELECT rs.id, d.title as deck_title, rs.total_reviewed as cards_reviewed, 
                       rs.accuracy_percentage as accuracy, rs.started_at 
                FROM review_sessions rs 
                JOIN decks d ON rs.deck_id = d.id 
                WHERE rs.user_id = ? AND rs.ended_at IS NOT NULL 
                ORDER BY rs.started_at DESC LIMIT ?";

        $sessions = [];
        if ($stmt = mysqli_prepare($conn, $sql)) {
            mysqli_stmt_bind_param($stmt, "ii", $userId, $limit);
            mysqli_stmt_execute($stmt);
            $result = mysqli_stmt_get_result($stmt);

            while ($row = mysqli_fetch_assoc($result)) {
                $timeDiff = time() - strtotime($row['started_at']);
                if ($timeDiff < 3600) {
                    $formatted = floor($timeDiff / 60) . ' mins ago';
                } elseif ($timeDiff < 86400) {
                    $formatted = floor($timeDiff / 3600) . ' hours ago';
                } else {
                    $formatted = floor($timeDiff / 86400) . ' days ago';
                }

                $sessions[] = [
                    'deckTitle' => $row['deck_title'],
                    'cardsReviewed' => (int) $row['cards_reviewed'],
                    'accuracy' => (int) $row['accuracy'],
                    'formattedTime' => $formatted
                ];
            }
            mysqli_stmt_close($stmt);
        }

        return $sessions;
    }

    private static function fetchCount(mysqli $conn, string $query, int $userId): int
    {
        if ($stmt = mysqli_prepare($conn, $query)) {
            mysqli_stmt_bind_param($stmt, "i", $userId);
            mysqli_stmt_execute($stmt);
            mysqli_stmt_bind_result($stmt, $count);
            mysqli_stmt_fetch($stmt);
            mysqli_stmt_close($stmt);
            return (int) $count;
        }
        return 0;
    }

    private static function emptyStreakState(): array
    {
        return [
            'currentStreak' => 0,
            'bestStreak' => 0,
            'streakAccuracy' => 0.0,
            'studyDays' => 0,
            'heatmapData' => [],
            'heatmapStart' => new DateTime('-364 days'),
            'heatmapEnd' => new DateTime('today'),
            'calendarStart' => (new DateTime('-364 days'))->modify('monday this week')
        ];
    }
}

<?php
/**
 * Deck Actions API Endpoint
 * 
 * Secure HTTP POST JSON endpoint handling asynchronous deck and flashcard mutation 
 * actions including creation, renaming, archiving, restoring, permanent deletion, 
 * and flashcard addition.
 */

session_start();
header('Content-Type: application/json; charset=utf-8');

/**
 * Send a structured JSON HTTP response and terminate request execution.
 * 
 * @param int $status HTTP status code.
 * @param array $body Response payload data.
 * @return void
 */
function respond(int $status, array $body): void
{
    http_response_code($status);
    echo json_encode($body);
    exit;
}

/**
 * Generate a unique deck title suggestion when a collision occurs.
 * 
 * @param callable $exists Callback function checking title existence.
 * @param string $title Target base deck title.
 * @param int $max Maximum allowed title character length.
 * @return string Available deck title candidate.
 */
function suggest_unique(callable $exists, string $title, int $max = 150): string
{
    $base = preg_replace('/\s*\(\d+\)$/u', '', $title);
    $base = mb_substr($base, 0, $max - 8, 'UTF-8');
    for ($n = 2; $n < 1000; $n++) {
        $candidate = $base . ' (' . $n . ')';
        if (!$exists($candidate)) {
            return $candidate;
        }
    }
    return $base . ' (' . bin2hex(random_bytes(2)) . ')';
}

}

require_once __DIR__ . '/../../config/db.php';

$user_id = (int) $_SESSION['user_id'];
$deck_id = filter_var($input['deck_id'] ?? null, FILTER_VALIDATE_INT);
$action  = $input['action'] ?? '';

/**
 * Helper Closure: Check if a live deck title exists for the authenticated user.
 * 
 * @param string $t Deck title string.
 * @param int $exclude_id Optional deck ID to ignore (for renaming).
 * @return bool True if collision exists, false otherwise.
 */
$deck_exists = function (string $t, int $exclude_id = 0) use ($conn, $user_id): bool {
    $s = mysqli_prepare($conn, "SELECT 1 FROM decks WHERE user_id = ? AND title = ? AND id <> ? AND deleted_at IS NULL LIMIT 1");
    mysqli_stmt_bind_param($s, "isi", $user_id, $t, $exclude_id);

<?php
/**
 * FlashcardService
 * 
 * Centralized service layer for fetching deck details, managing flashcard 
 * operations, and enforcing data integrity and user authorization.
 */
require_once __DIR__ . '/ArchiveService.php';

class FlashcardService
{
    /**
     * Retrieve single deck details and verify user ownership.
     * 
     * @param mysqli $conn Active database connection instance.
     * @param int $deckId Target deck ID.
     * @param int $userId Authenticated user ID.
     * @return array|null Returns deck array if authorized and active, null otherwise.
     */
    public static function getDeckDetails(mysqli $conn, int $deckId, int $userId): ?array
    {
        $sql = "SELECT d.id, 
                       d.title, 
                       d.created_at,
                       (SELECT COUNT(*) 
                          FROM flashcards f 
                         WHERE f.deck_id = d.id AND f.deleted_at IS NULL) AS card_count,
                       (SELECT COUNT(*) 
                          FROM flashcards f 
                         WHERE f.deck_id = d.id AND f.deleted_at IS NULL AND f.next_review_at <= NOW()) AS due_count,
                       (SELECT COUNT(*)
                          FROM flashcards f
                         WHERE f.deck_id = d.id AND f.deleted_at IS NOT NULL) AS archived_count
                  FROM decks d
                 WHERE d.id = ? AND d.user_id = ? AND d.deleted_at IS NULL";

        $stmt = mysqli_prepare($conn, $sql);
        if (!$stmt) {
            return null;
        }

        mysqli_stmt_bind_param($stmt, "ii", $deckId, $userId);
        mysqli_stmt_execute($stmt);
        $result = mysqli_stmt_get_result($stmt);
        $deck = mysqli_fetch_assoc($result);
        mysqli_stmt_close($stmt);

        return $deck ?: null;
    }

    /**
     * Fetch active flashcards belonging to a specific deck with optional search filtering.
     * 
     * @param mysqli $conn Active database connection instance.
     * @param int $deckId Target deck ID.
     * @param string $search Search query to match against front or back content.
     * @return array Array of flashcard records.
     */
        public static function getDeckFlashcards(mysqli $conn, int $deckId, string $search = '', bool $archived = false): array
    {
        $search      = trim($search);
        $deletedCond = $archived ? 'IS NOT NULL' : 'IS NULL';
        $orderBy     = $archived ? 'f.deleted_at DESC, f.id DESC' : 'f.id DESC';
        $retention   = ArchiveService::RETENTION_DAYS;

        $sql = "SELECT f.id, f.deck_id, f.front, f.back, f.interval_days, f.next_review_at,
                    f.review_count, f.lapses, f.created_at, f.updated_at, f.deleted_at,
                    GREATEST(0, ? - TIMESTAMPDIFF(DAY, f.deleted_at, NOW())) AS days_left,
                    (SELECT MAX(l.reviewed_at)
                        FROM review_logs l WHERE l.card_id = f.id) AS last_reviewed_at,
                    (SELECT ROUND(100 * SUM(l.rating = 'remembered') / COUNT(*))
                        FROM review_logs l WHERE l.card_id = f.id) AS accuracy,
                    (SELECT ROUND(AVG(l.response_time_ms))
                        FROM review_logs l WHERE l.card_id = f.id) AS avg_response_ms
                FROM flashcards f
                WHERE f.deck_id = ? AND f.deleted_at {$deletedCond}";

        if ($search !== '') {
            $sql .= " AND (f.front LIKE ? OR f.back LIKE ?)";
        }
        $sql .= " ORDER BY {$orderBy}";

        $stmt = mysqli_prepare($conn, $sql);
        if (!$stmt) {
            return [];
        }

        if ($search !== '') {
            $like = '%' . $search . '%';
            mysqli_stmt_bind_param($stmt, "iiss", $retention, $deckId, $like, $like);
        } else {
            mysqli_stmt_bind_param($stmt, "ii", $retention, $deckId);
        }

        mysqli_stmt_execute($stmt);
        $result = mysqli_stmt_get_result($stmt);

        $cards = [];
        while ($row = mysqli_fetch_assoc($result)) {
            $cards[] = [
                'id'               => (int) $row['id'],
                'deck_id'          => (int) $row['deck_id'],
                'front'            => $row['front'],
                'back'             => $row['back'],
                'interval_days'    => (int) $row['interval_days'],
                'next_review_at'   => $row['next_review_at'],
                'review_count'     => (int) $row['review_count'],
                'lapses'           => (int) $row['lapses'],
                'created_at'       => $row['created_at'],
                'updated_at'       => $row['updated_at'],
                'archived_at'      => $row['deleted_at'],
                'days_left'        => $row['days_left'] !== null ? (int) $row['days_left'] : null,
                'last_reviewed_at' => $row['last_reviewed_at'],
                'accuracy'         => $row['accuracy'] !== null ? (int) $row['accuracy'] : null,
                'avg_response_ms'  => $row['avg_response_ms'] !== null ? (int) $row['avg_response_ms'] : null,
            ];
        }
        mysqli_stmt_close($stmt);

        return $cards;
    }

    /**
     * Fetch a single flashcard record and verify deck ownership.
     * 
     * @param mysqli $conn Active database connection instance.
     * @param int $cardId Target flashcard ID.
     * @param int $userId Authenticated user ID.
     * @return array|null Returns flashcard array if found and authorized, null otherwise.
     */
    public static function getFlashcardById(mysqli $conn, int $cardId, int $userId): ?array
    {
        $sql = "SELECT f.id, f.deck_id, f.front, f.back, f.next_review_at, f.interval_days, f.review_count, f.lapses, f.created_at, f.updated_at
                  FROM flashcards f
                  JOIN decks d ON f.deck_id = d.id
                 WHERE f.id = ? AND d.user_id = ? AND f.deleted_at IS NULL AND d.deleted_at IS NULL";

        $stmt = mysqli_prepare($conn, $sql);
        if (!$stmt) {
            return null;
        }

        mysqli_stmt_bind_param($stmt, "ii", $cardId, $userId);
        mysqli_stmt_execute($stmt);
        $result = mysqli_stmt_get_result($stmt);
        $card = mysqli_fetch_assoc($result);
        mysqli_stmt_close($stmt);

        return $card ?: null;
    }

    /**
     * Safely escape HTML string values for view rendering.
     * 
     * @param mixed $value Raw string content.
     * @return string Escaped HTML string.
     */
    public static function h($value): string
    {
        return htmlspecialchars((string) $value, ENT_QUOTES, 'UTF-8');
    }
}

    // API Helper with Database Connectivity Verification
    async function sendCardAction(payload) {
        try {
            const res = await fetch('../api/flashcard-actions.php', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ ...payload, csrf_token: CSRF_TOKEN, deck_id: DECK_ID })
            });
            let data = {};
            try { data = await res.json(); } catch (_) { }

            if (!res.ok || !data.ok) {
                throw new Error(data.error || `Database communication error (HTTP ${res.status}).`);
            }
            return data;
        } catch (error) {
            console.error("Database connection failure:", error);
            throw error;
        }
    }

    function formatDateTimeForInput(dateString) {
        if (!dateString) return '';
        const d = new Date(dateString);
        if (isNaN(d.getTime())) return '';
        const pad = (n) => String(n).padStart(2, '0');
        return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
    }

    function formatDateTimeDisplay(dateString) {
        if (!dateString) return '';
        const d = new Date(dateString);
        if (isNaN(d.getTime())) return dateString;
        const pad = (n) => String(n).padStart(2, '0');
        return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())} ${pad(d.getHours())}:${pad(d.getMinutes())}`;
    }

    // Populate Sidebar Editor with selected card data & metadata details
    function populateSidebar(cardData, rowInstance = null, { expand = true } = {}) {
        if (!cardData) return;
        if (!rowInstance && tabulatorTable) rowInstance = tabulatorTable.getRow(Number(cardData.id)) || null;
        selectedSidebarCardId = cardData.id != null ? Number(cardData.id) : null;
        if (rowInstance) selectedTabulatorRow = rowInstance;

        if (sidebarCardId) sidebarCardId.value = cardData.id || '';
        if (sidebarCardFront) sidebarCardFront.value = cardData.front || '';
        if (sidebarCardBack) sidebarCardBack.value = cardData.back || '';
        if (sidebarCardDue) sidebarCardDue.value = formatDateTimeForInput(cardData.next_review_at);
        if (sidebarCardError) sidebarCardError.classList.add('hidden');

        if (detLastReviewed) detLastReviewed.textContent = cardData.last_reviewed_at ? formatDateTimeDisplay(cardData.last_reviewed_at) : 'Never';
        if (detAccuracy) detAccuracy.textContent = cardData.accuracy !== null && cardData.accuracy !== undefined ? cardData.accuracy + '%' : '—';
        if (detSpeed) detSpeed.textContent = cardData.avg_response_ms !== null && cardData.avg_response_ms !== undefined ? (cardData.avg_response_ms / 1000).toFixed(1) + 's' : '—';
        if (detInterval) detInterval.textContent = cardData.interval_days !== undefined ? cardData.interval_days + (Number(cardData.interval_days) === 1 ? ' day' : ' days') : '—';
        if (detReviews) detReviews.textContent = cardData.review_count !== undefined ? cardData.review_count : '0';
        if (detLapses) detLapses.textContent = cardData.lapses !== undefined ? cardData.lapses : '0';

        setEditorReadOnly(viewArchived);
        updatePositionLabel();
        if (isSidebarCollapsed && expand) setSidebarCollapsed(false);
    }

    function pageStep() {
        const holder = gridContainer.querySelector('.tabulator-tableholder');
        const first = tabulatorTable.getRows()[0];
        const rowH = first ? first.getElement().offsetHeight : 0;
        if (!holder || !rowH) return 10;
        return Math.max(1, Math.floor(holder.clientHeight / rowH) - 1);   // one screenful, minus one row of overlap
    }

    // Make sure row #index (0-based) is rendered, pulling 50 more at a time
    async function ensureLoaded(index) {
        while (tabulatorTable.getDataCount() <= index && pending.length) {
            if (loadingMore) { await new Promise((r) => setTimeout(r, 30)); continue; }
            await loadMore();
        }
    }

    // Single place that makes a row "the" selected row: one highlight, editor in sync
    function activateRow(row, { expand = true, position = 'nearest' } = {}) {
        tabulatorTable.deselectRow(tabulatorTable.getSelectedRows());
        row.select();
        if (isDesktopView()) populateSidebar(row.getData(), row, { expand });
        Promise.resolve(row.scrollTo(position, false)).catch(() => { });
    }

    async function navigateSidebarRow(direction, { expand = true, step = 1 } = {}) {
        if (!tabulatorTable) return;
        let rows = tabulatorTable.getRows();
        if (!rows.length) return;

        const idx = rows.findIndex(r => Number(r.getData().id) === selectedSidebarCardId);
        let targetIdx;
        if (idx === -1) targetIdx = direction === 'next' ? 0 : rows.length - 1;
        else targetIdx = direction === 'next' ? idx + step : idx - step;

        if (targetIdx > rows.length - 1) {                 // past the loaded slice: pull more first
            await ensureLoaded(targetIdx);
            rows = tabulatorTable.getRows();
        }
        targetIdx = Math.max(0, Math.min(targetIdx, rows.length - 1));   // clamp at first/last card
        if (targetIdx === idx) return;
        activateRow(rows[targetIdx], { expand });
    }

        sidebarPrevBtn.addEventListener('click', () => navigateSidebarRow('prev'));
    }
    if (sidebarNextBtn) {
        sidebarNextBtn.addEventListener('click', () => navigateSidebarRow('next'));

    // ↑ / ↓ browse cards on desktop (ignored while typing in a field or when a dialog is open)
    const NAV_KEYS = {
        ArrowDown: ['next', false], ArrowUp: ['prev', false],
        PageDown: ['next', true], PageUp: ['prev', true],
    };
    document.addEventListener('keydown', (e) => {
        const nav = NAV_KEYS[e.key];
        if (!nav) return;
        if (!isDesktopView() || e.altKey || e.ctrlKey || e.metaKey || e.shiftKey) return;
        if (document.querySelector('dialog[open]')) return;
        if (e.target instanceof Element && e.target.closest('input, textarea, select, [contenteditable="true"]')) return;
        e.preventDefault();
        navigateSidebarRow(nav[0], { expand: false, step: nav[1] ? pageStep() : 1 });
    });

    // ---------------------------------------------------------------------------
    // Lazy loading: render 50 cards at a time
    // ---------------------------------------------------------------------------
    const PAGE_SIZE = 50;
    let pending = [];          // cards of the current view that aren't rendered yet
    let loadingMore = false;
    let gridReady = false;

    function updateLoadMoreUI() {
        if (!gridFooter || !tabulatorTable) return;
        const loaded = tabulatorTable.getDataCount();
        const remaining = pending.length;
        const total = loaded + remaining;
        const noun = total === 1 ? 'card' : 'cards';

        gridFooter.hidden = total === 0;
        rangeLabel.textContent = remaining ? `Showing ${loaded} of ${total} ${noun}` : `All ${total} ${noun} loaded`;
        progressBar.style.width = total ? `${Math.round((loaded / total) * 100)}%` : '0%';
        renderJumpRail();
    }

    function loadFirstPage(list) {
        pending = list.slice(PAGE_SIZE);
        return tabulatorTable.setData(list.slice(0, PAGE_SIZE)).then(updateLoadMoreUI);
    }

    async function loadMore() {
        if (loadingMore || !pending.length || !tabulatorTable) return;
        loadingMore = true;
        updateLoadMoreUI();
        try {
            await tabulatorTable.addData(pending.splice(0, PAGE_SIZE), false); // false = append at bottom
        } finally {
            loadingMore = false;
            updateLoadMoreUI();
        }
    }

    function bindGridScroll() {
        const holder = gridContainer.querySelector('.tabulator-tableholder');
        if (!holder) return;
        holder.addEventListener('scroll', () => {
            if (holder.scrollTop + holder.clientHeight >= holder.scrollHeight - 160) loadMore();
        }, { passive: true });
    }
    // ---- Jump rail: clickable record ranges beside the grid ----
    const MAX_RAIL_BUTTONS = 16;
    let railStep = PAGE_SIZE;
    let railTotal = -1;

    function renderJumpRail() {
        if (!jumpRail) return;
        const total = tabulatorTable ? tabulatorTable.getDataCount() + pending.length : 0;
        if (total === railTotal) return;
        railTotal = total;
        jumpRail.textContent = '';
        if (total <= PAGE_SIZE) { jumpRail.hidden = true; return; }

        railStep = PAGE_SIZE * Math.ceil(total / PAGE_SIZE / MAX_RAIL_BUTTONS);
        for (let start = 0; start < total; start += railStep) {
            const b = document.createElement('button');
            b.type = 'button';
            b.dataset.start = start;
            b.textContent = start + 1;
            b.title = `Jump to card ${start + 1}`;
            b.setAttribute('aria-label', `Jump to card ${start + 1}`);
            jumpRail.appendChild(b);
        }
        jumpRail.hidden = false;
        const idx = selectedSidebarCardId
            ? tabulatorTable.getRows().findIndex(r => Number(r.getData().id) === selectedSidebarCardId)
            : -1;
        highlightJumpRail(idx);
    }

    function highlightJumpRail(idx) {
        if (!jumpRail || jumpRail.hidden) return;
        const start = idx < 0 ? -1 : Math.floor(idx / railStep) * railStep;
        jumpRail.querySelectorAll('button').forEach((b) =>
            b.classList.toggle('is-active', Number(b.dataset.start) === start));
    }

    async function jumpToRecord(index) {
        if (!tabulatorTable) return;
        await ensureLoaded(index);
        const row = tabulatorTable.getRows()[index];
        if (!row) return;
        activateRow(row, { expand: false, position: 'top' });
        highlightJumpRail(index);
    }

    if (jumpRail) jumpRail.addEventListener('click', (e) => {
        const b = e.target.closest('button[data-start]');
        if (b) jumpToRecord(Number(b.dataset.start));
    });

    // 2. Save Form Handler (Shared by Sidebar & Modal)
    async function handleCardSave(cardId, front, back, nextReviewAt, errorEl, saveBtnEl, isSidebar) {
        if (front === '' || back === '') {
            if (errorEl) {
                errorEl.textContent = 'Both Front and Back content are required.';
                errorEl.classList.remove('hidden');
            }
            return;
        }

        if (saveBtnEl) saveBtnEl.disabled = true;

        try {
            const action = cardId ? 'update' : 'add';
            const res = await sendCardAction({ action, card_id: cardId, front, back, next_review_at: nextReviewAt });

            if (action === 'add') {
                totalCards++;
                if (tabulatorTable) {
                    const row = await tabulatorTable.addRow(res.card, true);
                    if (isSidebar && row) {                 // bind the editor to the new card so Save can't duplicate it
                        row.select();
                        populateSidebar(row.getData(), row);
                    }
                }
            } else {
                const row = tabulatorTable ? tabulatorTable.getRow(Number(res.card.id)) : null;
                if (row) await row.update(res.card);
                if (isSidebar && isDesktopView()) {
                    populateSidebar(row ? row.getData() : res.card, row || null);   // row data keeps accuracy/speed/etc.
                }
            }

            updateCardCountLabel();
            if (gridReady) updateLoadMoreUI();
            if (!isSidebar && cardModal) cardModal.close();
            Archive.toast(action === 'add' ? 'Card created successfully' : 'Card updated successfully');
        } catch (err) {
            if (errorEl) {
                errorEl.textContent = err.message;
                errorEl.classList.remove('hidden');
            }
        } finally {
            if (saveBtnEl) saveBtnEl.disabled = false;
        }
    }

        sidebarForm.addEventListener('submit', async (e) => {
            e.preventDefault();
            const cardId = sidebarCardId ? sidebarCardId.value.trim() : '';
            const front = sidebarCardFront ? sidebarCardFront.value.trim() : '';
            const back = sidebarCardBack ? sidebarCardBack.value.trim() : '';
            const nextReviewAt = sidebarCardDue ? sidebarCardDue.value.trim() : '';
            await handleCardSave(cardId, front, back, nextReviewAt, sidebarCardError, sidebarSaveBtn, true);
        });
    }

    async function archiveCard(card) {
        try {
            await sendCardAction({ action: 'archive', card_id: card.id });
            removeRow(card.id);
            offscreenList.unshift({ ...card, archived_at: new Date().toISOString(), days_left: Archive.RETENTION_DAYS });
            totalCards--;

            cardFront.focus();
        } else if (cardDeckSelect) {
            cardDeckSelect.focus();
        }
    }

    if (cardDeckSelect) {
        cardDeckSelect.addEventListener('change', () => {
            if (cardDeckSelect.value === '__create__') {
                cardDeckSelect.value = cardDeckId ? String(cardDeckId) : '';
                openCreateDialog(true);
                return;
            }

            const selectedId = Number(cardDeckSelect.value);
            cardDeckId = Number.isInteger(selectedId) && selectedId > 0 ? selectedId : null;
            if (cardError) cardError.classList.add('hidden');
        });
    }

    document.addEventListener('click', (e) => {
        const btn = e.target.closest('[data-action="add-card"]');
        if (btn) openCardDialog(btn.dataset.deckId);
    });

    [cardFront, cardBack].forEach((field) => {
        if (field) {
            field.addEventListener('keydown', (e) => {
                if (e.key === 'Enter' && (e.ctrlKey || e.metaKey)) {
                    e.preventDefault();
                    if (cardForm) cardForm.requestSubmit();
                }
            });
        }
    });

    if (cardForm) {
        cardForm.addEventListener('submit', async (e) => {
            e.preventDefault();
            const front = cardFront ? cardFront.value.trim() : '';
            const back = cardBack ? cardBack.value.trim() : '';
            const selectedDeckId = cardDeckSelect ? Number(cardDeckSelect.value) : null;

            if (cardSuccess) cardSuccess.classList.add('hidden');

            if (!Number.isInteger(selectedDeckId) || selectedDeckId <= 0) {
                cardDeckId = null;
                if (cardError) {
                    cardError.textContent = 'Please choose a deck before adding the card.';
                    cardError.classList.remove('hidden');
                }
                if (cardDeckSelect) cardDeckSelect.focus();
                return;
            }

            cardDeckId = selectedDeckId;

            if (front === '' || back === '') {
                if (cardError) {
                    cardError.textContent = 'Both the front and back are required.';
                    cardError.classList.remove('hidden');
                }
                return;
            }
            if (cardError) cardError.classList.add('hidden');

            if (cardSave) cardSave.disabled = true;
            try {
                await sendDeckAction({ action: 'add_card', deck_id: cardDeckId, front, back });
                cardsAdded = true;
                if (cardFront) cardFront.value = '';
                if (cardBack) cardBack.value = '';
                if (cardSuccess) cardSuccess.classList.remove('hidden');
                if (cardFront) cardFront.focus();
            } catch (err) {
                if (cardError) {
                    cardError.textContent = err.message;
                    cardError.classList.remove('hidden');
                }
            } finally {
                if (cardSave) cardSave.disabled = false;
            }
        });
    }

    function showFieldError(errorEl, input, err) {
        if (!errorEl) return;
        errorEl.textContent = err.message;
        if (err.suggestion && input) {
            const btn = document.createElement('button');
            btn.type = 'button';
            btn.className = 'ml-2 underline font-semibold';
            btn.textContent = 'Use "' + err.suggestion + '"';
            btn.addEventListener('click', () => {
                input.value = err.suggestion;
                errorEl.classList.add('hidden');
                input.focus();
            });
            errorEl.appendChild(btn);
        }
        errorEl.classList.remove('hidden');
    }

    if (cardDialog) {
        cardDialog.addEventListener('close', () => {
            if (cardsAdded) {
                const cleanUrl = window.location.pathname;
                window.history.replaceState({}, document.title, cleanUrl);
                window.location.reload();
            }
        });
    }

    // Auto-open modals via URL params
    if (config.createRequested) {
        openCreateDialog();
    } else if (config.addCardRequested) {
        openCardDialog(config.requestedDeckId);
    }

    // ---------------------------------------------------------------
    // 7. Custom Deck Reordering via LocalStorage & SortableJS
    // ---------------------------------------------------------------
    const ORDER_KEY = 'mempulse_deck_order';

    function applySavedDeckOrder(container) {
        if (!container) return;

        let savedOrder = null;
        try {
            savedOrder = JSON.parse(localStorage.getItem(ORDER_KEY));
        } catch (_) { }

        if (!Array.isArray(savedOrder) || savedOrder.length === 0) return;

        const items = Array.from(container.querySelectorAll('.deck-card[data-deck-id]'));
        const itemMap = new Map();
        items.forEach(item => itemMap.set(String(item.dataset.deckId), item));

        savedOrder.forEach(deckId => {
            const el = itemMap.get(String(deckId));
            if (el) {
                container.appendChild(el);
                itemMap.delete(String(deckId));
            }
        });

        const createBtn = container.querySelector('#new-deck-tile, #new-deck-row');
        itemMap.forEach(el => {
            if (createBtn) {
                container.insertBefore(el, createBtn);
            } else {
                container.appendChild(el);
            }
        });

        if (createBtn) {
            container.appendChild(createBtn);
        }
    }

    function saveCurrentDeckOrder(container) {
        if (!container) return;
        const items = Array.from(container.querySelectorAll('.deck-card[data-deck-id]'));
        const order = items.map(el => Number(el.dataset.deckId));
        try {
            localStorage.setItem(ORDER_KEY, JSON.stringify(order));
        } catch (_) { }
    }

    // Initialize Drag & Drop
    const gridContainer = document.getElementById('decks-grid-view');
    const listContainer = document.getElementById('decks-list-view');

    applySavedDeckOrder(gridContainer);
    applySavedDeckOrder(listContainer);

    if (gridContainer && typeof Sortable !== 'undefined') {
        new Sortable(gridContainer, {
            animation: 150,
            draggable: '.deck-card',
            filter: '[data-deck-menu], dialog, button, a',
            preventOnFilter: false,
            onEnd: () => {
                saveCurrentDeckOrder(gridContainer);
                applySavedDeckOrder(listContainer);
            }
        });
    }

    if (listContainer && typeof Sortable !== 'undefined') {
        new Sortable(listContainer, {
            animation: 150,
            draggable: '.deck-card',
            filter: '[data-deck-menu], dialog, button, a',
            preventOnFilter: false,
            onEnd: () => {
                saveCurrentDeckOrder(gridContainer);
                applySavedDeckOrder(listContainer);
            }
        });
    }
});

    document.addEventListener("DOMContentLoaded", () => {

                    mainContent.classList.toggle("collapsed", isCollapsed);
                }
                if (toggleIcon) {

            });

            sidebar.classList.remove("collapsed"); 
            mobileOverlay.classList.remove("hidden");

            setTimeout(() => mobileOverlay.classList.remove("opacity-0"), 10);

            sidebar.classList.add("-translate-x-full");
            mobileOverlay.classList.add("opacity-0");

            if (toggleIcon) {
                toggleIcon.textContent = sidebar.classList.contains("collapsed") ? "left_panel_open" : "left_panel_close";

            setTimeout(() => mobileOverlay.classList.add("hidden"), 300);

            mobileToggleBtn.addEventListener("click", openMobileSidebar);

        if (mobileOverlay) {

        const logoutBtn = document.getElementById('logout-btn');
