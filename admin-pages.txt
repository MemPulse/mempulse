CREATE DATABASE IF NOT EXISTS mempulse_db
CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

USE mempulse_db;


-- 1. USERS TABLE
CREATE TABLE users (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    email VARCHAR(255) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    password_changed_at DATETIME NULL,
    role ENUM('admin', 'user') NOT NULL DEFAULT 'user',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL,
    account_status ENUM('pending', 'approved', 'rejected') NOT NULL DEFAULT 'approved',
    INDEX idx_users_active (deleted_at)
) ENGINE=InnoDB;


-- 2. USER SETTINGS TABLE (SRS Configuration)
CREATE TABLE user_settings (
    user_id BIGINT UNSIGNED PRIMARY KEY,
    remember_multiplier DECIMAL(4,2) NOT NULL DEFAULT 2.00,
    forgot_multiplier DECIMAL(4,2) NOT NULL DEFAULT 0.50,
    initial_interval_days INT UNSIGNED NOT NULL DEFAULT 1,
    maximum_interval_days INT UNSIGNED NOT NULL DEFAULT 365,
    theme ENUM('light', 'dark', 'system')NOT NULL DEFAULT 'system',
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE
) ENGINE=InnoDB;


-- 3. DECKS TABLE
CREATE TABLE decks (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT UNSIGNED NOT NULL,
    title VARCHAR(150) NOT NULL,
    cover_image VARCHAR(255) DEFAULT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL,
    FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE,
    INDEX idx_decks_user_active (user_id, deleted_at)
) ENGINE=InnoDB;


-- 4. FLASHCARDS TABLE
CREATE TABLE flashcards (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    deck_id BIGINT UNSIGNED NOT NULL,
    front TEXT NOT NULL,
    back TEXT NOT NULL,
    interval_days INT UNSIGNED NOT NULL DEFAULT 1,
    next_review_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    review_count INT UNSIGNED NOT NULL DEFAULT 0,
    lapses INT UNSIGNED NOT NULL DEFAULT 0,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL,
    FOREIGN KEY (deck_id)
        REFERENCES decks(id)
        ON DELETE CASCADE,
    INDEX idx_cards_deck_due (deck_id, deleted_at, next_review_at)
) ENGINE=InnoDB;


-- 5. REVIEW SESSIONS TABLE
CREATE TABLE review_sessions (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT UNSIGNED NOT NULL,
    deck_id BIGINT UNSIGNED NOT NULL,
    started_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    ended_at DATETIME NULL,
    total_reviewed INT UNSIGNED NOT NULL DEFAULT 0,
    remembered_count INT UNSIGNED NOT NULL DEFAULT 0,
    accuracy_percentage DECIMAL(5,2)
        GENERATED ALWAYS AS (
            CASE
                WHEN total_reviewed > 0
                    THEN (remembered_count / total_reviewed) * 100
                ELSE 0.00
            END
        ) STORED,
    FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE,
    FOREIGN KEY (deck_id)
        REFERENCES decks(id)
        ON DELETE CASCADE,
    INDEX idx_user_sessions (user_id, started_at)
) ENGINE=InnoDB;


-- 6. REVIEW LOGS TABLE
CREATE TABLE review_logs (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    session_id BIGINT UNSIGNED NOT NULL,
    card_id BIGINT UNSIGNED NULL,
    rating ENUM('remembered', 'forgot') NOT NULL,
    previous_interval INT UNSIGNED NOT NULL,
    new_interval INT UNSIGNED NOT NULL,
    response_time_ms INT UNSIGNED NULL,
    reviewed_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (session_id)
        REFERENCES review_sessions(id)
        ON DELETE CASCADE,
    FOREIGN KEY (card_id)
        REFERENCES flashcards(id)
        ON DELETE SET NULL,
    INDEX idx_session_logs (session_id)
) ENGINE=InnoDB;

                return;

            const rect = target.getBoundingClientRect();
            const tipRect = floatingTooltip.getBoundingClientRect();
            const gap = 12;
            let left = rect.right + gap;
            // If there is not enough room to the right, place the tooltip to the left.
            if (left + tipRect.width > window.innerWidth - 8) {
                left = Math.max(8, rect.left - tipRect.width - gap);
                floatingTooltip.classList.add("tooltip-left");
            } else {
                floatingTooltip.classList.remove("tooltip-left");

            const top = Math.min(window.innerHeight - 8, Math.max(8, rect.top + rect.height / 2));

            settingsStatus.classList.toggle('text-error', isError);
            settingsStatus.classList.toggle('text-primary', !isError);

            Object.entries(settingsFields).forEach(([key, input]) => {
                if (input) input.value = settings[key];
            });

        async function loadSettings() {
            showSettingsStatus('Loading settings...');

            try {
                const response = await fetch(settingsApi, {
                    credentials: 'same-origin'
                });

                const result = await response.json();

                if (!response.ok || !result.success) {
                    throw new Error(result.message || 'Could not load settings.');
                }

                displaySettings(result.settings);
                showSettingsStatus('');
            } catch (error) {
                showSettingsStatus(error.message, true);
            }
        }

        async function saveSettings() {
            const payload = {};

            for (const [key, input] of Object.entries(settingsFields)) {
                if (!input || !input.checkValidity()) {
                    input?.reportValidity();
                    return;
                }

                payload[key] = input.value;
            }

            const initialValue = Number(payload.initial_interval_days);
            const maximumValue = Number(payload.maximum_interval_days);
            if (maximumValue < initialValue) {
                settingsFields.maximum_interval_days.setCustomValidity('Maximum interval must be at least the initial interval.');
                settingsFields.maximum_interval_days.reportValidity();
                settingsFields.maximum_interval_days.setCustomValidity('');
                return;
            }

            settingsSave.disabled = true;
            showSettingsStatus('Saving settings...');

            try {
                const response = await fetch(settingsApi, {
                    method: 'POST',
                    credentials: 'same-origin',
                    headers: {
                        'Content-Type': 'application/json',
                        'X-CSRF-Token': settingsDialog.dataset.csrf
                    },
                    body: JSON.stringify(payload)
                });

                const result = await response.json();

                if (!response.ok || !result.success) {
                    throw new Error(result.message || 'Could not save settings.');
                }

                displaySettings(result.settings);
                showSettingsStatus(result.message);
            } catch (error) {
                showSettingsStatus(error.message, true);
            } finally {
                settingsSave.disabled = false;
            }
        }

        settingsSave?.addEventListener('click', saveSettings);


        settingsOpeners.forEach((button) => {
            button.addEventListener('click', () => {
                const tabName = button.dataset.openSettings;

                setAccountPopoverOpen(false);
                switchSettingsTab(tabName);

                if (settingsDialog) {
                    if (!settingsDialog.open) {
                        settingsDialog.showModal();
                    }
                    loadSettings();
                }
            });
        });

        settingsTabs.forEach((tab) => {
            tab.addEventListener('click', () => {
                switchSettingsTab(tab.dataset.settingsTab);
            });
        });

        function closeSettings() {
            if (settingsDialog?.open) {
                settingsDialog.close();
            }
        }

        settingsClose?.addEventListener('click', closeSettings);
        settingsCancel?.addEventListener('click', closeSettings);

        // Close when clicking the backdrop
        settingsDialog?.addEventListener('click', (event) => {
            if (event.target === settingsDialog) {
                closeSettings();
            }
        });

        // Logout Dialog Logic
        const logoutBtn = document.getElementById('logout-btn');
        const logoutDialog = document.getElementById('logout-dialog');
        const logoutCancel = document.getElementById('logout-cancel');

        if (logoutBtn && logoutDialog) {
            logoutBtn.addEventListener('click', () => {
                setAccountPopoverOpen(false); // Close account menu
                logoutDialog.showModal();
            });

            if (logoutCancel) {
                logoutCancel.addEventListener('click', () => logoutDialog.close());
            }

            logoutDialog.addEventListener('click', (e) => {
                const rect = logoutDialog.getBoundingClientRect();
                if (e.clientX < rect.left || e.clientX > rect.right || e.clientY < rect.top || e.clientY > rect.bottom) {
                    logoutDialog.close();
                }
            });
        }
    });
</script>

<?php
/**
 * AdminOverviewService
 *
 * Read-only data for the admin Overview page. Aggregation and clock math happen
 * in SQL (no PHP/MySQL timezone drift). Every method returns a safe empty
 * structure on failure, never null.
 */
class AdminOverviewService
{
    /** Allowed chart ranges in days (fixed allow-list). */
    public const RANGES = [7, 30, 90];

    /**
     * Platform-wide headline numbers. Only approved, non-deleted users are
     * counted; decks and cards belonging to deleted users are excluded.
     *
     * @return array Keys: available, totalUsers, newUsers7d, pendingUsers, activeToday,
     *               active7d, decks, cards, reviews7d, reviewsPrev7d, retention30d.
     */
    public static function getMetrics(mysqli $conn): array
    {
        $empty = [
            'available' => false, 'totalUsers' => 0, 'newUsers7d' => 0, 'pendingUsers' => 0,
            'activeToday' => 0, 'active7d' => 0, 'decks' => 0, 'cards' => 0,
            'reviews7d' => 0, 'reviewsPrev7d' => 0, 'retention30d' => 0.0,
        ];

        $live = "deleted_at IS NULL AND account_status = 'approved'";
        $sql = "SELECT
            (SELECT COUNT(*) FROM users WHERE $live) AS total_users,
            (SELECT COUNT(*) FROM users WHERE $live AND created_at >= NOW() - INTERVAL 7 DAY) AS new_users_7d,
            (SELECT COUNT(*) FROM users WHERE deleted_at IS NULL AND account_status = 'pending') AS pending_users,
            (SELECT COUNT(DISTINCT user_id) FROM review_sessions WHERE started_at >= CURDATE()) AS active_today,
            (SELECT COUNT(DISTINCT user_id) FROM review_sessions WHERE started_at >= NOW() - INTERVAL 7 DAY) AS active_7d,
            (SELECT COUNT(*) FROM decks d JOIN users u ON u.id = d.user_id
              WHERE d.deleted_at IS NULL AND u.deleted_at IS NULL) AS decks,
            (SELECT COUNT(*) FROM flashcards f JOIN decks d ON d.id = f.deck_id JOIN users u ON u.id = d.user_id
              WHERE f.deleted_at IS NULL AND d.deleted_at IS NULL AND u.deleted_at IS NULL) AS cards,
            (SELECT COALESCE(SUM(total_reviewed), 0) FROM review_sessions
              WHERE ended_at IS NOT NULL AND started_at >= NOW() - INTERVAL 7 DAY) AS reviews_7d,
            (SELECT COALESCE(SUM(total_reviewed), 0) FROM review_sessions
              WHERE ended_at IS NOT NULL AND started_at >= NOW() - INTERVAL 14 DAY
                AND started_at < NOW() - INTERVAL 7 DAY) AS reviews_prev_7d,
            (SELECT COALESCE(ROUND(SUM(remembered_count) / NULLIF(SUM(total_reviewed), 0) * 100, 1), 0)
               FROM review_sessions
              WHERE ended_at IS NOT NULL AND started_at >= NOW() - INTERVAL 30 DAY) AS retention_30d";

        $res = mysqli_query($conn, $sql); // static SQL, no user input
        if (!$res) {
            return $empty;
        }
        $r = mysqli_fetch_assoc($res);
        mysqli_free_result($res);
        if (!$r) {
            return $empty;
        }

        return [
            'available'     => true,
            'totalUsers'    => (int) $r['total_users'],
            'newUsers7d'    => (int) $r['new_users_7d'],
            'pendingUsers'  => (int) $r['pending_users'],
            'activeToday'   => (int) $r['active_today'],
            'active7d'      => (int) $r['active_7d'],
            'decks'         => (int) $r['decks'],
            'cards'         => (int) $r['cards'],
            'reviews7d'     => (int) $r['reviews_7d'],
            'reviewsPrev7d' => (int) $r['reviews_prev_7d'],
            'retention30d'  => (float) $r['retention_30d'],
        ];
    }

    /**
     * Daily series with empty days filled so the chart axis is continuous.
     *
     * @param int $days One of self::RANGES (anything else falls back to 30).
     * @return array Keys: labels (Y-m-d), reviews, activeUsers, signups (parallel int arrays).
     */
    public static function getTrend(mysqli $conn, int $days = 30): array
    {
        $days = in_array($days, self::RANGES, true) ? $days : 30;
        $out = ['labels' => [], 'reviews' => [], 'activeUsers' => [], 'signups' => []];

        $today = null;
        if ($res = mysqli_query($conn, 'SELECT CURDATE()')) {
            $row = mysqli_fetch_row($res);
            $today = $row[0] ?? null;
            mysqli_free_result($res);
        }
        if (!$today) {
            return $out;
        }

        $offset = $days - 1;
        $sessions = self::fetchRows($conn,
            "SELECT DATE(started_at) AS d, SUM(total_reviewed) AS reviews, COUNT(DISTINCT user_id) AS active
             FROM review_sessions
             WHERE ended_at IS NOT NULL AND started_at >= CURDATE() - INTERVAL ? DAY
             GROUP BY d", 'i', [$offset]);
        $signups = self::fetchRows($conn,
            "SELECT DATE(created_at) AS d, COUNT(*) AS signups
             FROM users
             WHERE deleted_at IS NULL AND created_at >= CURDATE() - INTERVAL ? DAY
             GROUP BY d", 'i', [$offset]);

        $byDay = [];
        foreach ($sessions as $s) {
            $byDay[$s['d']]['reviews'] = (int) $s['reviews'];
            $byDay[$s['d']]['active']  = (int) $s['active'];
        }
        foreach ($signups as $s) {
            $byDay[$s['d']]['signups'] = (int) $s['signups'];
        }

        $end = new DateTimeImmutable($today);
        for ($i = $offset; $i >= 0; $i--) {
            $key = $end->modify("-$i days")->format('Y-m-d');
            $out['labels'][]      = $key;
            $out['reviews'][]     = $byDay[$key]['reviews'] ?? 0;
            $out['activeUsers'][] = $byDay[$key]['active'] ?? 0;
            $out['signups'][]     = $byDay[$key]['signups'] ?? 0;
        }
        return $out;
    }

    /**
     * Newest accounts (any status), for the sign-ups feed.
     *
     * @return array<int, array{id:int,name:string,email:string,role:string,status:string,age:string}>
     */
    public static function getRecentSignups(mysqli $conn, int $limit = 6): array
    {
        return self::userRows($conn, '', 'u.created_at DESC', $limit);
    }

    /**
     * Accounts waiting for approval, oldest first (the longest-waiting person is first in the queue).
     *
     * @return array<int, array{id:int,name:string,email:string,role:string,status:string,age:string}>
     */
    public static function getPendingUsers(mysqli $conn, int $limit = 5): array
    {
        return self::userRows($conn, "AND u.account_status = 'pending'", 'u.created_at ASC', $limit);
    }

    /**
     * Latest completed sessions platform-wide. Aggregates only: no deck titles or card content.
     *
     * @return array<int, array{name:string,cardsReviewed:int,accuracy:int,age:string}>
     */
    public static function getRecentSessions(mysqli $conn, int $limit = 6): array
    {
        $limit = max(1, min($limit, 50));
        $rows = self::fetchRows($conn,
            "SELECT u.first_name, u.last_name, u.email, rs.total_reviewed, rs.accuracy_percentage,
                    TIMESTAMPDIFF(SECOND, rs.started_at, NOW()) AS age_seconds
             FROM review_sessions rs
             JOIN users u ON u.id = rs.user_id
             WHERE rs.ended_at IS NOT NULL AND u.deleted_at IS NULL
             ORDER BY rs.started_at DESC LIMIT ?", 'i', [$limit]);

        $out = [];
        foreach ($rows as $r) {
            $out[] = [
                'name'          => self::displayName($r),
                'cardsReviewed' => (int) $r['total_reviewed'],
                'accuracy'      => (int) round((float) $r['accuracy_percentage']),
                'age'           => self::timeAgo((int) $r['age_seconds']),
            ];
        }
        return $out;
    }

    /**
     * Percentage change chip data, or null when there is nothing to compare.
     *
     * @return array{text:string,tone:string}|null tone is up, down or flat.
     */
    public static function percentChip(int $current, int $previous, string $suffix): ?array
    {
        if ($previous === 0) {
            return $current > 0 ? ['text' => 'New activity', 'tone' => 'up'] : null;
        }
        $pct = (int) round((($current - $previous) / $previous) * 100);
        if ($pct === 0) {
            return ['text' => '0% ' . $suffix, 'tone' => 'flat'];
        }
        return ['text' => ($pct > 0 ? '+' : '') . $pct . '% ' . $suffix, 'tone' => $pct > 0 ? 'up' : 'down'];
    }

    /** Human-friendly relative time from a seconds-ago value computed in SQL. */
    public static function timeAgo(int $seconds): string
    {
        if ($seconds < 60)    return 'Just now';
        if ($seconds < 3600)  return floor($seconds / 60) . ' min ago';
        if ($seconds < 86400) return floor($seconds / 3600) . ' hr ago';
        $d = (int) floor($seconds / 86400);
        return $d . ($d === 1 ? ' day ago' : ' days ago');
    }

    // ---- internals --------------------------------------------------------

    private static function userRows(mysqli $conn, string $extraWhere, string $order, int $limit): array
    {
        $limit = max(1, min($limit, 50));
        // $extraWhere and $order are fixed strings from the two callers above, never user input.
        $rows = self::fetchRows($conn,
            "SELECT u.id, u.first_name, u.last_name, u.email, u.role, u.account_status,
                    TIMESTAMPDIFF(SECOND, u.created_at, NOW()) AS age_seconds
             FROM users u
             WHERE u.deleted_at IS NULL $extraWhere
             ORDER BY $order LIMIT ?", 'i', [$limit]);

        $out = [];
        foreach ($rows as $r) {
            $out[] = [
                'id'     => (int) $r['id'],
                'name'   => self::displayName($r),
                'email'  => (string) $r['email'],
                'role'   => (string) $r['role'],
                'status' => (string) $r['account_status'],
                'age'    => self::timeAgo((int) $r['age_seconds']),
            ];
        }
        return $out;
    }

    private static function displayName(array $r): string
    {
        $n = trim(($r['first_name'] ?? '') . ' ' . ($r['last_name'] ?? ''));
        return $n !== '' ? $n : (string) ($r['email'] ?? 'Unknown user');
    }

    private static function fetchRows(mysqli $conn, string $sql, string $types = '', array $params = []): array
    {
        $rows = [];
        $stmt = mysqli_prepare($conn, $sql);
        if (!$stmt) {
            return $rows;
        }
        if ($types !== '') {
            mysqli_stmt_bind_param($stmt, $types, ...$params);
        }
        if (mysqli_stmt_execute($stmt)) {
            $res = mysqli_stmt_get_result($stmt);
            while ($res && ($row = mysqli_fetch_assoc($res))) {
                $rows[] = $row;
            }
        }
        mysqli_stmt_close($stmt);
        return $rows;
    }
}

<?php
/**
 * AdminUsersService
 *
 * Read-only user listing for the admin Users page. Sort columns and filter
 * values come from fixed allow-lists; all values are bound parameters.
 * Per-user aggregates only: no deck titles or card content.
 */
class AdminUsersService
{
    public const PER_PAGE = 15;
    public const STATUSES = ['pending', 'approved', 'rejected'];
    public const ROLES    = ['admin', 'user'];

    /** Sort key => SQL column (never taken from input directly). */
    private const SORTS = [
        'name'   => 'u.first_name',
        'email'  => 'u.email',
        'role'   => 'u.role',
        'status' => 'u.account_status',
        'joined' => 'u.created_at',
    ];

    /**
     * Validate and normalise GET input.
     *
     * @return array Keys: q, status, role, from, to, sort, dir, page.
     */
    public static function parseFilters(array $get): array
    {
        $str  = fn($k) => is_string($get[$k] ?? null) ? trim($get[$k]) : '';
        $date = function (string $v): string {
            $d = DateTimeImmutable::createFromFormat('!Y-m-d', $v);
            return ($d && $d->format('Y-m-d') === $v) ? $v : '';
        };
        $sort = $str('sort');

        return [
            'q'      => mb_substr($str('q'), 0, 100),
            'status' => in_array($str('status'), self::STATUSES, true) ? $str('status') : '',
            'role'   => in_array($str('role'), self::ROLES, true) ? $str('role') : '',
            'from'   => $date($str('from')),
            'to'     => $date($str('to')),
            'sort'   => isset(self::SORTS[$sort]) ? $sort : 'joined',
            'dir'    => $str('dir') === 'asc' ? 'asc' : 'desc',
            'page'   => max(1, (int) ($get['page'] ?? 1)),
        ];
    }

    /** Query string for links, keeping only non-default values. */
    public static function buildQuery(array $f, array $override = []): string
    {
        $f = array_merge($f, $override);
        $q = array_filter([
            'q' => $f['q'], 'status' => $f['status'], 'role' => $f['role'],
            'from' => $f['from'], 'to' => $f['to'],
            'sort' => $f['sort'] !== 'joined' ? $f['sort'] : '',
            'dir' => $f['dir'] !== 'desc' ? $f['dir'] : '',
            'page' => $f['page'] > 1 ? $f['page'] : '',
        ], fn($v) => $v !== '');
        return $q ? '?' . http_build_query($q) : '';
    }

    /**
     * Account counts per status (for the tabs), ignoring soft-deleted users.
     *
     * @return array Keys: all, pending, approved, rejected.
     */
    public static function getStatusCounts(mysqli $conn): array
    {
        $out = ['all' => 0, 'pending' => 0, 'approved' => 0, 'rejected' => 0];
        foreach (self::fetchRows($conn, "SELECT account_status, COUNT(*) AS n FROM users WHERE deleted_at IS NULL GROUP BY account_status") as $r) {
            if (isset($out[$r['account_status']])) {
                $out[$r['account_status']] = (int) $r['n'];
            }
            $out['all'] += (int) $r['n'];
        }
        return $out;
    }

    /**
     * One page of users with per-user aggregates.
     *
     * @return array Keys: available, rows, total, page, pages, first, last.
     */
    public static function getList(mysqli $conn, array $f): array
    {
        $empty = ['available' => false, 'rows' => [], 'total' => 0, 'page' => 1, 'pages' => 1, 'first' => 0, 'last' => 0];

        [$where, $types, $params] = self::buildWhere($f);

        $c = self::fetchRows($conn, "SELECT COUNT(*) AS n FROM users u WHERE $where", $types, $params);
        if (!$c) {
            return $empty;
        }
        $total = (int) $c[0]['n'];
        $pages = max(1, (int) ceil($total / self::PER_PAGE));
        $page  = min($f['page'], $pages);
        $offset = ($page - 1) * self::PER_PAGE;

        $col = self::SORTS[$f['sort']];
        $dir = $f['dir'] === 'asc' ? 'ASC' : 'DESC';
        $done = "s.user_id = u.id AND s.ended_at IS NOT NULL";

        $sql = "SELECT u.id, u.first_name, u.last_name, u.email, u.role, u.account_status, u.created_at,
                  (SELECT COUNT(*) FROM decks d WHERE d.user_id = u.id AND d.deleted_at IS NULL) AS decks,
                  (SELECT COUNT(*) FROM flashcards fc JOIN decks d ON d.id = fc.deck_id
                    WHERE d.user_id = u.id AND d.deleted_at IS NULL AND fc.deleted_at IS NULL) AS cards,
                  (SELECT COUNT(*) FROM review_sessions s WHERE $done) AS sessions,
                  (SELECT COALESCE(SUM(s.total_reviewed), 0) FROM review_sessions s WHERE $done) AS reviews,
                  (SELECT ROUND(SUM(s.remembered_count) / NULLIF(SUM(s.total_reviewed), 0) * 100, 1)
                     FROM review_sessions s WHERE $done) AS retention,
                  (SELECT TIMESTAMPDIFF(SECOND, MAX(s.started_at), NOW())
                     FROM review_sessions s WHERE s.user_id = u.id) AS last_active_seconds
                FROM users u
                WHERE $where
                ORDER BY $col $dir, u.id DESC
                LIMIT ? OFFSET ?";

        $rows = self::fetchRows($conn, $sql, $types . 'ii', array_merge($params, [self::PER_PAGE, $offset]));

        $out = [];
        foreach ($rows as $r) {
            $name = trim($r['first_name'] . ' ' . $r['last_name']);
            $out[] = [
                'id'         => (int) $r['id'],
                'name'       => $name !== '' ? $name : $r['email'],
                'email'      => (string) $r['email'],
                'role'       => (string) $r['role'],
                'status'     => (string) $r['account_status'],
                'joined'     => date('M j, Y', strtotime($r['created_at'])),
                'decks'      => (int) $r['decks'],
                'cards'      => (int) $r['cards'],
                'sessions'   => (int) $r['sessions'],
                'reviews'    => (int) $r['reviews'],
                'retention'  => $r['retention'] === null ? null : (float) $r['retention'],
                'lastActive' => $r['last_active_seconds'] === null ? 'Never' : AdminOverviewService::timeAgo((int) $r['last_active_seconds']),
            ];
        }

        return [
            'available' => true, 'rows' => $out, 'total' => $total, 'page' => $page, 'pages' => $pages,
            'first' => $total ? $offset + 1 : 0, 'last' => $offset + count($out),
        ];
    }

    // ---- internals --------------------------------------------------------

    /** @return array{0:string,1:string,2:array} where-clause, bind types, params. */
    private static function buildWhere(array $f): array
    {
        $where = ['u.deleted_at IS NULL'];
        $types = '';
        $params = [];

        if ($f['q'] !== '') {
            $like = '%' . addcslashes($f['q'], '%_\\') . '%';
            $where[] = "(u.first_name LIKE ? OR u.last_name LIKE ? OR u.email LIKE ? OR CONCAT(u.first_name, ' ', u.last_name) LIKE ?)";
            $types .= 'ssss';
            array_push($params, $like, $like, $like, $like);
        }
        if ($f['status'] !== '') { $where[] = 'u.account_status = ?'; $types .= 's'; $params[] = $f['status']; }
        if ($f['role'] !== '')   { $where[] = 'u.role = ?';           $types .= 's'; $params[] = $f['role']; }
        if ($f['from'] !== '')   { $where[] = 'u.created_at >= ?';    $types .= 's'; $params[] = $f['from'] . ' 00:00:00'; }
        if ($f['to'] !== '') {
            $where[] = 'u.created_at < ?';
            $types .= 's';
            $params[] = (new DateTimeImmutable($f['to']))->modify('+1 day')->format('Y-m-d') . ' 00:00:00';
        }
        return [implode(' AND ', $where), $types, $params];
    }

    private static function fetchRows(mysqli $conn, string $sql, string $types = '', array $params = []): array
    {
        $rows = [];
        $stmt = mysqli_prepare($conn, $sql);
        if (!$stmt) {
            return $rows;
        }
        if ($types !== '') {
            mysqli_stmt_bind_param($stmt, $types, ...$params);
        }
        if (mysqli_stmt_execute($stmt)) {
            $res = mysqli_stmt_get_result($stmt);
            while ($res && ($row = mysqli_fetch_assoc($res))) {
                $rows[] = $row;
            }
        }
        mysqli_stmt_close($stmt);
        return $rows;
    }
}
