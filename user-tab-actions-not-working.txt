/**

(function () {
    'use strict';

    var data = window.MemPulseAdminUsers || { users: {}, order: [] };
    var ui = window.MemPulseAdminUsersUI || { statusBadge: {}, roleBadge: {} };
    var FLASH_KEY = 'adminUsersFlash';

    var drawer = document.getElementById('user-drawer');
    var overlay = document.getElementById('user-drawer-overlay');
    var closeBtn = document.getElementById('user-drawer-close');
    var dialog = document.getElementById('user-action-dialog');
    var createDialog = document.getElementById('create-user-dialog');
    var createForm = document.getElementById('create-user-form');
    var toastEl = document.getElementById('admin-toast');

    var opener = null;
    var currentId = null;
    var busy = false;
    var toastTimer = null;
    var pending = null;

    // ---- Toast --------------------------------------------------------------
    function toast(message, ok) {
        if (!toastEl) return;
        toastEl.textContent = message;
        toastEl.className = (ui.toastBase || '') + ' ' + (ok ? ui.toastOk : ui.toastErr);
        clearTimeout(toastTimer);
        toastTimer = setTimeout(function () { toastEl.className = 'hidden'; }, 4500);
    }

    try {
        var flash = sessionStorage.getItem(FLASH_KEY);
        if (flash) {
            sessionStorage.removeItem(FLASH_KEY);
            toast(flash, true);
        }
    } catch (e) { }

    // ---- Auto-submit select filters -----------------------------------------
    document.querySelectorAll('[data-autosubmit]').forEach(function (el) {
        el.addEventListener('change', function () { el.form.submit(); });
    });

    // ---- Bulk Selection Checkboxes ------------------------------------------
    var selectAllBox = document.getElementById('select-all-users');
    var bulkBar = document.getElementById('bulk-action-bar');
    var bulkCountEl = document.getElementById('bulk-selected-count');

    function getSelectedIds() {
        var ids = [];
        document.querySelectorAll('.user-row-checkbox:checked').forEach(function (cb) {
            ids.push(Number(cb.value));
        });
        return ids;
    }

    function syncBulkBar() {
        var ids = getSelectedIds();
        if (!bulkBar) return;
        if (ids.length > 0) {
            bulkBar.classList.remove('hidden');
            bulkBar.classList.add('flex');
            if (bulkCountEl) bulkCountEl.textContent = String(ids.length);
        } else {
            bulkBar.classList.add('hidden');
            bulkBar.classList.remove('flex');
        }
    }

    if (selectAllBox) {
        selectAllBox.addEventListener('change', function () {
            document.querySelectorAll('.user-row-checkbox').forEach(function (cb) {
                cb.checked = selectAllBox.checked;
            });
            syncBulkBar();
        });
    }

    document.querySelectorAll('.user-row-checkbox').forEach(function (cb) {
        cb.addEventListener('change', syncBulkBar);
    });

    // ---- Drawer -------------------------------------------------------------
    function field(name) { return drawer ? drawer.querySelector('[data-field="' + name + '"]') : null; }
    function setText(name, value) { var el = field(name); if (el) el.textContent = value; }
    // `value` is the text shown; `key` (optional) is the raw value used to look up the badge colours.
    function setBadge(name, value, map, key) {
        var el = field(name);
        if (!el) return;
        el.textContent = value;
        el.className = (ui.badgeBase || '') + ' ' + (map[key || value] || '');
    }
    function num(n) { return Number(n || 0).toLocaleString(); }

    var AUDIT_LABELS = {
        user_approved: 'Approved account',
        user_rejected: 'Suspended account',
        user_deleted: 'Archived account',
        user_restored: 'Restored account',
        role_promoted_admin: 'Promoted to Admin',
        role_demoted_user: 'Changed role to User',
        user_created_by_admin: 'Created by Admin'
    };

    function renderAudit(items) {
        var container = document.getElementById('drawer-audit-list');
        if (!container) return;
        container.textContent = '';

        if (!items || items.length === 0) {
            var empty = document.createElement('p');
            empty.className = 'text-xs text-on-surface-variant italic py-2';
            empty.textContent = 'No admin actions recorded for this account yet.';
            container.appendChild(empty);
            return;
        }

        items.forEach(function (item) {
            var row = document.createElement('div');
            row.className = 'p-2.5 rounded-xl bg-surface-container-low border border-outline-variant/20 flex items-center justify-between gap-2 text-xs';

            var left = document.createElement('div');
            left.className = 'flex flex-col min-w-0';

            var title = document.createElement('span');
            title.className = 'font-bold text-on-surface truncate';
            title.textContent = AUDIT_LABELS[item.action] || item.action;

            var sub = document.createElement('span');
            sub.className = 'text-[11px] text-on-surface-variant truncate';
            sub.textContent = 'By ' + (item.by || 'Admin');

            left.appendChild(title);
            left.appendChild(sub);

            var right = document.createElement('span');
            right.className = 'text-[10px] text-on-surface-variant whitespace-nowrap';
            right.textContent = item.when || '';

            row.appendChild(left);
            row.appendChild(right);
            container.appendChild(row);
        });
    }

    function fill(u) {
        setText('initial', (u.name || '?').trim().charAt(0).toUpperCase());
        setText('name', u.name);
        setText('email', u.email);
        setBadge('status', (ui.statusLabels && ui.statusLabels[u.status]) || u.status, ui.statusBadge, u.status);
        setBadge('role', u.role, ui.roleBadge);
        var selfEl = field('self');
        if (selfEl) selfEl.classList.toggle('hidden', !u.isSelf);

        setText('decks', num(u.decks));
        setText('cards', num(u.cards));
        setText('sessions', num(u.sessions));
        setText('reviews', num(u.reviews) + (u.accuracy ? ' (' + u.accuracy + '%)' : ''));
        setText('lastActive', u.lastActive);
        setText('joined', u.joined);

        renderAudit(u.audit || []);

        var allowed = u.actions || ['approve', 'reject', 'delete', 'restore', 'make_admin', 'make_user'];
        drawer.querySelectorAll('#drawer-actions [data-action]').forEach(function (b) {
            b.classList.toggle('hidden', u.isSelf || allowed.indexOf(b.getAttribute('data-action')) === -1);
            b.disabled = false;
        });

        var locked = field('locked');
        if (locked) {
            if (u.isSelf) {
                locked.textContent = 'You cannot modify or archive your own account.';
                locked.classList.remove('hidden');
            } else {
                locked.classList.add('hidden');
            }
        }
    }

    function openDrawer(id, trigger) {
        if (!drawer || !overlay) return;
        var u = data.users[id];
        if (!u) return;
        opener = trigger || null;
        currentId = id;
        fill(u);
        drawer.classList.remove('hidden');
        overlay.classList.remove('hidden');
        requestAnimationFrame(function () {
            drawer.classList.add('is-open');
            overlay.classList.remove('opacity-0');
        });
        if (closeBtn) closeBtn.focus();
    }

    function closeDrawer() {
        if (!drawer || !overlay) return;
        drawer.classList.remove('is-open');
        overlay.classList.add('opacity-0');
        setTimeout(function () {
            drawer.classList.add('hidden');
            overlay.classList.add('hidden');
        }, 250);
        if (opener) opener.focus();
        currentId = null;
    }

    // ---- Server Requests ----------------------------------------------------
    function postJson(payload) {
        return fetch(data.endpoint, {
            method: 'POST',
            credentials: 'same-origin',
            headers: {
                'Content-Type': 'application/json',
                'Accept': 'application/json',
                'X-CSRF-Token': data.csrf
            },
            body: JSON.stringify(payload)
        }).then(function (res) {
            return res.json().catch(function () { return {}; }).then(function (body) {
                return { ok: res.ok && body.success === true, status: res.status, message: body.message };
            });
        });
    }

    function setBusy(on, trigger) {
        busy = on;
        if (trigger) trigger.disabled = on;
        var confirmBtn = document.getElementById('user-action-confirm');
        if (confirmBtn) confirmBtn.disabled = on;
    }

    function showDialogError(message) {
        var el = document.getElementById('user-action-error');
        if (!el) return;
        el.textContent = message || '';
        el.classList.toggle('hidden', !message);
    }

    function executePayload(payload, trigger) {
        if (busy) return;
        setBusy(true, trigger);
        postJson(payload).then(function (r) {
            if (r.ok) {
                try { sessionStorage.setItem(FLASH_KEY, r.message || 'Saved.'); } catch (e) { }
                window.location.reload();
                return;
            }
            setBusy(false, trigger);
            var msg = r.status === 401
                ? 'Your session has ended. Refresh the page and sign in again.'
                : (r.message || 'That action could not be completed.');
            if (dialog && dialog.open) { showDialogError(msg); } else { toast(msg, false); }
        }).catch(function () {
            setBusy(false, trigger);
            var msg = 'Could not reach the server. Check your connection and try again.';
            if (dialog && dialog.open) { showDialogError(msg); } else { toast(msg, false); }
        });
    }

    var COPY = {
        reject: {
            title: 'Suspend this account?',
            message: function (n) { return n + ' will not be able to sign in.'; },
            button: 'Suspend'
        },
        delete: {
            title: 'Archive this account?',
            message: function (n) { return n + ' will be archived and hidden from active lists.'; },
            button: 'Archive'
        },
        make_admin: {
            title: 'Promote to Administrator?',
            message: function (n) { return n + ' will have full access to the MemPulse Admin dashboard.'; },
            button: 'Promote to Admin'
        },
        make_user: {
            title: 'Demote to regular User?',
            message: function (n) { return n + ' will lose administrator privileges.'; },
            button: 'Change to User'
        }
    };

    function handleSingleAction(action, id, trigger) {
        var u = data.users[id];
        if (!u) return;

        if (action === 'approve' || action === 'restore' || !dialog || !COPY[action]) {
            executePayload({ action: action, user_id: Number(id) }, trigger);
            return;
        }

        pending = { payload: { action: action, user_id: Number(id) }, trigger: trigger };
        document.getElementById('user-action-title').textContent = COPY[action].title;
        document.getElementById('user-action-message').textContent = COPY[action].message(u.name);
        document.getElementById('user-action-confirm').textContent = COPY[action].button;
        showDialogError('');
        dialog.showModal();
    }

    function handleBulkAction(action, trigger) {
        var ids = getSelectedIds();
        if (ids.length === 0) return;

        if (action === 'approve' || action === 'restore' || !dialog) {
            executePayload({ action: action, user_ids: ids }, trigger);
            return;
        }

        var label = ids.length + (ids.length === 1 ? ' selected account' : ' selected accounts');
        pending = { payload: { action: action, user_ids: ids }, trigger: trigger };
        document.getElementById('user-action-title').textContent = COPY[action].title;
        document.getElementById('user-action-message').textContent = COPY[action].message(label);
        document.getElementById('user-action-confirm').textContent = COPY[action].button;
        showDialogError('');
        dialog.showModal();
    }

    if (dialog) {
        document.getElementById('user-action-cancel').addEventListener('click', function () { dialog.close(); });
        document.getElementById('user-action-confirm').addEventListener('click', function () {
            if (pending) executePayload(pending.payload, pending.trigger);
        });
        dialog.addEventListener('close', function () { pending = null; showDialogError(''); });
    }

    // ---- Create User Modal --------------------------------------------------
    var openCreateBtn = document.getElementById('open-create-user-btn');
    var createErrorEl = document.getElementById('create-user-error');

    if (openCreateBtn && createDialog) {
        openCreateBtn.addEventListener('click', function () {
            if (createForm) createForm.reset();
            if (createErrorEl) createErrorEl.classList.add('hidden');
            createDialog.showModal();
        });
        document.getElementById('create-user-close')?.addEventListener('click', function () { createDialog.close(); });
        document.getElementById('create-user-cancel')?.addEventListener('click', function () { createDialog.close(); });

        createForm?.addEventListener('submit', function (e) {
            e.preventDefault();
            if (busy) return;
            var submitBtn = document.getElementById('create-user-submit');
            var payload = {
                action: 'create_user',
                first_name: document.getElementById('cu-first').value.trim(),
                last_name: document.getElementById('cu-last').value.trim(),
                email: document.getElementById('cu-email').value.trim(),
                password: document.getElementById('cu-password').value,
                role: document.getElementById('cu-role').value,
                status: document.getElementById('cu-status').value
            };
            setBusy(true, submitBtn);
            postJson(payload).then(function (r) {
                if (r.ok) {
                    try { sessionStorage.setItem(FLASH_KEY, r.message || 'User created.'); } catch (err) { }
                    window.location.reload();
                    return;
                }
                setBusy(false, submitBtn);
                if (createErrorEl) {
                    createErrorEl.textContent = r.message || 'Could not create user.';
                    createErrorEl.classList.remove('hidden');
                }
            }).catch(function () {
                setBusy(false, submitBtn);
                if (createErrorEl) {
                    createErrorEl.textContent = 'Network error. Please try again.';
                    createErrorEl.classList.remove('hidden');
                }
            });
        });
    }

    // ---- Event Delegation ---------------------------------------------------
    document.addEventListener('click', function (e) {
        var bulkBtn = e.target.closest('[data-bulk-action]');
        if (bulkBtn) {
            e.preventDefault();
            handleBulkAction(bulkBtn.getAttribute('data-bulk-action'), bulkBtn);
            return;
        }

        var actBtn = e.target.closest('[data-action]');
        if (actBtn) {
            e.preventDefault();
            handleSingleAction(
                actBtn.getAttribute('data-action'),
                actBtn.getAttribute('data-user-id') || currentId,
                actBtn
            );
            return;
        }

        var manageBtn = e.target.closest('[data-user-id]');
        if (manageBtn) {
            e.preventDefault();
            openDrawer(manageBtn.getAttribute('data-user-id'), manageBtn);
            return;
        }

        var row = e.target.closest('[data-row-user-id]');
        if (row && !e.target.closest('button, a, input, label, [data-no-row]')) {
            openDrawer(row.getAttribute('data-row-user-id'), row);
        }
    });

    closeBtn?.addEventListener('click', closeDrawer);
    overlay?.addEventListener('click', closeDrawer);
    document.addEventListener('keydown', function (e) {
        if (e.key === 'Escape' && drawer && !drawer.classList.contains('hidden') && !(dialog && dialog.open)) {
            closeDrawer();
        }
    });
})();

        <script>
        (function () {
            var REGIONS = ['users-filter', 'users-table'];
            var tableEl = document.getElementById('users-table');
            if (!tableEl) return;

            var controller = null;
            history.replaceState({ usersAjax: true }, '', location.href);

            function load(url, push) {
                if (controller) controller.abort();
                var ctrl = controller = new AbortController();
                tableEl.classList.add('is-loading');

                fetch(url, {
                    credentials: 'same-origin',
                    headers: { 'X-Requested-With': 'fetch' },
                    signal: ctrl.signal
                })
                    .then(function (res) {
                        if (!res.ok) throw new Error('HTTP ' + res.status);
                        return res.text();
                    })
                    .then(function (html) {
                        var doc = new DOMParser().parseFromString(html, 'text/html');
                        var fresh = REGIONS.map(function (id) { return doc.getElementById(id); });
                        if (fresh.some(function (el) { return !el; })) throw new Error('Region not found');

                        // Keep the tab bar's horizontal scroll position
                        var nav = document.querySelector('#users-filter nav');
                        var tabScroll = nav ? nav.scrollLeft : 0;

                        REGIONS.forEach(function (id, i) {
                            document.getElementById(id).innerHTML = fresh[i].innerHTML;
                        });

                        var newNav = document.querySelector('#users-filter nav');
                        if (newNav) newNav.scrollLeft = tabScroll;

                        if (push) history.pushState({ usersAjax: true }, '', url);

                        // Lets your existing JS re-bind to the new rows (see note below)
                        document.dispatchEvent(new CustomEvent('users:refreshed'));
                    })
                    .catch(function (err) {
                        if (err.name !== 'AbortError') window.location.href = url; // fall back to a normal page load
                    })
                    .finally(function () {
                        if (controller === ctrl) {
                            controller = null;
                            tableEl.classList.remove('is-loading');
                        }
                    });
            }

            // Tabs, sort headers, pagination, Reset links
            document.addEventListener('click', function (e) {
                var a = e.target.closest('a[href]');
                if (!a) return;
                if (!a.closest('#users-filter, #users-table')) return;
                if (e.button !== 0 || e.metaKey || e.ctrlKey || e.shiftKey || e.altKey) return;
                if (a.target && a.target !== '_self') return;

                var url = new URL(a.href, location.href);
                if (url.origin !== location.origin || !/users\.php$/.test(url.pathname)) return;

                e.preventDefault();
                load(url.href, true);
            });

            // Search / filter form
            document.addEventListener('submit', function (e) {
                var form = e.target;
                if (!form.matches || !form.matches('form[data-ajax]')) return;
                e.preventDefault();

                var url = new URL(form.getAttribute('action'), location.href);
                url.search = new URLSearchParams(new FormData(form)).toString();
                load(url.href, true);
            });

            // Role dropdown (data-autosubmit): submit through the AJAX path instead of a full reload.
            // Capture phase so it runs before any existing listener that calls form.submit().
            document.addEventListener('change', function (e) {
                var el = e.target;
                if (!el.matches || !el.matches('[data-autosubmit]')) return;
                var form = el.form;

<?php
/**
 * AdminUsersService
 *
 * Industry-standard read & query service for the Admin Users page.
 * Supports active & soft-deleted accounts, role/status/date/search filters,
 * multi-column sorting, per-user SRS aggregate metrics, and per-user audit history.
 */
class AdminUsersService
{
    public const PER_PAGE = 15;
    public const STATUSES = ['pending', 'approved', 'rejected', 'deleted'];
    public const ROLES    = ['admin', 'user'];

    /** Sort key => SQL expression (strict allow-list). */
    private const SORTS = [
        'name'    => 'u.first_name',
        'email'   => 'u.email',
        'role'    => 'u.role',
        'status'  => 'u.account_status',
        'decks'   => 'decks',
        'reviews' => 'reviews',
        'joined'  => 'u.created_at',
    ];

    /**
     * Validate and normalise GET input.
     */
    public static function parseFilters(array $get): array
    {
        $str  = fn($k) => is_string($get[$k] ?? null) ? trim($get[$k]) : '';
        $date = function (string $v): string {
            $d = DateTimeImmutable::createFromFormat('!Y-m-d', $v);
            return ($d && $d->format('Y-m-d') === $v) ? $v : '';
        };
        $sort = $str('sort');

        return [
            'q'      => mb_substr($str('q'), 0, 100),
            'status' => in_array($str('status'), self::STATUSES, true) ? $str('status') : '',
            'role'   => in_array($str('role'), self::ROLES, true) ? $str('role') : '',
            'from'   => $date($str('from')),
            'to'     => $date($str('to')),
            'sort'   => isset(self::SORTS[$sort]) ? $sort : 'joined',
            'dir'    => $str('dir') === 'asc' ? 'asc' : 'desc',
            'page'   => max(1, (int) ($get['page'] ?? 1)),
        ];
    }

    /** Query string builder keeping only non-default parameters. */
    public static function buildQuery(array $f, array $override = []): string
    {
        $f = array_merge($f, $override);
        $q = array_filter([
            'q'      => $f['q'],
            'status' => $f['status'],
            'role'   => $f['role'],
            'from'   => $f['from'],
            'to'     => $f['to'],
            'sort'   => $f['sort'] !== 'joined' ? $f['sort'] : '',
            'dir'    => $f['dir'] !== 'desc' ? $f['dir'] : '',
            'page'   => $f['page'] > 1 ? $f['page'] : '',
        ], fn($v) => $v !== '');
        return $q ? '?' . http_build_query($q) : '';
    }

    /**
     * Account counts per status tab + headline KPIs.
     */
    public static function getStatusCounts(mysqli $conn): array
    {
        $out = [
            'all'      => 0,
            'pending'  => 0,
            'approved' => 0,
            'rejected' => 0,
            'deleted'  => 0,
            'admins'   => 0,
            'active7d' => 0,
        ];

        $rows = self::fetchRows(
            $conn,
            "SELECT
                SUM(CASE WHEN deleted_at IS NULL THEN 1 ELSE 0 END) AS all_live,
                SUM(CASE WHEN deleted_at IS NULL AND account_status = 'pending' THEN 1 ELSE 0 END) AS pending,
                SUM(CASE WHEN deleted_at IS NULL AND account_status = 'approved' THEN 1 ELSE 0 END) AS approved,
                SUM(CASE WHEN deleted_at IS NULL AND account_status = 'rejected' THEN 1 ELSE 0 END) AS rejected,
                SUM(CASE WHEN deleted_at IS NOT NULL THEN 1 ELSE 0 END) AS deleted,
                SUM(CASE WHEN deleted_at IS NULL AND role = 'admin' THEN 1 ELSE 0 END) AS admins,
                (SELECT COUNT(DISTINCT rs.user_id)
                   FROM review_sessions rs
                   JOIN users u2 ON u2.id = rs.user_id
                  WHERE u2.deleted_at IS NULL AND rs.started_at >= NOW() - INTERVAL 7 DAY) AS active_7d
             FROM users"
        );

        if ($rows) {
            $r = $rows[0];
            $out['all']      = (int) ($r['all_live'] ?? 0);
            $out['pending']  = (int) ($r['pending'] ?? 0);
            $out['approved'] = (int) ($r['approved'] ?? 0);
            $out['rejected'] = (int) ($r['rejected'] ?? 0);
            $out['deleted']  = (int) ($r['deleted'] ?? 0);
            $out['admins']   = (int) ($r['admins'] ?? 0);
            $out['active7d'] = (int) ($r['active_7d'] ?? 0);
        }

        return $out;
    }

    /**
     * Paginated (or unpaginated for CSV export) user list with per-user aggregates.
     */
    public static function getList(mysqli $conn, array $f, bool $exportAll = false): array
    {
        $empty = [
            'available' => false, 'rows' => [], 'total' => 0,
            'page' => 1, 'pages' => 1, 'first' => 0, 'last' => 0,
        ];

        [$where, $types, $params] = self::buildWhere($f);

        $c = self::fetchRows($conn, "SELECT COUNT(*) AS n FROM users u WHERE $where", $types, $params);
        if (!$c) {
            return $empty;
        }
        $total  = (int) $c[0]['n'];
        $pages  = max(1, (int) ceil($total / self::PER_PAGE));
        $page   = min($f['page'], $pages);
        $offset = ($page - 1) * self::PER_PAGE;

        $col  = self::SORTS[$f['sort']] ?? 'u.created_at';
        $dir  = $f['dir'] === 'asc' ? 'ASC' : 'DESC';
        $done = "s.user_id = u.id AND s.ended_at IS NOT NULL";

        $limitSql = $exportAll ? '' : 'LIMIT ? OFFSET ?';
        $sql = "SELECT u.id, u.first_name, u.last_name, u.email, u.role, u.account_status,
                       u.created_at, u.deleted_at,
                  (SELECT COUNT(*) FROM decks d WHERE d.user_id = u.id AND d.deleted_at IS NULL) AS decks,
                  (SELECT COUNT(*) FROM flashcards fc JOIN decks d ON d.id = fc.deck_id
                    WHERE d.user_id = u.id AND d.deleted_at IS NULL AND fc.deleted_at IS NULL) AS cards,
                  (SELECT COUNT(*) FROM review_sessions s WHERE $done) AS sessions,
                  (SELECT COALESCE(SUM(s.total_reviewed), 0) FROM review_sessions s WHERE $done) AS reviews,
                  (SELECT COALESCE(ROUND(SUM(s.remembered_count) / NULLIF(SUM(s.total_reviewed), 0) * 100, 1), 0)
                     FROM review_sessions s WHERE $done) AS accuracy,
                  (SELECT TIMESTAMPDIFF(SECOND, MAX(s.started_at), NOW())
                     FROM review_sessions s WHERE s.user_id = u.id) AS last_active_seconds
                FROM users u
                WHERE $where
                ORDER BY $col $dir, u.id DESC
                $limitSql";

        $queryTypes  = $exportAll ? $types : $types . 'ii';
        $queryParams = $exportAll ? $params : array_merge($params, [self::PER_PAGE, $offset]);
        $rows        = self::fetchRows($conn, $sql, $queryTypes, $queryParams);

        // Fetch recent audit logs for the returned users in one query
        $userIds   = array_map(fn($r) => (int) $r['id'], $rows);
        $auditByUser = self::getAuditHistoryForUsers($conn, $userIds);

        $out = [];
        foreach ($rows as $r) {
            $id        = (int) $r['id'];
            $name      = trim($r['first_name'] . ' ' . $r['last_name']);
            $isDeleted = $r['deleted_at'] !== null;
            $effStatus = $isDeleted ? 'deleted' : (string) $r['account_status'];

            $out[] = [
                'id'         => $id,
                'firstName'  => (string) $r['first_name'],
                'lastName'   => (string) $r['last_name'],
                'name'       => $name !== '' ? $name : (string) $r['email'],
                'email'      => (string) $r['email'],
                'role'       => (string) $r['role'],
                'status'     => $effStatus,
                'rawStatus'  => (string) $r['account_status'],
                'isDeleted'  => $isDeleted,
                'joined'     => date('M j, Y', strtotime($r['created_at'])),
                'joinedIso'  => date('Y-m-d', strtotime($r['created_at'])),
                'deletedAt'  => $isDeleted ? date('M j, Y', strtotime($r['deleted_at'])) : null,
                'decks'      => (int) $r['decks'],
                'cards'      => (int) $r['cards'],
                'sessions'   => (int) $r['sessions'],
                'reviews'    => (int) $r['reviews'],
                'accuracy'   => (float) $r['accuracy'],
                'lastActive' => $r['last_active_seconds'] === null
                    ? 'Never'
                    : AdminOverviewService::timeAgo((int) $r['last_active_seconds']),
                'audit'      => $auditByUser[$id] ?? [],
            ];
        }

        return [
            'available' => true,
            'rows'      => $out,
            'total'     => $total,
            'page'      => $page,
            'pages'     => $pages,
            'first'     => $total ? $offset + 1 : 0,
            'last'      => $offset + count($out),
        ];
    }

    /**
     * Fetch up to 8 recent audit events per user for the drawer timeline.
     */
    private static function getAuditHistoryForUsers(mysqli $conn, array $userIds): array
    {
        if (!$userIds) {
            return [];
        }
        $placeholders = implode(',', array_fill(0, count($userIds), '?'));
        $types        = str_repeat('i', count($userIds));
        $sql          = "SELECT a.target_user_id, a.action, a.previous_status, a.new_status, a.created_at,
                                CONCAT(COALESCE(adm.first_name,''), ' ', COALESCE(adm.last_name,'')) AS admin_name
                           FROM admin_audit_log a
                           LEFT JOIN users adm ON adm.id = a.admin_id
                          WHERE a.target_user_id IN ($placeholders)
                          ORDER BY a.created_at DESC";

        $rows = self::fetchRows($conn, $sql, $types, $userIds);
        $map  = [];
        foreach ($rows as $r) {
            $uid = (int) $r['target_user_id'];
            if (count($map[$uid] ?? []) >= 6) {
                continue;
            }
            $adminName = trim((string) $r['admin_name']);
            $map[$uid][] = [
                'action' => (string) $r['action'],
                'from'   => $r['previous_status'],
                'to'     => $r['new_status'],
                'by'     => $adminName !== '' ? $adminName : 'Admin',
                'when'   => date('M j, Y g:i A', strtotime($r['created_at'])),
            ];
        }
        return $map;
    }

    /** @return array{0:string,1:string,2:array} */
    private static function buildWhere(array $f): array
    {
        $where  = [];
        $types  = '';
        $params = [];

        if ($f['status'] === 'deleted') {
            $where[] = 'u.deleted_at IS NOT NULL';
        } else {
            $where[] = 'u.deleted_at IS NULL';
            if ($f['status'] !== '') {
                $where[]  = 'u.account_status = ?';
                $types   .= 's';
                $params[] = $f['status'];
            }
        }

        if ($f['q'] !== '') {
            $like    = '%' . addcslashes($f['q'], '%_\\') . '%';
            $where[] = "(u.first_name LIKE ? OR u.last_name LIKE ? OR u.email LIKE ? OR CONCAT(u.first_name, ' ', u.last_name) LIKE ?)";
            $types  .= 'ssss';
            array_push($params, $like, $like, $like, $like);
        }
        if ($f['role'] !== '') {
            $where[]  = 'u.role = ?';
            $types   .= 's';
            $params[] = $f['role'];
        }
        if ($f['from'] !== '') {
            $where[]  = 'u.created_at >= ?';
            $types   .= 's';
            $params[] = $f['from'] . ' 00:00:00';
        }
        if ($f['to'] !== '') {
            $where[]  = 'u.created_at < ?';
            $types   .= 's';
            $params[] = (new DateTimeImmutable($f['to']))->modify('+1 day')->format('Y-m-d') . ' 00:00:00';
        }

        return [implode(' AND ', $where), $types, $params];
    }

    private static function fetchRows(mysqli $conn, string $sql, string $types = '', array $params = []): array
    {
        $rows = [];
        $stmt = mysqli_prepare($conn, $sql);
        if (!$stmt) {
            return $rows;
        }
        if ($types !== '') {
            mysqli_stmt_bind_param($stmt, $types, ...$params);
        }
        if (mysqli_stmt_execute($stmt)) {
            $res = mysqli_stmt_get_result($stmt);
            while ($res && ($row = mysqli_fetch_assoc($res))) {
                $rows[] = $row;
            }
        }
        mysqli_stmt_close($stmt);
        return $rows;
    }
}

            'status' => in_array($str('status'), self::STATUSES, true) ? $str('status') : '',
            'role'   => in_array($str('role'), self::ROLES, true) ? $str('role') : '',
            'from'   => $date($str('from')),
            'to'     => $date($str('to')),
            'sort'   => isset(self::SORTS[$sort]) ? $sort : 'joined',
            'dir'    => $str('dir') === 'asc' ? 'asc' : 'desc',
            'page'   => max(1, (int) ($get['page'] ?? 1)),
        ];
    }

    /** Query string builder keeping only non-default parameters. */
    public static function buildQuery(array $f, array $override = []): string
    {
        $f = array_merge($f, $override);
        $q = array_filter([
            'q'      => $f['q'],
            'status' => $f['status'],
            'role'   => $f['role'],
            'from'   => $f['from'],
            'to'     => $f['to'],
            'sort'   => $f['sort'] !== 'joined' ? $f['sort'] : '',
            'dir'    => $f['dir'] !== 'desc' ? $f['dir'] : '',
            'page'   => $f['page'] > 1 ? $f['page'] : '',
        ], fn($v) => $v !== '');
        return $q ? '?' . http_build_query($q) : '';
    }

    /**
     * Account counts per status tab + headline KPIs.
     */
    public static function getStatusCounts(mysqli $conn): array
    {
        $out = [
            'all'      => 0,
            'pending'  => 0,
            'approved' => 0,
            'rejected' => 0,
            'deleted'  => 0,
            'admins'   => 0,
            'active7d' => 0,
        ];

        $rows = self::fetchRows(
            $conn,
            "SELECT
                SUM(CASE WHEN deleted_at IS NULL THEN 1 ELSE 0 END) AS all_live,
                SUM(CASE WHEN deleted_at IS NULL AND account_status = 'pending' THEN 1 ELSE 0 END) AS pending,
                SUM(CASE WHEN deleted_at IS NULL AND account_status = 'approved' THEN 1 ELSE 0 END) AS approved,
                SUM(CASE WHEN deleted_at IS NULL AND account_status = 'rejected' THEN 1 ELSE 0 END) AS rejected,
                SUM(CASE WHEN deleted_at IS NOT NULL THEN 1 ELSE 0 END) AS deleted,
                SUM(CASE WHEN deleted_at IS NULL AND role = 'admin' THEN 1 ELSE 0 END) AS admins,
                (SELECT COUNT(DISTINCT rs.user_id)
                   FROM review_sessions rs
                   JOIN users u2 ON u2.id = rs.user_id
                  WHERE u2.deleted_at IS NULL AND rs.started_at >= NOW() - INTERVAL 7 DAY) AS active_7d
             FROM users"
        );

        if ($rows) {
            $r = $rows[0];
            $out['all']      = (int) ($r['all_live'] ?? 0);
            $out['pending']  = (int) ($r['pending'] ?? 0);
            $out['approved'] = (int) ($r['approved'] ?? 0);
            $out['rejected'] = (int) ($r['rejected'] ?? 0);
            $out['deleted']  = (int) ($r['deleted'] ?? 0);
            $out['admins']   = (int) ($r['admins'] ?? 0);
            $out['active7d'] = (int) ($r['active_7d'] ?? 0);
        }

        return $out;
    }

    /**
     * Paginated (or unpaginated for CSV export) user list with per-user aggregates.
     */
    public static function getList(mysqli $conn, array $f, bool $exportAll = false): array
    {
        $empty = [
            'available' => false, 'rows' => [], 'total' => 0,
            'page' => 1, 'pages' => 1, 'first' => 0, 'last' => 0,
        ];

        [$where, $types, $params] = self::buildWhere($f);

        $c = self::fetchRows($conn, "SELECT COUNT(*) AS n FROM users u WHERE $where", $types, $params);
        if (!$c) {
            return $empty;
        }
        $total  = (int) $c[0]['n'];
        $pages  = max(1, (int) ceil($total / self::PER_PAGE));
        $page   = min($f['page'], $pages);
        $offset = ($page - 1) * self::PER_PAGE;

        $col  = self::SORTS[$f['sort']] ?? 'u.created_at';
        $dir  = $f['dir'] === 'asc' ? 'ASC' : 'DESC';
        $done = "s.user_id = u.id AND s.ended_at IS NOT NULL";

        $limitSql = $exportAll ? '' : 'LIMIT ? OFFSET ?';
        $sql = "SELECT u.id, u.first_name, u.last_name, u.email, u.role, u.account_status,
                       u.created_at, u.deleted_at,
                  (SELECT COUNT(*) FROM decks d WHERE d.user_id = u.id AND d.deleted_at IS NULL) AS decks,
                  (SELECT COUNT(*) FROM flashcards fc JOIN decks d ON d.id = fc.deck_id
                    WHERE d.user_id = u.id AND d.deleted_at IS NULL AND fc.deleted_at IS NULL) AS cards,
                  (SELECT COUNT(*) FROM review_sessions s WHERE $done) AS sessions,
                  (SELECT COALESCE(SUM(s.total_reviewed), 0) FROM review_sessions s WHERE $done) AS reviews,
                  (SELECT COALESCE(ROUND(SUM(s.remembered_count) / NULLIF(SUM(s.total_reviewed), 0) * 100, 1), 0)
                     FROM review_sessions s WHERE $done) AS accuracy,
                  (SELECT TIMESTAMPDIFF(SECOND, MAX(s.started_at), NOW())
                     FROM review_sessions s WHERE s.user_id = u.id) AS last_active_seconds
                FROM users u
                WHERE $where
                ORDER BY $col $dir, u.id DESC
                $limitSql";

        $queryTypes  = $exportAll ? $types : $types . 'ii';
        $queryParams = $exportAll ? $params : array_merge($params, [self::PER_PAGE, $offset]);
        $rows        = self::fetchRows($conn, $sql, $queryTypes, $queryParams);

        // Fetch recent audit logs for the returned users in one query
        $userIds   = array_map(fn($r) => (int) $r['id'], $rows);
        $auditByUser = self::getAuditHistoryForUsers($conn, $userIds);

        $out = [];
        foreach ($rows as $r) {
            $id        = (int) $r['id'];
            $name      = trim($r['first_name'] . ' ' . $r['last_name']);
            $isDeleted = $r['deleted_at'] !== null;
            $effStatus = $isDeleted ? 'deleted' : (string) $r['account_status'];

            $out[] = [
                'id'         => $id,
                'firstName'  => (string) $r['first_name'],
                'lastName'   => (string) $r['last_name'],
                'name'       => $name !== '' ? $name : (string) $r['email'],
                'email'      => (string) $r['email'],
                'role'       => (string) $r['role'],
                'status'     => $effStatus,
                'rawStatus'  => (string) $r['account_status'],
                'isDeleted'  => $isDeleted,
                'joined'     => date('M j, Y', strtotime($r['created_at'])),
                'joinedIso'  => date('Y-m-d', strtotime($r['created_at'])),
                'deletedAt'  => $isDeleted ? date('M j, Y', strtotime($r['deleted_at'])) : null,
                'decks'      => (int) $r['decks'],
                'cards'      => (int) $r['cards'],
                'sessions'   => (int) $r['sessions'],
                'reviews'    => (int) $r['reviews'],
                'accuracy'   => (float) $r['accuracy'],
                'lastActive' => $r['last_active_seconds'] === null
                    ? 'Never'
                    : AdminOverviewService::timeAgo((int) $r['last_active_seconds']),
                'audit'      => $auditByUser[$id] ?? [],
            ];
        }

        return [
            'available' => true,
            'rows'      => $out,
            'total'     => $total,
            'page'      => $page,
            'pages'     => $pages,
            'first'     => $total ? $offset + 1 : 0,
            'last'      => $offset + count($out),
        ];
    }

    /**
     * Fetch up to 8 recent audit events per user for the drawer timeline.
     */
    private static function getAuditHistoryForUsers(mysqli $conn, array $userIds): array
    {
        if (!$userIds) {
            return [];
        }
        $placeholders = implode(',', array_fill(0, count($userIds), '?'));
        $types        = str_repeat('i', count($userIds));
        $sql          = "SELECT a.target_user_id, a.action, a.previous_status, a.new_status, a.created_at,
                                CONCAT(COALESCE(adm.first_name,''), ' ', COALESCE(adm.last_name,'')) AS admin_name
                           FROM admin_audit_log a
                           LEFT JOIN users adm ON adm.id = a.admin_id
                          WHERE a.target_user_id IN ($placeholders)
                          ORDER BY a.created_at DESC";

        $rows = self::fetchRows($conn, $sql, $types, $userIds);
        $map  = [];
        foreach ($rows as $r) {
            $uid = (int) $r['target_user_id'];
            if (count($map[$uid] ?? []) >= 6) {
                continue;
            }
            $adminName = trim((string) $r['admin_name']);
            $map[$uid][] = [
                'action' => (string) $r['action'],
                'from'   => $r['previous_status'],
                'to'     => $r['new_status'],
                'by'     => $adminName !== '' ? $adminName : 'Admin',
                'when'   => date('M j, Y g:i A', strtotime($r['created_at'])),
            ];
        }
        return $map;
    }

    /** @return array{0:string,1:string,2:array} */
    private static function buildWhere(array $f): array
    {
        $where  = [];
        $types  = '';
        $params = [];

        if ($f['status'] === 'deleted') {
            $where[] = 'u.deleted_at IS NOT NULL';
        } else {
            $where[] = 'u.deleted_at IS NULL';
            if ($f['status'] !== '') {
                $where[]  = 'u.account_status = ?';
                $types   .= 's';
                $params[] = $f['status'];
            }
        }

        if ($f['q'] !== '') {
            $like    = '%' . addcslashes($f['q'], '%_\\') . '%';
            $where[] = "(u.first_name LIKE ? OR u.last_name LIKE ? OR u.email LIKE ? OR CONCAT(u.first_name, ' ', u.last_name) LIKE ?)";
            $types  .= 'ssss';
            array_push($params, $like, $like, $like, $like);
        }
        if ($f['role'] !== '') {
            $where[]  = 'u.role = ?';
            $types   .= 's';
            $params[] = $f['role'];
        }
        if ($f['from'] !== '') {
            $where[]  = 'u.created_at >= ?';
            $types   .= 's';
            $params[] = $f['from'] . ' 00:00:00';
        }
        if ($f['to'] !== '') {
            $where[]  = 'u.created_at < ?';
            $types   .= 's';
            $params[] = (new DateTimeImmutable($f['to']))->modify('+1 day')->format('Y-m-d') . ' 00:00:00';
        }

        return [implode(' AND ', $where), $types, $params];
    }

    private static function fetchRows(mysqli $conn, string $sql, string $types = '', array $params = []): array
    {
        $rows = [];
        $stmt = mysqli_prepare($conn, $sql);
        if (!$stmt) {
            return $rows;
        }
        if ($types !== '') {
            mysqli_stmt_bind_param($stmt, $types, ...$params);
        }
        if (mysqli_stmt_execute($stmt)) {
            $res = mysqli_stmt_get_result($stmt);
            while ($res && ($row = mysqli_fetch_assoc($res))) {
                $rows[] = $row;
            }
        }
        mysqli_stmt_close($stmt);
        return $rows;
    }
