    return '<span class="inline-block px-2 py-0.5 rounded-full border text-xs font-bold ' . $tone . '">' . $txt . '</span>';
};

$chartHeight = function (?array $chart): int {
    return $chart ? max(160, 56 + 30 * count($chart['labels'])) : 0;

/**
 * reports.js (core)
 *
 * - Filter form behaviour (auto-submit, custom date toggle)
 * - Accessible tabs (click, arrow keys, URL ?tab= sync)
 * - Chart registry so each tab builds its charts lazily, and ALL tabs
 *   build/resize their charts before printing.
 *
 * Tab scripts call: RP.register('overview', function () { ...build charts... });
 * and return Chart instances from RP.track(chart).
 */
(function () {
    'use strict';

    var TAB_ORDER = ['overview', 'decks', 'cards', 'sessions'];
    var initFns = {};      // tab -> init function
    var initDone = {};     // tab -> true once initialised
    var charts = [];       // every Chart instance, for resize on print

    var RP = {
        register: function (tab, fn) {
            initFns[tab] = fn;
            // If the tab is already visible when its script loads, init now.
            var panel = document.getElementById('panel-' + tab);
            if (panel && !panel.classList.contains('hidden')) {
                RP.init(tab);
            }
        },
        init: function (tab) {
            if (initDone[tab] || !initFns[tab]) return;
            initDone[tab] = true;
            try { initFns[tab](); } catch (e) { console.error('Reports: init failed for ' + tab, e); }
        },
        track: function (chart) {
            if (chart) charts.push(chart);
            return chart;
        },
        resizeAll: function () {
            charts.forEach(function (c) { try { c.resize(); } catch (e) { /* ignore */ } });
        },
        json: function (id) {
            var el = document.getElementById(id);
            if (!el) return null;
            try { return JSON.parse(el.textContent); } catch (e) { return null; }
        }
    };
    window.RP = RP;

    /* ---------------- Filters ---------------- */
    var form = document.getElementById('rp-filters');
    if (form) {
        var range = document.getElementById('rp-range');
        var custom = document.getElementById('rp-custom-dates');
        var from = document.getElementById('rp-from');
        var to = document.getElementById('rp-to');

        function setCustomVisible(show) {
            custom.classList.toggle('hidden', !show);
            from.disabled = !show;
            to.disabled = !show;
        }

        range.addEventListener('change', function () {
            if (range.value === 'custom') {
                setCustomVisible(true);   // wait for the user to press Apply
            } else {
                setCustomVisible(false);
                form.submit();
            }
        });

        ['rp-deck', 'rp-archived'].forEach(function (id) {
            var el = document.getElementById(id);
            if (el) el.addEventListener('change', function () { form.submit(); });
        });
    }

    /* ---------------- Tabs ---------------- */
    var tabs = Array.prototype.slice.call(document.querySelectorAll('[role="tab"][data-tab]'));
    var panels = {};
    document.querySelectorAll('.rp-panel').forEach(function (p) { panels[p.dataset.tab] = p; });
    var tabInput = document.getElementById('rp-filter-tab');

    var ACTIVE = ['border-primary', 'text-primary'];
    var INACTIVE = ['border-transparent', 'text-on-surface-variant', 'hover:text-on-surface'];

    function showTab(name, opts) {
        opts = opts || {};
        if (!panels[name]) return;

        tabs.forEach(function (t) {
            var on = t.dataset.tab === name;
            t.setAttribute('aria-selected', on ? 'true' : 'false');
            t.tabIndex = on ? 0 : -1;
            ACTIVE.forEach(function (c) { t.classList.toggle(c, on); });
            INACTIVE.forEach(function (c) { t.classList.toggle(c, !on); });
        });

        Object.keys(panels).forEach(function (key) {
            panels[key].classList.toggle('hidden', key !== name);
        });

        if (tabInput) tabInput.value = name;

        // Keep the URL shareable / refresh-safe without a reload.
        var url = new URL(window.location.href);
        url.searchParams.set('tab', name);
        window.history.replaceState(null, '', url.toString());

        // Build charts the first time a tab is shown; resize otherwise
        // (charts drawn in a hidden panel measure 0px wide).
        RP.init(name);
        RP.resizeAll();

        if (opts.focus) {
            var t = tabs.filter(function (x) { return x.dataset.tab === name; })[0];
            if (t) t.focus();
        }
    }

    tabs.forEach(function (t, i) {
        t.addEventListener('click', function (e) {
            e.preventDefault();
            showTab(t.dataset.tab);
        });
        t.addEventListener('keydown', function (e) {
            var idx = null;
            if (e.key === 'ArrowRight') idx = (i + 1) % tabs.length;
            else if (e.key === 'ArrowLeft') idx = (i - 1 + tabs.length) % tabs.length;
            else if (e.key === 'Home') idx = 0;
            else if (e.key === 'End') idx = tabs.length - 1;
            if (idx !== null) {
                e.preventDefault();
                showTab(tabs[idx].dataset.tab, { focus: true });
            }
        });
    });

    /* ---------------- Print ---------------- */
    var printBtn = document.getElementById('rp-print-btn');
    if (printBtn) printBtn.addEventListener('click', function () { window.print(); });

    // Print shows every panel. Canvases drawn at screen size (or inside hidden tabs)
    // print badly, so for printing we: reveal all panels, draw every tab's charts
    // without animation, and swap each canvas for a scalable <img> snapshot.
    var printState = null;

    window.addEventListener('beforeprint', function () {
        if (printState) return;

        var prevAnimation = window.Chart ? Chart.defaults.animation : undefined;
        if (window.Chart) Chart.defaults.animation = false;

        var revealed = [];
        Object.keys(panels).forEach(function (key) {
            if (panels[key].classList.contains('hidden')) {
                panels[key].classList.remove('hidden');
                revealed.push(key);
            }
        });

        TAB_ORDER.forEach(function (t) { RP.init(t); });

        var imgs = [];
        charts.forEach(function (c) {
            try {
                c.stop();
                c.resize();
                c.update('none');
                var img = document.createElement('img');
                img.className = 'rp-chart-img';
                img.alt = c.canvas.getAttribute('aria-label') || 'Chart';
                img.src = c.toBase64Image();
                c.canvas.parentNode.appendChild(img);
                imgs.push(img);
            } catch (e) { /* skip this chart */ }
        });

        document.documentElement.classList.add('rp-printing');
        printState = { revealed: revealed, imgs: imgs, prevAnimation: prevAnimation };
    });

    window.addEventListener('afterprint', function () {
        if (!printState) return;

        printState.imgs.forEach(function (img) { img.remove(); });
        printState.revealed.forEach(function (key) { panels[key].classList.add('hidden'); });
        document.documentElement.classList.remove('rp-printing');
        if (window.Chart) Chart.defaults.animation = printState.prevAnimation;
        printState = null;
        RP.resizeAll();
    });
})();

    (function initSort() {
        var table = document.getElementById('rp-decks-table');
        if (!table || !table.tBodies.length) return;

        var tbody = table.tBodies[0];
        var headers = Array.prototype.slice.call(table.querySelectorAll('thead th[data-sort]'));

        function cellValue(row, idx, type) {
            var cell = row.cells[idx];
            var v = cell ? cell.getAttribute('data-value') : '';
            if (type === 'num') {
                var n = parseFloat(v);
                return isNaN(n) ? -Infinity : n;
            }
            return (v || '').toString();
        }

        headers.forEach(function (th) {
            var btn = th.querySelector('button');
            if (!btn) return;

            btn.addEventListener('click', function () {
                var idx = th.cellIndex;
                var type = th.getAttribute('data-sort');
                // Text columns start ascending, numeric columns start descending.
                var current = th.getAttribute('aria-sort');
                var asc;
                if (current === 'none') asc = (type === 'text');
                else asc = (current === 'descending');

                headers.forEach(function (h) { h.setAttribute('aria-sort', 'none'); });
                th.setAttribute('aria-sort', asc ? 'ascending' : 'descending');

                var rows = Array.prototype.slice.call(tbody.rows);
                rows.sort(function (a, b) {
                    var va = cellValue(a, idx, type);
                    var vb = cellValue(b, idx, type);
                    var cmp = (type === 'num') ? (va - vb) : va.localeCompare(vb);
                    if (cmp === 0 || isNaN(cmp)) {
                        cmp = (a.cells[0].getAttribute('data-value') || '').localeCompare(b.cells[0].getAttribute('data-value') || '');
                    }
                    return asc ? cmp : -cmp;
                });
                rows.forEach(function (r) { tbody.appendChild(r); });
            });
        });
    })();

    /* ---------------- Charts ---------------- */
    function truncate(label, max) {
        label = String(label);
        return label.length > max ? label.slice(0, max - 1) + '…' : label;
    }

            indexAxis: 'y',

<?php
/**
 * ReportsService
 *
 * Read-only analytics queries for the Reports page.
 * Every query is scoped to the authenticated user and uses prepared statements.
 *
 * Conventions (same as DashboardService):
 *  - A "completed" session is one with ended_at IS NOT NULL.
 *  - Totals and accuracy come from review_sessions (they survive a card Reset).
 *  - review_logs is only for card-level detail (used by later tabs).
 */
class ReportsService
{
    public const RANGES = ['7d' => 7, '30d' => 30, '90d' => 90, 'all' => 0, 'custom' => 0];
    public const TABS   = ['overview', 'decks', 'cards', 'sessions'];

    /* ------------------------------------------------------------------ */
    /*  FILTERS                                                            */
    /* ------------------------------------------------------------------ */

    /**
     * Validate $_GET input into a safe filter array.
     */
    public static function parseFilters(mysqli $conn, int $userId, array $get): array
    {
        $range = (string) ($get['range'] ?? '30d');
        if (!array_key_exists($range, self::RANGES)) {
            $range = '30d';
        }

        $today = new DateTimeImmutable('today');
        $from  = $today->modify('-29 days');
        $to    = $today;

        if ($range === '7d' || $range === '30d' || $range === '90d') {
            $from = $today->modify('-' . (self::RANGES[$range] - 1) . ' days');
        } elseif ($range === 'all') {
            $first = self::firstSessionDate($conn, $userId);
            if ($first !== null) {
                $from = $first;
            }
        } else { // custom
            $f = self::parseDate((string) ($get['from'] ?? ''));
            $t = self::parseDate((string) ($get['to'] ?? ''));
            if ($f !== null && $t !== null) {
                if ($f > $t) {
                    [$f, $t] = [$t, $f];
                }
                if ($t > $today) {
                    $t = $today;
                }
                if ($f > $t) {
                    $f = $t;
                }
                $from = $f;
                $to   = $t;
            } else {
                $range = '30d'; // incomplete custom range falls back safely
            }
        }

        // Deck filter: must belong to this user. An archived deck switches the archived toggle on.
        $deckId          = 0;
        $deckTitle       = 'All decks';
        $includeArchived = (($get['archived'] ?? '') === '1');
        $rawDeck         = (string) ($get['deck'] ?? '');

        if ($rawDeck !== '' && ctype_digit($rawDeck) && (int) $rawDeck > 0) {
            $deck = self::fetchDeckOwned($conn, $userId, (int) $rawDeck);
            if ($deck !== null) {
                $deckId    = (int) $deck['id'];
                $deckTitle = $deck['title'];
                if ($deck['deleted_at'] !== null) {
                    $includeArchived = true;
                }
            }
        }

        $tab = (string) ($get['tab'] ?? 'overview');
        if (!in_array($tab, self::TABS, true)) {
            $tab = 'overview';
        }

        return [
            'range'           => $range,
            'from'            => $from->format('Y-m-d'),
            'to'              => $to->format('Y-m-d'),
            'fromDt'          => $from->format('Y-m-d') . ' 00:00:00',
            'toExclusive'     => $to->modify('+1 day')->format('Y-m-d') . ' 00:00:00',
            'days'            => (int) $from->diff($to)->days + 1,
            'deckId'          => $deckId,
            'deckTitle'       => $deckTitle,
            'includeArchived' => $includeArchived,
            'tab'             => $tab,
        ];
    }

    /**
     * Decks for the filter dropdown.
     */
    public static function getDeckOptions(mysqli $conn, int $userId, bool $includeArchived): array
    {
        $sql = "SELECT id, title, deleted_at FROM decks WHERE user_id = ?"
             . ($includeArchived ? "" : " AND deleted_at IS NULL")
             . " ORDER BY title ASC";
        $rows = self::fetchAll($conn, $sql, 'i', [$userId]);

        return array_map(fn($r) => [
            'id'       => (int) $r['id'],
            'title'    => $r['title'],
            'archived' => $r['deleted_at'] !== null,
        ], $rows);
    }

    /**
     * Build a query string that keeps the current filters (used by tab links and exports).
     */
    public static function buildQuery(array $f, array $override = []): string
    {
        $q = ['range' => $f['range']];
        if ($f['range'] === 'custom') {
            $q['from'] = $f['from'];
            $q['to']   = $f['to'];
        }
        if ($f['deckId'] > 0) {
            $q['deck'] = $f['deckId'];
        }
        if ($f['includeArchived']) {
            $q['archived'] = '1';
        }
        $q['tab'] = $f['tab'];

        return http_build_query(array_merge($q, $override));
    }

    /* ------------------------------------------------------------------ */
    /*  OVERVIEW TAB                                                       */
    /* ------------------------------------------------------------------ */

    public static function getOverview(mysqli $conn, int $userId, array $f): array
    {
        return [
            'kpis'     => self::getKpis($conn, $userId, $f),
            'timeline' => self::getTimeline($conn, $userId, $f),
            'forecast' => self::getForecast($conn, $userId, $f),
            'streak'   => self::getStreak($conn, $userId),
        ];
    }

    private static function getKpis(mysqli $conn, int $userId, array $f): array
    {
        [$where, $types, $params] = self::sessionScope($userId, $f);

        $sql = "SELECT
                    COALESCE(SUM(CASE WHEN s.ended_at IS NOT NULL THEN s.total_reviewed END), 0)   AS reviewed,
                    COALESCE(SUM(CASE WHEN s.ended_at IS NOT NULL THEN s.remembered_count END), 0) AS remembered,
                    COALESCE(SUM(CASE WHEN s.ended_at IS NOT NULL
                                      THEN TIMESTAMPDIFF(SECOND, s.started_at, s.ended_at) END), 0) AS seconds,
                    COUNT(CASE WHEN s.ended_at IS NOT NULL THEN 1 END) AS completed,
                    COUNT(CASE WHEN s.ended_at IS NULL THEN 1 END)     AS abandoned,
                    COUNT(DISTINCT CASE WHEN s.ended_at IS NOT NULL AND s.total_reviewed > 0
                                        THEN DATE(s.started_at) END)   AS active_days
                FROM review_sessions s
                JOIN decks d ON d.id = s.deck_id
                WHERE $where";

        $row = self::fetchAll($conn, $sql, $types, $params)[0] ?? [];

        $reviewed   = (int) ($row['reviewed'] ?? 0);
        $remembered = (int) ($row['remembered'] ?? 0);
        $completed  = (int) ($row['completed'] ?? 0);
        $abandoned  = (int) ($row['abandoned'] ?? 0);
        $started    = $completed + $abandoned;

        return [
            'totalReviews'   => $reviewed,
            'remembered'     => $remembered,
            'forgotten'      => max(0, $reviewed - $remembered),
            'accuracy'       => $reviewed > 0 ? round(($remembered / $reviewed) * 100, 1) : 0.0,
            'studySeconds'   => (int) ($row['seconds'] ?? 0),
            'sessions'       => $completed,
            'abandoned'      => $abandoned,
            'completionRate' => $started > 0 ? round(($completed / $started) * 100, 1) : 0.0,
            'activeDays'     => (int) ($row['active_days'] ?? 0),
            'rangeDays'      => (int) $f['days'],
        ];
    }

    /**
     * Reviews + accuracy per day (range <= 31 days) or per week (longer ranges).
     * Empty buckets are filled so the chart has a continuous axis.
     */
    private static function getTimeline(mysqli $conn, int $userId, array $f): array
    {
        [$where, $types, $params] = self::sessionScope($userId, $f);

        $granularity = ($f['days'] <= 31) ? 'day' : 'week';
        $bucketExpr  = ($granularity === 'day')
            ? "DATE(s.started_at)"
            : "DATE_SUB(DATE(s.started_at), INTERVAL WEEKDAY(s.started_at) DAY)";

        $sql = "SELECT $bucketExpr AS bucket,
                       SUM(s.total_reviewed)   AS reviewed,
                       SUM(s.remembered_count) AS remembered
                FROM review_sessions s
                JOIN decks d ON d.id = s.deck_id
                WHERE $where AND s.ended_at IS NOT NULL
                GROUP BY bucket
                ORDER BY bucket ASC";

        $byBucket = [];
        foreach (self::fetchAll($conn, $sql, $types, $params) as $r) {
            $byBucket[$r['bucket']] = [(int) $r['reviewed'], (int) $r['remembered']];
        }

        $from = new DateTimeImmutable($f['from']);
        $to   = new DateTimeImmutable($f['to']);

        if ($granularity === 'week') {
            // Snap to Monday so keys match MySQL WEEKDAY() bucketing.
            $from = $from->modify('-' . ((int) $from->format('N') - 1) . ' days');
        }

        $step = ($granularity === 'day') ? '+1 day' : '+7 days';
        $labels = $reviewed = $accuracy = [];

        for ($cursor = $from; $cursor <= $to; $cursor = $cursor->modify($step)) {
            $key = $cursor->format('Y-m-d');
            [$rv, $rm] = $byBucket[$key] ?? [0, 0];

            $labels[]   = $cursor->format('M j');
            $reviewed[] = $rv;
            $accuracy[] = $rv > 0 ? round(($rm / $rv) * 100, 1) : null;
        }

        return [
            'granularity' => $granularity,
            'labels'      => $labels,
            'reviewed'    => $reviewed,
            'accuracy'    => $accuracy,
        ];
    }

    /**
     * Cards coming due. Forward-looking, so the date range does not apply
     * (deck and archived filters do).
     */
    private static function getForecast(mysqli $conn, int $userId, array $f): array
    {
        $where  = "d.user_id = ? AND f.deleted_at IS NULL";
        $types  = 'i';
        $params = [$userId];

        if ($f['deckId'] > 0) {
            $where   .= " AND d.id = ?";
            $types   .= 'i';
            $params[] = $f['deckId'];
        }
        if (!$f['includeArchived']) {
            $where .= " AND d.deleted_at IS NULL";
        }

        $sql = "SELECT CASE WHEN f.next_review_at <= NOW() THEN -1
                            ELSE DATEDIFF(DATE(f.next_review_at), CURDATE()) END AS day_offset,
                       COUNT(*) AS cnt
                FROM flashcards f
                JOIN decks d ON d.id = f.deck_id
                WHERE $where AND f.next_review_at < (CURDATE() + INTERVAL 30 DAY)
                GROUP BY day_offset";

        $counts = [];
        foreach (self::fetchAll($conn, $sql, $types, $params) as $r) {
            $counts[(int) $r['day_offset']] = (int) $r['cnt'];
        }

        $dueNow = $counts[-1] ?? 0;
        $next7  = $dueNow;
        $next30 = $dueNow;
        $labels = ['Due now'];
        $values = [$dueNow];
        $today  = new DateTimeImmutable('today');

        for ($i = 0; $i < 30; $i++) {
            $c = $counts[$i] ?? 0;
            $next30 += $c;
            if ($i < 7) {
                $next7   += $c;
                $labels[] = ($i === 0) ? 'Later today' : (($i === 1) ? 'Tomorrow' : $today->modify("+$i days")->format('D j'));
                $values[] = $c;
            }
        }

        return [
            'dueNow' => $dueNow,
            'next7'  => $next7,
            'next30' => $next30,
            'labels' => $labels,
            'values' => $values,
        ];
    }

    /**
     * One row per deck: card stats (current state, not date-bound) plus
     * review activity inside the selected period.
     */
    public static function getDecksReport(mysqli $conn, int $userId, array $f): array
    {
        // 1) Card stats per deck. LEFT JOIN so empty decks still appear.
        $where  = "d.user_id = ?";
        $types  = 'i';
        $params = [$userId];
 
        if ($f['deckId'] > 0) {
            $where   .= " AND d.id = ?";
            $types   .= 'i';
            $params[] = $f['deckId'];
        }
        if (!$f['includeArchived']) {
            $where .= " AND d.deleted_at IS NULL";
        }
 
        $sql = "SELECT d.id, d.title, d.deleted_at,
                       COUNT(f.id) AS cards,
                       COALESCE(SUM(f.next_review_at <= NOW()), 0) AS due_now,
                       COALESCE(SUM(f.interval_days >= 21), 0)     AS mastered,
                       COALESCE(SUM(f.lapses >= 8), 0)             AS leeches
                FROM decks d
                LEFT JOIN flashcards f ON f.deck_id = d.id AND f.deleted_at IS NULL
                WHERE $where
                GROUP BY d.id, d.title, d.deleted_at
                ORDER BY d.title ASC";
 
        $decks = [];
        foreach (self::fetchAll($conn, $sql, $types, $params) as $r) {
            $id = (int) $r['id'];
            $decks[$id] = [
                'id'          => $id,
                'title'       => $r['title'],
                'archived'    => $r['deleted_at'] !== null,
                'cards'       => (int) $r['cards'],
                'dueNow'      => (int) $r['due_now'],
                'mastered'    => (int) $r['mastered'],
                'leeches'     => (int) $r['leeches'],
                'masteryPct'  => ((int) $r['cards']) > 0 ? (int) round(((int) $r['mastered'] / (int) $r['cards']) * 100) : 0,
                'reviews'     => 0,
                'remembered'  => 0,
                'sessions'    => 0,
                'seconds'     => 0,
                'accuracy'    => null,
                'lastStudied' => null,
            ];
        }
 
        if (empty($decks)) {
            return ['rows' => [], 'totals' => null, 'charts' => ['reviews' => null, 'mastery' => null]];
        }
 
        // 2) Completed-session activity inside the period.
        [$sWhere, $sTypes, $sParams] = self::sessionScope($userId, $f);
        $sql = "SELECT s.deck_id,
                       SUM(s.total_reviewed)   AS reviewed,
                       SUM(s.remembered_count) AS remembered,
                       COUNT(*)                AS sessions,
                       SUM(TIMESTAMPDIFF(SECOND, s.started_at, s.ended_at)) AS seconds
                FROM review_sessions s
                JOIN decks d ON d.id = s.deck_id
                WHERE $sWhere AND s.ended_at IS NOT NULL
                GROUP BY s.deck_id";
 
        foreach (self::fetchAll($conn, $sql, $sTypes, $sParams) as $r) {
            $id = (int) $r['deck_id'];
            if (!isset($decks[$id])) {
                continue;
            }
            $reviewed   = (int) $r['reviewed'];
            $remembered = (int) $r['remembered'];
 
            $decks[$id]['reviews']    = $reviewed;
            $decks[$id]['remembered'] = $remembered;
            $decks[$id]['sessions']   = (int) $r['sessions'];
            $decks[$id]['seconds']    = (int) $r['seconds'];
            $decks[$id]['accuracy']   = $reviewed > 0 ? round(($remembered / $reviewed) * 100, 1) : null;
        }
 
        // 3) Last studied: all time, not limited by the period.
        $sql = "SELECT s.deck_id, MAX(s.ended_at) AS last_at
                FROM review_sessions s
                WHERE s.user_id = ? AND s.ended_at IS NOT NULL
                GROUP BY s.deck_id";
 
        foreach (self::fetchAll($conn, $sql, 'i', [$userId]) as $r) {
            $id = (int) $r['deck_id'];
            if (isset($decks[$id])) {
                $decks[$id]['lastStudied'] = $r['last_at'];
            }
        }
 
        $rows = array_values($decks);
 
        // Default order: most reviewed first, then by title.
        usort($rows, function ($a, $b) {
            return [$b['reviews'], $a['title']] <=> [$a['reviews'], $b['title']];
        });
 
        // Totals row.
        $totals = ['decks' => count($rows), 'cards' => 0, 'dueNow' => 0, 'mastered' => 0, 'leeches' => 0,
                   'reviews' => 0, 'remembered' => 0, 'sessions' => 0, 'seconds' => 0];
        foreach ($rows as $r) {
            foreach (['cards', 'dueNow', 'mastered', 'leeches', 'reviews', 'remembered', 'sessions', 'seconds'] as $key) {
                $totals[$key] += $r[$key];
            }
        }
        $totals['accuracy']   = $totals['reviews'] > 0 ? round(($totals['remembered'] / $totals['reviews']) * 100, 1) : null;
        $totals['masteryPct'] = $totals['cards'] > 0 ? (int) round(($totals['mastered'] / $totals['cards']) * 100) : 0;
 
        // Chart data (top 10 each).
        $byReviews = array_values(array_filter($rows, fn($r) => $r['reviews'] > 0));
        $byReviews = array_slice($byReviews, 0, 10);
 
        $byCards = array_values(array_filter($rows, fn($r) => $r['cards'] > 0));
        usort($byCards, fn($a, $b) => [$b['cards'], $a['title']] <=> [$a['cards'], $b['title']]);
        $byCards = array_slice($byCards, 0, 10);
 
        return [
            'rows'   => $rows,
            'totals' => $totals,
            'charts' => [
                'reviews' => $byReviews ? [
                    'labels'   => array_column($byReviews, 'title'),
                    'reviews'  => array_column($byReviews, 'reviews'),
                    'accuracy' => array_column($byReviews, 'accuracy'),
                ] : null,
                'mastery' => $byCards ? [
                    'labels'   => array_column($byCards, 'title'),
                    'mastered' => array_column($byCards, 'mastered'),
                    'rest'     => array_map(fn($r) => $r['cards'] - $r['mastered'], $byCards),
                ] : null,
            ],
        ];
    }

    public static function formatDuration(int $seconds): string
    {
        if ($seconds < 60) {
            return $seconds . 's';
        }
        $h = intdiv($seconds, 3600);
        $m = intdiv($seconds % 3600, 60);

        return $h > 0 ? "{$h}h {$m}m" : "{$m}m";
    }

    public static function h($value): string
    {
        return htmlspecialchars((string) $value, ENT_QUOTES, 'UTF-8');
    }

    /**
     * Shared WHERE clause for review_sessions s (joined to decks d).
     * Returns [sql, bindTypes, bindParams].
     */
    private static function sessionScope(int $userId, array $f): array
    {
        $sql    = "s.user_id = ? AND s.started_at >= ? AND s.started_at < ?";
        $types  = 'iss';
        $params = [$userId, $f['fromDt'], $f['toExclusive']];

        if ($f['deckId'] > 0) {
            $sql    .= " AND s.deck_id = ?";
            $types  .= 'i';
            $params[] = $f['deckId'];
        }
        if (!$f['includeArchived']) {
            $sql .= " AND d.deleted_at IS NULL";
        }

        return [$sql, $types, $params];
    }

    private static function parseDate(string $value): ?DateTimeImmutable
    {
        $d = DateTimeImmutable::createFromFormat('!Y-m-d', $value);

        return ($d !== false && $d->format('Y-m-d') === $value) ? $d : null;
    }

    private static function firstSessionDate(mysqli $conn, int $userId): ?DateTimeImmutable
    {
        $rows = self::fetchAll($conn, "SELECT MIN(DATE(started_at)) AS first_day FROM review_sessions WHERE user_id = ?", 'i', [$userId]);
        $first = $rows[0]['first_day'] ?? null;

        return $first !== null ? self::parseDate((string) $first) : null;
    }

    private static function fetchDeckOwned(mysqli $conn, int $userId, int $deckId): ?array
    {
        $rows = self::fetchAll($conn, "SELECT id, title, deleted_at FROM decks WHERE id = ? AND user_id = ?", 'ii', [$deckId, $userId]);

        return $rows[0] ?? null;
    }

    private static function fetchAll(mysqli $conn, string $sql, string $types, array $params): array
    {
        $stmt = mysqli_prepare($conn, $sql);
        if (!$stmt) {
            return [];
        }
        if ($types !== '') {
            mysqli_stmt_bind_param($stmt, $types, ...$params);
        }
        mysqli_stmt_execute($stmt);
        $result = mysqli_stmt_get_result($stmt);
        $rows   = $result ? mysqli_fetch_all($result, MYSQLI_ASSOC) : [];
        mysqli_stmt_close($stmt);

        return $rows;
    }

CREATE DATABASE IF NOT EXISTS mempulse_db
CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

USE mempulse_db;


-- 1. USERS TABLE
CREATE TABLE users (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    email VARCHAR(255) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    role ENUM('admin', 'user') NOT NULL DEFAULT 'user',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL,
    INDEX idx_users_active (deleted_at)
) ENGINE=InnoDB;


-- 2. USER SETTINGS TABLE (SRS Configuration)
CREATE TABLE user_settings (
    user_id BIGINT UNSIGNED PRIMARY KEY,
    remember_multiplier DECIMAL(4,2) NOT NULL DEFAULT 2.00,
    forgot_multiplier DECIMAL(4,2) NOT NULL DEFAULT 0.50,
    initial_interval_days INT UNSIGNED NOT NULL DEFAULT 1,
    maximum_interval_days INT UNSIGNED NOT NULL DEFAULT 365,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE
) ENGINE=InnoDB;


-- 3. DECKS TABLE
CREATE TABLE decks (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT UNSIGNED NOT NULL,
    title VARCHAR(150) NOT NULL,
    cover_image VARCHAR(255) DEFAULT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL,
    FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE,
    INDEX idx_decks_user_active (user_id, deleted_at)
) ENGINE=InnoDB;


-- 4. FLASHCARDS TABLE
CREATE TABLE flashcards (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    deck_id BIGINT UNSIGNED NOT NULL,
    front TEXT NOT NULL,
    back TEXT NOT NULL,
    interval_days INT UNSIGNED NOT NULL DEFAULT 1,
    next_review_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    review_count INT UNSIGNED NOT NULL DEFAULT 0,
    lapses INT UNSIGNED NOT NULL DEFAULT 0,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL,
    FOREIGN KEY (deck_id)
        REFERENCES decks(id)
        ON DELETE CASCADE,
    INDEX idx_cards_deck_due (deck_id, deleted_at, next_review_at)
) ENGINE=InnoDB;


-- 5. REVIEW SESSIONS TABLE
CREATE TABLE review_sessions (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT UNSIGNED NOT NULL,
    deck_id BIGINT UNSIGNED NOT NULL,
    started_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    ended_at DATETIME NULL,
    total_reviewed INT UNSIGNED NOT NULL DEFAULT 0,
    remembered_count INT UNSIGNED NOT NULL DEFAULT 0,
    accuracy_percentage DECIMAL(5,2)
        GENERATED ALWAYS AS (
            CASE
                WHEN total_reviewed > 0
                    THEN (remembered_count / total_reviewed) * 100
                ELSE 0.00
            END
        ) STORED,
    FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE,
    FOREIGN KEY (deck_id)
        REFERENCES decks(id)
        ON DELETE CASCADE,
    INDEX idx_user_sessions (user_id, started_at)
) ENGINE=InnoDB;


-- 6. REVIEW LOGS TABLE
CREATE TABLE review_logs (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    session_id BIGINT UNSIGNED NOT NULL,
    card_id BIGINT UNSIGNED NULL,
    rating ENUM('remembered', 'forgot') NOT NULL,
    previous_interval INT UNSIGNED NOT NULL,
    new_interval INT UNSIGNED NOT NULL,
    response_time_ms INT UNSIGNED NULL,
    reviewed_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (session_id)
        REFERENCES review_sessions(id)
        ON DELETE CASCADE,
    FOREIGN KEY (card_id)
        REFERENCES flashcards(id)
        ON DELETE SET NULL,
    INDEX idx_session_logs (session_id)
) ENGINE=InnoDB;
