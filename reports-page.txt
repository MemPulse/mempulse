        return Math.max(1, Math.floor(holder.clientHeight / rowH) - 1);   // one screenful, minus one row of overlap

    async function ensureLoaded(index) {
        while (tabulatorTable.getDataCount() <= index && pending.length) {
            if (loadingMore) { await new Promise((r) => setTimeout(r, 30)); continue; }
            await loadMore();
        }

    // Single place that makes a row "the" selected row: one highlight, editor in sync

        tabulatorTable.deselectRow(tabulatorTable.getSelectedRows());
        row.select();
        if (isDesktopView()) populateSidebar(row.getData(), row, { expand });
        Promise.resolve(row.scrollTo(position, false)).catch(() => { });

    async function navigateSidebarRow(direction, { expand = true, step = 1 } = {}) {
        if (!tabulatorTable) return;
        let rows = tabulatorTable.getRows();
        if (!rows.length) return;

        const idx = rows.findIndex(r => Number(r.getData().id) === selectedSidebarCardId);
        let targetIdx;
        if (idx === -1) targetIdx = direction === 'next' ? 0 : rows.length - 1;
        else targetIdx = direction === 'next' ? idx + step : idx - step;

        if (targetIdx > rows.length - 1) {                 // past the loaded slice: pull more first
            await ensureLoaded(targetIdx);
            rows = tabulatorTable.getRows();
        }
        targetIdx = Math.max(0, Math.min(targetIdx, rows.length - 1));   // clamp at first/last card
        if (targetIdx === idx) return;
        activateRow(rows[targetIdx], { expand });
    }

        sidebarNextBtn.addEventListener('click', () => navigateSidebarRow('next'));

    // ↑ / ↓ browse cards on desktop (ignored while typing in a field or when a dialog is open)
    const NAV_KEYS = {
        ArrowDown: ['next', false], ArrowUp: ['prev', false],
        PageDown: ['next', true], PageUp: ['prev', true],
    };
    document.addEventListener('keydown', (e) => {
        // 1. Ctrl + Return to save changes
        if ((e.ctrlKey || e.metaKey) && e.key === 'Enter') {
            if (isDesktopView() && !isSidebarCollapsed && !viewArchived && (checkedIds.size > 0 || !selectedSidebarCardId)) {
                e.preventDefault();
                // If the save button isn't hidden (meaning it's not locked), trigger the click
                if (sidebarSaveBtn && !sidebarSaveBtn.classList.contains('hidden')) {
                    sidebarSaveBtn.click();
                }
            }
            return;
        }

        // Guard against intercepting keystrokes while typing in an input or a modal
        if (document.querySelector('dialog[open]')) return;
        if (e.target instanceof Element && e.target.closest('input, textarea, select, [contenteditable="true"]')) return;

        // 2. Return key to toggle sidebar collapse
        if (e.key === 'Enter' && !e.ctrlKey && !e.metaKey && !e.shiftKey && !e.altKey) {
            if (!isDesktopView() || viewArchived) return;
            e.preventDefault();
            setSidebarCollapsed(!isSidebarCollapsed);
            return;
        }

        // 3. Arrow Up/Down Navigation
        const nav = NAV_KEYS[e.key];
        if (!nav) return;
        if (!isDesktopView() || viewArchived || e.altKey || e.ctrlKey || e.metaKey || e.shiftKey) return;
        e.preventDefault();
        navigateSidebarRow(nav[0], { expand: false, step: nav[1] ? pageStep() : 1 });
    });

    function renderJumpRail() {
        if (!jumpRail) return;
        const total = tabulatorTable ? tabulatorTable.getDataCount() + pending.length : 0;
        if (total === railTotal) return;
        railTotal = total;
        jumpRail.textContent = '';
        if (total <= PAGE_SIZE) { jumpRail.hidden = true; return; }

        railStep = PAGE_SIZE * Math.ceil(total / PAGE_SIZE / MAX_RAIL_BUTTONS);
        for (let start = 0; start < total; start += railStep) {
            const b = document.createElement('button');
            b.type = 'button';
            b.dataset.start = start;
            b.textContent = start + 1;
            b.title = `Jump to card ${start + 1}`;
            b.setAttribute('aria-label', `Jump to card ${start + 1}`);
            jumpRail.appendChild(b);
        }
        jumpRail.hidden = false;
        const idx = selectedSidebarCardId
            ? tabulatorTable.getRows().findIndex(r => Number(r.getData().id) === selectedSidebarCardId)
            : -1;
        highlightJumpRail(idx);
    }

        jumpRail.querySelectorAll('button').forEach((b) =>
            b.classList.toggle('is-active', Number(b.dataset.start) === start));
    }

    async function handleCardSave(cardId, front, back, nextReviewAt, errorEl, saveBtnEl, isSidebar) {
        if (front === '' || back === '') {
            if (errorEl) {
                errorEl.textContent = 'Both Front and Back content are required.';
                errorEl.classList.remove('hidden');
            }
            return;
        }

        if (saveBtnEl) saveBtnEl.disabled = true;

        try {
            const action = cardId ? 'update' : 'add';
            const res = await sendCardAction({ action, card_id: cardId, front, back, next_review_at: nextReviewAt });

            if (action === 'add') {
                totalCards++;
                if (tabulatorTable) {
                    const row = await tabulatorTable.addRow(res.card, true);
                    if (isSidebar && row) {                 // bind the editor to the new card so Save can't duplicate it
                        row.select();
                        populateSidebar(row.getData(), row);
                    }
                }
            } else {
                const row = tabulatorTable ? tabulatorTable.getRow(Number(res.card.id)) : null;
                if (row) await row.update(res.card);
                if (isSidebar && isDesktopView()) {
                    populateSidebar(row ? row.getData() : res.card, row || null);   // row data keeps accuracy/speed/etc.
                }
            }

            updateCardCountLabel();
            if (gridReady) updateLoadMoreUI();
            if (!isSidebar && cardModal) cardModal.close();
            Archive.toast(action === 'add' ? 'Card created successfully' : 'Card updated successfully');
        } catch (err) {
            if (errorEl) {
                errorEl.textContent = err.message;
                errorEl.classList.remove('hidden');
            }
        } finally {
            if (saveBtnEl) saveBtnEl.disabled = false;
        }
    }

    async function archiveCard(card) {
        try {
            await sendCardAction({ action: 'archive', card_id: card.id });
            removeRow(card.id);
            offscreenList.unshift({ ...card, archived_at: new Date().toISOString(), days_left: Archive.RETENTION_DAYS });
            totalCards--;
            archivedCards++;
            updateCardCountLabel();
            Archive.toast('Card archived', { actionLabel: 'Undo', onAction: () => restoreCard(card.id) });
        } catch (err) {
            Archive.toast(err.message, { type: 'error' });
        }
    }

    async function restoreCard(cardId) {
        try {
            await sendCardAction({ action: 'restore', card_id: cardId });
            const id = Number(cardId);

            if (viewArchived) {
                const row = tabulatorTable.getRow(id);
                const data = row ? row.getData() : null;
                if (row) row.delete();
                if (data) offscreenList.unshift({ ...data, archived_at: null, days_left: null });
                Archive.toast('Card restored');
            } else {
                const idx = offscreenList.findIndex((c) => Number(c.id) === id);
                if (idx !== -1) {
                    const [data] = offscreenList.splice(idx, 1);
                    await tabulatorTable.addRow({ ...data, archived_at: null, days_left: null }, true);
                }
            }

            totalCards++;
            archivedCards--;
            updateCardCountLabel();
        } catch (err) {
            Archive.toast(err.message, { type: 'error' });
        }
    }

                        last_reviewed_at: null

            setEditMode(false);
            resetDialog.close();
        } catch (err) {
            Archive.toast(err.message, { type: 'error' });
        } finally {
            document.getElementById('confirm-reset-btn').disabled = false;
        }

    // ---------------------------------------------------------------------------
    // Archive view: bulk Restore / Delete permanently
    // ---------------------------------------------------------------------------
    function takeCards(ids) {       // removes cards from the grid AND the not-yet-rendered list
        const set = new Set(ids);
        const taken = [];
        ids.forEach((id) => {
            const row = tabulatorTable.getRow(id);
            if (row) { taken.push(row.getData()); row.delete(); }
        });
        pending = pending.filter((p) => {
            if (!set.has(Number(p.id))) return true;
            taken.push(p);
            return false;
        });
        return taken;
    }

    document.getElementById('bulk-restore-btn').addEventListener('click', async () => {
        const ids = Array.from(checkedIds);
        if (!viewArchived || !ids.length) return;
        try {
            archiveActionBtns.forEach((b) => b.disabled = true);
            const res = await sendCardAction({ action: 'bulk_restore', card_ids: ids });
            const taken = takeCards(ids);
            // while viewing the archive, offscreenList holds the active cards
            offscreenList.unshift(...taken.map((c) => ({ ...c, archived_at: null, days_left: null })));
            totalCards += res.affected;
            archivedCards -= res.affected;
            checkedIds.clear();
            updateCardCountLabel();
            if (gridReady) updateLoadMoreUI();
            Archive.toast(`${res.affected} ${res.affected === 1 ? 'card' : 'cards'} restored`);
        } catch (err) {
            Archive.toast(err.message, { type: 'error' });
        } finally {
            updateBulkUI();
        }
    });

    document.getElementById('bulk-delete-btn').addEventListener('click', () => {
        const n = checkedIds.size;
        if (!viewArchived || !n) return;

        if (deleteCardConfirm) deleteCardConfirm.textContent = n === 1 ? 'Delete Card' : `Delete ${n} Cards`;
        if (deleteCardError) deleteCardError.classList.add('hidden');
        if (deleteCardDialog) deleteCardDialog.showModal();
    });

    if (deleteCardConfirm) deleteCardConfirm.addEventListener('click', async () => {
        const ids = Array.from(checkedIds);
        if (!ids.length) return;
        deleteCardConfirm.disabled = true;
        try {
            const res = await sendCardAction({ action: 'bulk_delete_permanent', card_ids: ids });
            takeCards(ids);
            archivedCards -= res.affected;
            checkedIds.clear();
            updateCardCountLabel();
            if (gridReady) updateLoadMoreUI();
            if (deleteCardDialog) deleteCardDialog.close();
            Archive.toast(`${res.affected} ${res.affected === 1 ? 'card' : 'cards'} deleted permanently`, { type: 'error' });
        } catch (err) {
            if (deleteCardError) {
                deleteCardError.textContent = err.message;
                deleteCardError.classList.remove('hidden');
            }
        } finally {
            deleteCardConfirm.disabled = false;
            updateBulkUI();
        }
    });

    updateBulkUI();   // initial render of the panel (must stay last)
});

    // Read the Tailwind design tokens from the probe element and expose them as CSS variables
    function applyTokens(probe) {
        if (!probe) return;
        var q = function (t) { return probe.querySelector('[data-t="' + t + '"]'); };
        var s = document.documentElement.style;
        s.setProperty('--mp-head-bg', getComputedStyle(q('head')).backgroundColor);
        s.setProperty('--mp-head-text', getComputedStyle(q('head')).color);
        s.setProperty('--mp-text', getComputedStyle(q('text')).color);

        s.setProperty('--mp-border', getComputedStyle(q('border')).borderTopColor);
        s.setProperty('--mp-divider', getComputedStyle(q('divider')).borderTopColor);

    function esc(v) {
        return String(v == null ? '' : v).replace(/[&<>"']/g, function (c) {
            return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c];

        sc.src = TABULATOR_JS;
        sc.onload = done;
        sc.onerror = done; // make() shows the fallback message

    // Read-only defaults (headerSort / resizable are COLUMN options in Tabulator 5)
    var base = {

    function make(selector, rows, columns, extra) {
        var el = document.querySelector(selector);
        if (!el) return null;
        if (typeof Tabulator === 'undefined') {

            return null;
        }
        var cols = columns.map(function (c) { return Object.assign({ headerSort: false, resizable: false }, c); });
        return new Tabulator(el, Object.assign({}, base, { data: rows, columns: cols }, extra || {}));
    }

    window.MPTable = { applyTokens: applyTokens, esc: esc, ensure: ensure, make: make };
})();

<?php
/**
 * DashboardService
 * 
 * Centralized service layer for handling database reads, user profiling,
 * spaced repetition metric analytics, and streak tracking.
 */
class DashboardService
{
    /**
     * Retrieve authenticated user profile details.
     * 
     * @param mysqli $conn Active database connection instance.
     * @param int $userId Authenticated user ID.
     * @return array Profile metadata containing display name.
     */
    public static function getUserProfile(mysqli $conn, int $userId): array
    {
        $sql = "SELECT first_name, last_name FROM users WHERE id = ? AND deleted_at IS NULL";
        $stmt = mysqli_prepare($conn, $sql);
        
        if (!$stmt) {
            return ['displayName' => 'Active Learner'];
        }

        mysqli_stmt_bind_param($stmt, "i", $userId);
        mysqli_stmt_execute($stmt);
        mysqli_stmt_bind_result($stmt, $firstName, $lastName);
        mysqli_stmt_fetch($stmt);
        mysqli_stmt_close($stmt);

        $fullName = trim(($firstName ?? '') . ' ' . ($lastName ?? ''));
        
        return [
            'displayName' => !empty($fullName) ? $fullName : 'Active Learner'
        ];
    }

    /**
     * Fetch core flashcard and study session metrics.
     * 
     * @param mysqli $conn Active database connection instance.
     * @param int $userId Authenticated user ID.
     * @return array Calculated user metrics.
     */
    public static function getMetrics(mysqli $conn, int $userId): array
    {
        // 1. Cards Due Today
        $due = self::fetchCount($conn, 
            "SELECT COUNT(*) FROM flashcards f 
             JOIN decks d ON f.deck_id = d.id
             WHERE d.user_id = ? AND d.deleted_at IS NULL AND f.deleted_at IS NULL AND f.next_review_at <= NOW()", 
            $userId
        );

        // 2. Mastered Cards (Interval >= 21 days)
        $mastered = self::fetchCount($conn, 
            "SELECT COUNT(*) FROM flashcards f 
             JOIN decks d ON f.deck_id = d.id 
             WHERE d.user_id = ? AND d.deleted_at IS NULL AND f.deleted_at IS NULL AND f.interval_days >= 21", 
            $userId
        );

        // 3. Tough Cards / Leeches (Lapses >= 8)
        $tough = self::fetchCount($conn, 
            "SELECT COUNT(*) FROM flashcards f 
             JOIN decks d ON f.deck_id = d.id 
             WHERE d.user_id = ? AND d.deleted_at IS NULL AND f.deleted_at IS NULL AND f.lapses >= 8", 
            $userId
        );

        // 4. Cards Reviewed Today
        $reviewedToday = self::fetchCount($conn, 
            "SELECT COALESCE(SUM(total_reviewed), 0) FROM review_sessions 
             WHERE user_id = ? AND ended_at IS NOT NULL AND DATE(started_at) = CURDATE()", 
            $userId
        );

        // 5. Retention Rate Percentage
        $retentionRate = 0.0;
        $retentionSql = "SELECT COALESCE(SUM(remembered_count), 0), COALESCE(SUM(total_reviewed), 0) 
                         FROM review_sessions 
                         WHERE user_id = ? AND ended_at IS NOT NULL";
                         
        if ($stmt = mysqli_prepare($conn, $retentionSql)) {
            mysqli_stmt_bind_param($stmt, "i", $userId);
            mysqli_stmt_execute($stmt);
            mysqli_stmt_bind_result($stmt, $remembered, $reviewed);
            if (mysqli_stmt_fetch($stmt) && $reviewed > 0) {
                $retentionRate = round(($remembered / $reviewed) * 100, 1);
            }
            mysqli_stmt_close($stmt);
        }

        return [
            'cardsDue' => $due,
            'masteredCards' => $mastered,
            'toughCards' => $tough,
            'cardsReviewedToday' => $reviewedToday,
            'retentionRate' => $retentionRate
        ];
    }

    /**
     * Compute activity history, streaks, and heatmap data.
     * 
     * @param mysqli $conn Active database connection instance.
     * @param int $userId Authenticated user ID.
     * @return array Streak counters and calendar heatmap data.
     */
    public static function getStreakAndHeatmapData(mysqli $conn, int $userId): array
    {
        $sessionSql = "SELECT DATE(started_at) as session_date, 
                              SUM(remembered_count) as total_remembered, 
                              SUM(total_reviewed) as total_reviewed
                       FROM review_sessions 
                       WHERE user_id = ? AND ended_at IS NOT NULL
                       GROUP BY DATE(started_at)
                       HAVING SUM(total_reviewed) > 0
                       ORDER BY session_date DESC";

        $stmt = mysqli_prepare($conn, $sessionSql);
        if (!$stmt) {
            return self::emptyStreakState();
        }

        mysqli_stmt_bind_param($stmt, "i", $userId);
        mysqli_stmt_execute($stmt);
        $result = mysqli_stmt_get_result($stmt);

        $activityHistory = [];
        while ($row = mysqli_fetch_assoc($result)) {
            $activityHistory[$row['session_date']] = [
                'remembered' => (int) $row['total_remembered'],
                'reviewed' => (int) $row['total_reviewed']
            ];
        }
        mysqli_stmt_close($stmt);

        if (empty($activityHistory)) {
            return self::emptyStreakState();
        }

        // Streak Analytics Processing
        $todayStr = date('Y-m-d');
        $yesterdayStr = date('Y-m-d', strtotime('-1 day'));
        
        $currentStreak = 0;
        $bestStreak = 0;
        $streakAccuracy = 0.0;
        
        $dates = array_keys($activityHistory);
        $latestDate = $dates[0];

        // Active Streak calculation
        if ($latestDate === $todayStr || $latestDate === $yesterdayStr) {
            $checkDate = new DateTime($latestDate);
            $streakRemembered = 0;
            $streakReviewed = 0;

            while (true) {
                $dateKey = $checkDate->format('Y-m-d');
                if (isset($activityHistory[$dateKey])) {
                    $currentStreak++;
                    $streakRemembered += $activityHistory[$dateKey]['remembered'];
                    $streakReviewed += $activityHistory[$dateKey]['reviewed'];
                    $checkDate->modify('-1 day');
                } else {
                    break;
                }
            }

            if ($streakReviewed > 0) {
                $streakAccuracy = round(($streakRemembered / $streakReviewed) * 100, 1);
            }
        }

        // Best Historical Streak calculation
        $tempStreak = 0;
        $expectedDate = null;

        foreach (array_reverse($dates) as $dateStr) {
            $currentDate = new DateTime($dateStr);
            if ($expectedDate === null || $currentDate == $expectedDate) {
                $tempStreak++;
            } else {
                if ($tempStreak > $bestStreak) {
                    $bestStreak = $tempStreak;
                }
                $tempStreak = 1;
            }
            $expectedDate = clone $currentDate;
            $expectedDate->modify('+1 day');
        }

        if ($tempStreak > $bestStreak) {
            $bestStreak = $tempStreak;
        }

        // Heatmap bounds calculation
        $heatmapEnd = new DateTime('today');
        $heatmapStart = (clone $heatmapEnd)->modify('-364 days');
        $heatmapData = [];
        $studyDays = 0;

        foreach ($activityHistory as $dateStr => $metrics) {
            $activityDate = DateTime::createFromFormat('!Y-m-d', $dateStr);
            if ($activityDate && $activityDate >= $heatmapStart && $activityDate <= $heatmapEnd && $metrics['reviewed'] > 0) {
                $heatmapData[$dateStr] = $metrics['reviewed'];
                $studyDays++;
            }
        }

        $calendarStart = clone $heatmapStart;

    {
        if ($cardCount === 0) {
            return 'No cards yet';
        }
        if ($dueCount > 0) {
            return $dueCount . ' due now';
        }
        if ($nextDue === null) {
            return 'Nothing due';
        }
        $diff = (new DateTime())->diff(new DateTime($nextDue));
        if ($diff->days >= 1) {
            return 'Due in ' . $diff->days . ($diff->days === 1 ? ' day' : ' days');
        }
        if ($diff->h >= 1) {
            return 'Due in ' . $diff->h . ($diff->h === 1 ? ' hr' : ' hrs');
        }
        return 'Due in under 1 hr';
    }

     * Generate human-readable relative time string for last review.
     * 
     * @param string|null $lastReviewed Timestamp of last review session.
     * @return string Relative time string.
     */
    public static function lastReviewedLabel(?string $lastReviewed): string
    {
        if ($lastReviewed === null) {
            return 'Never reviewed';
        }
        $reviewedDate = new DateTime($lastReviewed);
        $now = new DateTime();
        $diff = $now->diff($reviewedDate);

        if ($diff->days === 0) {
            if ($diff->h === 0) {
                return $diff->i <= 1 ? 'Reviewed just now' : 'Reviewed ' . $diff->i . ' mins ago';
            }
            return 'Reviewed ' . $diff->h . ($diff->h === 1 ? ' hr ago' : ' hrs ago');
        }
        if ($diff->days === 1) {
            return 'Reviewed yesterday';
        }
        if ($diff->days < 30) {
            return 'Reviewed ' . $diff->days . ' days ago';
        }
        return 'Reviewed ' . $reviewedDate->format('M j, Y');
    }

<?php
/**
 * ReviewHubService
 *
 * Read-only data + formatting helpers for the Review Hub page (review.php, no deck_id).
 * All "seconds" values are computed in SQL (TIMESTAMPDIFF) so PHP/MySQL timezone
 * differences cannot skew them.
 *
 * Place next to FlashcardService.php (the api/ folder).
 */
class ReviewHubService
{
    /** Cards scheduled within this many days count as "upcoming". */
    public const UPCOMING_DAYS = 2;

    // -----------------------------------------------------------------------
    // Queries
    // -----------------------------------------------------------------------

    /**
     * Every non-archived deck that has cards due now or coming up soon.
     *
     * @return array<int, array{
     *   id:int, title:string, totalCards:int, dueCount:int, upcomingCount:int,
     *   overdueSeconds:?int, nextDueSeconds:?int
     * }>
     */
    public static function getHubDecks(mysqli $conn, int $userId): array
    {
        $days = (int) self::UPCOMING_DAYS;

        $sql = "SELECT d.id, d.title,
                       COUNT(f.id) AS total_cards,
                       COALESCE(SUM(f.next_review_at <= NOW()), 0) AS due_count,
                       COALESCE(SUM(f.next_review_at > NOW()
                                    AND f.next_review_at <= DATE_ADD(NOW(), INTERVAL $days DAY)), 0) AS upcoming_count,
                       TIMESTAMPDIFF(SECOND,
                           MIN(CASE WHEN f.next_review_at <= NOW() THEN f.next_review_at END),
                           NOW()) AS overdue_seconds,
                       TIMESTAMPDIFF(SECOND, NOW(),
                           MIN(CASE WHEN f.next_review_at > NOW()
                                     AND f.next_review_at <= DATE_ADD(NOW(), INTERVAL $days DAY)
                                    THEN f.next_review_at END)) AS next_due_seconds
                FROM decks d
                LEFT JOIN flashcards f ON f.deck_id = d.id AND f.deleted_at IS NULL
                WHERE d.user_id = ? AND d.deleted_at IS NULL
                GROUP BY d.id, d.title
                HAVING due_count > 0 OR upcoming_count > 0";

        $decks = [];
        if ($stmt = mysqli_prepare($conn, $sql)) {
            mysqli_stmt_bind_param($stmt, "i", $userId);
            mysqli_stmt_execute($stmt);
            $result = mysqli_stmt_get_result($stmt);
            while ($row = mysqli_fetch_assoc($result)) {
                $decks[] = [
                    'id'             => (int) $row['id'],
                    'title'          => (string) $row['title'],
                    'totalCards'     => (int) $row['total_cards'],
                    'dueCount'       => (int) $row['due_count'],
                    'upcomingCount'  => (int) $row['upcoming_count'],
                    'overdueSeconds' => $row['overdue_seconds'] === null ? null : (int) $row['overdue_seconds'],
                    'nextDueSeconds' => $row['next_due_seconds'] === null ? null : (int) $row['next_due_seconds'],
                ];
            }
            mysqli_stmt_close($stmt);
        }
        return $decks;
    }

    /**
     * Seconds until the user's next card becomes due (no 48h limit), or null
     * when there are no scheduled cards at all.
     */
    public static function getNextDueSeconds(mysqli $conn, int $userId): ?int
    {
        $sql = "SELECT TIMESTAMPDIFF(SECOND, NOW(), MIN(f.next_review_at))
                FROM flashcards f
                JOIN decks d ON d.id = f.deck_id
                WHERE d.user_id = ? AND d.deleted_at IS NULL
                  AND f.deleted_at IS NULL AND f.next_review_at > NOW()";

        $secs = null;
        if ($stmt = mysqli_prepare($conn, $sql)) {
            mysqli_stmt_bind_param($stmt, "i", $userId);
            mysqli_stmt_execute($stmt);
            mysqli_stmt_bind_result($stmt, $secs);
            mysqli_stmt_fetch($stmt);
            mysqli_stmt_close($stmt);
        }
        return $secs === null ? null : (int) $secs;
    }

    /**
     * Latest completed review sessions.
     *
     * @return array<int, array{
     *   deckId:int, deckTitle:string, cardsReviewed:int, rememberedCount:int,
     *   accuracy:int, durationSeconds:?int, formattedTime:string
     * }>
     */
    public static function getRecentSessions(mysqli $conn, int $userId, int $limit = 8): array
    {
        $limit = max(1, min($limit, 50));

        $sql = "SELECT rs.id AS session_id, rs.deck_id, d.title AS deck_title, rs.total_reviewed, rs.remembered_count,
                       rs.accuracy_percentage,
                       TIMESTAMPDIFF(SECOND, rs.started_at, rs.ended_at) AS duration_seconds,
                       TIMESTAMPDIFF(SECOND, rs.ended_at, NOW()) AS ended_ago_seconds
                FROM review_sessions rs
                JOIN decks d ON d.id = rs.deck_id
                WHERE rs.user_id = ? AND rs.ended_at IS NOT NULL
                ORDER BY rs.ended_at DESC, rs.id DESC
                LIMIT ?";

        $sessions = [];
        if ($stmt = mysqli_prepare($conn, $sql)) {
            mysqli_stmt_bind_param($stmt, "ii", $userId, $limit);
            mysqli_stmt_execute($stmt);
            $result = mysqli_stmt_get_result($stmt);
            while ($row = mysqli_fetch_assoc($result)) {
                $sessions[] = [
                    'sessionID'       => (int) $row['session_id'],
                    'deckId'          => (int) $row['deck_id'],
                    'deckTitle'       => (string) $row['deck_title'],
                    'cardsReviewed'   => (int) $row['total_reviewed'],
                    'rememberedCount' => (int) $row['remembered_count'],
                    'accuracy'        => (int) $row['accuracy_percentage'],
                    'durationSeconds' => $row['duration_seconds'] === null ? null : (int) $row['duration_seconds'],
                    'formattedTime'   => self::formatAgo((int) $row['ended_ago_seconds']),
                ];
            }
            mysqli_stmt_close($stmt);
        }
        return $sessions;
    }

    /**
     * Cards reviewed today and the current study-day streak.
     * Same rules as the dashboard: a day counts when a completed session
     * reviewed at least one card; the streak is alive if the latest study day
     * is today or yesterday.
     *
     * @return array{reviewedToday:int, currentStreak:int}
     */
    public static function getStreakInfo(mysqli $conn, int $userId): array
    {
        $sql = "SELECT DATE(started_at) AS study_date, SUM(total_reviewed) AS reviewed
                FROM review_sessions
                WHERE user_id = ? AND ended_at IS NOT NULL
                GROUP BY DATE(started_at)
                HAVING SUM(total_reviewed) > 0
                ORDER BY study_date DESC";

        $days = [];
        if ($stmt = mysqli_prepare($conn, $sql)) {
            mysqli_stmt_bind_param($stmt, "i", $userId);
            mysqli_stmt_execute($stmt);
            $result = mysqli_stmt_get_result($stmt);
            while ($row = mysqli_fetch_assoc($result)) {
                $days[$row['study_date']] = (int) $row['reviewed'];
            }
            mysqli_stmt_close($stmt);
        }

        $today     = date('Y-m-d');
        $yesterday = date('Y-m-d', strtotime('-1 day'));

        $streak = 0;
        if (isset($days[$today]) || isset($days[$yesterday])) {
            $cursor = new DateTime(isset($days[$today]) ? $today : $yesterday);
            while (isset($days[$cursor->format('Y-m-d')])) {
                $streak++;
                $cursor->modify('-1 day');
            }
        }

        return [
            'reviewedToday' => $days[$today] ?? 0,
            'currentStreak' => $streak,
        ];
    }

    // -----------------------------------------------------------------------
    // Table helpers (pure PHP, no DB)
    // -----------------------------------------------------------------------

    /** Decks with cards due, most due first (oldest overdue breaks ties). */
    public static function dueDecks(array $hubDecks): array
    {
        $due = array_values(array_filter($hubDecks, fn($d) => $d['dueCount'] > 0));
        usort($due, function ($a, $b) {
            return [$b['dueCount'], $b['overdueSeconds'] ?? 0]
               <=> [$a['dueCount'], $a['overdueSeconds'] ?? 0];
        });
        return $due;
    }

    /** Decks with cards coming up soon, soonest first. */
    public static function upcomingDecks(array $hubDecks): array
    {
        $up = array_values(array_filter($hubDecks, fn($d) => $d['upcomingCount'] > 0));
        usort($up, fn($a, $b) => ($a['nextDueSeconds'] ?? PHP_INT_MAX) <=> ($b['nextDueSeconds'] ?? PHP_INT_MAX));
        return $up;
    }

    // -----------------------------------------------------------------------
    // Formatting
    // -----------------------------------------------------------------------

    /** "5 min", "3 h", "2 d" (never shows 0). */
    public static function formatSpan(int $seconds): string
    {
        $seconds = max(0, $seconds);
        if ($seconds < 3600)  return max(1, (int) floor($seconds / 60)) . ' min';
        if ($seconds < 86400) return (int) floor($seconds / 3600) . ' h';
        return (int) floor($seconds / 86400) . ' d';
    }

    /** "just now", "5 mins ago", "1 hour ago", "3 days ago". */
    public static function formatAgo(int $seconds): string
    {
        $seconds = max(0, $seconds);
        if ($seconds < 60) return 'just now';

        if ($seconds < 3600) {
            $n = (int) floor($seconds / 60);
            return $n . ($n === 1 ? ' min ago' : ' mins ago');
        }
        if ($seconds < 86400) {
            $n = (int) floor($seconds / 3600);
            return $n . ($n === 1 ? ' hour ago' : ' hours ago');
        }
        $n = (int) floor($seconds / 86400);
        return $n . ($n === 1 ? ' day ago' : ' days ago');
    }

    /** "45s", "4m 12s", "1h 5m", or an em dash when unknown. */
    public static function formatDuration(?int $seconds): string
    {
        if ($seconds === null || $seconds <= 0) return '—';
        if ($seconds < 60) return $seconds . 's';
        if ($seconds < 3600) {
            $m = (int) floor($seconds / 60);
            $s = $seconds % 60;
            return $s > 0 ? "{$m}m {$s}s" : "{$m}m";
        }
        $h = (int) floor($seconds / 3600);
        $m = (int) floor(($seconds % 3600) / 60);
        return $m > 0 ? "{$h}h {$m}m" : "{$h}h";
    }

    /** Tailwind classes for an accuracy badge (matches the deck grid thresholds). */
    public static function accuracyBadgeClass(int $accuracy): string
    {
        if ($accuracy >= 80) return 'text-emerald-700 bg-emerald-50 border-emerald-200/80';
        if ($accuracy >= 50) return 'text-amber-700 bg-amber-50 border-amber-200/80';
        return 'text-rose-700 bg-rose-50 border-rose-200/80';
    }

    /** Shorthand for htmlspecialchars with the project's flags. */
    public static function h($value): string
    {
        return htmlspecialchars((string) $value, ENT_QUOTES, 'UTF-8');
    }
}

CREATE DATABASE IF NOT EXISTS mempulse_db
CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

USE mempulse_db;


-- 1. USERS TABLE
CREATE TABLE users (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    email VARCHAR(255) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    role ENUM('admin', 'user') NOT NULL DEFAULT 'user',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL,
    INDEX idx_users_active (deleted_at)
) ENGINE=InnoDB;


-- 2. USER SETTINGS TABLE (SRS Configuration)
CREATE TABLE user_settings (
    user_id BIGINT UNSIGNED PRIMARY KEY,
    remember_multiplier DECIMAL(4,2) NOT NULL DEFAULT 2.00,
    forgot_multiplier DECIMAL(4,2) NOT NULL DEFAULT 0.50,
    initial_interval_days INT UNSIGNED NOT NULL DEFAULT 1,
    maximum_interval_days INT UNSIGNED NOT NULL DEFAULT 365,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE
) ENGINE=InnoDB;


-- 3. DECKS TABLE
CREATE TABLE decks (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT UNSIGNED NOT NULL,
    title VARCHAR(150) NOT NULL,
    cover_image VARCHAR(255) DEFAULT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL,
    FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE,
    INDEX idx_decks_user_active (user_id, deleted_at)
) ENGINE=InnoDB;


-- 4. FLASHCARDS TABLE
CREATE TABLE flashcards (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    deck_id BIGINT UNSIGNED NOT NULL,
    front TEXT NOT NULL,
    back TEXT NOT NULL,
    interval_days INT UNSIGNED NOT NULL DEFAULT 1,
    next_review_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    review_count INT UNSIGNED NOT NULL DEFAULT 0,
    lapses INT UNSIGNED NOT NULL DEFAULT 0,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL,
    FOREIGN KEY (deck_id)
        REFERENCES decks(id)
        ON DELETE CASCADE,
    INDEX idx_cards_deck_due (deck_id, deleted_at, next_review_at)
) ENGINE=InnoDB;


-- 5. REVIEW SESSIONS TABLE
CREATE TABLE review_sessions (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT UNSIGNED NOT NULL,
    deck_id BIGINT UNSIGNED NOT NULL,
    started_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    ended_at DATETIME NULL,
    total_reviewed INT UNSIGNED NOT NULL DEFAULT 0,
    remembered_count INT UNSIGNED NOT NULL DEFAULT 0,
    accuracy_percentage DECIMAL(5,2)
        GENERATED ALWAYS AS (
            CASE
                WHEN total_reviewed > 0
                    THEN (remembered_count / total_reviewed) * 100
                ELSE 0.00
            END
        ) STORED,
    FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE,
    FOREIGN KEY (deck_id)
        REFERENCES decks(id)
        ON DELETE CASCADE,
    INDEX idx_user_sessions (user_id, started_at)
) ENGINE=InnoDB;


-- 6. REVIEW LOGS TABLE
CREATE TABLE review_logs (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    session_id BIGINT UNSIGNED NOT NULL,
    card_id BIGINT UNSIGNED NULL,
    rating ENUM('remembered', 'forgot') NOT NULL,
    previous_interval INT UNSIGNED NOT NULL,
    new_interval INT UNSIGNED NOT NULL,
    response_time_ms INT UNSIGNED NULL,
    reviewed_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (session_id)
        REFERENCES review_sessions(id)
        ON DELETE CASCADE,
    FOREIGN KEY (card_id)
        REFERENCES flashcards(id)
        ON DELETE SET NULL,
    INDEX idx_session_logs (session_id)
) ENGINE=InnoDB;
