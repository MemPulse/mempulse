<?php
session_start();
header('Content-Type: application/json; charset=utf-8');

// Make every failed query throw, no matter how config/db.php set things up.
// Without this, a failed INSERT INTO review_logs is silently ignored.
mysqli_report(MYSQLI_REPORT_ERROR | MYSQLI_REPORT_STRICT);

if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
    http_response_code(405);
    echo json_encode(['error' => 'Method not allowed.']);
    exit;
}

if (!isset($_SESSION['logged_in']) || $_SESSION['logged_in'] !== true || !isset($_SESSION['user_id'])) {
    http_response_code(401);
    echo json_encode(['error' => 'Unauthorized']);
    exit;
}

require_once __DIR__ . '/../../config/db.php';
mysqli_report(MYSQLI_REPORT_ERROR | MYSQLI_REPORT_STRICT); // db.php may have switched it off

$userId = (int) $_SESSION['user_id'];
$input = json_decode(file_get_contents('php://input'), true);

if (!is_array($input)) {
    http_response_code(400);
    echo json_encode(['error' => 'Invalid request.']);
    exit;
}

$token = $input['csrf_token'] ?? '';
if (empty($_SESSION['csrf_token']) || !is_string($token) || !hash_equals($_SESSION['csrf_token'], $token)) {
    http_response_code(403);
    echo json_encode(['error' => 'Security check failed. Refresh the page and try again.']);
    exit;
}

$deckId = filter_var($input['deck_id'] ?? null, FILTER_VALIDATE_INT);
$results = $input['results'] ?? null;
$durationSeconds = filter_var($input['duration_seconds'] ?? 0, FILTER_VALIDATE_INT);
$durationSeconds = ($durationSeconds === false) ? 0 : max(0, min($durationSeconds, 6 * 3600));

if (!$deckId || $deckId < 1 || !is_array($results) || empty($results)) {
    echo json_encode(['ok' => true, 'saved' => 0, 'msg' => 'No data to save or invalid deck.']);
    exit;
}
if (count($results) > 1000) {
    http_response_code(400);
    echo json_encode(['error' => 'Batch size limited to 1000.']);
    exit;
}

try {
    // Verify ownership
    $ownStmt = mysqli_prepare($conn, "SELECT id FROM decks WHERE id = ? AND user_id = ? AND deleted_at IS NULL");
    mysqli_stmt_bind_param($ownStmt, "ii", $deckId, $userId);
    mysqli_stmt_execute($ownStmt);
    mysqli_stmt_store_result($ownStmt);
    $ownsDeck = mysqli_stmt_num_rows($ownStmt) === 1;
    mysqli_stmt_close($ownStmt);

    if (!$ownsDeck) {
        http_response_code(404);
        echo json_encode(['error' => 'Deck not found or access denied.']);
        exit;
    }

    // 1. Sanitize IDs
    $requested = [];
    foreach ($results as $cardId => $status) {
        $cid = filter_var($cardId, FILTER_VALIDATE_INT);
        if ($cid && ($status === 'remembered' || $status === 'forgotten')) {
            $requested[$cid] = $status;
        }
    }
    if (empty($requested)) {
        echo json_encode(['ok' => true, 'saved' => 0, 'msg' => 'No valid cards to save.']);
        exit;
    }

    // 2. Keep only cards that really belong to this deck (and fetch their intervals)
    $idString = implode(',', array_map('intval', array_keys($requested)));
    $cardData = [];
    $res = mysqli_query($conn, "SELECT id, interval_days FROM flashcards
                                WHERE id IN ($idString) AND deck_id = " . (int) $deckId . " AND deleted_at IS NULL");
    while ($row = mysqli_fetch_assoc($res)) {
        $cardData[(int) $row['id']] = (int) $row['interval_days'];
    }

    $toSave = array_intersect_key($requested, $cardData);
    if (empty($toSave)) {
        echo json_encode(['ok' => true, 'saved' => 0, 'msg' => 'No matching cards found in this deck.']);
        exit;
    }

    // Totals come from what will actually be saved (not from what the browser sent)
    $savedCount      = count($toSave);
    $rememberedCount = count(array_filter($toSave, fn($s) => $s === 'remembered'));
    $accuracy        = (int) round(($rememberedCount / $savedCount) * 100);

    mysqli_begin_transaction($conn);

    // 3. Insert session record
    $sessStmt = mysqli_prepare($conn, "INSERT INTO review_sessions
        (user_id, deck_id, total_reviewed, remembered_count, accuracy_percentage, started_at, ended_at)
        VALUES (?, ?, ?, ?, ?, DATE_SUB(NOW(), INTERVAL ? SECOND), NOW())");
    mysqli_stmt_bind_param($sessStmt, "iiiiii", $userId, $deckId, $savedCount, $rememberedCount, $accuracy, $durationSeconds);
    mysqli_stmt_execute($sessStmt);
    $sessionId = mysqli_insert_id($conn);
    mysqli_stmt_close($sessStmt);

    // 4. Update each card + write its log row
    $stmtCard = mysqli_prepare($conn, "UPDATE flashcards
        SET review_count = review_count + 1, lapses = lapses + ?, interval_days = ?,
            next_review_at = DATE_ADD(NOW(), INTERVAL ? DAY), updated_at = NOW()
        WHERE id = ? AND deck_id = ?");

    $stmtLog = mysqli_prepare($conn, "INSERT INTO review_logs
        (session_id, card_id, rating, previous_interval, new_interval)
        VALUES (?, ?, ?, ?, ?)");

    foreach ($toSave as $cid => $status) {
        $prevInterval = $cardData[$cid];
        $isLapse      = ($status === 'forgotten') ? 1 : 0;
        $newInterval  = ($status === 'remembered') ? max(1, $prevInterval * 2) : 0;

        // The review_logs.rating ENUM is ('remembered', 'forgot'), NOT 'forgotten'
        $dbRating = ($status === 'remembered') ? 'remembered' : 'forgot';

        mysqli_stmt_bind_param($stmtCard, "iiiii", $isLapse, $newInterval, $newInterval, $cid, $deckId);
        mysqli_stmt_execute($stmtCard);

        mysqli_stmt_bind_param($stmtLog, "iisii", $sessionId, $cid, $dbRating, $prevInterval, $newInterval);
        mysqli_stmt_execute($stmtLog);
    }

    mysqli_stmt_close($stmtCard);
    mysqli_stmt_close($stmtLog);

    mysqli_commit($conn);
    echo json_encode(['ok' => true, 'saved' => $savedCount, 'session_id' => $sessionId]);

} catch (Throwable $e) {
    // Only roll back if a transaction was started; harmless otherwise
    try { mysqli_rollback($conn); } catch (Throwable $ignored) {}
    error_log('save-session.php error: ' . $e->getMessage());
    http_response_code(500);
    echo json_encode(['error' => 'Database exception occurred']);
}

session_start();

header('Content-Type: application/json; charset=utf-8');

if (

$userId = (int) $_SESSION['user_id'];
$method = $_SERVER['REQUEST_METHOD'];

function respond(int $status, array $data): void {
    http_response_code($status);
    echo json_encode($data);
    exit;
}

function getSettings(mysqli $conn, int $userId): array {
    $sql = "SELECT remember_multiplier, forgot_multiplier,
                   initial_interval_days, maximum_interval_days
            FROM user_settings WHERE user_id = ?";

    $stmt = mysqli_prepare($conn, $sql);
    mysqli_stmt_bind_param($stmt, 'i', $userId);
    mysqli_stmt_execute($stmt);
    mysqli_stmt_bind_result(
        $stmt, $remember, $forgot, $initial, $maximum
    );

    if (!mysqli_stmt_fetch($stmt)) {
        mysqli_stmt_close($stmt);

        return [
            'remember_multiplier' => 2.00,
            'forgot_multiplier' => 0.50,
            'initial_interval_days' => 1,
            'maximum_interval_days' => 365
        ];
    }

    mysqli_stmt_close($stmt);

    return [
        'remember_multiplier' => (float) $remember,
        'forgot_multiplier' => (float) $forgot,
        'initial_interval_days' => (int) $initial,
        'maximum_interval_days' => (int) $maximum
    ];
}

if ($method === 'GET') {
    respond(200, [
        'success' => true,
        'settings' => getSettings($conn, $userId)
    ]);
}

if ($method !== 'POST') {
    header('Allow: GET, POST');
    respond(405, ['success' => false, 'message' => 'Method not allowed.']);
}

if (
    !isset($_SESSION['settings_csrf']) ||
    !hash_equals(
        $_SESSION['settings_csrf'],
        $_SERVER['HTTP_X_CSRF_TOKEN'] ?? ''
    )
) {
    respond(403, ['success' => false, 'message' => 'Security token invalid. Refresh the page and try again.']);
}

$data = json_decode(file_get_contents('php://input'), true);

if (!is_array($data)) {
    respond(400, ['success' => false, 'message' => 'Invalid request data.']);
}

$required = [
    'remember_multiplier',
    'forgot_multiplier',
    'initial_interval_days',
    'maximum_interval_days'
];

foreach ($required as $key) {
    if (!isset($data[$key]) || !is_numeric($data[$key])) {
        respond(422, ['success' => false, 'message' => 'Enter valid values for all settings.']);
    }
}

$remember = (float) $data['remember_multiplier'];
$forgot = (float) $data['forgot_multiplier'];
$initial = filter_var($data['initial_interval_days'], FILTER_VALIDATE_INT);
$maximum = filter_var($data['maximum_interval_days'], FILTER_VALIDATE_INT);

if (
    !is_finite($remember) || !is_finite($forgot) ||
    $remember < 0.01 || $remember > 10 ||
    $forgot < 0.01 || $forgot > 10 ||
    $initial === false || $maximum === false ||
    $initial < 1 || $maximum < $initial || $maximum > 36500
) {
    respond(422, [
        'success' => false,
        'message' => 'Check the multipliers and interval limits.'
    ]);
}

$sql = "INSERT INTO user_settings
            (user_id, remember_multiplier, forgot_multiplier,
             initial_interval_days, maximum_interval_days)
        VALUES (?, ?, ?, ?, ?)
        ON DUPLICATE KEY UPDATE
            remember_multiplier = VALUES(remember_multiplier),
            forgot_multiplier = VALUES(forgot_multiplier),
            initial_interval_days = VALUES(initial_interval_days),
            maximum_interval_days = VALUES(maximum_interval_days)";

$stmt = mysqli_prepare($conn, $sql);
mysqli_stmt_bind_param(
    $stmt, 'iddii', $userId, $remember, $forgot, $initial, $maximum
);

if (!mysqli_stmt_execute($stmt)) {
    mysqli_stmt_close($stmt);
    respond(500, ['success' => false, 'message' => 'Unable to save settings.']);
}

mysqli_stmt_close($stmt);

respond(200, [
    'success' => true,
    'message' => 'Learning settings saved.',
    'settings' => getSettings($conn, $userId)
]);

<script>
    document.addEventListener("DOMContentLoaded", () => {
        const sidebar = document.querySelector("#sidebar");
        const mainContent = document.querySelector("main");
        const toggleBtn = document.querySelector("#sidebar-toggle");
        
        // Mobile elements
        const floatingToggleBtn = document.querySelector("#floating-mobile-toggle");
        const mobileOverlay = document.querySelector("#mobile-sidebar-overlay");
        const reviewDot = document.querySelector("#review-dot");

        // Popover Elements
        const profileTrigger = document.querySelector("#profile-popover-trigger");
        const settingsPopover = document.querySelector("#settings-popover");

        if (!sidebar) return;

        // A body-level fixed tooltip is not clipped by the sidebar/nav scroll area.
        let floatingTooltip = document.getElementById("sidebar-floating-tooltip");
        if (!floatingTooltip) {
            floatingTooltip = document.createElement("div");
            floatingTooltip.id = "sidebar-floating-tooltip";
            floatingTooltip.setAttribute("role", "tooltip");
            document.body.appendChild(floatingTooltip);
        }

        let tooltipTarget = null;
        function hideSidebarTooltip() {
            floatingTooltip.classList.remove("is-visible");
            tooltipTarget = null;
        }
        function showSidebarTooltip(target) {
            const label = target.getAttribute("data-tooltip") || target.getAttribute("aria-label");
            if (!label || !sidebar.classList.contains("collapsed") || window.innerWidth < 768) {
                hideSidebarTooltip();
                return;
            }
            tooltipTarget = target;
            floatingTooltip.textContent = label;
            floatingTooltip.style.left = "0px";
            floatingTooltip.style.top = "0px";
            floatingTooltip.classList.add("is-visible");
            const rect = target.getBoundingClientRect();
            const tipRect = floatingTooltip.getBoundingClientRect();
            const gap = 12;
            let left = rect.right + gap;
            // If there is not enough room to the right, place the tooltip to the left.
            if (left + tipRect.width > window.innerWidth - 8) {
                left = Math.max(8, rect.left - tipRect.width - gap);
                floatingTooltip.classList.add("tooltip-left");
            } else {
                floatingTooltip.classList.remove("tooltip-left");
            }
            const top = Math.min(window.innerHeight - 8, Math.max(8, rect.top + rect.height / 2));
            floatingTooltip.style.left = `${left}px`;
            floatingTooltip.style.top = `${top}px`;
        }

        sidebar.querySelectorAll(".sidebar-tooltip").forEach((item) => {
            item.addEventListener("mouseenter", () => showSidebarTooltip(item));
            item.addEventListener("mouseleave", hideSidebarTooltip);
            item.addEventListener("focus", () => showSidebarTooltip(item));
            item.addEventListener("blur", hideSidebarTooltip);
        });
        sidebar.addEventListener("scroll", hideSidebarTooltip, true);
        window.addEventListener("resize", hideSidebarTooltip);
        window.addEventListener("scroll", () => {
            if (tooltipTarget) showSidebarTooltip(tooltipTarget);
        }, true);

        const toggleIcon = toggleBtn ? toggleBtn.querySelector(".material-symbols-outlined") : null;

        // Apply visual collapsed state
        const sidebarCollapsed = localStorage.getItem("sidebarCollapsed") === "true";
        if (sidebarCollapsed && window.innerWidth >= 768) {
            sidebar.classList.add("collapsed");
            if (mainContent) mainContent.classList.add("collapsed");
            if (toggleIcon) toggleIcon.textContent = "left_panel_open";
            if (reviewDot) reviewDot.classList.remove("hidden");
        }

        // Desktop Border Toggle Interaction
        if (toggleBtn) {
            toggleBtn.addEventListener("click", () => {
                const isCollapsed = sidebar.classList.toggle("collapsed");
                hideSidebarTooltip();
                if (mainContent) {
                    mainContent.classList.toggle("collapsed", isCollapsed);
                }
                if (toggleIcon) {
                    toggleIcon.textContent = isCollapsed ? "left_panel_open" : "left_panel_close";
                }
                if (reviewDot) {
                    isCollapsed ? reviewDot.classList.remove("hidden") : reviewDot.classList.add("hidden");
                }
                // Close popover if sidebar collapses
                setAccountPopoverOpen(false);
                
                localStorage.setItem("sidebarCollapsed", isCollapsed);
            });
        }

        // Mobile Toggle Interactions
        function openMobileSidebar() {
            sidebar.classList.remove("-translate-x-full", "collapsed"); 
            mobileOverlay.classList.remove("hidden");
            setTimeout(() => mobileOverlay.classList.remove("opacity-0"), 10);
        }

        function closeMobileSidebar() {
            sidebar.classList.add("-translate-x-full");
            mobileOverlay.classList.add("opacity-0");
            setTimeout(() => mobileOverlay.classList.add("hidden"), 300);
        }

        if (floatingToggleBtn) floatingToggleBtn.addEventListener("click", openMobileSidebar);
        if (mobileOverlay) mobileOverlay.addEventListener("click", closeMobileSidebar);

        // Popover Toggle Logic
        function setAccountPopoverOpen(isOpen) {
            if (!settingsPopover || !profileTrigger) return;
            settingsPopover.classList.toggle('hidden', !isOpen);
            profileTrigger.setAttribute('aria-expanded', String(isOpen));
        }

        function togglePopover(e) {
            e.stopPropagation();
            setAccountPopoverOpen(settingsPopover?.classList.contains('hidden'));
        }

        if (profileTrigger) profileTrigger.addEventListener('click', togglePopover);

        // Click outside the avatar/menu closes the account popover.
        document.addEventListener('click', (e) => {
            if (
                settingsPopover &&
                !settingsPopover.contains(e.target) &&
                !profileTrigger?.contains(e.target)
            ) {
                setAccountPopoverOpen(false);
            }
        });

        
        // Settings Modal
        const settingsDialog = document.getElementById('settings-dialog');
        const settingsClose = document.getElementById('settings-close');
        const settingsCancel = document.getElementById('settings-cancel');

        const settingsTabs = document.querySelectorAll('[data-settings-tab]');
        const settingsOpeners = document.querySelectorAll('[data-open-settings]');

        function switchSettingsTab(tabName) {
            settingsTabs.forEach((tab) => {
                const active = tab.dataset.settingsTab === tabName;

                tab.setAttribute('aria-selected', String(active));
                tab.classList.toggle('border-primary', active);
                tab.classList.toggle('text-primary', active);
                tab.classList.toggle('font-semibold', active);
                tab.classList.toggle('border-transparent', !active);
                tab.classList.toggle('text-on-surface-variant', !active);
                tab.classList.toggle('font-medium', !active);

                const panel = document.getElementById(
                    `settings-panel-${tab.dataset.settingsTab}`
                );

                if (panel) {
                    panel.classList.toggle('hidden', !active);
                }
            });
        }

        
        const settingsSave = document.getElementById('settings-save');
        const settingsStatus = document.getElementById('settings-status');

        const settingsFields = {
            remember_multiplier: document.getElementById('remember-multiplier'),
            forgot_multiplier: document.getElementById('forgot-multiplier'),
            initial_interval_days: document.getElementById('initial-interval'),
            maximum_interval_days: document.getElementById('maximum-interval')
        };

        const settingsApi = <?= json_encode(BASE_URL . 'api/settings-actions.php') ?>;

        function showSettingsStatus(message, isError = false) {
            if (!settingsStatus) return;

            settingsStatus.textContent = message;
            settingsStatus.classList.toggle('text-error', isError);
            settingsStatus.classList.toggle('text-primary', !isError);
        }

        function displaySettings(settings) {
            Object.entries(settingsFields).forEach(([key, input]) => {
                if (input) input.value = settings[key];
            });
        }

        async function loadSettings() {
            showSettingsStatus('Loading settings...');

            try {
                const response = await fetch(settingsApi, {
                    credentials: 'same-origin'
                });

                const result = await response.json();

                if (!response.ok || !result.success) {
                    throw new Error(result.message || 'Could not load settings.');
                }

                displaySettings(result.settings);
                showSettingsStatus('');
            } catch (error) {
                showSettingsStatus(error.message, true);
            }
        }

        async function saveSettings() {
            const payload = {};

            for (const [key, input] of Object.entries(settingsFields)) {
                if (!input || !input.checkValidity()) {
                    input?.reportValidity();
                    return;
                }

                payload[key] = input.value;
            }

            const initialValue = Number(payload.initial_interval_days);
            const maximumValue = Number(payload.maximum_interval_days);
            if (maximumValue < initialValue) {
                settingsFields.maximum_interval_days.setCustomValidity('Maximum interval must be at least the initial interval.');
                settingsFields.maximum_interval_days.reportValidity();
                settingsFields.maximum_interval_days.setCustomValidity('');
                return;
            }

            settingsSave.disabled = true;
            showSettingsStatus('Saving settings...');

            try {
                const response = await fetch(settingsApi, {
                    method: 'POST',
                    credentials: 'same-origin',
                    headers: {
                        'Content-Type': 'application/json',
                        'X-CSRF-Token': settingsDialog.dataset.csrf
                    },
                    body: JSON.stringify(payload)
                });

                const result = await response.json();

                if (!response.ok || !result.success) {
                    throw new Error(result.message || 'Could not save settings.');
                }

                displaySettings(result.settings);
                showSettingsStatus(result.message);
            } catch (error) {
                showSettingsStatus(error.message, true);
            } finally {
                settingsSave.disabled = false;
            }
        }

        settingsSave?.addEventListener('click', saveSettings);


        settingsOpeners.forEach((button) => {
            button.addEventListener('click', () => {
                const tabName = button.dataset.openSettings;

                setAccountPopoverOpen(false);
                switchSettingsTab(tabName);

                if (settingsDialog) {
                    if (!settingsDialog.open) {
                        settingsDialog.showModal();
                    }
                    loadSettings();
                }
            });
        });

        settingsTabs.forEach((tab) => {
            tab.addEventListener('click', () => {
                switchSettingsTab(tab.dataset.settingsTab);
            });
        });

        function closeSettings() {
            if (settingsDialog?.open) {
                settingsDialog.close();
            }
        }

        settingsClose?.addEventListener('click', closeSettings);
        settingsCancel?.addEventListener('click', closeSettings);

        // Close when clicking the backdrop
        settingsDialog?.addEventListener('click', (event) => {
            if (event.target === settingsDialog) {
                closeSettings();
            }
        });

        // Logout Dialog Logic
        const logoutBtn = document.getElementById('logout-btn');
        const logoutDialog = document.getElementById('logout-dialog');
        const logoutCancel = document.getElementById('logout-cancel');

        if (logoutBtn && logoutDialog) {
            logoutBtn.addEventListener('click', () => {
                setAccountPopoverOpen(false); // Close account menu
                logoutDialog.showModal();
            });

            if (logoutCancel) {
                logoutCancel.addEventListener('click', () => logoutDialog.close());
            }

            logoutDialog.addEventListener('click', (e) => {
                const rect = logoutDialog.getBoundingClientRect();
                if (e.clientX < rect.left || e.clientX > rect.right || e.clientY < rect.top || e.clientY > rect.bottom) {
                    logoutDialog.close();
                }
            });
        }
    });

CREATE DATABASE IF NOT EXISTS mempulse_db
CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

USE mempulse_db;


-- 1. USERS TABLE
CREATE TABLE users (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    email VARCHAR(255) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    role ENUM('admin', 'user') NOT NULL DEFAULT 'user',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL,
    INDEX idx_users_active (deleted_at)
) ENGINE=InnoDB;


-- 2. USER SETTINGS TABLE (SRS Configuration)
CREATE TABLE user_settings (
    user_id BIGINT UNSIGNED PRIMARY KEY,
    remember_multiplier DECIMAL(4,2) NOT NULL DEFAULT 2.00,
    forgot_multiplier DECIMAL(4,2) NOT NULL DEFAULT 0.50,
    initial_interval_days INT UNSIGNED NOT NULL DEFAULT 1,
    maximum_interval_days INT UNSIGNED NOT NULL DEFAULT 365,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE
) ENGINE=InnoDB;


-- 3. DECKS TABLE
CREATE TABLE decks (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT UNSIGNED NOT NULL,
    title VARCHAR(150) NOT NULL,
    cover_image VARCHAR(255) DEFAULT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL,
    FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE,
    INDEX idx_decks_user_active (user_id, deleted_at)
) ENGINE=InnoDB;


-- 4. FLASHCARDS TABLE
CREATE TABLE flashcards (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    deck_id BIGINT UNSIGNED NOT NULL,
    front TEXT NOT NULL,
    back TEXT NOT NULL,
    interval_days INT UNSIGNED NOT NULL DEFAULT 1,
    next_review_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    review_count INT UNSIGNED NOT NULL DEFAULT 0,
    lapses INT UNSIGNED NOT NULL DEFAULT 0,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL,
    FOREIGN KEY (deck_id)
        REFERENCES decks(id)
        ON DELETE CASCADE,
    INDEX idx_cards_deck_due (deck_id, deleted_at, next_review_at)
) ENGINE=InnoDB;


-- 5. REVIEW SESSIONS TABLE
CREATE TABLE review_sessions (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT UNSIGNED NOT NULL,
    deck_id BIGINT UNSIGNED NOT NULL,
    started_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    ended_at DATETIME NULL,
    total_reviewed INT UNSIGNED NOT NULL DEFAULT 0,
    remembered_count INT UNSIGNED NOT NULL DEFAULT 0,
    accuracy_percentage DECIMAL(5,2)
        GENERATED ALWAYS AS (
            CASE
                WHEN total_reviewed > 0
                    THEN (remembered_count / total_reviewed) * 100
                ELSE 0.00
            END
        ) STORED,
    FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE,
    FOREIGN KEY (deck_id)
        REFERENCES decks(id)
        ON DELETE CASCADE,
    INDEX idx_user_sessions (user_id, started_at)
) ENGINE=InnoDB;


-- 6. REVIEW LOGS TABLE
CREATE TABLE review_logs (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    session_id BIGINT UNSIGNED NOT NULL,
    card_id BIGINT UNSIGNED NULL,
    rating ENUM('remembered', 'forgot') NOT NULL,
    previous_interval INT UNSIGNED NOT NULL,
    new_interval INT UNSIGNED NOT NULL,
    response_time_ms INT UNSIGNED NULL,
    reviewed_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (session_id)
        REFERENCES review_sessions(id)
        ON DELETE CASCADE,
    FOREIGN KEY (card_id)
        REFERENCES flashcards(id)
        ON DELETE SET NULL,
    INDEX idx_session_logs (session_id)
) ENGINE=InnoDB;
