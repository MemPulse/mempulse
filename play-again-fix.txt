    // API Helper with Database Connectivity Verification
    async function sendCardAction(payload) {
        try {
            const res = await fetch('../api/flashcard-actions.php', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ ...payload, csrf_token: CSRF_TOKEN, deck_id: DECK_ID })
            });
            let data = {};
            try { data = await res.json(); } catch (_) { }

            if (!res.ok || !data.ok) {
                throw new Error(data.error || `Database communication error (HTTP ${res.status}).`);
            }
            return data;
        } catch (error) {
            console.error("Database connection failure:", error);
            throw error;
        }
    }

    function formatDateTimeForInput(dateString) {
        if (!dateString) return '';
        const d = new Date(dateString);
        if (isNaN(d.getTime())) return '';
        const pad = (n) => String(n).padStart(2, '0');
        return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
    }

        return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())} ${pad(d.getHours())}:${pad(d.getMinutes())}`;

    function pageStep() {
        const holder = gridContainer.querySelector('.tabulator-tableholder');
        const first = tabulatorTable.getRows()[0];
        const rowH = first ? first.getElement().offsetHeight : 0;
        if (!holder || !rowH) return 10;
        return Math.max(1, Math.floor(holder.clientHeight / rowH) - 1);   // one screenful, minus one row of overlap
    }

    // Make sure row #index (0-based) is rendered, pulling 50 more at a time
    async function ensureLoaded(index) {
        while (tabulatorTable.getDataCount() <= index && pending.length) {
            if (loadingMore) { await new Promise((r) => setTimeout(r, 30)); continue; }
            await loadMore();
        }
    }

    // Single place that makes a row "the" selected row: one highlight, editor in sync
    function activateRow(row, { expand = true, position = 'nearest' } = {}) {
        tabulatorTable.deselectRow(tabulatorTable.getSelectedRows());
        row.select();
        if (isDesktopView()) populateSidebar(row.getData(), row, { expand });
        Promise.resolve(row.scrollTo(position, false)).catch(() => { });
    }

    async function navigateSidebarRow(direction, { expand = true, step = 1 } = {}) {
        if (!tabulatorTable) return;
        let rows = tabulatorTable.getRows();
        if (!rows.length) return;

        const idx = rows.findIndex(r => Number(r.getData().id) === selectedSidebarCardId);
        let targetIdx;
        if (idx === -1) targetIdx = direction === 'next' ? 0 : rows.length - 1;
        else targetIdx = direction === 'next' ? idx + step : idx - step;

        if (targetIdx > rows.length - 1) {                 // past the loaded slice: pull more first
            await ensureLoaded(targetIdx);
            rows = tabulatorTable.getRows();
        }
        targetIdx = Math.max(0, Math.min(targetIdx, rows.length - 1));   // clamp at first/last card
        if (targetIdx === idx) return;
        activateRow(rows[targetIdx], { expand });
    }

    // ↑ / ↓ browse cards on desktop (ignored while typing in a field or when a dialog is open)

        PageDown: ['next', true], PageUp: ['prev', true],

    document.addEventListener('keydown', (e) => {
        // 1. Ctrl + Return to save changes
        if ((e.ctrlKey || e.metaKey) && e.key === 'Enter') {
            if (isDesktopView() && !isSidebarCollapsed && !viewArchived && (checkedIds.size > 0 || !selectedSidebarCardId)) {
                e.preventDefault();
                // If the save button isn't hidden (meaning it's not locked), trigger the click
                if (sidebarSaveBtn && !sidebarSaveBtn.classList.contains('hidden')) {
                    sidebarSaveBtn.click();
                }
            }
            return;
        }

        // Guard against intercepting keystrokes while typing in an input or a modal
        if (document.querySelector('dialog[open]')) return;
        if (e.target instanceof Element && e.target.closest('input, textarea, select, [contenteditable="true"]')) return;

        // 2. Return key to toggle sidebar collapse
        if (e.key === 'Enter' && !e.ctrlKey && !e.metaKey && !e.shiftKey && !e.altKey) {
            if (!isDesktopView() || viewArchived) return;
            e.preventDefault();
            setSidebarCollapsed(!isSidebarCollapsed);
            return;
        }

        // 3. Arrow Up/Down Navigation
        const nav = NAV_KEYS[e.key];
        if (!nav) return;
        if (!isDesktopView() || viewArchived || e.altKey || e.ctrlKey || e.metaKey || e.shiftKey) return;
        e.preventDefault();
        navigateSidebarRow(nav[0], { expand: false, step: nav[1] ? pageStep() : 1 });
    });

    const PAGE_SIZE = 50;
    let pending = [];          // cards of the current view that aren't rendered yet
    let loadingMore = false;

    function updateLoadMoreUI() {
        if (!gridFooter || !tabulatorTable) return;
        const loaded = tabulatorTable.getDataCount();
        const remaining = pending.length;
        const total = loaded + remaining;
        const noun = total === 1 ? 'card' : 'cards';

        gridFooter.hidden = total === 0;
        rangeLabel.textContent = remaining ? `Showing ${loaded} of ${total} ${noun}` : `All ${total} ${noun} loaded`;
        progressBar.style.width = total ? `${Math.round((loaded / total) * 100)}%` : '0%';
        renderJumpRail();
    }

    function loadFirstPage(list) {
        pending = list.slice(PAGE_SIZE);
        return tabulatorTable.setData(list.slice(0, PAGE_SIZE)).then(updateLoadMoreUI);
    }

    async function loadMore() {
        if (loadingMore || !pending.length || !tabulatorTable) return;
        loadingMore = true;
        updateLoadMoreUI();
        try {
            await tabulatorTable.addData(pending.splice(0, PAGE_SIZE), false); // false = append at bottom
        } finally {
            loadingMore = false;
            updateLoadMoreUI();
        }
    }

    function bindGridScroll() {
        const holder = gridContainer.querySelector('.tabulator-tableholder');
        if (!holder) return;
        holder.addEventListener('scroll', () => {
            if (holder.scrollTop + holder.clientHeight >= holder.scrollHeight - 160) loadMore();
        }, { passive: true });
    }
    // ---- Jump rail: clickable record ranges beside the grid ----
    const MAX_RAIL_BUTTONS = 16;
    let railStep = PAGE_SIZE;
    let railTotal = -1;

    function renderJumpRail() {
        if (!jumpRail) return;
        const total = tabulatorTable ? tabulatorTable.getDataCount() + pending.length : 0;
        if (total === railTotal) return;
        railTotal = total;
        jumpRail.textContent = '';
        if (total <= PAGE_SIZE) { jumpRail.hidden = true; return; }

        railStep = PAGE_SIZE * Math.ceil(total / PAGE_SIZE / MAX_RAIL_BUTTONS);
        for (let start = 0; start < total; start += railStep) {
            const b = document.createElement('button');
            b.type = 'button';
            b.dataset.start = start;
            b.textContent = start + 1;
            b.title = `Jump to card ${start + 1}`;
            b.setAttribute('aria-label', `Jump to card ${start + 1}`);
            jumpRail.appendChild(b);
        }
        jumpRail.hidden = false;
        const idx = selectedSidebarCardId
            ? tabulatorTable.getRows().findIndex(r => Number(r.getData().id) === selectedSidebarCardId)
            : -1;
        highlightJumpRail(idx);
    }

    function highlightJumpRail(idx) {
        if (!jumpRail || jumpRail.hidden) return;
        const start = idx < 0 ? -1 : Math.floor(idx / railStep) * railStep;
        jumpRail.querySelectorAll('button').forEach((b) =>
            b.classList.toggle('is-active', Number(b.dataset.start) === start));
    }

    async function jumpToRecord(index) {
        if (!tabulatorTable) return;
        await ensureLoaded(index);
        const row = tabulatorTable.getRows()[index];
        if (!row) return;
        activateRow(row, { expand: false, position: 'top' });
        highlightJumpRail(index);
    }

    if (jumpRail) jumpRail.addEventListener('click', (e) => {
        const b = e.target.closest('button[data-start]');
        if (b) jumpToRecord(Number(b.dataset.start));

<?php
/**
 * PracticeService: READ-ONLY card lookups for Practice mode.
 * Nothing in this file writes to the database (SELECT statements only).
 * Place next to ReviewHubService.php (app/api/).
 */
class PracticeService
{
    const MAX_CARDS = 300;

    /** Which cards fields review.js needs (same shape as the due-cards query). */
    private const CARD_COLS = 'f.id, f.front, f.back, f.interval_days, f.next_review_at';

    /**
     * Parse "5,9,14" into positive unique ints, original order, first MAX_CARDS only.
     */
    public static function parseIds($raw): array
    {
        if (!is_string($raw) || $raw === '') {
            return [];
        }
        $ids = [];
        foreach (explode(',', $raw) as $part) {
            $part = trim($part);
            if ($part !== '' && ctype_digit($part)) {
                $id = (int) $part;
                if ($id > 0) {
                    $ids[$id] = true; // key = dedupe, keeps first-seen order
                }
            }
            if (count($ids) >= self::MAX_CARDS) {
                break;
            }
        }
        return array_keys($ids);
    }

    /**
     * Cards by id: only this user's, in this deck, not archived/deleted.
     * Returned in the same order as $ids (grid order from the deck page).
     */
    public static function cardsByIds($conn, int $userId, int $deckId, array $ids): array
    {
        $ids = array_slice(array_values(array_unique(array_filter(array_map('intval', $ids), fn($i) => $i > 0))), 0, self::MAX_CARDS);
        if (!$ids) {
            return [];
        }
        $marks = implode(',', array_fill(0, count($ids), '?'));
        $sql = "SELECT " . self::CARD_COLS . "
                FROM flashcards f
                JOIN decks d ON d.id = f.deck_id
                WHERE f.id IN ($marks) AND f.deck_id = ? AND d.user_id = ?
                  AND d.deleted_at IS NULL AND f.deleted_at IS NULL";
        $params = array_merge($ids, [$deckId, $userId]);
        $types  = str_repeat('i', count($params));

        $byId = [];
        if ($stmt = mysqli_prepare($conn, $sql)) {
            mysqli_stmt_bind_param($stmt, $types, ...$params);
            mysqli_stmt_execute($stmt);
            $res = mysqli_stmt_get_result($stmt);
            while ($row = mysqli_fetch_assoc($res)) {
                $byId[(int) $row['id']] = $row;
            }
            mysqli_stmt_close($stmt);
        }
        $ordered = [];
        foreach ($ids as $id) {
            if (isset($byId[$id])) {
                $ordered[] = $byId[$id];
            }
        }
        return $ordered;
    }

    /**
     * The session's deck and how many distinct cards it logged (including ones whose
     * card_id is now NULL because the card was permanently deleted).
     * Returns null if the session does not exist or is not this user's.
     */
    public static function sessionInfo($conn, int $userId, int $sessionId): ?array
    {
        $sql = "SELECT rs.deck_id, d.title,
                       COUNT(DISTINCT rl.card_id) + SUM(rl.card_id IS NULL) AS requested
                FROM review_sessions rs
                JOIN decks d ON d.id = rs.deck_id
                LEFT JOIN review_logs rl ON rl.session_id = rs.id
                WHERE rs.id = ? AND rs.user_id = ? AND d.user_id = ?
                GROUP BY rs.id, rs.deck_id, d.title";
        $info = null;
        if ($stmt = mysqli_prepare($conn, $sql)) {
            mysqli_stmt_bind_param($stmt, "iii", $sessionId, $userId, $userId);
            mysqli_stmt_execute($stmt);
            $row = mysqli_fetch_assoc(mysqli_stmt_get_result($stmt));
            mysqli_stmt_close($stmt);
            if ($row) {
                $info = [
                    'deckId'    => (int) $row['deck_id'],
                    'deckTitle' => $row['title'],
                    'requested' => (int) $row['requested'],
                ];
            }
        }
        return $info;
    }

    /**
     * Cards of one of the user's sessions, in the order they were reviewed.
     * Archived/deleted cards are left out. A card logged twice appears once.
     */
    public static function cardsFromSession($conn, int $userId, int $sessionId): array
    {
        $sql = "SELECT " . self::CARD_COLS . "
                FROM review_sessions rs
                JOIN review_logs rl ON rl.session_id = rs.id
                JOIN flashcards f ON f.id = rl.card_id
                JOIN decks d ON d.id = f.deck_id
                WHERE rs.id = ? AND rs.user_id = ? AND d.user_id = ?
                  AND d.deleted_at IS NULL AND f.deleted_at IS NULL
                ORDER BY rl.id ASC";
        $cards = [];
        if ($stmt = mysqli_prepare($conn, $sql)) {
            mysqli_stmt_bind_param($stmt, "iii", $sessionId, $userId, $userId);
            mysqli_stmt_execute($stmt);
            $res = mysqli_stmt_get_result($stmt);
            while ($row = mysqli_fetch_assoc($res)) {
                $cards[(int) $row['id']] ??= $row; // first occurrence wins
            }
            mysqli_stmt_close($stmt);
        }
        return array_slice(array_values($cards), 0, self::MAX_CARDS);
    }

    /**
     * Where "Back" goes. The server decides from an allow-list; never a free URL.
     */
    public static function returnUrl(string $from, int $deckId, int $sessionId): string
    {
        switch ($from) {
            case 'deck':
                return $deckId > 0 ? 'deck.php?deck_id=' . $deckId : 'review.php';
            case 'sessions':
                return 'sessions.php';
            case 'reports':
                return 'reports.php';
            case 'session':
                return $sessionId > 0 ? 'session-details.php?id=' . $sessionId : 'sessions.php';
            default:
                return 'review.php';
        }
    }

    function stampStat(el, tile, ringClass, { scale = 3, duration = 520, shakePx = 5, hero = false } = {}) {

    function impactRing(tile, colorClass) {

    function shake(el, px = 6, duration = 380) {
        el.animate([

            { transform: `translate(${px * 0.4}px, 0)` },

    const CONFETTI_COLORS = ['#10b981', '#34d399', '#6366f1', '#f59e0b', '#f43f5e', '#38bdf8'];
    let confettiCanvas = null, confettiCtx = null, confettiParts = [], confettiRaf = null;

    function confettiBurst({ x = 0.5, y = 0.4, count = 120, angle = -Math.PI / 2, spread = Math.PI * 2, power = 15 } = {}) {
        if (reduceMotion) return;

        const dpr = window.devicePixelRatio || 1;
        if (!confettiCanvas) {
            confettiCanvas = document.createElement('canvas');
            confettiCanvas.className = 'fixed inset-0 w-full h-full pointer-events-none z-[60]';
            document.body.appendChild(confettiCanvas);
            confettiCtx = confettiCanvas.getContext('2d');
        }
        if (confettiCanvas.width !== Math.round(innerWidth * dpr)) {
            confettiCanvas.width = Math.round(innerWidth * dpr);
            confettiCanvas.height = Math.round(innerHeight * dpr);
        }

        for (let i = 0; i < count; i++) {
            const a = angle + (Math.random() - 0.5) * spread;
            const v = power * (0.5 + Math.random() * 0.7);
            confettiParts.push({
                x: x * innerWidth, y: y * innerHeight,
                vx: Math.cos(a) * v, vy: Math.sin(a) * v,
                w: 6 + Math.random() * 6, h: 4 + Math.random() * 5,
                rot: Math.random() * Math.PI * 2, vr: (Math.random() - 0.5) * 0.35,
                wobble: Math.random() * Math.PI * 2,
                color: CONFETTI_COLORS[(Math.random() * CONFETTI_COLORS.length) | 0],
                life: 0, ttl: 110 + Math.random() * 60
            });
        }
        if (!confettiRaf) confettiRaf = requestAnimationFrame(stepConfetti);
    }

    function stepConfetti() {
        const dpr = window.devicePixelRatio || 1;
        confettiCtx.setTransform(dpr, 0, 0, dpr, 0, 0);
        confettiCtx.clearRect(0, 0, innerWidth, innerHeight);

        confettiParts = confettiParts.filter(p => p.life < p.ttl && p.y < innerHeight + 40);
        for (const p of confettiParts) {
            p.life++;
            p.vx *= 0.985;
            p.vy = p.vy * 0.985 + 0.38;                 // drag + gravity
            p.wobble += 0.12;
            p.x += p.vx + Math.sin(p.wobble) * 0.6;
            p.y += p.vy;
            p.rot += p.vr;

            confettiCtx.save();
            confettiCtx.globalAlpha = Math.min(1, (p.ttl - p.life) / 30);
            confettiCtx.translate(p.x, p.y);
            confettiCtx.rotate(p.rot);
            confettiCtx.scale(1, Math.abs(Math.cos(p.wobble)));   // flutter
            confettiCtx.fillStyle = p.color;
            confettiCtx.fillRect(-p.w / 2, -p.h / 2, p.w, p.h);
            confettiCtx.restore();
        }

        if (confettiParts.length) {
            confettiRaf = requestAnimationFrame(stepConfetti);
        } else {
            confettiRaf = null;
            confettiCanvas.remove();
            confettiCanvas = null;
        }
    }
