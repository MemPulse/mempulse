<?php
// 1. Initialize user session tracking
session_start();

// 2. Route Protection: Redirect to login view if user credentials are uninitialized

    header("Location: ../../authentication/login.php");
    exit;
}

// 3. Connect to database module and load business service
require_once __DIR__ . '/../../config/db.php';
require_once __DIR__ . '/../api/DashboardService.php';

$userId = (int) $_SESSION['user_id'];

// --- DATA ORCHESTRATION ---
$profileData    = DashboardService::getUserProfile($conn, $userId);
$metricsData    = DashboardService::getMetrics($conn, $userId);
$streakData     = DashboardService::getStreakAndHeatmapData($conn, $userId);
$recentSessions = DashboardService::getRecentSessions($conn, $userId, 4);

// Extracted variables for template includes
$displayName    = $profileData['displayName'];
$current_streak = $streakData['currentStreak'];
$pageTitle      = "Home Dashboard";
$currentPage    = 'home';

/**
 * Helper function for Heatmap grid CSS class matching review intensity (Green Palette).
 */
if (!function_exists('heatmap_level_style')) {
    function heatmap_level_style(int $reviews): string {
        if ($reviews <= 0) return 'background-color: #F6F6FD; border: 1px solid rgba(150,150,150,0.2);';
        if ($reviews <= 7) return 'background-color: #86efac;';
        if ($reviews <= 14) return 'background-color: #4ade80;';
        if ($reviews <= 21) return 'background-color: #16a34a;';
        return 'background-color: #15803d;';
    }
}
?>
<!DOCTYPE html>
<html lang="en">

<?php include __DIR__ . '/../includes/head.php'; ?>

<!-- BODY: Main App Background & Styling Root -->
<body class="bg-background font-body-md text-on-surface antialiased min-h-screen selection:bg-primary-fixed selection:text-on-primary-fixed">

    <?php include __DIR__ . '/../includes/sidebar.php'; ?>

    <!-- MAIN: Viewport Shell Offset for Fixed Top Header (pt-16) and Sidebar (md:pl-72) -->
    <main id="main-content" 
        class="pl-0 md:pl-72 pt-16 transition-all duration-300 ease-in-out">
        <?php include __DIR__ . '/dashboard.php'; ?>                
    </main>

    <!-- 1. DATA INJECTION: Server-side PHP metrics passed safely to client-side JS -->
    <script>
        window.MemPulseHomeData = {
            stats: {
                cardsDue: <?php echo (int) $metricsData['cardsDue']; ?>,
                masteredCards: <?php echo (int) $metricsData['masteredCards']; ?>,
                toughCards: <?php echo (int) $metricsData['toughCards']; ?>,
                currentStreak: <?php echo (int) $streakData['currentStreak']; ?>,
                bestStreak: <?php echo (int) $streakData['bestStreak']; ?>,
                streakAccuracy: <?php echo (float) $streakData['streakAccuracy']; ?>
            }
        };

        /**
         * Global Dashboard UI Manager
         * Synchronizes dynamic metrics across header, sidebar, and metric panels.
         */
        const MemPulseDashboard = {
            updateAll(payload) {
                if (!payload) return;
                this.syncHeader(payload.user);
                this.syncSidebarProfile(payload.user);
                this.syncMetrics(payload.stats);
            },

            syncHeader(user) {
                if (!user || !user.displayName) return;
                document.querySelectorAll('[data-user-name]').forEach((el) => {
                    el.textContent = user.displayName;
                });
            },

            syncSidebarProfile(user) {
                if (!user || !user.displayName) return;
                document.querySelectorAll('[data-user-name]').forEach((el) => {
                    el.textContent = user.displayName;
                });
            },

            syncMetrics(stats) {
                if (!stats) return;
                this.setElText('stat-cards-due', stats.cardsDue);
                this.setElText('stat-mastered', stats.masteredCards);
                this.setElText('stat-tough', stats.toughCards);
                this.setElText('stat-current-streak', stats.currentStreak);
                this.setElText('stat-best-streak', stats.bestStreak);
                this.setElText('stat-streak-accuracy', stats.streakAccuracy);
            },

            setElText(id, value) {
                const el = document.getElementById(id);
                if (el) el.textContent = value ?? '--';
            }
        };

        document.addEventListener("DOMContentLoaded", () => {
            MemPulseDashboard.updateAll(window.MemPulseHomeData);

            const heatmapScroller = document.getElementById("heatmap-scroll");
            if (heatmapScroller) {
                requestAnimationFrame(() => {
                    heatmapScroller.scrollLeft = heatmapScroller.scrollWidth - heatmapScroller.clientWidth;
                });
            }
        });
    </script>

    <!-- 2. CLIENT-SIDE ASSET IMPORTS: Separated standalone JavaScript modules -->
    <script src="../assets/js/hero-carousel.js" defer></script>
</body>
</html>

        const mainContent = document.querySelector("main");
        const toggleBtn = document.querySelector("#sidebar-toggle");

        const floatingToggleBtn = document.querySelector("#floating-mobile-toggle");
        const mobileToggleIcon = document.querySelector("#mobile-toggle-icon");
        const mobileOverlay = document.querySelector("#mobile-sidebar-overlay");
        const reviewDot = document.querySelector("#review-dot");

        // Header Profile Popover Elements
        const profileTrigger = document.querySelector("#profile-popover-trigger");
        const profileChevron = document.querySelector("#profile-chevron");
        const settingsPopover = document.querySelector("#settings-popover");

        if (!sidebar) return;

        // Body-level fixed tooltip for collapsed desktop sidebar
        let floatingTooltip = document.getElementById("sidebar-floating-tooltip");
        if (!floatingTooltip) {
            floatingTooltip = document.createElement("div");
            floatingTooltip.id = "sidebar-floating-tooltip";
            floatingTooltip.setAttribute("role", "tooltip");
            document.body.appendChild(floatingTooltip);
        }

        let tooltipTarget = null;
        function hideSidebarTooltip() {
            floatingTooltip.classList.remove("is-visible");
            tooltipTarget = null;
        }
        function showSidebarTooltip(target) {
            const label = target.getAttribute("data-tooltip") || target.getAttribute("aria-label");
            if (!label || !sidebar.classList.contains("collapsed") || window.innerWidth < 768) {
                hideSidebarTooltip();
                return;
            }
            tooltipTarget = target;
            floatingTooltip.textContent = label;
            floatingTooltip.style.left = "0px";
            floatingTooltip.style.top = "0px";
            floatingTooltip.classList.add("is-visible");
            const rect = target.getBoundingClientRect();
            const tipRect = floatingTooltip.getBoundingClientRect();
            const gap = 12;
            let left = rect.right + gap;
            if (left + tipRect.width > window.innerWidth - 8) {
                left = Math.max(8, rect.left - tipRect.width - gap);
                floatingTooltip.classList.add("tooltip-left");
            } else {
                floatingTooltip.classList.remove("tooltip-left");
            }
            const top = Math.min(window.innerHeight - 8, Math.max(8, rect.top + rect.height / 2));
            floatingTooltip.style.left = `${left}px`;
            floatingTooltip.style.top = `${top}px`;
        }

        sidebar.querySelectorAll(".sidebar-tooltip").forEach((item) => {
            item.addEventListener("mouseenter", () => showSidebarTooltip(item));
            item.addEventListener("mouseleave", hideSidebarTooltip);
            item.addEventListener("focus", () => showSidebarTooltip(item));
            item.addEventListener("blur", hideSidebarTooltip);
        });
        sidebar.addEventListener("scroll", hideSidebarTooltip, true);
        window.addEventListener("resize", hideSidebarTooltip);
        window.addEventListener("scroll", () => {
            if (tooltipTarget) showSidebarTooltip(tooltipTarget);
        }, true);

        const toggleIcon = toggleBtn ? toggleBtn.querySelector(".material-symbols-outlined") : null;

        // Apply visual collapsed state on desktop
        const sidebarCollapsed = localStorage.getItem("sidebarCollapsed") === "true";
        if (sidebarCollapsed && window.innerWidth >= 768) {
            sidebar.classList.add("collapsed");
            if (mainContent) mainContent.classList.add("collapsed");
            if (toggleIcon) toggleIcon.textContent = "left_panel_open";
            if (reviewDot) reviewDot.classList.remove("hidden");
        }

        // Desktop Border Toggle Interaction
        if (toggleBtn) {
            toggleBtn.addEventListener("click", () => {
                const isCollapsed = sidebar.classList.toggle("collapsed");
                
                document.documentElement.classList.toggle(
                    "sidebar-collapsed",
                    isCollapsed
                );
                
                hideSidebarTooltip();
                if (mainContent) {
                    mainContent.classList.toggle("collapsed", isCollapsed);
                }
                if (toggleIcon) {
                    toggleIcon.textContent = isCollapsed ? "left_panel_open" : "left_panel_close";
                }
                if (reviewDot) {
                    isCollapsed ? reviewDot.classList.remove("hidden") : reviewDot.classList.add("hidden");
                }
                setAccountPopoverOpen(false);
                localStorage.setItem("sidebarCollapsed", isCollapsed);
            });
        }

        // Mobile Sidebar Drawer Toggle (Inside Header)
        function isMobileSidebarOpen() {
            return !sidebar.classList.contains("-translate-x-full");
        }

        function openMobileSidebar() {
            sidebar.classList.remove("-translate-x-full", "collapsed");
            mobileOverlay.classList.remove("hidden");
            if (floatingToggleBtn) floatingToggleBtn.setAttribute("aria-expanded", "true");
            if (mobileToggleIcon) mobileToggleIcon.textContent = "close";
            setTimeout(() => mobileOverlay.classList.remove("opacity-0"), 10);
        }

        function closeMobileSidebar() {
            sidebar.classList.add("-translate-x-full");
            mobileOverlay.classList.add("opacity-0");
            if (floatingToggleBtn) floatingToggleBtn.setAttribute("aria-expanded", "false");
            if (mobileToggleIcon) mobileToggleIcon.textContent = "menu";
            setTimeout(() => mobileOverlay.classList.add("hidden"), 300);
        }

        if (floatingToggleBtn) {
            floatingToggleBtn.addEventListener("click", () => {
                if (isMobileSidebarOpen()) {
                    closeMobileSidebar();

                }
            });
        }
        if (mobileOverlay) mobileOverlay.addEventListener("click", closeMobileSidebar);

        // Header Profile Popover Toggle Logic
        function setAccountPopoverOpen(isOpen) {

        }

        function renderSearchResults(query) {
            if (!searchResults) return;
            const q = query.toLowerCase();
            activeSearchIndex = -1;

            const matchedPages = q === ''
                ? quickPages.slice(0, 5)
                : quickPages.filter(p => p.title.toLowerCase().includes(q) || p.subtitle.toLowerCase().includes(q));

            const matchedDecks = q === ''
                ? (searchData.decks || []).slice(0, 4)

            const matchedCards = q === ''

                : (searchData.cards || []).filter(c =>

                            Decks

                        <div class="flex flex-col gap-0.5">

        async function loadSettings() {
            showSettingsStatus('Loading settings...');

            try {
                const response = await fetch(settingsApi, {
                    credentials: 'same-origin'
                });

                const result = await response.json();

                if (!response.ok || !result.success) {
                    throw new Error(result.message || 'Could not load settings.');
                }

                displaySettings(result.settings);
                showSettingsStatus('');
            } catch (error) {
                showSettingsStatus(error.message, true);
            }
        }

        async function saveSettings() {
            const payload = {};

            for (const [key, input] of Object.entries(settingsFields)) {
                if (!input || !input.checkValidity()) {
                    input?.reportValidity();
                    return;
                }

                payload[key] = input.value;
            }

            const initialValue = Number(payload.initial_interval_days);
            const maximumValue = Number(payload.maximum_interval_days);
            if (maximumValue < initialValue) {
                settingsFields.maximum_interval_days.setCustomValidity('Maximum interval must be at least the initial interval.');
                settingsFields.maximum_interval_days.reportValidity();
                settingsFields.maximum_interval_days.setCustomValidity('');
                return;
            }

            settingsSave.disabled = true;
            showSettingsStatus('Saving settings...');

            try {
                const response = await fetch(settingsApi, {
                    method: 'POST',
                    credentials: 'same-origin',
                    headers: {
                        'Content-Type': 'application/json',
                        'X-CSRF-Token': settingsDialog.dataset.csrf
                    },
                    body: JSON.stringify(payload)
                });

                const result = await response.json();

                if (!response.ok || !result.success) {
                    throw new Error(result.message || 'Could not save settings.');
                }

                displaySettings(result.settings);
                showSettingsStatus(result.message);
            } catch (error) {
                showSettingsStatus(error.message, true);
            } finally {
                settingsSave.disabled = false;
            }
        }

            });
