/**
 * PDF Reviewer: layout switch, remembered options and print button.
 * Works without a network; nothing is sent to the server.
 *
 * Rules for the layout:
 *   1. ?layout=beside|end in the URL wins (server marks it with data-layout-source="url")
 *   2. otherwise the last choice saved in localStorage
 *   3. otherwise "beside"
 */
(function () {
    'use strict';

    var KEY_LAYOUT = 'mempulse.pdfReviewer.layout';
    var KEY_LINES = 'mempulse.pdfReviewer.lines';

    var sheet = document.getElementById('pr-sheet');
    if (!sheet) return;

    var radios = Array.prototype.slice.call(document.querySelectorAll('input[name="pr-layout"]'));
    var linesBox = document.getElementById('pr-lines');
    var linesWrap = document.getElementById('pr-lines-wrap');
    var printBtn = document.getElementById('pr-print-btn');

    // localStorage can throw (private mode, blocked storage): never let that break the page
    function readStore(key) {
        try { return window.localStorage.getItem(key); } catch (e) { return null; }
    }
    function writeStore(key, value) {
        try { window.localStorage.setItem(key, value); } catch (e) { /* ignore */ }
    }

    function applyLayout(layout, persist) {
        if (layout !== 'beside' && layout !== 'end') layout = 'beside';
        sheet.setAttribute('data-layout', layout);

        radios.forEach(function (r) { r.checked = (r.value === layout); });
        if (linesWrap) linesWrap.hidden = (layout === 'beside');   // writing space only matters in "end"

        if (persist) {
            writeStore(KEY_LAYOUT, layout);
            // keep the choice if the student refreshes or shares the link
            try {
                var url = new URL(window.location.href);
                url.searchParams.set('layout', layout);
                window.history.replaceState(null, '', url.toString());
            } catch (e) { /* ignore */ }
        }
    }

    function applyLines(on, persist) {
        sheet.setAttribute('data-lines', on ? '1' : '0');
        if (linesBox) linesBox.checked = on;
        if (persist) writeStore(KEY_LINES, on ? '1' : '0');
    }

    // ---- initial state -------------------------------------------------------
    var layoutFromUrl = document.body.getAttribute('data-layout-source') === 'url';
    var startLayout = sheet.getAttribute('data-layout');
    if (!layoutFromUrl) {
        var saved = readStore(KEY_LAYOUT);
        if (saved === 'beside' || saved === 'end') startLayout = saved;
    }
    applyLayout(startLayout, false);
    applyLines(readStore(KEY_LINES) !== '0', false);

    // ---- events --------------------------------------------------------------
    radios.forEach(function (r) {
        r.addEventListener('change', function () {
            if (r.checked) applyLayout(r.value, true);
        });
    });

    if (linesBox) {
        linesBox.addEventListener('change', function () {
            applyLines(linesBox.checked, true);
        });
    }

    if (printBtn) {
        printBtn.addEventListener('click', function () { window.print(); });
    }

    // Optional: ?autoprint=1 opens the print dialog once the page is ready
    if (document.body.getAttribute('data-autoprint') === '1') {
        var ready = (document.fonts && document.fonts.ready) ? document.fonts.ready : Promise.resolve();
        ready.then(function () { window.setTimeout(function () { window.print(); }, 250); });
    }
})();

    header("Location: decks.php");
    exit;

    $part = trim($part);

        $ids[(int) $part] = true;

    async function sendCardAction(payload) {
        try {
            const res = await fetch('../api/flashcard-actions.php', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ ...payload, csrf_token: CSRF_TOKEN, deck_id: DECK_ID })
            });
            let data = {};
            try { data = await res.json(); } catch (_) { }

            if (!res.ok || !data.ok) {
                throw new Error(data.error || `Database communication error (HTTP ${res.status}).`);
            }
            return data;
        } catch (error) {
            console.error("Database connection failure:", error);
            throw error;
        }
    }

    function formatDateTimeForInput(dateString) {

        return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
    }

    function pageStep() {
        const holder = gridContainer.querySelector('.tabulator-tableholder');
        const first = tabulatorTable.getRows()[0];
        const rowH = first ? first.getElement().offsetHeight : 0;
        if (!holder || !rowH) return 10;
        let perRow = 1;
        if (layoutMode === 'tiles') {
            const table = gridContainer.querySelector('.tabulator-table');
            if (table) perRow = Math.max(1, getComputedStyle(table).gridTemplateColumns.split(' ').length);
        }
        return Math.max(1, (Math.floor(holder.clientHeight / rowH) - 1) * perRow);   // one screenful, minus one row of overlap
    }



    // Single place that makes a row "the" selected row: one highlight, editor in sync
    function activateRow(row, { expand = true, position = 'nearest' } = {}) {
        tabulatorTable.deselectRow(tabulatorTable.getSelectedRows());
        row.select();
        if (isDesktopView()) populateSidebar(row.getData(), row, { expand });
        Promise.resolve(row.scrollTo(position, false)).catch(() => { });
    }

    async function navigateSidebarRow(direction, { expand = true, step = 1 } = {}) {
        if (!tabulatorTable) return;

        let rows = tabulatorTable.getRows('all');
        if (!rows.length) return;

        const idx = rows.findIndex(
            r => Number(r.getData().id) === selectedSidebarCardId
        );

        let targetIdx;
        if (idx === -1) {
            targetIdx = direction === 'next' ? 0 : rows.length - 1;
        } else {
            targetIdx = direction === 'next' ? idx + step : idx - step;
        }

        targetIdx = Math.max(0, Math.min(targetIdx, rows.length - 1));
        if (targetIdx === idx) return;

        const targetPage =
            Math.floor(targetIdx / Math.max(1, tabulatorTable.getPageSize())) + 1;

        if (targetPage !== tabulatorTable.getPage()) {
            await tabulatorTable.setPage(targetPage);
        }

        rows = tabulatorTable.getRows('all');
        activateRow(rows[targetIdx], { expand });
    }

    // ↑ / ↓ browse cards on desktop (ignored while typing in a field or when a dialog is open)
    const NAV_KEYS = {
        ArrowDown: ['next', false], ArrowUp: ['prev', false],
        PageDown: ['next', true], PageUp: ['prev', true],
    };
    document.addEventListener('keydown', (e) => {
        // 1. Ctrl + Return to save changes
        if ((e.ctrlKey || e.metaKey) && e.key === 'Enter') {
            if (viewArchived) return;

            // Mobile / tablet modal: create or save from the keyboard
            if (cardModal && cardModal.open) {
                e.preventDefault();
                if (modalSaveBtn && !modalSaveBtn.classList.contains('hidden') && !modalSaveBtn.disabled) {
                    modalSaveBtn.click();
                }
                return;
            }

            // Desktop sidebar (editing is only unlocked for checked cards; creating is always allowed)
            if (isDesktopView() && !isSidebarCollapsed && (checkedIds.size > 0 || !selectedSidebarCardId)) {
                e.preventDefault();
                // If the save button isn't hidden (meaning it's not locked) or mid-save, trigger the click
                if (sidebarSaveBtn && !sidebarSaveBtn.classList.contains('hidden') && !sidebarSaveBtn.disabled) {
                    sidebarSaveBtn.click();
                }
            }
            return;
        }

        // Guard against intercepting keystrokes while typing in an input or a modal
        if (document.querySelector('dialog[open]')) return;
        if (e.target instanceof Element && e.target.closest('input, textarea, select, [contenteditable="true"]')) return;

        // 2. Return key to toggle sidebar collapse
        if (e.key === 'Enter' && !e.ctrlKey && !e.metaKey && !e.shiftKey && !e.altKey) {
            if (!isDesktopView() || viewArchived) return;
            e.preventDefault();
            setSidebarCollapsed(!isSidebarCollapsed);
            return;
        }

        // 3. Arrow Up/Down Navigation
        const nav = NAV_KEYS[e.key];
        if (!nav) return;
        if (!isDesktopView() || viewArchived || e.altKey || e.ctrlKey || e.metaKey || e.shiftKey) return;
        e.preventDefault();
        navigateSidebarRow(nav[0], { expand: false, step: nav[1] ? pageStep() : 1 });
    });

    function updatePaginationUI() {
        if (!gridFooter || !tabulatorTable) return;

        const total = tabulatorTable.getDataCount('active');
        const currentPage = tabulatorTable.getPage() || 1;
        const size = tabulatorTable.getPageSize() || pageSize;
        const pages = Math.max(1, tabulatorTable.getPageMax() || 1);

        if (currentPage > pages) {
            tabulatorTable.setPage(pages).then(updatePaginationUI);
            return;
        }

        const start = total ? ((currentPage - 1) * size) + 1 : 0;
        const end = Math.min(currentPage * size, total);

        gridFooter.hidden = total === 0;

        if (rangeLabel) {
            rangeLabel.textContent = total
                ? `Showing ${start}–${end} of ${total} ${total === 1 ? 'card' : 'cards'} · Page ${currentPage} of ${pages}`
                : 'No cards to display';
        }

        if (pageSizeSelect && Number(pageSizeSelect.value) !== size) {
            pageSizeSelect.value = String(size);
        }

        if (paginationControls) {
            paginationControls.setAttribute(
                'aria-label',
                `Pagination, page ${currentPage} of ${pages}`
            );

            paginationControls.querySelectorAll('.tabulator-page').forEach((button) => {
                const isCurrent =
                    button.getAttribute('aria-current') === 'page' ||
                    button.classList.contains('active');

                if (isCurrent) button.setAttribute('aria-current', 'page');
            });
        }
    }

    if (pageSizeSelect) {
        pageSizeSelect.addEventListener('change', async () => {
            const requested = Number(pageSizeSelect.value);

            if (![10, 25, 50, 100].includes(requested) || !tabulatorTable) {
                return;
            }

            pageSize = requested;
            await tabulatorTable.setPageSize(pageSize);
            await tabulatorTable.setPage(1);
            updatePaginationUI();
        });
    }

    let layoutMode = 'table';
    let needsRenumber = false;      // only set if this Tabulator build cannot watch row positions

    function readSavedLayout() {
        try {
            const v = window.localStorage.getItem(LAYOUT_KEY);
            return v === 'tiles' || v === 'table' ? v : null;
        } catch (_) { return null; }
    }
    function saveLayout(mode) {
        try { window.localStorage.setItem(LAYOUT_KEY, mode); } catch (_) { /* storage unavailable: choice lasts for this visit only */ }
    }
    let layoutChoice = readSavedLayout();   // explicit choice; null = follow the width-based default
    function resolveLayout() {
        return layoutChoice || (mobileMQ.matches ? 'tiles' : 'table');
    }
    function syncLayoutToggle() {
        if (!layoutToggle) return;
        layoutToggle.querySelectorAll('button[data-layout]').forEach((b) => {
            b.setAttribute('aria-pressed', String(b.dataset.layout === layoutMode));
        });
    }

    // ---- Tile rendering (DOM nodes + textContent, so card text is never parsed as HTML) ----
    function mk(tag, cls, text) {
        const n = document.createElement(tag);
        if (cls) n.className = cls;
        if (text !== undefined) n.textContent = text;
        return n;
    }

    function parseDbDate(value) {
        if (!value) return null;
        const d = new Date(String(value).replace(' ', 'T'));
        return isNaN(d.getTime()) ? null : d;
    }

    function dueChipInfo(value) {
        const t = parseDbDate(value);
        const diff = t ? t.getTime() - Date.now() : 0;
        if (diff <= 0) return { text: 'Due now', due: true };
        const mins = Math.max(1, Math.round(diff / 60000));
        if (mins < 60) return { text: `Due in ${mins}m`, due: false };
        const hours = Math.round(mins / 60);
        if (hours < 24) return { text: `Due in ${hours}h`, due: false };
        return { text: `Due in ${Math.round(hours / 24)}d`, due: false };
    }

    function mkChip(text, cls, icon) {
        const c = mk('span', 'mp-chip' + (cls ? ' ' + cls : ''));
        if (icon) c.appendChild(mk('span', 'material-symbols-outlined', icon));
        c.appendChild(document.createTextNode(text));
        return c;
    }

    function buildTile(cell) {
        const d = cell.getData();
        const row = cell.getRow();
        const tile = mk('div', 'mp-tile');

        const top = mk('div', 'mp-tile-top');
        const num = mk('span', 'mp-tile-num');
        if (typeof row.watchPosition === 'function') {
            row.watchPosition((pos) => { num.textContent = '#' + pos; });   // stays correct as cards are added / removed
        } else {
            needsRenumber = true;
            num.textContent = '#' + row.getPosition(true);
        }

        // Same hooks as the table's checkbox cell: .row-check feeds the delegated change handler,
        // .mp-check-cell keeps taps on the checkbox from opening the card.
        const checkWrap = mk('label', 'mp-check-cell mp-tile-check');
        const box = mk('input', 'mp-check row-check');
        box.type = 'checkbox';
        box.dataset.id = String(d.id);
        box.checked = checkedIds.has(Number(d.id));
        box.setAttribute('aria-label', 'Select card');
        checkWrap.appendChild(box);
        top.append(num, checkWrap);

        const meta = mk('div', 'mp-tile-meta');
        if (viewArchived) {
            const left = d.days_left;
            const urgent = left !== null && left !== undefined && Number(left) <= 3;
            meta.appendChild(mkChip(Archive.retentionLabel(left), urgent ? 'is-warn' : '', 'inventory_2'));
        } else {
            const due = dueChipInfo(d.next_review_at);
            meta.appendChild(mkChip(due.text, due.due ? 'is-due' : '', 'schedule'));
        }
        if (d.interval_days !== undefined && d.interval_days !== null) {
            const n = Number(d.interval_days);
            meta.appendChild(mkChip(`Every ${n} ${n === 1 ? 'day' : 'days'}`, '', 'update'));
        }
        if (Number(d.lapses) > 0) {
            const n = Number(d.lapses);
            meta.appendChild(mkChip(`${n} ${n === 1 ? 'lapse' : 'lapses'}`, 'is-due', 'replay'));
        }

        tile.append(top, mk('div', 'mp-tile-front', d.front || ''), mk('div', 'mp-tile-back', d.back || ''), meta);
        return tile;
    }

    const tileColumns = [{
        title: "",
        field: "id",
        resizable: false,
        headerSort: false,
        formatter: (cell) => buildTile(cell)
    }];

    function columnsFor(mode, archived) {
        if (mode === 'tiles') return tileColumns;
        return allColumns.filter((c) => c.field !== (archived ? 'next_review_at' : 'days_left'));
    }

    // ---- Grid construction (rebuildable) ----
    let renumberQueued = false;
    function renumberRows() {
        if (layoutMode === 'tiles' && !needsRenumber) return;   // tiles watch their own position
        if (renumberQueued || !tabulatorTable) return;
        renumberQueued = true;
        requestAnimationFrame(() => {
            renumberQueued = false;
            if (tabulatorTable) tabulatorTable.getRows().forEach((r) => r.reformat());
        });
    }

    // The table keeps Tabulator's virtual rendering; tiles need "basic" rendering because they are a CSS grid.
    // That option is fixed at construction, so changing layout means building a new grid from the same data.
    function createGrid(mode, list, onLoaded) {
        layoutMode = mode;
        gridReady = false;
        gridContainer.classList.toggle('mp-tiles', mode === 'tiles');
        syncLayoutToggle();

        tabulatorTable = new Tabulator("#flashcards-tabulator-grid", {
            data: list,
            layout: mode === 'tiles' ? "fitColumns" : "fitDataFill",
            renderVertical: mode === 'tiles' ? "basic" : "virtual",
            height: "100%",                        // fills .mp-pane (replaces maxHeight)
            responsiveLayout: false,
            pagination: "local",
            paginationSize: pageSize,
            paginationButtonCount: 5,
            paginationElement: paginationControls,
            selectable: 1,                         // select-only: the grid has no cell editors
            selectableRollingSelection: true,
            resizableColumnFit: false,
            columnDefaults: { headerSort: false }, // keep navigation order predictable            placeholder: "No Flashcards Available",
            columns: columnsFor(mode, viewArchived),
        });

        tabulatorTable.on("tableBuilt", () => {
            gridReady = true;
            toggleStateVisibility(list.length > 0);
            updatePaginationUI();
            if (onLoaded) onLoaded();
        });

        tabulatorTable.on("pageLoaded", updatePaginationUI);
        tabulatorTable.on("dataProcessed", updatePaginationUI);

        // Desktop: row click fills the editor. Mobile/tablet: row click opens the modal.
        tabulatorTable.on("rowClick", (e, row) => {
            // Ignore clicks anywhere within the checkbox cell, not just the input
            if (e.target.closest('.mp-check-cell')) return;

            const cardData = row.getData();
            if (isDesktopView()) populateSidebar(cardData, row);
            else openModalForEdit(cardData);
        });

        tabulatorTable.on("rowAdded", () => {
            renumberRows();
            updatePaginationUI();
        });

        tabulatorTable.on("rowDeleted", () => {
            renumberRows();
            updatePaginationUI();
        });

        // A tile is drawn from the whole card, so redraw it whenever any field of the row changes
        tabulatorTable.on("rowUpdated", (row) => { if (layoutMode === 'tiles') row.reformat(); });
    }

    function selectFirstCardOnDesktop() {
        if (!isDesktopView()) return;
        const firstRow = tabulatorTable.getRows()[0];
        if (firstRow) {
            firstRow.select();
            populateSidebar(firstRow.getData(), firstRow, { expand: false });
        }
    }

    async function switchLayout(mode) {
        if (!gridContainer || !tabulatorTable || mode === layoutMode) { syncLayoutToggle(); return; }

        const list = tabulatorTable.getData();
        const selected = tabulatorTable.getSelectedRows();
        const keepId = selected.length ? Number(selected[0].getData().id) : null;

        tabulatorTable.destroy();
        selectedTabulatorRow = null;                               // pointed at a row of the destroyed grid

        // Put the highlight back on the same card. The editor is NOT repopulated, so unsaved typing survives.
        const restoreSelection = async () => {
            if (keepId === null) return;
            const idx = list.findIndex((c) => Number(c.id) === keepId);
            if (idx === -1) return;
            const targetPage =
                Math.floor(idx / Math.max(1, tabulatorTable.getPageSize())) + 1;

            if (targetPage !== tabulatorTable.getPage()) {
                await tabulatorTable.setPage(targetPage);
            }

            const row = tabulatorTable.getRows('all')[idx];
            if (!row) return;
            row.select();
            if (selectedSidebarCardId === keepId) selectedTabulatorRow = row;
            Promise.resolve(row.scrollTo('nearest', false)).catch(() => { });
            updatePositionLabel();
        };
        createGrid(mode, list, restoreSelection);
    }

    if (layoutToggle) layoutToggle.addEventListener('click', (e) => {
        const btn = e.target.closest('button[data-layout]');
        if (!btn) return;
        layoutChoice = btn.dataset.layout;      // an explicit choice always wins over the width-based default
        saveLayout(layoutChoice);
        switchLayout(layoutChoice);
    });

    // Without an explicit choice, follow the screen: tiles at phone width, table otherwise
    mobileMQ.addEventListener('change', () => {
        if (layoutChoice === null) switchLayout(resolveLayout());
    });

    if (gridContainer) createGrid(resolveLayout(), initialFlashcards, selectFirstCardOnDesktop);

    desktopMQ.addEventListener('change', (e) => {
        if (e.matches && cardModal && cardModal.open) cardModal.close();
        if (tabulatorTable) requestAnimationFrame(() => tabulatorTable.redraw(true));
    });

    // 2. Save Form Handler (Shared by Sidebar & Modal)
    async function handleCardSave(cardId, front, back, nextReviewAt, errorEl, saveBtnEl, isSidebar) {
        if (front === '' || back === '') {
            if (errorEl) {
                errorEl.textContent = 'Both Front and Back content are required.';
                errorEl.classList.remove('hidden');
            }
            return;
        }

        if (saveBtnEl) saveBtnEl.disabled = true;

        try {
            const action = cardId ? 'update' : 'add';
            const res = await sendCardAction({ action, card_id: cardId, front, back, next_review_at: nextReviewAt });

            if (action === 'add') {
                totalCards++;
                if (tabulatorTable) await tabulatorTable.addRow(res.card, true);
            } else {
                const row = tabulatorTable ? tabulatorTable.getRow(Number(res.card.id)) : null;
                if (row) await row.update(res.card);
                if (isSidebar && isDesktopView()) {
                    populateSidebar(row ? row.getData() : res.card, row || null);   // row data keeps accuracy/speed/etc.
                }
            }

            applyCounts(res.counts);
            if (gridReady) updatePaginationUI();
            if (action === 'add') {
                // Fast entry: clear the fields and put the cursor back on the front side
                if (isSidebar) resetSidebarForCreate();
                else resetModalForCreate();
            } else if (!isSidebar && cardModal) {
                cardModal.close();
            }
            Archive.toast(action === 'add' ? 'Card created successfully' : 'Card updated successfully');
        } catch (err) {
            if (errorEl) {
                errorEl.textContent = err.message;
                errorEl.classList.remove('hidden');
            }
        } finally {
            if (saveBtnEl) saveBtnEl.disabled = false;
        }
    }

    if (sidebarForm) {
        sidebarForm.addEventListener('submit', async (e) => {
            e.preventDefault();
            const cardId = sidebarCardId ? sidebarCardId.value.trim() : '';
            const front = sidebarCardFront ? sidebarCardFront.value.trim() : '';
            const back = sidebarCardBack ? sidebarCardBack.value.trim() : '';
            const nextReviewAt = sidebarCardDue ? sidebarCardDue.value.trim() : '';
            await handleCardSave(cardId, front, back, nextReviewAt, sidebarCardError, sidebarSaveBtn, true);
        });
    }

    if (modalForm) {
        modalForm.addEventListener('submit', async (e) => {
            e.preventDefault();
            const cardId = modalCardId ? modalCardId.value.trim() : '';
            const front = modalCardFront ? modalCardFront.value.trim() : '';
            const back = modalCardBack ? modalCardBack.value.trim() : '';
            const nextReviewAt = modalCardDue ? modalCardDue.value.trim() : '';
            await handleCardSave(cardId, front, back, nextReviewAt, modalError, modalSaveBtn, false);
        });
    }

    // ---------------------------------------------------------------------------
    // Archive: archive / restore / view switching
    // ---------------------------------------------------------------------------
    function removeRow(id) {
        if (!tabulatorTable) return;
        const row = tabulatorTable.getRow(Number(id));
        if (row) row.delete();
    }

    async function archiveCard(card) {
        try {
            const res = await sendCardAction({ action: 'archive', card_id: card.id });
            removeRow(card.id);
            offscreenList.unshift({ ...card, archived_at: new Date().toISOString(), days_left: Archive.RETENTION_DAYS });
            totalCards--;
            archivedCards++;
            applyCounts(res.counts);
            Archive.toast('Card archived', { actionLabel: 'Undo', onAction: () => restoreCard(card.id) });
        } catch (err) {
            Archive.toast(err.message, { type: 'error' });
        }
    }

    async function restoreCard(cardId) {
        try {
            const res = await sendCardAction({ action: 'restore', card_id: cardId });
            const id = Number(cardId);

            if (viewArchived) {
                const row = tabulatorTable.getRow(id);
                const data = row ? row.getData() : null;
                if (row) row.delete();
                if (data) offscreenList.unshift({ ...data, archived_at: null, days_left: null });
                Archive.toast('Card restored');
            } else {
                const idx = offscreenList.findIndex((c) => Number(c.id) === id);
                if (idx !== -1) {
                    const [data] = offscreenList.splice(idx, 1);
                    await tabulatorTable.addRow({ ...data, archived_at: null, days_left: null }, true);
                }
            }

            totalCards++;
            archivedCards--;
            applyCounts(res.counts);
        } catch (err) {
            Archive.toast(err.message, { type: 'error' });
        }
    }

    function setViewMode(archived) {
        viewArchived = archived;

        if (archiveBanner) archiveBanner.classList.toggle('hidden', !archived);
        if (openAddCardBtn) openAddCardBtn.classList.toggle('hidden', archived);
        if (toggleArchiveIcon) toggleArchiveIcon.textContent = archived ? 'arrow_back' : 'inventory_2';
        if (toggleArchiveLabel) toggleArchiveLabel.textContent = archived ? 'Back to cards' : 'Archive';
        if (toggleArchiveBtn) {
            const label = archived ? 'Back to cards' : 'View archive';
            toggleArchiveBtn.title = label;
            toggleArchiveBtn.setAttribute('aria-label', label);
            toggleArchiveBtn.setAttribute('aria-pressed', String(archived));
        }
        if (emptyTitle) emptyTitle.textContent = archived ? 'Archive is empty' : 'No flashcards yet';
        if (emptyText) {
            emptyText.textContent = archived
                ? `Archived cards appear here for ${Archive.RETENTION_DAYS} days before they are permanently deleted.`
                : 'Click "Create Card" to add your first card.';
        }

        if (tabulatorTable) {
            if (layoutMode === 'table') tabulatorTable.setColumns(columnsFor('table', archived));

            const current = tabulatorTable.getData();
            const next = offscreenList;
            offscreenList = current;
            selectedSidebarCardId = null;
            selectedTabulatorRow = null;

            tabulatorTable.setData(next).then(() => {
                tabulatorTable.setPage(1);
                tabulatorTable.redraw(true);
                updatePaginationUI();

                const first = tabulatorTable.getRows('all')[0];

                if (first && isDesktopView() && !viewArchived) {
                    first.select();
                    populateSidebar(first.getData(), first, { expand: false });
                }
            });
        }
        updateCardCountLabel();
    }

    // 1. Toggle Edit Mode
    function setEditMode(active) {
        editMode = active;
        if (!active) {
            checkedIds.clear();

        updateBulkUI();

        if (bulkPanel) bulkPanel.classList.toggle('is-open', active);
        if (workspace) workspace.classList.toggle('workspace-editing', active);

            tabulatorTable.getRows().forEach(r => r.reformat());
        }

        checkedIds.clear();                 // ids differ between the two views
        originalSetViewMode(archived);
        if (archived) setSidebarCollapsed(true);
        updateBulkUI();
    };

    // 2. Selection Logic
    function updateBulkUI() {
        const total = tabulatorTable ? tabulatorTable.getDataCount('active') : 0; const selected = checkedIds.size;
        const hasSelection = selected > 0;

        const selectAll = document.getElementById('bulk-select-all');
        const selectWrapper = document.getElementById('bulk-select-wrapper');
        const selectSeparator = document.getElementById('bulk-select-separator');

        // Select all / divider / Clear only appear once something is ticked
        if (selectWrapper) selectWrapper.classList.toggle('hidden', !hasSelection);
        if (selectSeparator) selectSeparator.classList.toggle('hidden', !hasSelection);
        if (selectAll) {
            selectAll.indeterminate = false;
            selectAll.checked = hasSelection && selected === total;
        }
        if (bulkClearBtn) bulkClearBtn.hidden = !hasSelection;
        if (bulkStatus) bulkStatus.textContent = `${selected} selected`;

        // Panel buttons depend on the view
        bulkActionBtns.forEach((btn) => { if (btn) { btn.hidden = viewArchived; btn.disabled = !hasSelection; } });
        archiveActionBtns.forEach((btn) => { if (btn) { btn.hidden = !viewArchived; btn.disabled = !hasSelection; } });

        // Practice: needs a selection, and at most PRACTICE_MAX cards
        // PDF: needs a selection, and at most PDF_MAX cards
        if (pdfBtn) {
            if (selected > PDF_MAX) {
                pdfBtn.disabled = true;
                pdfBtn.title = `Select up to ${PDF_MAX} cards`;
            } else {
                pdfBtn.title = 'Download a printable PDF of the selected cards';
            }
        }

        if (practiceBtn) {
            if (selected > PRACTICE_MAX) {
                practiceBtn.disabled = true;
                practiceBtn.title = `Select up to ${PRACTICE_MAX} cards`;
            } else {
                practiceBtn.title = 'Practice selected cards (nothing is saved)';
            }
        }


        const moveBtn = document.getElementById('bulk-move-btn');
        if (moveBtn) {
            if (otherDecks.length === 0) {
                moveBtn.disabled = true;
                moveBtn.title = "No other decks available";
            } else {
                moveBtn.title = "Move to deck...";
            }
        }

        // Card editor: locked in the archive, and until a checkbox is ticked
        if (selectedSidebarCardId !== null) setEditorReadOnly(viewArchived || !hasSelection);
    }

    document.addEventListener('change', (e) => {
        // Handle Bulk Panel "Select All"
        if (e.target.id === 'bulk-select-all') {
            const isChecked = e.target.checked;
            checkedIds.clear();

            if (isChecked && tabulatorTable) {
                const allIds = tabulatorTable.getData().map(r => Number(r.id));
                allIds.forEach(id => checkedIds.add(id));
            }

            updateBulkUI();
            if (tabulatorTable) tabulatorTable.getRows().forEach(r => r.reformat());
        }
        // Handle Individual Row Checks
        else if (e.target.classList.contains('row-check')) {
            const id = Number(e.target.dataset.id);
            if (e.target.checked) checkedIds.add(id);
            else checkedIds.delete(id);
            updateBulkUI();
        }
    });

        const ids = Array.from(checkedIds);
        if (!ids.length) return;
        try {
            bulkActionBtns.forEach(b => b.disabled = true);
            const res = await sendCardAction({ action: 'bulk_duplicate', card_ids: ids });

            if (tabulatorTable) {
                res.cards.reverse().forEach(card => tabulatorTable.addRow(card, true));

            totalCards += res.cards.length;
            applyCounts(res.counts);
            Archive.toast(`${res.cards.length} cards duplicated`);

            Archive.toast(err.message, { type: 'error' });

                if (row) row.delete();

        const ids = Array.from(checkedIds);
        if (!ids.length) return;

            bulkActionBtns.forEach(b => b.disabled = true);

                const row = tabulatorTable.getRow(id);

                    const firstRow = tabulatorTable.getRows()[0];
                    if (firstRow) activateRow(firstRow, { expand: false });
                    else setSidebarCollapsed(true);
                }
            });

            totalCards -= res.affected;
            archivedCards += res.affected;
            applyCounts(res.counts);
            Archive.toast(`${res.affected} cards archived`);
            setEditMode(false);
        } catch (err) {
            Archive.toast(err.message, { type: 'error' });

            updateBulkUI();

/**

    /**
     * Retrieve single deck details and verify user ownership.
     * 
     * @param mysqli $conn Active database connection instance.
     * @param int $deckId Target deck ID.
     * @param int $userId Authenticated user ID.
     * @return array|null Returns deck array if authorized and active, null otherwise.
     */
    public static function getDeckDetails(mysqli $conn, int $deckId, int $userId): ?array
    {
        $sql = "SELECT d.id, 
                       d.title, 
                       d.created_at,
                       (SELECT COUNT(*) 
                          FROM flashcards f 
                         WHERE f.deck_id = d.id AND f.deleted_at IS NULL) AS card_count,
                       (SELECT COUNT(*) 
                          FROM flashcards f 
                         WHERE f.deck_id = d.id AND f.deleted_at IS NULL AND f.next_review_at <= NOW()) AS due_count,
                       (SELECT COUNT(*)
                          FROM flashcards f
                         WHERE f.deck_id = d.id AND f.deleted_at IS NOT NULL) AS archived_count
                  FROM decks d
                 WHERE d.id = ? AND d.user_id = ? AND d.deleted_at IS NULL";

        $stmt = mysqli_prepare($conn, $sql);
        if (!$stmt) {
            return null;
        }

        mysqli_stmt_bind_param($stmt, "ii", $deckId, $userId);
        mysqli_stmt_execute($stmt);
        $result = mysqli_stmt_get_result($stmt);
        $deck = mysqli_fetch_assoc($result);
        mysqli_stmt_close($stmt);

        return $deck ?: null;
    }

    /**
     * Live counters for a deck: active cards, cards due right now, and archived cards.
     * Returned by every card action so the deck page never shows a stale "Due Now".
     * Callers must already have verified that the user owns the deck.
     *
     * @param mysqli $conn Active database connection instance.
     * @param int $deckId Target deck ID.
     * @return array{card_count:int, due_count:int, archived_count:int}
     */
    public static function getDeckCounts(mysqli $conn, int $deckId): array
    {
        $counts = ['card_count' => 0, 'due_count' => 0, 'archived_count' => 0];

        $sql = "SELECT COUNT(CASE WHEN deleted_at IS NULL THEN 1 END) AS card_count,
                       COUNT(CASE WHEN deleted_at IS NULL AND next_review_at <= NOW() THEN 1 END) AS due_count,
                       COUNT(CASE WHEN deleted_at IS NOT NULL THEN 1 END) AS archived_count
                  FROM flashcards
                 WHERE deck_id = ?";

        $stmt = mysqli_prepare($conn, $sql);
        if (!$stmt) {
            return $counts;
        }

        mysqli_stmt_bind_param($stmt, "i", $deckId);
        mysqli_stmt_execute($stmt);
        $row = mysqli_fetch_assoc(mysqli_stmt_get_result($stmt));
        mysqli_stmt_close($stmt);

        if ($row) {
            $counts = [
                'card_count'     => (int) $row['card_count'],
                'due_count'      => (int) $row['due_count'],
                'archived_count' => (int) $row['archived_count'],
            ];
        }

        return $counts;
    }

    /**
     * Fetch active flashcards belonging to a specific deck with optional search filtering.
     * 
     * @param mysqli $conn Active database connection instance.
     * @param int $deckId Target deck ID.
     * @param string $search Search query to match against front or back content.
     * @return array Array of flashcard records.
     */
        public static function getDeckFlashcards(mysqli $conn, int $deckId, string $search = '', bool $archived = false): array
    {
        $search      = trim($search);
        $deletedCond = $archived ? 'IS NOT NULL' : 'IS NULL';
        $orderBy     = $archived ? 'f.deleted_at DESC, f.id DESC' : 'f.id DESC';
        $retention   = ArchiveService::RETENTION_DAYS;

        $sql = "SELECT f.id, f.deck_id, f.front, f.back, f.interval_days, f.next_review_at,
                    f.review_count, f.lapses, f.created_at, f.updated_at, f.deleted_at,
                    GREATEST(0, ? - TIMESTAMPDIFF(DAY, f.deleted_at, NOW())) AS days_left,
                    (SELECT MAX(l.reviewed_at)
                        FROM review_logs l WHERE l.card_id = f.id) AS last_reviewed_at,
                    (SELECT ROUND(100 * SUM(l.rating = 'remembered') / COUNT(*))
                        FROM review_logs l WHERE l.card_id = f.id) AS accuracy,
                    (SELECT ROUND(AVG(l.response_time_ms))
                        FROM review_logs l WHERE l.card_id = f.id) AS avg_response_ms
                FROM flashcards f
                WHERE f.deck_id = ? AND f.deleted_at {$deletedCond}";

        if ($search !== '') {
            $sql .= " AND (f.front LIKE ? OR f.back LIKE ?)";
        }
        $sql .= " ORDER BY {$orderBy}";

        $stmt = mysqli_prepare($conn, $sql);
        if (!$stmt) {
            return [];
        }

        if ($search !== '') {
            $like = '%' . $search . '%';
            mysqli_stmt_bind_param($stmt, "iiss", $retention, $deckId, $like, $like);
        } else {
            mysqli_stmt_bind_param($stmt, "ii", $retention, $deckId);
        }

        mysqli_stmt_execute($stmt);
        $result = mysqli_stmt_get_result($stmt);

        $cards = [];
        while ($row = mysqli_fetch_assoc($result)) {
            $cards[] = [
                'id'               => (int) $row['id'],
                'deck_id'          => (int) $row['deck_id'],
                'front'            => $row['front'],
                'back'             => $row['back'],
                'interval_days'    => (int) $row['interval_days'],
                'next_review_at'   => $row['next_review_at'],
                'review_count'     => (int) $row['review_count'],
                'lapses'           => (int) $row['lapses'],
                'created_at'       => $row['created_at'],
                'updated_at'       => $row['updated_at'],
                'archived_at'      => $row['deleted_at'],
                'days_left'        => $row['days_left'] !== null ? (int) $row['days_left'] : null,
                'last_reviewed_at' => $row['last_reviewed_at'],
                'accuracy'         => $row['accuracy'] !== null ? (int) $row['accuracy'] : null,
                'avg_response_ms'  => $row['avg_response_ms'] !== null ? (int) $row['avg_response_ms'] : null,
            ];
        }
        mysqli_stmt_close($stmt);

        return $cards;
    }

        /**
     * Cards for the printable PDF reviewer.
     * Only ACTIVE cards of a deck the user owns, in the order of $cardIds
     * (grid order from the deck page). Read-only.
     *
     * @param int[] $cardIds
     * @return array<int, array{id:int, front:string, back:string}>
     */
    public static function getCardsForPrint(mysqli $conn, int $deckId, int $userId, array $cardIds): array
    {
        $cardIds = array_values(array_unique(array_filter(
            array_map('intval', $cardIds),
            fn($id) => $id > 0
        )));
        if (!$cardIds) {
            return [];
        }

        $placeholders = implode(',', array_fill(0, count($cardIds), '?'));
        $orderList    = implode(',', $cardIds);   // integers only (cast above), safe to inline

        $sql = "SELECT f.id, f.front, f.back
                  FROM flashcards f
                  JOIN decks d ON d.id = f.deck_id
                 WHERE f.deck_id = ?
                   AND d.user_id = ?
                   AND d.deleted_at IS NULL
                   AND f.deleted_at IS NULL
                   AND f.id IN ($placeholders)
                 ORDER BY FIELD(f.id, $orderList)";

        $stmt = mysqli_prepare($conn, $sql);
        if (!$stmt) {
            return [];
        }

        $types  = 'ii' . str_repeat('i', count($cardIds));
        $params = array_merge([$deckId, $userId], $cardIds);
        mysqli_stmt_bind_param($stmt, $types, ...$params);
        mysqli_stmt_execute($stmt);
        $result = mysqli_stmt_get_result($stmt);

        $cards = [];
        while ($row = mysqli_fetch_assoc($result)) {
            $cards[] = [
                'id'    => (int) $row['id'],
                'front' => (string) $row['front'],
                'back'  => (string) $row['back'],

    /**
     * Bulk Duplicate: Copies front/back of specified cards into the same deck as new cards.
     */
    public static function bulkDuplicate(mysqli $conn, int $deckId, array $cardIds, int $userId): array
    {
        if (empty($cardIds)) return [];

        $placeholders = implode(',', array_fill(0, count($cardIds), '?'));
        $types = str_repeat('i', count($cardIds)) . 'ii';
        $params = array_merge($cardIds, [$deckId, $userId]);

        // Fetch original content first to ensure we only copy owned, non-deleted cards
        $fetchSql = "SELECT f.front, f.back FROM flashcards f 
                     JOIN decks d ON f.deck_id = d.id 
                     WHERE f.id IN ($placeholders) AND f.deck_id = ? AND d.user_id = ? AND f.deleted_at IS NULL";

        $stmt = mysqli_prepare($conn, $fetchSql);
        $bindParams = [$types];
        foreach ($params as $k => $v) {
            $bindParams[] = &$params[$k];
        }
        call_user_func_array([$stmt, 'bind_param'], $bindParams);
        mysqli_stmt_execute($stmt);
        $result = mysqli_stmt_get_result($stmt);

        $cardsToInsert = [];
        while ($row = mysqli_fetch_assoc($result)) {
            $cardsToInsert[] = $row;
        }
        mysqli_stmt_close($stmt);

        if (empty($cardsToInsert)) return [];

        mysqli_begin_transaction($conn);
        $newIds = [];
        try {
            $insertStmt = mysqli_prepare($conn, "INSERT INTO flashcards (deck_id, front, back) VALUES (?, ?, ?)");
            foreach ($cardsToInsert as $card) {
                mysqli_stmt_bind_param($insertStmt, "iss", $deckId, $card['front'], $card['back']);
                mysqli_stmt_execute($insertStmt);
                $newIds[] = mysqli_insert_id($conn);
            }
            mysqli_stmt_close($insertStmt);
            mysqli_commit($conn);
        } catch (Exception $e) {
            mysqli_rollback($conn);
            throw $e;
        }

        // Return full card objects for the UI
        $newPlaceholders = implode(',', array_fill(0, count($newIds), '?'));
        $newTypes = str_repeat('i', count($newIds));
        $newStmt = mysqli_prepare($conn, "SELECT id, deck_id, front, back, interval_days, next_review_at, review_count, lapses, created_at, updated_at, deleted_at AS archived_at FROM flashcards WHERE id IN ($newPlaceholders) ORDER BY id DESC");
        mysqli_stmt_bind_param($newStmt, $newTypes, ...$newIds);
        mysqli_stmt_execute($newStmt);
        $newResult = mysqli_stmt_get_result($newStmt);

        $insertedCards = [];
        while ($row = mysqli_fetch_assoc($newResult)) {
            $row['accuracy'] = null;
            $row['avg_response_ms'] = null;
            $row['last_reviewed_at'] = null;
            $row['days_left'] = null;
            $insertedCards[] = $row;
        }
        mysqli_stmt_close($newStmt);
        return $insertedCards;
    }

    /**
     * Bulk Move: Moves specified cards to another owned deck.
     */
    public static function bulkMove(mysqli $conn, int $sourceDeckId, int $targetDeckId, array $cardIds, int $userId): int
    {
        if (empty($cardIds) || $sourceDeckId === $targetDeckId) return 0;

        // Verify target deck ownership
        $checkStmt = mysqli_prepare($conn, "SELECT id FROM decks WHERE id = ? AND user_id = ? AND deleted_at IS NULL");
        mysqli_stmt_bind_param($checkStmt, "ii", $targetDeckId, $userId);
        mysqli_stmt_execute($checkStmt);
        mysqli_stmt_store_result($checkStmt);
        $ownsTarget = mysqli_stmt_num_rows($checkStmt) === 1;
        mysqli_stmt_close($checkStmt);

        $placeholders = implode(',', array_fill(0, count($cardIds), '?'));
        $types = 'i' . str_repeat('i', count($cardIds)) . 'ii';
        $params = array_merge([$targetDeckId], $cardIds, [$sourceDeckId, $userId]);

        $sql = "UPDATE flashcards f 
                JOIN decks d ON f.deck_id = d.id 
                SET f.deck_id = ? 
                WHERE f.id IN ($placeholders) AND f.deck_id = ? AND d.user_id = ? AND f.deleted_at IS NULL";

        $stmt = mysqli_prepare($conn, $sql);
        $bindParams = [$types];
        foreach ($params as $k => $v) {
            $bindParams[] = &$params[$k];
        }
        call_user_func_array([$stmt, 'bind_param'], $bindParams);
        mysqli_stmt_execute($stmt);
        $affected = mysqli_stmt_affected_rows($stmt);
        mysqli_stmt_close($stmt);
        return $affected;
    }

    /**
     * Bulk Reset: Resets progress metrics for specified cards.
     */
    public static function bulkReset(mysqli $conn, int $deckId, array $cardIds, int $userId): int
    {
        if (empty($cardIds)) return 0;

        $placeholders = implode(',', array_fill(0, count($cardIds), '?'));
        $types = 'i' . str_repeat('i', count($cardIds)) . 'ii';
        $params = array_merge([$userId], $cardIds, [$deckId, $userId]);

        $sql = "UPDATE flashcards f 
                JOIN decks d ON f.deck_id = d.id 
                SET f.interval_days = COALESCE((SELECT initial_interval_days FROM user_settings WHERE user_id = ?), 1),
                    f.next_review_at = NOW(),
                    f.review_count = 0,
                    f.lapses = 0
                WHERE f.id IN ($placeholders) AND f.deck_id = ? AND d.user_id = ? AND f.deleted_at IS NULL";

        $bindParams = [$types];
        foreach ($params as $k => $v) {
            $bindParams[] = &$params[$k];
        }
        call_user_func_array([$stmt, 'bind_param'], $bindParams);
        mysqli_stmt_execute($stmt);
        $affected = mysqli_stmt_affected_rows($stmt);
        mysqli_stmt_close($stmt);

        return $affected;
    }

<?php
/**
 * ReviewHubService
 *
 * Read-only data + formatting helpers for the Review Hub page (review.php, no deck_id).
 * All "seconds" values are computed in SQL (TIMESTAMPDIFF) so PHP/MySQL timezone
 * differences cannot skew them.
 *
 * Place next to FlashcardService.php (the api/ folder).
 */
class ReviewHubService
{
    /** Cards scheduled within this many days count as "upcoming". */
    public const UPCOMING_DAYS = 2;

    /**
     * Schema-derived card buckets used by the Due Now table.
     *
     * The schema has no explicit learning/young/mastered state column, so these
     * buckets are inferred from review_count and interval_days:
     * - new: never reviewed
     * - learning: reviewed and interval_days <= 1
     * - young: interval_days 2..20
     * - mastered: interval_days >= 21
     *
     * Adjust these thresholds if the application defines different SRS rules.
     */
    public const YOUNG_MAX_INTERVAL_DAYS = 20;
    public const MASTERED_MIN_INTERVAL_DAYS = 21;

    // -----------------------------------------------------------------------
    // Queries
    // -----------------------------------------------------------------------

    /**
     * Every non-archived deck that has cards due now or coming up soon.
     *
     * @return array<int, array{
     *   id:int, title:string, totalCards:int, dueCount:int, upcomingCount:int,
     *   newDueCount:int, learningDueCount:int, youngDueCount:int, masteredDueCount:int,
     *   overdueSeconds:?int, nextDueSeconds:?int
     * }>
     */
    public static function getHubDecks(mysqli $conn, int $userId): array
    {
        $days = (int) self::UPCOMING_DAYS;
        $youngMax = (int) self::YOUNG_MAX_INTERVAL_DAYS;
        $masteredMin = (int) self::MASTERED_MIN_INTERVAL_DAYS;

        $sql = "SELECT d.id, d.title,
                       COUNT(f.id) AS total_cards,
                       COALESCE(SUM(f.next_review_at <= NOW()), 0) AS due_count,
                       COALESCE(SUM(f.next_review_at <= NOW() AND f.review_count = 0), 0) AS new_due_count,
                       COALESCE(SUM(f.next_review_at <= NOW()
                                    AND f.review_count > 0 AND f.interval_days <= 1), 0) AS learning_due_count,
                       COALESCE(SUM(f.next_review_at <= NOW()
                                    AND f.review_count > 0 AND f.interval_days BETWEEN 2 AND $youngMax), 0) AS young_due_count,
                       COALESCE(SUM(f.next_review_at <= NOW()
                                    AND f.review_count > 0 AND f.interval_days >= $masteredMin), 0) AS mastered_due_count,
                       COALESCE(SUM(f.next_review_at > NOW()
                                    AND f.next_review_at <= DATE_ADD(NOW(), INTERVAL $days DAY)), 0) AS upcoming_count,
                       TIMESTAMPDIFF(SECOND,
                           MIN(CASE WHEN f.next_review_at <= NOW() THEN f.next_review_at END),
                           NOW()) AS overdue_seconds,
                       TIMESTAMPDIFF(SECOND, NOW(),
                           MIN(CASE WHEN f.next_review_at > NOW()
                                     AND f.next_review_at <= DATE_ADD(NOW(), INTERVAL $days DAY)
                                    THEN f.next_review_at END)) AS next_due_seconds
                FROM decks d
                LEFT JOIN flashcards f ON f.deck_id = d.id AND f.deleted_at IS NULL
                WHERE d.user_id = ? AND d.deleted_at IS NULL
                GROUP BY d.id, d.title
                HAVING due_count > 0 OR upcoming_count > 0";

        $decks = [];
        if ($stmt = mysqli_prepare($conn, $sql)) {
            mysqli_stmt_bind_param($stmt, "i", $userId);
            mysqli_stmt_execute($stmt);
            $result = mysqli_stmt_get_result($stmt);
            while ($row = mysqli_fetch_assoc($result)) {
                $decks[] = [
                    'id'             => (int) $row['id'],
                    'title'          => (string) $row['title'],
                    'totalCards'     => (int) $row['total_cards'],
                    'dueCount'       => (int) $row['due_count'],
                    'newDueCount'    => (int) $row['new_due_count'],
                    'learningDueCount' => (int) $row['learning_due_count'],
                    'youngDueCount'  => (int) $row['young_due_count'],
                    'masteredDueCount' => (int) $row['mastered_due_count'],
                    'upcomingCount'  => (int) $row['upcoming_count'],
                    'overdueSeconds' => $row['overdue_seconds'] === null ? null : (int) $row['overdue_seconds'],
                    'nextDueSeconds' => $row['next_due_seconds'] === null ? null : (int) $row['next_due_seconds'],
                ];
            }
            mysqli_stmt_close($stmt);
        }
        return $decks;
    }

    /**
     * Decks that have at least one active card, for the Random Practice picker.
     *
     * @return array<int, array{id:int, title:string, activeCount:int}>
     */
    public static function getPracticeDecks(mysqli $conn, int $userId): array
    {
        $sql = "SELECT d.id, d.title, COUNT(f.id) AS active_count
                FROM decks d
                JOIN flashcards f ON f.deck_id = d.id AND f.deleted_at IS NULL
                WHERE d.user_id = ? AND d.deleted_at IS NULL
                GROUP BY d.id, d.title
                HAVING active_count > 0
                ORDER BY d.title ASC";

        $decks = [];
        if ($stmt = mysqli_prepare($conn, $sql)) {
            mysqli_stmt_bind_param($stmt, "i", $userId);
            mysqli_stmt_execute($stmt);
            $result = mysqli_stmt_get_result($stmt);
            while ($row = mysqli_fetch_assoc($result)) {
                $decks[] = [
                    'id'          => (int) $row['id'],
                    'title'       => (string) $row['title'],
                    'activeCount' => (int) $row['active_count'],
                ];
            }
            mysqli_stmt_close($stmt);
        }
        return $decks;
    }

    /**
     * Seconds until the user's next card becomes due (no 48h limit), or null
     * when there are no scheduled cards at all.
     */
    public static function getNextDueSeconds(mysqli $conn, int $userId): ?int
    {
        $sql = "SELECT TIMESTAMPDIFF(SECOND, NOW(), MIN(f.next_review_at))
                FROM flashcards f
                JOIN decks d ON d.id = f.deck_id
                WHERE d.user_id = ? AND d.deleted_at IS NULL
                  AND f.deleted_at IS NULL AND f.next_review_at > NOW()";

        $secs = null;
        if ($stmt = mysqli_prepare($conn, $sql)) {
            mysqli_stmt_bind_param($stmt, "i", $userId);
            mysqli_stmt_execute($stmt);
            mysqli_stmt_bind_result($stmt, $secs);
            mysqli_stmt_fetch($stmt);
            mysqli_stmt_close($stmt);
        }
        return $secs === null ? null : (int) $secs;
    }

    /**
     * Latest completed sessions: scheduled reviews plus Random Practice
     * (stored separately in random_practice_sessions).
     *
     * @return array<int, array{
     *   sessionType:string, sessionLabel:string, sessionID:int, deckId:int,
     *   deckTitle:string, cardsReviewed:int, rememberedCount:int,
     *   accuracy:int, durationSeconds:?int, formattedTime:string
     * }>
     */
    public static function getRecentSessions(mysqli $conn, int $userId, int $limit = 8): array
    {
        $limit = max(1, min($limit, 50));

        $sql = "SELECT * FROM (
                    SELECT 'srs' AS session_type, 'Review' AS session_label,
                           rs.id AS session_id, rs.deck_id, d.title AS deck_title,
                           rs.total_reviewed AS cards, rs.remembered_count,
                           rs.accuracy_percentage AS accuracy,
                           TIMESTAMPDIFF(SECOND, rs.started_at, rs.ended_at) AS duration_seconds,
                           rs.ended_at,
                           TIMESTAMPDIFF(SECOND, rs.ended_at, NOW()) AS ended_ago_seconds
                    FROM review_sessions rs
                    JOIN decks d ON d.id = rs.deck_id
                    WHERE rs.user_id = ? AND rs.ended_at IS NOT NULL
                    UNION ALL
                    SELECT 'random', rp.label,
                           rp.id, rp.deck_id, d.title,
                           rp.total_rated, rp.remembered_count,
                           rp.accuracy_percentage,
                           rp.duration_seconds,
                           rp.ended_at,
                           TIMESTAMPDIFF(SECOND, rp.ended_at, NOW())
                    FROM random_practice_sessions rp
                    JOIN decks d ON d.id = rp.deck_id
                    WHERE rp.user_id = ?
                ) t
                ORDER BY ended_at DESC, session_id DESC
                LIMIT ?";

        $sessions = [];
        if ($stmt = mysqli_prepare($conn, $sql)) {
            mysqli_stmt_bind_param($stmt, "iii", $userId, $userId, $limit);
            mysqli_stmt_execute($stmt);
            $result = mysqli_stmt_get_result($stmt);
            while ($row = mysqli_fetch_assoc($result)) {
                $sessions[] = [
                    'sessionType'     => (string) $row['session_type'],
                    'sessionLabel'    => (string) $row['session_label'],
                    'sessionID'       => (int) $row['session_id'],
                    'deckId'          => (int) $row['deck_id'],
                    'deckTitle'       => (string) $row['deck_title'],
                    'cardsReviewed'   => (int) $row['cards'],
                    'rememberedCount' => (int) $row['remembered_count'],
                    'accuracy'        => (int) $row['accuracy'],
                    'durationSeconds' => $row['duration_seconds'] === null ? null : (int) $row['duration_seconds'],
                    'formattedTime'   => self::formatAgo((int) $row['ended_ago_seconds']),
                ];
            }
            mysqli_stmt_close($stmt);
        }
        return $sessions;
    }

    /**
     * Cards reviewed today and the current study-day streak.
     * Same rules as the dashboard: a day counts when a completed session
     * reviewed at least one card; the streak is alive if the latest study day
     * is today or yesterday.
     *
     * @return array{reviewedToday:int, currentStreak:int}
     */
    public static function getStreakInfo(mysqli $conn, int $userId): array
    {
        $sql = "SELECT DATE(started_at) AS study_date, SUM(total_reviewed) AS reviewed
                FROM review_sessions
                WHERE user_id = ? AND ended_at IS NOT NULL
                GROUP BY DATE(started_at)
                HAVING SUM(total_reviewed) > 0
                ORDER BY study_date DESC";

        $days = [];
        if ($stmt = mysqli_prepare($conn, $sql)) {
            mysqli_stmt_bind_param($stmt, "i", $userId);
            mysqli_stmt_execute($stmt);
            $result = mysqli_stmt_get_result($stmt);
            while ($row = mysqli_fetch_assoc($result)) {
                $days[$row['study_date']] = (int) $row['reviewed'];
            }
            mysqli_stmt_close($stmt);
        }

        $today     = date('Y-m-d');
        $yesterday = date('Y-m-d', strtotime('-1 day'));

        $streak = 0;
        if (isset($days[$today]) || isset($days[$yesterday])) {
            $cursor = new DateTime(isset($days[$today]) ? $today : $yesterday);
            while (isset($days[$cursor->format('Y-m-d')])) {
                $streak++;
                $cursor->modify('-1 day');
            }
        }

        return [
            'reviewedToday' => $days[$today] ?? 0,
            'currentStreak' => $streak,
        ];
    }

    // -----------------------------------------------------------------------
    // Table helpers (pure PHP, no DB)
    // -----------------------------------------------------------------------

    /** Decks with cards due, most due first (oldest overdue breaks ties). */
    public static function dueDecks(array $hubDecks): array
    {
        $due = array_values(array_filter($hubDecks, fn($d) => $d['dueCount'] > 0));
        usort($due, function ($a, $b) {
            return [$b['dueCount'], $b['overdueSeconds'] ?? 0]
               <=> [$a['dueCount'], $a['overdueSeconds'] ?? 0];
        });
        return $due;
    }

    /** Decks with cards coming up soon, soonest first. */
    public static function upcomingDecks(array $hubDecks): array
    {
        $up = array_values(array_filter($hubDecks, fn($d) => $d['upcomingCount'] > 0));
        usort($up, fn($a, $b) => ($a['nextDueSeconds'] ?? PHP_INT_MAX) <=> ($b['nextDueSeconds'] ?? PHP_INT_MAX));
        return $up;
    }

    // -----------------------------------------------------------------------
    // Formatting
    // -----------------------------------------------------------------------

    /** "5 min", "3 h", "2 d" (never shows 0). */
    public static function formatSpan(int $seconds): string
    {
        $seconds = max(0, $seconds);
        if ($seconds < 3600)  return max(1, (int) floor($seconds / 60)) . ' min';
        if ($seconds < 86400) return (int) floor($seconds / 3600) . ' h';
        return (int) floor($seconds / 86400) . ' d';
    }

    /** "just now", "5 mins ago", "1 hour ago", "3 days ago". */
    public static function formatAgo(int $seconds): string
    {
        $seconds = max(0, $seconds);
        if ($seconds < 60) return 'just now';

        if ($seconds < 3600) {
            $n = (int) floor($seconds / 60);
            return $n . ($n === 1 ? ' min ago' : ' mins ago');
        }
        if ($seconds < 86400) {
            $n = (int) floor($seconds / 3600);
            return $n . ($n === 1 ? ' hour ago' : ' hours ago');
        }
        $n = (int) floor($seconds / 86400);
        return $n . ($n === 1 ? ' day ago' : ' days ago');
    }

    /** "45s", "4m 12s", "1h 5m", or an em dash when unknown. */
    public static function formatDuration(?int $seconds): string
    {
        if ($seconds === null || $seconds <= 0) return '—';
        if ($seconds < 60) return $seconds . 's';
        if ($seconds < 3600) {
            $m = (int) floor($seconds / 60);
            $s = $seconds % 60;
            return $s > 0 ? "{$m}m {$s}s" : "{$m}m";
        }
        $h = (int) floor($seconds / 3600);
        $m = (int) floor(($seconds % 3600) / 60);
        return $m > 0 ? "{$h}h {$m}m" : "{$h}h";
    }

    /** Tailwind classes for an accuracy badge (matches the deck grid thresholds). */
    public static function accuracyBadgeClass(int $accuracy): string
    {
        if ($accuracy >= 80) return 'text-emerald-700 bg-emerald-50 border-emerald-200/80';
        if ($accuracy >= 50) return 'text-amber-700 bg-amber-50 border-amber-200/80';
        return 'text-rose-700 bg-rose-50 border-rose-200/80';
    }

    /** Shorthand for htmlspecialchars with the project's flags. */
    public static function h($value): string
    {
        return htmlspecialchars((string) $value, ENT_QUOTES, 'UTF-8');
    }
}
