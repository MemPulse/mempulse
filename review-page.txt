    async function showStats(rem, forg, accuracy) {
        const token = finishToken;
        const headline = document.getElementById('finished-headline');
        const bar = document.getElementById('summary-accuracy-bar');
        const tiles = document.querySelectorAll('#session-summary-stats [data-stat-tile]');

            return;

        // Empty frames fade in first; each number stays hidden until its stamp lands
        [sumAccuracy, sumRem, sumFor].forEach(el => { el.style.opacity = '0'; });
        bar.style.width = '0%';
        tiles.forEach(tile => tile.animate(
            [{ opacity: 0, transform: 'translateY(12px)' }, { opacity: 1, transform: 'translateY(0)' }],
            { duration: 350, easing: 'ease-out', fill: 'backwards' }));

        await sleep(500);
        if (token !== finishToken) return;
        await stampStat(sumRem, tileRem, 'border-emerald-400', { scale: 3, duration: 520, shakePx: 4 });
        if (token !== finishToken) return;
        await sleep(300);
        if (token !== finishToken) return;
        await stampStat(sumFor, tileFor, 'border-rose-400', { scale: 3, duration: 520, shakePx: 4 });
        if (token !== finishToken) return;
        await sleep(650);   // beat of suspense before the hero number
        if (token !== finishToken) return;
        await stampStat(sumAccuracy, tileAcc, 'border-primary', { scale: 4.5, duration: 720, shakePx: 10, hero: true });
        if (token !== finishToken) return;

        // Climax: bar fills, headline appears, confetti explodes
        bar.animate([{ width: '0%' }, { width: accuracy + '%' }],
            { duration: 900, easing: 'cubic-bezier(.2,.8,.2,1)' });
        bar.style.width = accuracy + '%';

        headline.textContent = headlineFor(accuracy);
        headline.animate(
            [{ opacity: 0, transform: 'translateY(10px)' }, { opacity: 1, transform: 'translateY(0)' }],
            { duration: 350, easing: 'ease-out' });

        confettiBurst({ x: 0.5, y: 0.3, count: accuracy >= 80 ? 170 : 120, power: 17 });
        const n = accuracy >= 80 ? 90 : 60;
        confettiBurst({ x: 0, y: 0.75, count: n, angle: -Math.PI / 3, spread: Math.PI / 3, power: 22 });
        confettiBurst({ x: 1, y: 0.75, count: n, angle: -2 * Math.PI / 3, spread: Math.PI / 3, power: 22 });
    }

    // A number slams in from huge and blurry, squashes on impact, then settles.
    // Resolves at the moment of impact so the caller can chain the next beat.
    function stampStat(el, tile, ringClass, { scale = 3, duration = 520, shakePx = 5, hero = false } = {}) {
        return new Promise(resolve => {
            el.style.opacity = '1';
            el.animate([
                { opacity: 0, transform: `scale(${scale}) rotate(-14deg)`, filter: 'blur(6px)', offset: 0, easing: 'cubic-bezier(.5,0,1,.6)' },
                { opacity: 1, transform: 'scale(0.9) rotate(2deg)', filter: 'blur(0px)', offset: 0.55 },
                { transform: 'scale(1.06) rotate(-1deg)', offset: 0.75 },
                { transform: 'scale(1) rotate(0deg)', offset: 1 }
            ], { duration, easing: 'linear' });

            setTimeout(() => {
                impactRing(tile, ringClass);
                if (hero) setTimeout(() => impactRing(tile, ringClass), 130);
                tile.animate(
                    [{ transform: 'scale(1)' }, { transform: 'scale(0.96)' }, { transform: 'scale(1)' }],
                    { duration: 260, easing: 'ease-out' });
                shake(viewFinished, shakePx);
                resolve();
            }, duration * 0.55);
        });
    }

    function impactRing(tile, colorClass) {
        tile.classList.add('relative');
        const ring = document.createElement('div');
        ring.className = `absolute inset-0 rounded-2xl border-2 pointer-events-none ${colorClass}`;
        tile.appendChild(ring);
        ring.animate(
            [{ opacity: 0.9, transform: 'scale(1)' }, { opacity: 0, transform: 'scale(1.35)' }],
            { duration: 520, easing: 'ease-out' }
        ).finished.then(() => ring.remove());
    }

    function shake(el, px = 6, duration = 380) {
        el.animate([
            { transform: 'translate(0, 0)' },
            { transform: `translate(${-px}px, ${px * 0.5}px)` },
            { transform: `translate(${px}px, ${-px * 0.5}px)` },
            { transform: `translate(${-px * 0.6}px, 0)` },
            { transform: `translate(${px * 0.4}px, 0)` },
            { transform: 'translate(0, 0)' }
        ], { duration, easing: 'ease-out' });
    }

    // -----------------------------------------------------------------------
    // Confetti (tiny canvas engine, no dependencies)
    // -----------------------------------------------------------------------
    const CONFETTI_COLORS = ['#10b981', '#34d399', '#6366f1', '#f59e0b', '#f43f5e', '#38bdf8'];
    let confettiCanvas = null, confettiCtx = null, confettiParts = [], confettiRaf = null;

    function confettiBurst({ x = 0.5, y = 0.4, count = 120, angle = -Math.PI / 2, spread = Math.PI * 2, power = 15 } = {}) {
        if (reduceMotion) return;

        const dpr = window.devicePixelRatio || 1;
        if (!confettiCanvas) {
            confettiCanvas = document.createElement('canvas');
            confettiCanvas.className = 'fixed inset-0 w-full h-full pointer-events-none z-[60]';
            document.body.appendChild(confettiCanvas);
            confettiCtx = confettiCanvas.getContext('2d');
        }
        if (confettiCanvas.width !== Math.round(innerWidth * dpr)) {
            confettiCanvas.width = Math.round(innerWidth * dpr);
            confettiCanvas.height = Math.round(innerHeight * dpr);
        }

        for (let i = 0; i < count; i++) {
            const a = angle + (Math.random() - 0.5) * spread;
            const v = power * (0.5 + Math.random() * 0.7);
            confettiParts.push({
                x: x * innerWidth, y: y * innerHeight,
                vx: Math.cos(a) * v, vy: Math.sin(a) * v,
                w: 6 + Math.random() * 6, h: 4 + Math.random() * 5,
                rot: Math.random() * Math.PI * 2, vr: (Math.random() - 0.5) * 0.35,
                wobble: Math.random() * Math.PI * 2,
                color: CONFETTI_COLORS[(Math.random() * CONFETTI_COLORS.length) | 0],
                life: 0, ttl: 110 + Math.random() * 60
            });
        }
        if (!confettiRaf) confettiRaf = requestAnimationFrame(stepConfetti);
    }

    function stepConfetti() {
        const dpr = window.devicePixelRatio || 1;
        confettiCtx.setTransform(dpr, 0, 0, dpr, 0, 0);
        confettiCtx.clearRect(0, 0, innerWidth, innerHeight);

        confettiParts = confettiParts.filter(p => p.life < p.ttl && p.y < innerHeight + 40);
        for (const p of confettiParts) {
            p.life++;
            p.vx *= 0.985;
            p.vy = p.vy * 0.985 + 0.38;                 // drag + gravity
            p.wobble += 0.12;
            p.x += p.vx + Math.sin(p.wobble) * 0.6;
            p.y += p.vy;
            p.rot += p.vr;

            confettiCtx.save();
            confettiCtx.globalAlpha = Math.min(1, (p.ttl - p.life) / 30);
            confettiCtx.translate(p.x, p.y);
            confettiCtx.rotate(p.rot);
            confettiCtx.scale(1, Math.abs(Math.cos(p.wobble)));   // flutter
            confettiCtx.fillStyle = p.color;
            confettiCtx.fillRect(-p.w / 2, -p.h / 2, p.w, p.h);
            confettiCtx.restore();
        }

        if (confettiParts.length) {
            confettiRaf = requestAnimationFrame(stepConfetti);
        } else {
            confettiRaf = null;
            confettiCanvas.remove();
            confettiCanvas = null;
        }
    }

    // Utilities
    function escapeHtml(text) {
        const div = document.createElement('div');
        div.textContent = text;
        return div.innerHTML;
    }

    // Init
    if (practice) {
        setPracticeMode(true);
        if (randomPractice) {

    renderCard();
});

<?php
/**
 * ReviewHubService
 *
 * Read-only data + formatting helpers for the Review Hub page (review.php, no deck_id).
 * All "seconds" values are computed in SQL (TIMESTAMPDIFF) so PHP/MySQL timezone
 * differences cannot skew them.
 *
 * Place next to FlashcardService.php (the api/ folder).
 */
class ReviewHubService
{
    /** Cards scheduled within this many days count as "upcoming". */
    public const UPCOMING_DAYS = 2;

    /**
     * Schema-derived card buckets used by the Due Now table.
     *
     * The schema has no explicit learning/young/mastered state column, so these
     * buckets are inferred from review_count and interval_days:
     * - new: never reviewed
     * - learning: reviewed and interval_days <= 1
     * - young: interval_days 2..20
     * - mastered: interval_days >= 21
     *
     * Adjust these thresholds if the application defines different SRS rules.
     */
    public const YOUNG_MAX_INTERVAL_DAYS = 20;
    public const MASTERED_MIN_INTERVAL_DAYS = 21;

    // -----------------------------------------------------------------------
    // Queries
    // -----------------------------------------------------------------------

    /**
     * Every non-archived deck that has cards due now or coming up soon.
     *
     * @return array<int, array{
     *   id:int, title:string, totalCards:int, dueCount:int, upcomingCount:int,
     *   newDueCount:int, learningDueCount:int, youngDueCount:int, masteredDueCount:int,
     *   overdueSeconds:?int, nextDueSeconds:?int
     * }>
     */
    public static function getHubDecks(mysqli $conn, int $userId): array
    {
        $days = (int) self::UPCOMING_DAYS;
        $youngMax = (int) self::YOUNG_MAX_INTERVAL_DAYS;
        $masteredMin = (int) self::MASTERED_MIN_INTERVAL_DAYS;

        $sql = "SELECT d.id, d.title,
                       COUNT(f.id) AS total_cards,
                       COALESCE(SUM(f.next_review_at <= NOW()), 0) AS due_count,
                       COALESCE(SUM(f.next_review_at <= NOW() AND f.review_count = 0), 0) AS new_due_count,
                       COALESCE(SUM(f.next_review_at <= NOW()
                                    AND f.review_count > 0 AND f.interval_days <= 1), 0) AS learning_due_count,
                       COALESCE(SUM(f.next_review_at <= NOW()
                                    AND f.review_count > 0 AND f.interval_days BETWEEN 2 AND $youngMax), 0) AS young_due_count,
                       COALESCE(SUM(f.next_review_at <= NOW()
                                    AND f.review_count > 0 AND f.interval_days >= $masteredMin), 0) AS mastered_due_count,
                       COALESCE(SUM(f.next_review_at > NOW()
                                    AND f.next_review_at <= DATE_ADD(NOW(), INTERVAL $days DAY)), 0) AS upcoming_count,
                       TIMESTAMPDIFF(SECOND,
                           MIN(CASE WHEN f.next_review_at <= NOW() THEN f.next_review_at END),
                           NOW()) AS overdue_seconds,
                       TIMESTAMPDIFF(SECOND, NOW(),
                           MIN(CASE WHEN f.next_review_at > NOW()
                                     AND f.next_review_at <= DATE_ADD(NOW(), INTERVAL $days DAY)
                                    THEN f.next_review_at END)) AS next_due_seconds
                FROM decks d
                LEFT JOIN flashcards f ON f.deck_id = d.id AND f.deleted_at IS NULL
                WHERE d.user_id = ? AND d.deleted_at IS NULL
                GROUP BY d.id, d.title
                HAVING due_count > 0 OR upcoming_count > 0";

        $decks = [];
        if ($stmt = mysqli_prepare($conn, $sql)) {
            mysqli_stmt_bind_param($stmt, "i", $userId);
            mysqli_stmt_execute($stmt);
            $result = mysqli_stmt_get_result($stmt);
            while ($row = mysqli_fetch_assoc($result)) {
                $decks[] = [
                    'id'             => (int) $row['id'],
                    'title'          => (string) $row['title'],
                    'totalCards'     => (int) $row['total_cards'],
                    'dueCount'       => (int) $row['due_count'],
                    'newDueCount'    => (int) $row['new_due_count'],
                    'learningDueCount' => (int) $row['learning_due_count'],
                    'youngDueCount'  => (int) $row['young_due_count'],
                    'masteredDueCount' => (int) $row['mastered_due_count'],
                    'upcomingCount'  => (int) $row['upcoming_count'],
                    'overdueSeconds' => $row['overdue_seconds'] === null ? null : (int) $row['overdue_seconds'],
                    'nextDueSeconds' => $row['next_due_seconds'] === null ? null : (int) $row['next_due_seconds'],
                ];
            }
            mysqli_stmt_close($stmt);
        }
        return $decks;
    }

    /**
     * Seconds until the user's next card becomes due (no 48h limit), or null
     * when there are no scheduled cards at all.
     */
    public static function getNextDueSeconds(mysqli $conn, int $userId): ?int
    {
        $sql = "SELECT TIMESTAMPDIFF(SECOND, NOW(), MIN(f.next_review_at))
                FROM flashcards f
                JOIN decks d ON d.id = f.deck_id
                WHERE d.user_id = ? AND d.deleted_at IS NULL
                  AND f.deleted_at IS NULL AND f.next_review_at > NOW()";

        $secs = null;
        if ($stmt = mysqli_prepare($conn, $sql)) {
            mysqli_stmt_bind_param($stmt, "i", $userId);
            mysqli_stmt_execute($stmt);
            mysqli_stmt_bind_result($stmt, $secs);
            mysqli_stmt_fetch($stmt);
            mysqli_stmt_close($stmt);
        }
        return $secs === null ? null : (int) $secs;
    }

    /**
     * Latest completed review sessions.
     *
     * @return array<int, array{
     *   deckId:int, deckTitle:string, cardsReviewed:int, rememberedCount:int,
     *   accuracy:int, durationSeconds:?int, formattedTime:string
     * }>
     */
    public static function getRecentSessions(mysqli $conn, int $userId, int $limit = 8): array
    {
        $limit = max(1, min($limit, 50));

        $sql = "SELECT rs.id AS session_id, rs.deck_id, d.title AS deck_title, rs.total_reviewed, rs.remembered_count,
                       rs.accuracy_percentage,
                       TIMESTAMPDIFF(SECOND, rs.started_at, rs.ended_at) AS duration_seconds,
                       TIMESTAMPDIFF(SECOND, rs.ended_at, NOW()) AS ended_ago_seconds
                FROM review_sessions rs
                JOIN decks d ON d.id = rs.deck_id
                WHERE rs.user_id = ? AND rs.ended_at IS NOT NULL
                ORDER BY rs.ended_at DESC, rs.id DESC
                LIMIT ?";

        $sessions = [];
        if ($stmt = mysqli_prepare($conn, $sql)) {
            mysqli_stmt_bind_param($stmt, "ii", $userId, $limit);
            mysqli_stmt_execute($stmt);
            $result = mysqli_stmt_get_result($stmt);
            while ($row = mysqli_fetch_assoc($result)) {
                $sessions[] = [
                    'sessionID'       => (int) $row['session_id'],
                    'deckId'          => (int) $row['deck_id'],
                    'deckTitle'       => (string) $row['deck_title'],
                    'cardsReviewed'   => (int) $row['total_reviewed'],
                    'rememberedCount' => (int) $row['remembered_count'],
                    'accuracy'        => (int) $row['accuracy_percentage'],
                    'durationSeconds' => $row['duration_seconds'] === null ? null : (int) $row['duration_seconds'],
                    'formattedTime'   => self::formatAgo((int) $row['ended_ago_seconds']),
                ];
            }
            mysqli_stmt_close($stmt);
        }
        return $sessions;
    }

    /**
     * Cards reviewed today and the current study-day streak.
     * Same rules as the dashboard: a day counts when a completed session
     * reviewed at least one card; the streak is alive if the latest study day
     * is today or yesterday.
     *
     * @return array{reviewedToday:int, currentStreak:int}
     */
    public static function getStreakInfo(mysqli $conn, int $userId): array
    {
        $sql = "SELECT DATE(started_at) AS study_date, SUM(total_reviewed) AS reviewed
                FROM review_sessions
                WHERE user_id = ? AND ended_at IS NOT NULL
                GROUP BY DATE(started_at)
                HAVING SUM(total_reviewed) > 0
                ORDER BY study_date DESC";

        $days = [];
        if ($stmt = mysqli_prepare($conn, $sql)) {
            mysqli_stmt_bind_param($stmt, "i", $userId);
            mysqli_stmt_execute($stmt);
            $result = mysqli_stmt_get_result($stmt);
            while ($row = mysqli_fetch_assoc($result)) {
                $days[$row['study_date']] = (int) $row['reviewed'];
            }
            mysqli_stmt_close($stmt);
        }

        $today     = date('Y-m-d');
        $yesterday = date('Y-m-d', strtotime('-1 day'));

        $streak = 0;
        if (isset($days[$today]) || isset($days[$yesterday])) {
            $cursor = new DateTime(isset($days[$today]) ? $today : $yesterday);
            while (isset($days[$cursor->format('Y-m-d')])) {
                $streak++;
                $cursor->modify('-1 day');
            }
        }

        return [
            'reviewedToday' => $days[$today] ?? 0,
            'currentStreak' => $streak,
        ];
    }

    // -----------------------------------------------------------------------
    // Table helpers (pure PHP, no DB)
    // -----------------------------------------------------------------------

    /** Decks with cards due, most due first (oldest overdue breaks ties). */
    public static function dueDecks(array $hubDecks): array
    {
        $due = array_values(array_filter($hubDecks, fn($d) => $d['dueCount'] > 0));
        usort($due, function ($a, $b) {
            return [$b['dueCount'], $b['overdueSeconds'] ?? 0]
               <=> [$a['dueCount'], $a['overdueSeconds'] ?? 0];
        });
        return $due;
    }

    /** Decks with cards coming up soon, soonest first. */
    public static function upcomingDecks(array $hubDecks): array
    {
        $up = array_values(array_filter($hubDecks, fn($d) => $d['upcomingCount'] > 0));
        usort($up, fn($a, $b) => ($a['nextDueSeconds'] ?? PHP_INT_MAX) <=> ($b['nextDueSeconds'] ?? PHP_INT_MAX));
        return $up;
    }

    // -----------------------------------------------------------------------
    // Formatting
    // -----------------------------------------------------------------------

    /** "5 min", "3 h", "2 d" (never shows 0). */
    public static function formatSpan(int $seconds): string
    {
        $seconds = max(0, $seconds);
        if ($seconds < 3600)  return max(1, (int) floor($seconds / 60)) . ' min';
        if ($seconds < 86400) return (int) floor($seconds / 3600) . ' h';
        return (int) floor($seconds / 86400) . ' d';
    }

    /** "just now", "5 mins ago", "1 hour ago", "3 days ago". */
    public static function formatAgo(int $seconds): string
    {
        $seconds = max(0, $seconds);
        if ($seconds < 60) return 'just now';

        if ($seconds < 3600) {
            $n = (int) floor($seconds / 60);
            return $n . ($n === 1 ? ' min ago' : ' mins ago');
        }
        if ($seconds < 86400) {
            $n = (int) floor($seconds / 3600);
            return $n . ($n === 1 ? ' hour ago' : ' hours ago');
        }
        $n = (int) floor($seconds / 86400);
        return $n . ($n === 1 ? ' day ago' : ' days ago');
    }

    /** "45s", "4m 12s", "1h 5m", or an em dash when unknown. */
    public static function formatDuration(?int $seconds): string
    {
        if ($seconds === null || $seconds <= 0) return '—';
        if ($seconds < 60) return $seconds . 's';
        if ($seconds < 3600) {
            $m = (int) floor($seconds / 60);
            $s = $seconds % 60;
            return $s > 0 ? "{$m}m {$s}s" : "{$m}m";
        }
        $h = (int) floor($seconds / 3600);
        $m = (int) floor(($seconds % 3600) / 60);
        return $m > 0 ? "{$h}h {$m}m" : "{$h}h";
    }

    /** Tailwind classes for an accuracy badge (matches the deck grid thresholds). */
    public static function accuracyBadgeClass(int $accuracy): string
    {
        if ($accuracy >= 80) return 'text-emerald-700 bg-emerald-50 border-emerald-200/80';
        if ($accuracy >= 50) return 'text-amber-700 bg-amber-50 border-amber-200/80';
        return 'text-rose-700 bg-rose-50 border-rose-200/80';
    }

    /** Shorthand for htmlspecialchars with the project's flags. */
    public static function h($value): string
    {
        return htmlspecialchars((string) $value, ENT_QUOTES, 'UTF-8');
    }
}

    mysqli_stmt_close($sessStmt);

    // 4. Update each card + write its log row
    $stmtCard = mysqli_prepare($conn, "UPDATE flashcards
        SET review_count = review_count + 1, lapses = lapses + ?, interval_days = ?,
            next_review_at = DATE_ADD(NOW(), INTERVAL ? DAY), updated_at = NOW()
        WHERE id = ? AND deck_id = ?");

    $stmtLog = mysqli_prepare($conn, "INSERT INTO review_logs
        (session_id, card_id, rating, previous_interval, new_interval)
        VALUES (?, ?, ?, ?, ?)");

    foreach ($toSave as $cid => $status) {
        $prevInterval = $cardData[$cid];
        $isLapse      = ($status === 'forgotten') ? 1 : 0;

        $multiplier  = ($status === 'remembered') ? $rememberMult : $forgotMult;
        $newInterval = (int) round($prevInterval * $multiplier);

        // Never below the initial interval, never above the maximum.
        $newInterval = max($minInterval, min($maxInterval, $newInterval));

        // The review_logs.rating ENUM is ('remembered', 'forgot'), NOT 'forgotten'
        $dbRating = ($status === 'remembered') ? 'remembered' : 'forgot';
        
        mysqli_stmt_bind_param($stmtCard, "iiiii", $isLapse, $newInterval, $newInterval, $cid, $deckId);
        mysqli_stmt_execute($stmtCard);

        mysqli_stmt_bind_param($stmtLog, "iisii", $sessionId, $cid, $dbRating, $prevInterval, $newInterval);
        mysqli_stmt_execute($stmtLog);
    }

    mysqli_stmt_close($stmtCard);
    mysqli_stmt_close($stmtLog);

    mysqli_commit($conn);
    echo json_encode(['ok' => true, 'saved' => $savedCount, 'session_id' => $sessionId]);

} catch (Throwable $e) {
    // Only roll back if a transaction was started; harmless otherwise
    try { mysqli_rollback($conn); } catch (Throwable $ignored) {}
    error_log('save-session.php error: ' . $e->getMessage());
    http_response_code(500);
    echo json_encode(['error' => 'Database exception occurred']);
}

    async function showStats(rem, forg, accuracy) {

    // A number slams in from huge and blurry, squashes on impact, then settles.
    // Resolves at the moment of impact so the caller can chain the next beat.
    function stampStat(el, tile, ringClass, { scale = 3, duration = 520, shakePx = 5, hero = false } = {}) {
        return new Promise(resolve => {
            el.style.opacity = '1';
            el.animate([
                { opacity: 0, transform: `scale(${scale}) rotate(-14deg)`, filter: 'blur(6px)', offset: 0, easing: 'cubic-bezier(.5,0,1,.6)' },
                { opacity: 1, transform: 'scale(0.9) rotate(2deg)', filter: 'blur(0px)', offset: 0.55 },
                { transform: 'scale(1.06) rotate(-1deg)', offset: 0.75 },
                { transform: 'scale(1) rotate(0deg)', offset: 1 }
            ], { duration, easing: 'linear' });

            setTimeout(() => {
                impactRing(tile, ringClass);
                if (hero) setTimeout(() => impactRing(tile, ringClass), 130);
                tile.animate(
                    [{ transform: 'scale(1)' }, { transform: 'scale(0.96)' }, { transform: 'scale(1)' }],
                    { duration: 260, easing: 'ease-out' });
                shake(viewFinished, shakePx);
                resolve();
            }, duration * 0.55);
        });
    }

    function impactRing(tile, colorClass) {
        tile.classList.add('relative');
        const ring = document.createElement('div');
        ring.className = `absolute inset-0 rounded-2xl border-2 pointer-events-none ${colorClass}`;
        tile.appendChild(ring);
        ring.animate(
            [{ opacity: 0.9, transform: 'scale(1)' }, { opacity: 0, transform: 'scale(1.35)' }],
            { duration: 520, easing: 'ease-out' }
        ).finished.then(() => ring.remove());
    }

    function shake(el, px = 6, duration = 380) {
        el.animate([
            { transform: 'translate(0, 0)' },
            { transform: `translate(${-px}px, ${px * 0.5}px)` },
            { transform: `translate(${px}px, ${-px * 0.5}px)` },
            { transform: `translate(${-px * 0.6}px, 0)` },
            { transform: `translate(${px * 0.4}px, 0)` },
            { transform: 'translate(0, 0)' }
        ], { duration, easing: 'ease-out' });
    }

    // -----------------------------------------------------------------------
    // Confetti (tiny canvas engine, no dependencies)
    // -----------------------------------------------------------------------
    const CONFETTI_COLORS = ['#10b981', '#34d399', '#6366f1', '#f59e0b', '#f43f5e', '#38bdf8'];
    let confettiCanvas = null, confettiCtx = null, confettiParts = [], confettiRaf = null;

    function confettiBurst({ x = 0.5, y = 0.4, count = 120, angle = -Math.PI / 2, spread = Math.PI * 2, power = 15 } = {}) {
        if (reduceMotion) return;

        const dpr = window.devicePixelRatio || 1;
        if (!confettiCanvas) {
            confettiCanvas = document.createElement('canvas');
            confettiCanvas.className = 'fixed inset-0 w-full h-full pointer-events-none z-[60]';
            document.body.appendChild(confettiCanvas);
            confettiCtx = confettiCanvas.getContext('2d');
        }
        if (confettiCanvas.width !== Math.round(innerWidth * dpr)) {
            confettiCanvas.width = Math.round(innerWidth * dpr);
            confettiCanvas.height = Math.round(innerHeight * dpr);
        }

        for (let i = 0; i < count; i++) {
            const a = angle + (Math.random() - 0.5) * spread;
            const v = power * (0.5 + Math.random() * 0.7);
            confettiParts.push({
                x: x * innerWidth, y: y * innerHeight,
                vx: Math.cos(a) * v, vy: Math.sin(a) * v,
                w: 6 + Math.random() * 6, h: 4 + Math.random() * 5,
                rot: Math.random() * Math.PI * 2, vr: (Math.random() - 0.5) * 0.35,
                wobble: Math.random() * Math.PI * 2,
                color: CONFETTI_COLORS[(Math.random() * CONFETTI_COLORS.length) | 0],
                life: 0, ttl: 110 + Math.random() * 60
            });
        }
        if (!confettiRaf) confettiRaf = requestAnimationFrame(stepConfetti);
    }

    function stepConfetti() {
        const dpr = window.devicePixelRatio || 1;
        confettiCtx.setTransform(dpr, 0, 0, dpr, 0, 0);
        confettiCtx.clearRect(0, 0, innerWidth, innerHeight);

        confettiParts = confettiParts.filter(p => p.life < p.ttl && p.y < innerHeight + 40);
        for (const p of confettiParts) {
            p.life++;
            p.vx *= 0.985;
            p.vy = p.vy * 0.985 + 0.38;                 // drag + gravity
            p.wobble += 0.12;
            p.x += p.vx + Math.sin(p.wobble) * 0.6;
            p.y += p.vy;
            p.rot += p.vr;

            confettiCtx.save();
            confettiCtx.globalAlpha = Math.min(1, (p.ttl - p.life) / 30);
            confettiCtx.translate(p.x, p.y);
            confettiCtx.rotate(p.rot);
            confettiCtx.scale(1, Math.abs(Math.cos(p.wobble)));   // flutter
            confettiCtx.fillStyle = p.color;
            confettiCtx.fillRect(-p.w / 2, -p.h / 2, p.w, p.h);
            confettiCtx.restore();
        }

        if (confettiParts.length) {
            confettiRaf = requestAnimationFrame(stepConfetti);
        } else {
            confettiRaf = null;
            confettiCanvas.remove();
            confettiCanvas = null;
        }
    }

/**
 * MemPulse read-only Tabulator helper (Tabulator 5.5.0, same build as deck-view.php).
 * Shared by sessions.php and session-details.php.
 *
 * Contains NO Tailwind classes on purpose: every class used in a table cell lives in the
 * PHP view that renders it, so Tailwind always sees it.
 *
 * Usage (from a view, on DOMContentLoaded):
 *   MPTable.applyTokens(document.getElementById('mp-token-probe'));
 *   MPTable.ensure(function () { MPTable.make('#my-table', rows, columns, { rowClick: ... }); });
 */
(function () {
    if (window.MPTable) return;

    var TABULATOR_JS = 'https://unpkg.com/tabulator-tables@5.5.0/dist/js/tabulator.min.js';

    // Structure + responsive layout adjustments
    var CSS = [
        '.mp-table .tabulator { background: transparent; border: 0; font-family: inherit; font-size: .875rem; color: var(--mp-text, inherit); }',
        '.mp-table .tabulator .tabulator-header { background: var(--mp-head-bg, #f1f1f1) !important; color: var(--mp-head-text, inherit); border-top: 0; border-bottom: 1px solid var(--mp-border, #ddd); }',
        '.mp-table .tabulator .tabulator-header .tabulator-col { background: transparent !important; border-right: 0; }',
        '.mp-table .tabulator .tabulator-header .tabulator-col .tabulator-col-content { padding: .75rem 1rem; }',
        '.mp-table .tabulator .tabulator-header .tabulator-col .tabulator-col-title { font-size: 11px; font-weight: 700; text-transform: uppercase; letter-spacing: .05em; }',
        '.mp-table .tabulator .tabulator-tableholder, .mp-table .tabulator .tabulator-tableholder .tabulator-table { background: transparent; }',
        '.mp-table .tabulator .tabulator-row, .mp-table .tabulator .tabulator-row.tabulator-row-even, .mp-table .tabulator .tabulator-row.tabulator-row-odd { background: transparent !important; color: inherit; border-bottom: 1px solid var(--mp-divider, #eee); min-height: 0; }',
        '.mp-table .tabulator .tabulator-row:hover { background: var(--mp-hover, rgba(0,0,0,.03)) !important; }',
        '.mp-table .tabulator .tabulator-table > .tabulator-row:last-child { border-bottom: 0; }',
        '.mp-table .tabulator .tabulator-row .tabulator-cell { padding: .75rem 1rem; border-right: 0; vertical-align: middle; }',
        '.mp-table .tabulator .tabulator-header .tabulator-col .tabulator-col-resize-handle { display: none; }',
        '@media (max-width: 640px) {',
        '  .mp-table .tabulator .tabulator-header .tabulator-col .tabulator-col-content,',
        '  .mp-table .tabulator .tabulator-row .tabulator-cell { padding: .5rem .625rem; font-size: .8125rem; }',
        '}'
    ].join('\n');

    var styleEl = document.createElement('style');
    styleEl.id = 'mp-table-styles';
    styleEl.textContent = CSS;
    document.head.appendChild(styleEl);

    // Read the Tailwind design tokens from the probe element and expose them as CSS variables
    function applyTokens(probe) {
        if (!probe) return;
        var q = function (t) { return probe.querySelector('[data-t="' + t + '"]'); };
        var s = document.documentElement.style;
        s.setProperty('--mp-head-bg', getComputedStyle(q('head')).backgroundColor);
        s.setProperty('--mp-head-text', getComputedStyle(q('head')).color);
        s.setProperty('--mp-text', getComputedStyle(q('text')).color);
        s.setProperty('--mp-hover', getComputedStyle(q('hover')).backgroundColor);
        s.setProperty('--mp-border', getComputedStyle(q('border')).borderTopColor);
        s.setProperty('--mp-divider', getComputedStyle(q('divider')).borderTopColor);
    }

    function esc(v) {
        return String(v == null ? '' : v).replace(/[&<>"']/g, function (c) {
            return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c];
        });
    }

    // Load Tabulator only if the page doesn't already have it
    function ensure(done) {
        if (window.Tabulator) return done();
        var sc = document.createElement('script');
        sc.src = TABULATOR_JS;
        sc.onload = done;
        sc.onerror = done;
        document.head.appendChild(sc);
    }

    // Read-only responsive defaults
    var base = {
        layout: 'fitColumns',
        responsiveLayout: 'hide',
        movableColumns: false,
        selectable: false,
        selectableRows: false,
        columnDefaults: { headerSort: false, resizable: false }
    };

    function make(selector, rows, columns, extra) {
        var el = document.querySelector(selector);
        if (!el) return null;
        if (typeof Tabulator === 'undefined') {
            console.error('Tabulator failed to load.');
            el.innerHTML = '<p style="padding:2rem;text-align:center;font-size:.875rem;opacity:.7">Table could not be loaded.</p>';
            return null;
        }
        var cols = columns.map(function (c) { return Object.assign({ headerSort: false, resizable: false }, c); });
        return new Tabulator(el, Object.assign({}, base, { data: rows, columns: cols }, extra || {}));
    }

    window.MPTable = { applyTokens: applyTokens, esc: esc, ensure: ensure, make: make };
})();

CREATE DATABASE IF NOT EXISTS mempulse_db
CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

USE mempulse_db;


-- 1. USERS TABLE
CREATE TABLE users (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    email VARCHAR(255) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    password_changed_at DATETIME NULL,
    role ENUM('admin', 'user') NOT NULL DEFAULT 'user',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL,
    account_status ENUM('pending', 'approved', 'rejected') NOT NULL DEFAULT 'pending',
    INDEX idx_users_active (deleted_at)
) ENGINE=InnoDB;


-- 2. USER SETTINGS TABLE (SRS Configuration)
CREATE TABLE user_settings (
    user_id BIGINT UNSIGNED PRIMARY KEY,
    remember_multiplier DECIMAL(4,2) NOT NULL DEFAULT 2.00,
    forgot_multiplier DECIMAL(4,2) NOT NULL DEFAULT 0.50,
    initial_interval_days INT UNSIGNED NOT NULL DEFAULT 1,
    maximum_interval_days INT UNSIGNED NOT NULL DEFAULT 365,
    theme ENUM('light', 'dark', 'system')NOT NULL DEFAULT 'system',
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE
) ENGINE=InnoDB;


-- 3. DECKS TABLE
CREATE TABLE decks (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT UNSIGNED NOT NULL,
    title VARCHAR(150) NOT NULL,
    cover_image VARCHAR(255) DEFAULT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL,
    FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE,
    INDEX idx_decks_user_active (user_id, deleted_at)
) ENGINE=InnoDB;


-- 4. FLASHCARDS TABLE
CREATE TABLE flashcards (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    deck_id BIGINT UNSIGNED NOT NULL,
    front TEXT NOT NULL,
    back TEXT NOT NULL,
    interval_days INT UNSIGNED NOT NULL DEFAULT 1,
    next_review_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    review_count INT UNSIGNED NOT NULL DEFAULT 0,
    lapses INT UNSIGNED NOT NULL DEFAULT 0,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL,
    FOREIGN KEY (deck_id)
        REFERENCES decks(id)
        ON DELETE CASCADE,
    INDEX idx_cards_deck_due (deck_id, deleted_at, next_review_at)
) ENGINE=InnoDB;


-- 5. REVIEW SESSIONS TABLE
CREATE TABLE review_sessions (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT UNSIGNED NOT NULL,
    deck_id BIGINT UNSIGNED NOT NULL,
    started_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    ended_at DATETIME NULL,
    total_reviewed INT UNSIGNED NOT NULL DEFAULT 0,
    remembered_count INT UNSIGNED NOT NULL DEFAULT 0,
    accuracy_percentage DECIMAL(5,2)
        GENERATED ALWAYS AS (
            CASE
                WHEN total_reviewed > 0
                    THEN (remembered_count / total_reviewed) * 100
                ELSE 0.00
            END
        ) STORED,
    FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE,
    FOREIGN KEY (deck_id)
        REFERENCES decks(id)
        ON DELETE CASCADE,
    INDEX idx_user_sessions (user_id, started_at)
) ENGINE=InnoDB;


-- 6. REVIEW LOGS TABLE
CREATE TABLE review_logs (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    session_id BIGINT UNSIGNED NOT NULL,
    card_id BIGINT UNSIGNED NULL,
    rating ENUM('remembered', 'forgot') NOT NULL,
    previous_interval INT UNSIGNED NOT NULL,
    new_interval INT UNSIGNED NOT NULL,
    response_time_ms INT UNSIGNED NULL,
    reviewed_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (session_id)
        REFERENCES review_sessions(id)
        ON DELETE CASCADE,
    FOREIGN KEY (card_id)
        REFERENCES flashcards(id)
        ON DELETE SET NULL,
    INDEX idx_session_logs (session_id)
) ENGINE=InnoDB;

-- 7. ADMIN AUDIT LOG TABLE (append-only record of admin actions)
CREATE TABLE admin_audit_log (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    admin_id BIGINT UNSIGNED NULL,
    action VARCHAR(50) NOT NULL,
    target_user_id BIGINT UNSIGNED NULL,
    target_email VARCHAR(255) NOT NULL,
    target_name VARCHAR(201) NOT NULL,
    previous_status VARCHAR(20) NULL,
    new_status VARCHAR(20) NULL,
    ip_address VARCHAR(45) NULL,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (admin_id)
        REFERENCES users(id)
        ON DELETE SET NULL,
    FOREIGN KEY (target_user_id)
        REFERENCES users(id)
        ON DELETE SET NULL,
    INDEX idx_audit_created (created_at),
    INDEX idx_audit_target (target_user_id, created_at),
    INDEX idx_audit_admin (admin_id, created_at)
) ENGINE=InnoDB;


-- Random Practice is deliberately stored outside review_sessions/review_logs.
-- Existing SRS reports, streaks and review statistics therefore remain unchanged.
CREATE TABLE random_practice_sessions (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,

    total_rated INT UNSIGNED NOT NULL DEFAULT 0,
    remembered_count INT UNSIGNED NOT NULL DEFAULT 0,
    accuracy_percentage DECIMAL(5,2) NOT NULL DEFAULT 0.00,
    duration_seconds INT UNSIGNED NOT NULL DEFAULT 0,
    started_at DATETIME NOT NULL,
    ended_at DATETIME NOT NULL,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_random_practice_user_ended (user_id, ended_at),
    INDEX idx_random_practice_deck (deck_id),
    CONSTRAINT fk_random_practice_user FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    CONSTRAINT fk_random_practice_deck FOREIGN KEY (deck_id) REFERENCES decks(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
