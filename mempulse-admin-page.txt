        let floatingTooltip = document.getElementById("sidebar-floating-tooltip");
        if (!floatingTooltip) {
            floatingTooltip = document.createElement("div");
            floatingTooltip.id = "sidebar-floating-tooltip";
            floatingTooltip.setAttribute("role", "tooltip");
            document.body.appendChild(floatingTooltip);
        }

        let tooltipTarget = null;
        function hideSidebarTooltip() {
            floatingTooltip.classList.remove("is-visible");
            tooltipTarget = null;
        }
        function showSidebarTooltip(target) {
            const label = target.getAttribute("data-tooltip") || target.getAttribute("aria-label");
            if (!label || !sidebar.classList.contains("collapsed") || window.innerWidth < 768) {
                hideSidebarTooltip();
                return;
            }
            tooltipTarget = target;
            floatingTooltip.textContent = label;
            floatingTooltip.style.left = "0px";
            floatingTooltip.style.top = "0px";
            floatingTooltip.classList.add("is-visible");
            const rect = target.getBoundingClientRect();
            const tipRect = floatingTooltip.getBoundingClientRect();
            const gap = 12;
            let left = rect.right + gap;
            // If there is not enough room to the right, place the tooltip to the left.
            if (left + tipRect.width > window.innerWidth - 8) {
                left = Math.max(8, rect.left - tipRect.width - gap);
                floatingTooltip.classList.add("tooltip-left");
            } else {
                floatingTooltip.classList.remove("tooltip-left");
            }
            const top = Math.min(window.innerHeight - 8, Math.max(8, rect.top + rect.height / 2));
            floatingTooltip.style.left = `${left}px`;
            floatingTooltip.style.top = `${top}px`;
        }

        sidebar.querySelectorAll(".sidebar-tooltip").forEach((item) => {
            item.addEventListener("mouseenter", () => showSidebarTooltip(item));
            item.addEventListener("mouseleave", hideSidebarTooltip);
            item.addEventListener("focus", () => showSidebarTooltip(item));
            item.addEventListener("blur", hideSidebarTooltip);
        });
        sidebar.addEventListener("scroll", hideSidebarTooltip, true);
        window.addEventListener("resize", hideSidebarTooltip);
        window.addEventListener("scroll", () => {
            if (tooltipTarget) showSidebarTooltip(tooltipTarget);
        }, true);

        const toggleIcon = toggleBtn ? toggleBtn.querySelector(".material-symbols-outlined") : null;

        // Apply visual collapsed state
        const sidebarCollapsed = localStorage.getItem("sidebarCollapsed") === "true";
        if (sidebarCollapsed && window.innerWidth >= 768) {
            sidebar.classList.add("collapsed");
            if (mainContent) mainContent.classList.add("collapsed");
            if (toggleIcon) toggleIcon.textContent = "left_panel_open";
            if (reviewDot) reviewDot.classList.remove("hidden");

        // Desktop Border Toggle Interaction
        if (toggleBtn) {

                const isCollapsed = sidebar.classList.toggle("collapsed");
                
                document.documentElement.classList.toggle(
                    "sidebar-collapsed",
                    isCollapsed
                );
                
                hideSidebarTooltip();
                if (mainContent) {
                    mainContent.classList.toggle("collapsed", isCollapsed);
                }
                if (toggleIcon) {
                    toggleIcon.textContent = isCollapsed ? "left_panel_open" : "left_panel_close";
                }
                if (reviewDot) {
                    isCollapsed ? reviewDot.classList.remove("hidden") : reviewDot.classList.add("hidden");
                }
                // Close popover if sidebar collapses
                setAccountPopoverOpen(false);
                
                localStorage.setItem("sidebarCollapsed", isCollapsed);
            });
        }

        // Mobile Toggle Interactions
        function openMobileSidebar() {
            sidebar.classList.remove("-translate-x-full", "collapsed"); 
            mobileOverlay.classList.remove("hidden");
            setTimeout(() => mobileOverlay.classList.remove("opacity-0"), 10);

            sidebar.classList.add("-translate-x-full");
            mobileOverlay.classList.add("opacity-0");
            setTimeout(() => mobileOverlay.classList.add("hidden"), 300);
        }

        if (floatingToggleBtn) floatingToggleBtn.addEventListener("click", openMobileSidebar);
        if (mobileOverlay) mobileOverlay.addEventListener("click", closeMobileSidebar);

        // Popover Toggle Logic
        function setAccountPopoverOpen(isOpen) {
            if (!settingsPopover || !profileTrigger) return;
            settingsPopover.classList.toggle('hidden', !isOpen);
            profileTrigger.setAttribute('aria-expanded', String(isOpen));
        }

        function togglePopover(e) {
            e.stopPropagation();
            setAccountPopoverOpen(settingsPopover?.classList.contains('hidden'));
        }

        if (profileTrigger) profileTrigger.addEventListener('click', togglePopover);

        // Click outside the avatar/menu closes the account popover.
        document.addEventListener('click', (e) => {

                settingsPopover &&
                !settingsPopover.contains(e.target) &&

        const settingsOpeners = document.querySelectorAll('[data-open-settings]');

        function switchSettingsTab(tabName) {
                document.getElementById('settings-save')?.classList.toggle('hidden', tabName !== 'system');
            settingsTabs.forEach((tab) => {
                const active = tab.dataset.settingsTab === tabName;

                tab.setAttribute('aria-selected', String(active));
                tab.classList.toggle('border-primary', active);

        const settingsSave = document.getElementById('settings-save');
        const settingsStatus = document.getElementById('settings-status');

        const settingsFields = {
            remember_multiplier: document.getElementById('remember-multiplier'),
            forgot_multiplier: document.getElementById('forgot-multiplier'),
            initial_interval_days: document.getElementById('initial-interval'),
            maximum_interval_days: document.getElementById('maximum-interval')
        };

        const settingsApi = <?= json_encode(BASE_URL . 'api/settings-actions.php') ?>;

        function showSettingsStatus(message, isError = false) {
            if (!settingsStatus) return;

            settingsStatus.textContent = message;
            settingsStatus.classList.toggle('text-error', isError);
            settingsStatus.classList.toggle('text-primary', !isError);
        }

        function displaySettings(settings) {
            Object.entries(settingsFields).forEach(([key, input]) => {
                if (input) input.value = settings[key];
            });
        }

        async function loadSettings() {
            showSettingsStatus('Loading settings...');

            try {
                const response = await fetch(settingsApi, {
                    credentials: 'same-origin'
                });

                const result = await response.json();

                if (!response.ok || !result.success) {
                    throw new Error(result.message || 'Could not load settings.');
                }

                displaySettings(result.settings);
                showSettingsStatus('');
            } catch (error) {
                showSettingsStatus(error.message, true);
            }
        }

        async function saveSettings() {
            const payload = {};

            for (const [key, input] of Object.entries(settingsFields)) {
                if (!input || !input.checkValidity()) {
                    input?.reportValidity();
                    return;
                }

                payload[key] = input.value;
            }

            const initialValue = Number(payload.initial_interval_days);
            const maximumValue = Number(payload.maximum_interval_days);
            if (maximumValue < initialValue) {
                settingsFields.maximum_interval_days.setCustomValidity('Maximum interval must be at least the initial interval.');
                settingsFields.maximum_interval_days.reportValidity();
                settingsFields.maximum_interval_days.setCustomValidity('');
                return;
            }

            settingsSave.disabled = true;
            showSettingsStatus('Saving settings...');

            try {
                const response = await fetch(settingsApi, {
                    method: 'POST',
                    credentials: 'same-origin',
                    headers: {
                        'Content-Type': 'application/json',
                        'X-CSRF-Token': settingsDialog.dataset.csrf
                    },
                    body: JSON.stringify(payload)
                });

                const result = await response.json();

                if (!response.ok || !result.success) {
                    throw new Error(result.message || 'Could not save settings.');
                }

                displaySettings(result.settings);
                showSettingsStatus(result.message);
            } catch (error) {
                showSettingsStatus(error.message, true);
            } finally {
                settingsSave.disabled = false;
            }
        }

        settingsSave?.addEventListener('click', saveSettings);


        settingsOpeners.forEach((button) => {
            button.addEventListener('click', () => {
                const tabName = button.dataset.openSettings;

                setAccountPopoverOpen(false);
                switchSettingsTab(tabName);

                if (settingsDialog) {
                    if (!settingsDialog.open) {
                        settingsDialog.showModal();
                    }
                    loadSettings();
                }
            });
        });

        settingsTabs.forEach((tab) => {
            tab.addEventListener('click', () => {
                switchSettingsTab(tab.dataset.settingsTab);
            });
        });

        function closeSettings() {
            if (settingsDialog?.open) {
                settingsDialog.close();
            }
        }

        settingsClose?.addEventListener('click', closeSettings);
        settingsCancel?.addEventListener('click', closeSettings);

        // Close when clicking the backdrop
        settingsDialog?.addEventListener('click', (event) => {
            if (event.target === settingsDialog) {
                closeSettings();
            }
        });

        // Logout Dialog Logic
        const logoutBtn = document.getElementById('logout-btn');
        const logoutDialog = document.getElementById('logout-dialog');
        const logoutCancel = document.getElementById('logout-cancel');

        if (logoutBtn && logoutDialog) {
            logoutBtn.addEventListener('click', () => {
                setAccountPopoverOpen(false); // Close account menu
                logoutDialog.showModal();
            });

            if (logoutCancel) {
                logoutCancel.addEventListener('click', () => logoutDialog.close());
            }

            logoutDialog.addEventListener('click', (e) => {
                const rect = logoutDialog.getBoundingClientRect();
                if (e.clientX < rect.left || e.clientX > rect.right || e.clientY < rect.top || e.clientY > rect.bottom) {
                    logoutDialog.close();
                }
            });
        }
    });

CREATE DATABASE IF NOT EXISTS mempulse_db
CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

USE mempulse_db;


-- 1. USERS TABLE
CREATE TABLE users (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    email VARCHAR(255) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    password_changed_at DATETIME NULL,
    role ENUM('admin', 'user') NOT NULL DEFAULT 'user',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL,
    account_status ENUM('pending', 'approved', 'rejected') NOT NULL DEFAULT 'approved';
    INDEX idx_users_active (deleted_at)
) ENGINE=InnoDB;


-- 2. USER SETTINGS TABLE (SRS Configuration)
CREATE TABLE user_settings (
    user_id BIGINT UNSIGNED PRIMARY KEY,
    remember_multiplier DECIMAL(4,2) NOT NULL DEFAULT 2.00,
    forgot_multiplier DECIMAL(4,2) NOT NULL DEFAULT 0.50,
    initial_interval_days INT UNSIGNED NOT NULL DEFAULT 1,
    maximum_interval_days INT UNSIGNED NOT NULL DEFAULT 365,
    theme ENUM('light', 'dark', 'system')NOT NULL DEFAULT 'system',
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE
) ENGINE=InnoDB;


-- 3. DECKS TABLE
CREATE TABLE decks (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT UNSIGNED NOT NULL,
    title VARCHAR(150) NOT NULL,
    cover_image VARCHAR(255) DEFAULT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL,
    FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE,
    INDEX idx_decks_user_active (user_id, deleted_at)
) ENGINE=InnoDB;


-- 4. FLASHCARDS TABLE
CREATE TABLE flashcards (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    deck_id BIGINT UNSIGNED NOT NULL,
    front TEXT NOT NULL,
    back TEXT NOT NULL,
    interval_days INT UNSIGNED NOT NULL DEFAULT 1,
    next_review_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    review_count INT UNSIGNED NOT NULL DEFAULT 0,
    lapses INT UNSIGNED NOT NULL DEFAULT 0,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL,
    FOREIGN KEY (deck_id)
        REFERENCES decks(id)
        ON DELETE CASCADE,
    INDEX idx_cards_deck_due (deck_id, deleted_at, next_review_at)
) ENGINE=InnoDB;


-- 5. REVIEW SESSIONS TABLE
CREATE TABLE review_sessions (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT UNSIGNED NOT NULL,
    deck_id BIGINT UNSIGNED NOT NULL,
    started_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    ended_at DATETIME NULL,
    total_reviewed INT UNSIGNED NOT NULL DEFAULT 0,
    remembered_count INT UNSIGNED NOT NULL DEFAULT 0,
    accuracy_percentage DECIMAL(5,2)
        GENERATED ALWAYS AS (
            CASE
                WHEN total_reviewed > 0
                    THEN (remembered_count / total_reviewed) * 100
                ELSE 0.00
            END
        ) STORED,
    FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE,
    FOREIGN KEY (deck_id)
        REFERENCES decks(id)
        ON DELETE CASCADE,
    INDEX idx_user_sessions (user_id, started_at)
) ENGINE=InnoDB;


-- 6. REVIEW LOGS TABLE
CREATE TABLE review_logs (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    session_id BIGINT UNSIGNED NOT NULL,
    card_id BIGINT UNSIGNED NULL,
    rating ENUM('remembered', 'forgot') NOT NULL,
    previous_interval INT UNSIGNED NOT NULL,
    new_interval INT UNSIGNED NOT NULL,
    response_time_ms INT UNSIGNED NULL,
    reviewed_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (session_id)
        REFERENCES review_sessions(id)
        ON DELETE CASCADE,
    FOREIGN KEY (card_id)
        REFERENCES flashcards(id)
        ON DELETE SET NULL,
    INDEX idx_session_logs (session_id)
) ENGINE=InnoDB;

/**
 * report-export.php
 *
 * GET report-export.php?dataset=sessions|decks|cards|logs|full
 *                      &range=..&from=..&to=..&deck=..&archived=1&meta=1
 *
 * Session-guarded, read-only, scoped to the signed-in user.
 * Streams a UTF-8 CSV (with BOM) named mempulse-{dataset}-{YYYY-MM-DD}.csv
 */
session_start();

if (!isset($_SESSION['logged_in']) || $_SESSION['logged_in'] !== true || !isset($_SESSION['user_id'])) {
    http_response_code(401);
    header('Content-Type: text/plain; charset=utf-8');
    exit('Please sign in again.');
}

if (($_SERVER['REQUEST_METHOD'] ?? 'GET') !== 'GET') {
    http_response_code(405);
    header('Content-Type: text/plain; charset=utf-8');
    exit('Method not allowed.');
}

require_once __DIR__ . '/../../config/db.php';
require_once __DIR__ . '/DashboardService.php';
require_once __DIR__ . '/ReportsService.php';
require_once __DIR__ . '/ReportsExportService.php';

$userId = (int) $_SESSION['user_id'];
session_write_close();   // do not hold the session lock during a long export

$dataset = (string) ($_GET['dataset'] ?? '');
if (!in_array($dataset, ReportsExportService::DATASETS, true)) {
    http_response_code(400);
    header('Content-Type: text/plain; charset=utf-8');
    exit('Unknown dataset.');
}

// Same validation as the page: bad dates, foreign deck ids, etc. fall back safely.

<?php
/**
 * ReportsExportService
 *
 * Builds the CSV datasets for the Reports page. Every query is scoped to the
 * signed-in user and reuses ReportsService's filter scopes, so a CSV always
 * matches what the page shows.
 *
 * Each section is ['title' => string, 'headers' => array, 'rows' => iterable].
 * Rows are produced lazily (generators) so large exports are streamed.
 */
class ReportsExportService
{
    public const DATASETS = ['sessions', 'decks', 'cards', 'logs', 'full'];

    /** First characters that make Excel/Sheets treat a cell as a formula. */
    private const FORMULA_CHARS = "=+-@\t\r";

    /**
     * CSV formula-injection guard. Numbers pass through untouched;
     * text starting with = + - @ (or tab / CR) gets an apostrophe prefix.
     */
    public static function safeCell($value)
    {
        if ($value === null) {
            return '';
        }
        if (is_int($value) || is_float($value)) {
            return $value;
        }
        $s = (string) $value;
        if ($s !== '' && strpbrk($s[0], self::FORMULA_CHARS) !== false) {
            return "'" . $s;
        }

        return $s;
    }

    public static function sections(string $dataset, mysqli $conn, int $userId, array $f): array
    {
        switch ($dataset) {
            case 'sessions':
                return [self::sessions($conn, $userId, $f)];
            case 'decks':
                return [self::decks($conn, $userId, $f)];
            case 'cards':
                return [self::cards($conn, $userId, $f)];
            case 'logs':
                return [self::logs($conn, $userId, $f)];
            case 'full':
                return [
                    self::summary($conn, $userId, $f),
                    self::sessions($conn, $userId, $f),
                    self::decks($conn, $userId, $f),
                    self::cards($conn, $userId, $f),

    private static function sessions(mysqli $conn, int $userId, array $f): array
    {
        [$where, $types, $params] = ReportsService::sessionScope($userId, $f);

        $sql = "SELECT s.id, s.started_at, s.ended_at, s.total_reviewed, s.remembered_count,
                       d.title AS deck_title, d.deleted_at AS deck_deleted
                FROM review_sessions s
                JOIN decks d ON d.id = s.deck_id
                WHERE $where
                ORDER BY s.started_at DESC, s.id DESC";

        $rows = (function () use ($conn, $sql, $types, $params) {
            foreach (self::stream($conn, $sql, $types, $params) as $r) {
                $reviewed   = (int) $r['total_reviewed'];
                $remembered = (int) $r['remembered_count'];

                yield [

            }
        })();

        return [
            'title'   => 'Card performance',
            'headers' => ['Card ID', 'Deck', 'Front', 'Back', 'Maturity', 'Interval (days)', 'Lapses',

    private static function logs(mysqli $conn, int $userId, array $f): array
    {
        [$where, $types, $params] = ReportsService::logScope($userId, $f);

        // LEFT JOIN: reviews of purged cards (card_id NULL) must stay in the export.
        $sql = "SELECT l.id, l.reviewed_at, l.session_id, d.title AS deck_title, l.card_id, f.front,
                       l.rating, l.previous_interval, l.new_interval, l.response_time_ms
                FROM review_logs l
                JOIN review_sessions s ON s.id = l.session_id
                JOIN decks d ON d.id = s.deck_id
                LEFT JOIN flashcards f ON f.id = l.card_id
                WHERE $where
                ORDER BY l.reviewed_at ASC, l.id ASC";

        $rows = (function () use ($conn, $sql, $types, $params) {
            foreach (self::stream($conn, $sql, $types, $params) as $r) {
                yield [
                    (int) $r['id'],
                    $r['reviewed_at'],
                    (int) $r['session_id'],
                    $r['deck_title'],
                    $r['card_id'] === null ? '' : (int) $r['card_id'],
                    $r['card_id'] === null ? 'Deleted card' : (string) $r['front'],
                    $r['rating'] === 'remembered' ? 'Remembered' : 'Forgot',
                    (int) $r['previous_interval'],
                    (int) $r['new_interval'],
                    $r['response_time_ms'] === null ? '' : (int) $r['response_time_ms'],
                ];
            }
        })();

        return [
            'title'   => 'Review log detail',
            'headers' => ['Review ID', 'Reviewed at', 'Session ID', 'Deck', 'Card ID', 'Card', 'Rating',
                          'Previous interval (days)', 'New interval (days)', 'Response time (ms)'],
            'rows'    => $rows,
        ];
    }

    /* ---------------------------------------------------------------- */
    /*  Helpers                                                          */
    /* ---------------------------------------------------------------- */

    /**
     * Run a prepared statement and yield one associative row at a time.
     */
    private static function stream(mysqli $conn, string $sql, string $types, array $params): Generator
    {
        $stmt = mysqli_prepare($conn, $sql);
        if (!$stmt) {
            return;
        }
        if ($types !== '') {
            mysqli_stmt_bind_param($stmt, $types, ...$params);
        }
        mysqli_stmt_execute($stmt);
        $result = mysqli_stmt_get_result($stmt);

        if ($result) {
            while ($row = mysqli_fetch_assoc($result)) {
                yield $row;
            }
            mysqli_free_result($result);
        }
        mysqli_stmt_close($stmt);
    }

        $mastered = self::fetchCount($conn,

             WHERE d.user_id = ? AND d.deleted_at IS NULL AND f.deleted_at IS NULL AND f.interval_days >= 21", 
            $userId
        );

        // 3. Tough Cards / Leeches (Lapses >= 8)
        $tough = self::fetchCount($conn, 
            "SELECT COUNT(*) FROM flashcards f 
             JOIN decks d ON f.deck_id = d.id 
             WHERE d.user_id = ? AND d.deleted_at IS NULL AND f.deleted_at IS NULL AND f.lapses >= 8", 
            $userId
        );

        $reviewedToday = self::fetchCount($conn,

             WHERE user_id = ? AND ended_at IS NOT NULL AND DATE(started_at) = CURDATE()",

        );

        // 5. Retention Rate Percentage
        $retentionRate = 0.0;
        $retentionSql = "SELECT COALESCE(SUM(remembered_count), 0), COALESCE(SUM(total_reviewed), 0) 
                         FROM review_sessions 
                         WHERE user_id = ? AND ended_at IS NOT NULL";
                         
        if ($stmt = mysqli_prepare($conn, $retentionSql)) {
            mysqli_stmt_bind_param($stmt, "i", $userId);
            mysqli_stmt_execute($stmt);
            mysqli_stmt_bind_result($stmt, $remembered, $reviewed);
            if (mysqli_stmt_fetch($stmt) && $reviewed > 0) {

            }
            mysqli_stmt_close($stmt);
        }

        return [

            'masteredCards' => $mastered,

    /**
     * Compute activity history, streaks, and heatmap data.
     * 
     * @param mysqli $conn Active database connection instance.
     * @param int $userId Authenticated user ID.
     * @return array Streak counters and calendar heatmap data.
     */
    public static function getStreakAndHeatmapData(mysqli $conn, int $userId): array
    {
        $sessionSql = "SELECT DATE(started_at) as session_date, 
                              SUM(remembered_count) as total_remembered, 
                              SUM(total_reviewed) as total_reviewed
                       FROM review_sessions 
                       WHERE user_id = ? AND ended_at IS NOT NULL
                       GROUP BY DATE(started_at)
                       HAVING SUM(total_reviewed) > 0
                       ORDER BY session_date DESC";

        $stmt = mysqli_prepare($conn, $sessionSql);
        if (!$stmt) {
            return self::emptyStreakState();
        }

        mysqli_stmt_bind_param($stmt, "i", $userId);
        mysqli_stmt_execute($stmt);
        $result = mysqli_stmt_get_result($stmt);

        $activityHistory = [];
        while ($row = mysqli_fetch_assoc($result)) {
            $activityHistory[$row['session_date']] = [
                'remembered' => (int) $row['total_remembered'],
                'reviewed' => (int) $row['total_reviewed']
            ];
        }
        mysqli_stmt_close($stmt);

        if (empty($activityHistory)) {
            return self::emptyStreakState();
        }

        // Streak Analytics Processing
        $todayStr = date('Y-m-d');
        $yesterdayStr = date('Y-m-d', strtotime('-1 day'));
        
        $currentStreak = 0;
        $bestStreak = 0;
        $streakAccuracy = 0.0;
        
        $dates = array_keys($activityHistory);
        $latestDate = $dates[0];

        // Active Streak calculation
        if ($latestDate === $todayStr || $latestDate === $yesterdayStr) {
            $checkDate = new DateTime($latestDate);
            $streakRemembered = 0;
            $streakReviewed = 0;

            while (true) {
                $dateKey = $checkDate->format('Y-m-d');
                if (isset($activityHistory[$dateKey])) {
                    $currentStreak++;
                    $streakRemembered += $activityHistory[$dateKey]['remembered'];
                    $streakReviewed += $activityHistory[$dateKey]['reviewed'];
                    $checkDate->modify('-1 day');
                } else {
                    break;
                }
            }

            if ($streakReviewed > 0) {
                $streakAccuracy = round(($streakRemembered / $streakReviewed) * 100, 1);
            }
        }

        // Best Historical Streak calculation
        $tempStreak = 0;
        $expectedDate = null;

        foreach (array_reverse($dates) as $dateStr) {
            $currentDate = new DateTime($dateStr);
            if ($expectedDate === null || $currentDate == $expectedDate) {
                $tempStreak++;
            } else {
                if ($tempStreak > $bestStreak) {
                    $bestStreak = $tempStreak;
                }
                $tempStreak = 1;
            }
            $expectedDate = clone $currentDate;
            $expectedDate->modify('+1 day');
        }

        if ($tempStreak > $bestStreak) {
            $bestStreak = $tempStreak;
        }

        // Heatmap bounds calculation
        $heatmapEnd = new DateTime('today');
        $heatmapStart = (clone $heatmapEnd)->modify('-364 days');
        $heatmapData = [];
        $studyDays = 0;

        foreach ($activityHistory as $dateStr => $metrics) {
            $activityDate = DateTime::createFromFormat('!Y-m-d', $dateStr);
            if ($activityDate && $activityDate >= $heatmapStart && $activityDate <= $heatmapEnd && $metrics['reviewed'] > 0) {
                $heatmapData[$dateStr] = $metrics['reviewed'];
                $studyDays++;
            }
        }

        $calendarStart = clone $heatmapStart;
        $calendarStart->modify('monday this week');

        return [
            'currentStreak' => $currentStreak,
            'bestStreak' => $bestStreak,
            'streakAccuracy' => $streakAccuracy,
            'studyDays' => $studyDays,
            'heatmapData' => $heatmapData,
            'heatmapStart' => $heatmapStart,
            'heatmapEnd' => $heatmapEnd,
            'calendarStart' => $calendarStart
        ];
    }
