```
<?php
session_start();
require_once '../config/db.php';

if ($_SERVER["REQUEST_METHOD"] == "POST") {
    
    $email = trim($_POST['login-email'] ?? '');
    $password = $_POST['login-password'] ?? '';

    if (empty($email) || empty($password)) {
        header("Location: login.php?error=empty_fields");
        exit;
    }

    $sql  = "SELECT id, first_name, last_name, password_hash FROM users WHERE email = ? AND deleted_at IS NULL";
    $stmt = mysqli_prepare($conn, $sql);
    
    mysqli_stmt_bind_param($stmt, "s", $email);
    mysqli_stmt_execute($stmt);
    mysqli_stmt_bind_result($stmt, $user_id, $first_name, $last_name, $hashed_password);
    
    if (mysqli_stmt_fetch($stmt)) {
        if (password_verify($password, $hashed_password)) {
            // Prevent Session Fixation
            session_regenerate_id(true);
            
            $_SESSION['user_id']    = $user_id;
            $_SESSION['first_name'] = $first_name;
            $_SESSION['last_name']  = $last_name;
            $_SESSION['logged_in']  = true;

            mysqli_stmt_close($stmt);
            mysqli_close($conn);

            header("Location: ../app/views/index.php");
            exit;
        }
    }

    mysqli_stmt_close($stmt);
    mysqli_close($conn);
    
    // Invalid credentials back-route
    header("Location: login.php?error=invalid_credentials");
    exit;
}
```

logout.php
```
<?php
// 1. Initialize access to the active session data layer
session_start();

// 2. Unset all internal session state context arrays 
$_SESSION = array();

// 3. Destroy the session cookie on the user client web browser if present
if (ini_get("session.use_cookies")) {
    $params = session_get_cookie_params();
    setcookie(
        session_name(), 
        '', 
        time() - 42000,
        $params["path"], 
        $params["domain"],
        $params["secure"], 
        $params["httponly"]
    );
}

// 4. Wipe session storage data on the server completely
session_destroy();

// 5. Redirect cleanly back to the authentication portal login form view
header("Location: login.php");
exit;
?>
```

```
        die("You must agree to the Terms & Conditions.");
    }

    // 2. Grab inputs
    $first_name = trim($_POST['first-name']); 
    $last_name  = trim($_POST['last-name']);  
    $email      = trim($_POST['email']);
    $password   = $_POST['password'];

    // 3. Validation
    if (empty($first_name) || empty($last_name) || empty($email) || empty($password)) {
        die("All required fields must be filled out.");
    }

    if (strlen($password) < 8) {
        die("Password must be at least 8 characters long.");
    }

    // 4. Check for existing email (Prepare -> Bind -> Execute)
    $check_sql  = "SELECT id FROM users WHERE email = ?";
    $check_stmt = mysqli_prepare($conn, $check_sql);
    
    mysqli_stmt_bind_param($check_stmt, "s", $email); // "s" means 1 string parameter
    mysqli_stmt_execute($check_stmt);
    mysqli_stmt_store_result($check_stmt); // Required to get row count in MySQLi
    
    if (mysqli_stmt_num_rows($check_stmt) > 0) {
        die("Error: This email is already registered.");
    }
    mysqli_stmt_close($check_stmt);

    // 5. Securely hash the password
    $hashed_password = password_hash($password, PASSWORD_BCRYPT);

    // 6. Insert new user (Prepare -> Bind -> Execute)
    $insert_sql  = "INSERT INTO users (first_name, last_name, email, password_hash) VALUES (?, ?, ?, ?)";
    $insert_stmt = mysqli_prepare($conn, $insert_sql);
    
    // "ssss" means 4 string parameters matching the 4 variables
    mysqli_stmt_bind_param($insert_stmt, "ssss", $first_name, $last_name, $email, $hashed_password);
```

```
const authShell = document.querySelector("#auth-shell");
const signupForm = document.querySelector("#signup-form");
const loginForm = document.querySelector("#login-form");
const forgotPasswordLink = document.querySelector("#forgot-password-link");

function setAuthMode(mode) {
    const isLogin = mode === "login";
    authShell.classList.toggle("is-login", isLogin);
    authShell.classList.toggle("is-signup", !isLogin);
    document.querySelector(".login-view").setAttribute("aria-hidden", String(!isLogin));
    document.querySelector(".signup-view").setAttribute("aria-hidden", String(isLogin));
    document.querySelector(".story-copy-login").setAttribute("aria-hidden", String(!isLogin));
    document.querySelector(".story-copy-signup").setAttribute("aria-hidden", String(isLogin));
    window.history.replaceState(null, "", isLogin ? "#login" : "#signup");
}

document.querySelectorAll(".switch-to-login").forEach((button) => button.addEventListener("click", () => setAuthMode("login")));
document.querySelectorAll(".switch-to-signup").forEach((button) => button.addEventListener("click", () => setAuthMode("signup")));
forgotPasswordLink.addEventListener("click", (event) => {
    event.preventDefault();
    window.alert("Forgot password is still in development.");
});

function setAuthModeFromHash() {
    setAuthMode(window.location.hash === "#signup" ? "signup" : "login");
}

window.addEventListener("hashchange", setAuthModeFromHash);
setAuthModeFromHash();
```
