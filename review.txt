    // by the user's saved settings as soon as they load (see loadSrsSettings).
    let srs = {
        remember_multiplier: 2,
        forgot_multiplier: 0.5,
        initial_interval_days: 1,
        maximum_interval_days: 365
    };

    function nextInterval(prevDays, status) {
        const multiplier = status === 'remembered' ? srs.remember_multiplier : srs.forgot_multiplier;
        const days = Math.round(prevDays * multiplier);
        return Math.max(srs.initial_interval_days, Math.min(srs.maximum_interval_days, days));
    }

    const formatDays = (d) => `in ${d} ${d === 1 ? 'day' : 'days'}`;

    function updateProjections() {
        if (practice) return;   // Forgot / Remember day labels are hidden in practice
        const card = queue[currentIndex];
        if (!card) return;
        const interval = parseInt(card.interval_days, 10) || 0;
        projForgot.textContent = formatDays(nextInterval(interval, 'forgotten'));
        projRemember.textContent = formatDays(nextInterval(interval, 'remembered'));
    }

    async function loadSrsSettings() {
        try {
            const res = await fetch('../api/settings-actions.php', { credentials: 'same-origin' });
            const body = await res.json();
            if (res.ok && body.success && body.settings) {
                srs = { ...srs, ...body.settings };
                updateProjections();
            }
        } catch (_) { /* keep the defaults */ }
    }

    async function advanceCard(dir, glow) {

            return;

        const out = stage.animate([
            { opacity: 1, transform: 'translateX(0) rotate(0deg)', boxShadow: `0 0 0 0 ${glow}` },

        renderCard();
        out.cancel();

                { opacity: 0, transform: 'translateY(14px) scale(0.98)' },
                { opacity: 1, transform: 'translateY(0) scale(1)' }
            ], { duration: 240, easing: 'ease-out' });
        }
        isAnimating = false;

    // Again: no rating is saved; a copy of the card comes back a few cards later.
    async function againCard() {
        if (sessionEnded || isAnimating) return;
        const card = queue[currentIndex];

        if (ratings[card.id]) {              // reached with Back: un-rate it
            delete ratings[card.id];
            updateTally();
            unsavedRatings = !practice && Object.keys(ratings).length > 0;
        }

        card._again = true;                  // this slot can be skipped with Next

        // Remove any copy already waiting ahead, then insert a fresh one
        queue = queue.filter((c, i) => i <= currentIndex || c.id !== card.id);
        const insertAt = Math.min(currentIndex + 1 + AGAIN_OFFSET, queue.length);
        queue.splice(insertAt, 0, { ...card, _again: false, _repeat: true });

        return advanceCard(0, 'rgba(245, 158, 11, 0.35)');
    }

    // A number slams in from huge and blurry, squashes on impact, then settles.

            el.style.opacity = '1';
            el.animate([

                { transform: 'scale(1.06) rotate(-1deg)', offset: 0.75 },
                { transform: 'scale(1) rotate(0deg)', offset: 1 }

    let confettiCanvas = null, confettiCtx = null, confettiParts = [], confettiRaf = null;

    function confettiBurst({ x = 0.5, y = 0.4, count = 120, angle = -Math.PI / 2, spread = Math.PI * 2, power = 15 } = {}) {
        if (reduceMotion) return;

        const dpr = window.devicePixelRatio || 1;
        if (!confettiCanvas) {
            confettiCanvas = document.createElement('canvas');
            confettiCanvas.className = 'fixed inset-0 w-full h-full pointer-events-none z-[60]';
            document.body.appendChild(confettiCanvas);
            confettiCtx = confettiCanvas.getContext('2d');
        }
        if (confettiCanvas.width !== Math.round(innerWidth * dpr)) {
            confettiCanvas.width = Math.round(innerWidth * dpr);
            confettiCanvas.height = Math.round(innerHeight * dpr);
        }

        for (let i = 0; i < count; i++) {
            const a = angle + (Math.random() - 0.5) * spread;
            const v = power * (0.5 + Math.random() * 0.7);
            confettiParts.push({
                x: x * innerWidth, y: y * innerHeight,
                vx: Math.cos(a) * v, vy: Math.sin(a) * v,
                w: 6 + Math.random() * 6, h: 4 + Math.random() * 5,
                rot: Math.random() * Math.PI * 2, vr: (Math.random() - 0.5) * 0.35,
                wobble: Math.random() * Math.PI * 2,
                color: CONFETTI_COLORS[(Math.random() * CONFETTI_COLORS.length) | 0],
                life: 0, ttl: 110 + Math.random() * 60
            });
        }
        if (!confettiRaf) confettiRaf = requestAnimationFrame(stepConfetti);
    }

    function stepConfetti() {
        const dpr = window.devicePixelRatio || 1;
        confettiCtx.setTransform(dpr, 0, 0, dpr, 0, 0);
        confettiCtx.clearRect(0, 0, innerWidth, innerHeight);

        confettiParts = confettiParts.filter(p => p.life < p.ttl && p.y < innerHeight + 40);
        for (const p of confettiParts) {
            p.life++;
            p.vx *= 0.985;
            p.vy = p.vy * 0.985 + 0.38;                 // drag + gravity
            p.wobble += 0.12;
            p.x += p.vx + Math.sin(p.wobble) * 0.6;
            p.y += p.vy;
            p.rot += p.vr;

            confettiCtx.save();
            confettiCtx.globalAlpha = Math.min(1, (p.ttl - p.life) / 30);
            confettiCtx.translate(p.x, p.y);
            confettiCtx.rotate(p.rot);
            confettiCtx.scale(1, Math.abs(Math.cos(p.wobble)));   // flutter
            confettiCtx.fillStyle = p.color;
            confettiCtx.fillRect(-p.w / 2, -p.h / 2, p.w, p.h);
            confettiCtx.restore();
        }

        if (confettiParts.length) {
            confettiRaf = requestAnimationFrame(stepConfetti);
        } else {
            confettiRaf = null;
            confettiCanvas.remove();
            confettiCanvas = null;
        }
    }

    /**
     * Get metadata for a specific session.
     */
    public static function getSessionDetails(mysqli $conn, int $sessionId, int $userId): ?array
    {
        $sql = "SELECT rs.*, d.title AS deck_title,
                       TIMESTAMPDIFF(SESerCOND, rs.started_at, rs.ended_at) AS duration_seconds
                FROM review_sessions rs
                JOIN decks d ON rs.deck_id = d.id
                WHERE rs.id = ? AND rs.user_id = ?";

        $session = null;
        if ($stmt = mysqli_prepare($conn, $sql)) {
            mysqli_stmt_bind_param($stmt, "ii", $sessionId, $userId);
            mysqli_stmt_execute($stmt);
            $result = mysqli_stmt_get_result($stmt);
            if ($row = mysqli_fetch_assoc($result)) {
                $session = [
                    'id'               => (int) $row['id'],
                    'deckId'           => (int) $row['deck_id'],
                    'deckTitle'        => (string) $row['deck_title'],
                    'startedAt'        => $row['started_at'],
                    'totalReviewed'    => (int) $row['total_reviewed'],
                    'rememberedCount'  => (int) $row['remembered_count'],
                    // int, because ReviewHubService::accuracyBadgeClass() expects an int
                    'accuracy'         => (int) round((float) $row['accuracy_percentage']),
                    'durationSeconds'  => $row['duration_seconds'] === null ? null : (int) $row['duration_seconds'],
                ];
            }
            mysqli_stmt_close($stmt);
        }
        return $session;
    }

                       f.front, f.back, f.next_review_at, f.deleted_at,
                       TIMESTAMPDIFF(SECOND, NOW(), f.next_review_at) AS next_due_seconds
                FROM review_logs rl
                LEFT JOIN flashcards f ON rl.card_id = f.id
                WHERE rl.session_id = ?
                ORDER BY rl.reviewed_at ASC";

        $logs = [];
        if ($stmt = mysqli_prepare($conn, $sql)) {
            mysqli_stmt_bind_param($stmt, "i", $sessionId);
            mysqli_stmt_execute($stmt);
            $result = mysqli_stmt_get_result($stmt);
            while ($row = mysqli_fetch_assoc($result)) {
                $logs[] = [
                    'front'             => (string) $row['front'],
                    'back'              => (string) $row['back'],
                    'rating'            => (string) $row['rating'],
                    'previousInterval'  => (int) $row['previous_interval'],
                    'newInterval'       => (int) $row['new_interval'],
                    'nextDueSeconds'    => $row['next_due_seconds'] !== null ? (int) $row['next_due_seconds'] : null,
                    'isDeleted'         => $row['deleted_at'] !== null
                ];
            }
            mysqli_stmt_close($stmt);
        }
        return $logs;
    }

    /**
     * Number of completed sessions for a user (used for pagination).
     * Uses the same "completed" rule as the Review Hub (ended_at IS NOT NULL).
     */
    public static function countCompletedSessions(mysqli $conn, int $userId): int
    {
        $sql = "SELECT COUNT(*) FROM review_sessions WHERE user_id = ? AND ended_at IS NOT NULL";

        $total = 0;
        if ($stmt = mysqli_prepare($conn, $sql)) {
            mysqli_stmt_bind_param($stmt, "i", $userId);
            mysqli_stmt_execute($stmt);
            mysqli_stmt_bind_result($stmt, $total);
            mysqli_stmt_fetch($stmt);
            mysqli_stmt_close($stmt);
        }
        return (int) $total;
    }

    /**
     * One page of completed sessions, newest first.
     *
     * @return array<int, array{
     *   id:int, deckId:int, deckTitle:string, startedAt:string,
     *   totalReviewed:int, rememberedCount:int, accuracy:int, durationSeconds:?int
     * }>
     */
    public static function getSessionHistory(mysqli $conn, int $userId, int $limit = 15, int $offset = 0): array
    {
        $limit  = max(1, min($limit, 100));
        $offset = max(0, $offset);

        $sql = "SELECT rs.id, rs.deck_id, d.title AS deck_title, rs.started_at,
                       rs.total_reviewed, rs.remembered_count, rs.accuracy_percentage,
                       TIMESTAMPDIFF(SECOND, rs.started_at, rs.ended_at) AS duration_seconds
                FROM review_sessions rs
                JOIN decks d ON d.id = rs.deck_id
                WHERE rs.user_id = ? AND rs.ended_at IS NOT NULL
                ORDER BY rs.ended_at DESC, rs.id DESC
                LIMIT ? OFFSET ?";

        $sessions = [];
        if ($stmt = mysqli_prepare($conn, $sql)) {
            mysqli_stmt_bind_param($stmt, "iii", $userId, $limit, $offset);
            mysqli_stmt_execute($stmt);
            $result = mysqli_stmt_get_result($stmt);
            while ($row = mysqli_fetch_assoc($result)) {
                $sessions[] = [
                    'id'              => (int) $row['id'],
                    'deckId'          => (int) $row['deck_id'],
                    'deckTitle'       => (string) $row['deck_title'],
                    'startedAt'       => (string) $row['started_at'],
                    'totalReviewed'   => (int) $row['total_reviewed'],
                    'rememberedCount' => (int) $row['remembered_count'],
                    'accuracy'        => (int) round((float) $row['accuracy_percentage']),
                    'durationSeconds' => $row['duration_seconds'] === null ? null : (int) $row['duration_seconds'],
                ];
            }
            mysqli_stmt_close($stmt);
        }
        return $sessions;
    }
