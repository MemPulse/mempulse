<script>
    /**
 * Admin sidebar behavior. Mirrors the inline script in app/includes/sidebar.php:
 * floating tooltips, desktop collapse, mobile drawer, account popover, logout dialog.
 * (Settings modal and review badge are intentionally left out.)
 */
document.addEventListener('DOMContentLoaded', () => {
    const sidebar = document.querySelector('#sidebar');
    if (!sidebar) return;

    const mainContent = document.querySelector('main');
    const toggleBtn = document.querySelector('#sidebar-toggle');
    const toggleIcon = toggleBtn ? toggleBtn.querySelector('.material-symbols-outlined') : null;
    const floatingToggleBtn = document.querySelector('#floating-mobile-toggle');
    const mobileOverlay = document.querySelector('#mobile-sidebar-overlay');
    const profileTrigger = document.querySelector('#profile-popover-trigger');
    const accountPopover = document.querySelector('#settings-popover');

    // ---- Floating tooltip (rendered under <body> so the sidebar cannot clip it)
    let tooltip = document.getElementById('sidebar-floating-tooltip');
    if (!tooltip) {
        tooltip = document.createElement('div');
        tooltip.id = 'sidebar-floating-tooltip';
        tooltip.setAttribute('role', 'tooltip');
        document.body.appendChild(tooltip);
    }

    let tooltipTarget = null;

    function hideTooltip() {
        tooltip.classList.remove('is-visible');
        tooltipTarget = null;
    }

    function showTooltip(target) {
        const label = target.getAttribute('data-tooltip') || target.getAttribute('aria-label');
        if (!label || !sidebar.classList.contains('collapsed') || window.innerWidth < 768) {
            hideTooltip();
            return;
        }
        tooltipTarget = target;
        tooltip.textContent = label;
        tooltip.style.left = '0px';
        tooltip.style.top = '0px';
        tooltip.classList.add('is-visible');

        const rect = target.getBoundingClientRect();
        const tipRect = tooltip.getBoundingClientRect();
        const gap = 12;
        let left = rect.right + gap;

        if (left + tipRect.width > window.innerWidth - 8) {
            left = Math.max(8, rect.left - tipRect.width - gap);
            tooltip.classList.add('tooltip-left');
        } else {
            tooltip.classList.remove('tooltip-left');
        }

        const top = Math.min(window.innerHeight - 8, Math.max(8, rect.top + rect.height / 2));
        tooltip.style.left = `${left}px`;
        tooltip.style.top = `${top}px`;
    }

    sidebar.querySelectorAll('.sidebar-tooltip').forEach((item) => {
        item.addEventListener('mouseenter', () => showTooltip(item));
        item.addEventListener('mouseleave', hideTooltip);
        item.addEventListener('focus', () => showTooltip(item));
        item.addEventListener('blur', hideTooltip);
    });
    sidebar.addEventListener('scroll', hideTooltip, true);
    window.addEventListener('resize', hideTooltip);

    // ---- Account popover
    function setAccountPopoverOpen(isOpen) {
        if (!accountPopover || !profileTrigger) return;
        accountPopover.classList.toggle('hidden', !isOpen);
        profileTrigger.setAttribute('aria-expanded', String(isOpen));
    }

    profileTrigger?.addEventListener('click', (e) => {
        e.stopPropagation();
        setAccountPopoverOpen(accountPopover?.classList.contains('hidden'));
    });

    document.addEventListener('click', (e) => {
        if (accountPopover && !accountPopover.contains(e.target) && !profileTrigger?.contains(e.target)) {
            setAccountPopoverOpen(false);
        }
    });

    document.addEventListener('keydown', (e) => {
        if (e.key === 'Escape') setAccountPopoverOpen(false);
    });

    // ---- Desktop collapse (state is shared with the app via localStorage)
    if (localStorage.getItem('sidebarCollapsed') === 'true' && window.innerWidth >= 768) {
        sidebar.classList.add('collapsed');
        mainContent?.classList.add('collapsed');
        if (toggleIcon) toggleIcon.textContent = 'left_panel_open';
    }

    toggleBtn?.addEventListener('click', () => {
        const isCollapsed = sidebar.classList.toggle('collapsed');
        document.documentElement.classList.toggle('sidebar-collapsed', isCollapsed);
        mainContent?.classList.toggle('collapsed', isCollapsed);
        if (toggleIcon) toggleIcon.textContent = isCollapsed ? 'left_panel_open' : 'left_panel_close';
        hideTooltip();
        setAccountPopoverOpen(false);
        localStorage.setItem('sidebarCollapsed', isCollapsed);
    });

    // ---- Mobile drawer
    function openMobileSidebar() {
        sidebar.classList.remove('-translate-x-full', 'collapsed');
        floatingToggleBtn?.setAttribute('aria-expanded', 'true');
        mobileOverlay?.classList.remove('hidden');
        setTimeout(() => mobileOverlay?.classList.remove('opacity-0'), 10);
    }

    function closeMobileSidebar() {
        sidebar.classList.add('-translate-x-full');
        floatingToggleBtn?.setAttribute('aria-expanded', 'false');
        mobileOverlay?.classList.add('opacity-0');
        setTimeout(() => mobileOverlay?.classList.add('hidden'), 300);
    }

    floatingToggleBtn?.addEventListener('click', openMobileSidebar);
    mobileOverlay?.addEventListener('click', closeMobileSidebar);
    document.addEventListener('keydown', (e) => { if (e.key === 'Escape') closeMobileSidebar(); });

    // ---- Logout dialog
    const logoutBtn = document.getElementById('logout-btn');
    const logoutDialog = document.getElementById('logout-dialog');
    const logoutCancel = document.getElementById('logout-cancel');

    if (logoutBtn && logoutDialog) {
        logoutBtn.addEventListener('click', () => {
            setAccountPopoverOpen(false);
            logoutDialog.showModal();
        });
        logoutCancel?.addEventListener('click', () => logoutDialog.close());
        logoutDialog.addEventListener('click', (e) => {
            const r = logoutDialog.getBoundingClientRect();
            if (e.clientX < r.left || e.clientX > r.right || e.clientY < r.top || e.clientY > r.bottom) {
                logoutDialog.close();
            }
        });
    }
});
</script>

    function makeElement(tag, className, text) {
        const el = document.createElement(tag);
        if (className) el.className = className;
        if (text !== undefined) el.textContent = text;
        return el;
    }

    function showModal() {
        previousFocus = document.activeElement;
        modal.classList.remove('hidden');
        modal.classList.add('flex');
        document.body.classList.add('overflow-hidden');
        input.focus();
    }

    function hideModal() {
        modal.classList.add('hidden');
        modal.classList.remove('flex');
        document.body.classList.remove('overflow-hidden');

        if (requestController) {
            requestController.abort();
            requestController = null;
        }

        if (previousFocus && typeof previousFocus.focus === 'function') {
            previousFocus.focus();
        }
    }

    function showMessage(message) {
        results.replaceChildren(
            makeElement(
                'div',
                'px-3 py-8 text-center text-sm text-on-surface-variant',
                message
            )
        );
    }

    function renderResults(data) {
        results.replaceChildren();

        let total = 0;

        sections.forEach(section => {
            const items = Array.isArray(data[section.key])
                ? data[section.key]
                : [];

            if (!items.length) return;

            total += items.length;

            const group = makeElement('section', 'mb-3 last:mb-0');
            const heading = makeElement(
                'h3',
                'px-3 pb-2 pt-3 text-[11px] font-bold uppercase tracking-wider text-on-surface-variant',
                section.label
            );

            group.appendChild(heading);

            items.forEach(item => {
                if (
                    !item ||
                    typeof item.title !== 'string' ||
                    typeof item.detail !== 'string' ||
                    typeof item.url !== 'string'
                ) {
                    return;
                }

                const link = makeElement(
                    'a',
                    'flex items-start gap-3 rounded-xl px-3 py-3 hover:bg-surface-container-low focus:bg-surface-container-low focus:outline-none'
                );

                // Use only same-origin paths returned by this admin API.
                let destination;
                try {
                    destination = new URL(item.url, window.location.origin);
                } catch (_) {
                    return;
                }

                if (destination.origin !== window.location.origin) return;

                const icon = makeElement(
                    'span',
                    'material-symbols-outlined mt-0.5 text-[22px] text-primary',
                    section.icon
                );

                const textWrap = makeElement('div', 'min-w-0 flex-1');
                const title = makeElement(
                    'div',
                    'truncate text-sm font-semibold text-on-surface',
                    item.title
                );
                const detail = makeElement(
                    'div',
                    'mt-1 text-xs leading-relaxed text-on-surface-variant',
                    item.detail
                );

                textWrap.append(title, detail);
                link.append(icon, textWrap);
                link.href = destination.pathname + destination.search;
                group.appendChild(link);
            });

            results.appendChild(group);
        });

        status.textContent = total
            ? total + ' result' + (total === 1 ? '' : 's') + ' shown'
            : 'No matching results found in the 4 sections.';

        if (!total) {
            showMessage('No matches found. Try another name, email, session ID, action, or IP address.');
        }
    }

    async function runSearch() {
        const query = input.value.trim();
        const currentRequest = ++requestNumber;

        if (requestController) requestController.abort();

        if (query.length < 2) {
            status.textContent = 'Type at least 2 characters to search all 4 sections.';
            showMessage('Search across Users, Content, Activity, and Audit Log.');
            return;
        }

        requestController = new AbortController();
        status.textContent = 'Searching all 4 sections...';
        showMessage('Searching...');

        try {
            const endpoint = modal.dataset.searchEndpoint;
            const url = new URL(endpoint, window.location.origin);
            url.searchParams.set('q', query);

            if (url.origin !== window.location.origin) {
                throw new Error('Invalid search endpoint.');
            }

            const response = await fetch(url.href, {
                method: 'GET',
                credentials: 'same-origin',
                headers: { 'Accept': 'application/json' },
                signal: requestController.signal
            });

            if (!response.ok) {
                throw new Error('Search request failed.');
            }

            const data = await response.json();

            if (currentRequest !== requestNumber) return;
            if (data.error) throw new Error(data.error);

            renderResults(data);
        } catch (error) {
            if (error.name === 'AbortError') return;
            status.textContent = 'Search could not be completed.';
            showMessage('Something went wrong. Check the browser console and PHP error log.');
        }
    }

    trigger.addEventListener('click', showModal);
    closeBtn?.addEventListener('click', hideModal);

    modal.addEventListener('click', event => {
        if (event.target === modal) hideModal();
    });

    input.addEventListener('input', () => {
        clearTimeout(debounceTimer);
        debounceTimer = setTimeout(runSearch, 250);
    });

    document.addEventListener('keydown', event => {
        if ((event.ctrlKey || event.metaKey) && event.key.toLowerCase() === 'k') {
            event.preventDefault();

            if (modal.classList.contains('hidden')) {
                showModal();
            } else {
                input.focus();
            }
        }

        if (event.key === 'Escape' && !modal.classList.contains('hidden')) {
            hideModal();
        }
    });
});

    /**
     * Daily series for the Activity chart (Remembered cards, Forgot cards, Retention %).
     *
     * @param int $days One of self::RANGES (defaults to 90 for client-side slicing).
     * @return array Keys: labels, remembered, forgot, retention.

    public static function getTrend(mysqli $conn, int $days = 90): array
    {
        $days = in_array($days, self::RANGES, true) ? $days : 90;
        $out  = ['labels' => [], 'remembered' => [], 'forgot' => [], 'retention' => []];

        $today = null;
        if ($res = mysqli_query($conn, 'SELECT CURDATE()')) {
            $row   = mysqli_fetch_row($res);
            $today = $row[0] ?? null;
            mysqli_free_result($res);
        }
        if (!$today) {
            return $out;
        }

        $offset = $days - 1;
        $rows = self::fetchRows(
            $conn,
            "SELECT DATE(started_at) AS d,
                    COALESCE(SUM(remembered_count), 0) AS rem,
                    COALESCE(SUM(GREATEST(total_reviewed - remembered_count, 0)), 0) AS fgt,
                    COALESCE(ROUND(SUM(remembered_count) / NULLIF(SUM(total_reviewed), 0) * 100, 1), 0) AS ret
               FROM review_sessions
              WHERE ended_at IS NOT NULL AND started_at >= CURDATE() - INTERVAL ? DAY
              GROUP BY d",
            'i',
            [$offset]
        );

        $byDay = [];
        foreach ($rows as $r) {
            $byDay[$r['d']] = [
                'rem' => (int) $r['rem'],
                'fgt' => (int) $r['fgt'],
                'ret' => (float) $r['ret'],
            ];
        }

        $end = new DateTimeImmutable($today);
        for ($i = $offset; $i >= 0; $i--) {
            $key = $end->modify("-$i days")->format('Y-m-d');
            $out['labels'][]     = $key;
            $out['remembered'][] = $byDay[$key]['rem'] ?? 0;
            $out['forgot'][]     = $byDay[$key]['fgt'] ?? 0;

    private static function fetchRows(mysqli $conn, string $sql, string $types = '', array $params = []): array
    {
        $rows = [];
        $stmt = mysqli_prepare($conn, $sql);
        if (!$stmt) {
            return $rows;
        }
        if ($types !== '') {
            mysqli_stmt_bind_param($stmt, $types, ...$params);
        }
        if (mysqli_stmt_execute($stmt)) {
            $res = mysqli_stmt_get_result($stmt);
            while ($res && ($row = mysqli_fetch_assoc($res))) {
                $rows[] = $row;
            }
        }
        mysqli_stmt_close($stmt);
        return $rows;
    }

    /** @return array{0:string,1:string,2:array} */
    private static function buildWhere(array $f): array
    {
        $where  = ['1=1'];
        $types  = '';
        $params = [];

        if ($f['q'] !== '') {
            $like    = '%' . addcslashes($f['q'], '%_\\') . '%';
            $where[] = "(a.target_name LIKE ? OR a.target_email LIKE ? OR a.ip_address LIKE ? OR CONCAT(COALESCE(adm.first_name,''), ' ', COALESCE(adm.last_name,'')) LIKE ?)";
            $types  .= 'ssss';
            array_push($params, $like, $like, $like, $like);
        }
        if ($f['action'] !== '') {
            $where[]  = 'a.action = ?';
            $types   .= 's';
            $params[] = $f['action'];
        }
        if ($f['admin_id'] > 0) {
            $where[]  = 'a.admin_id = ?';
            $types   .= 'i';
            $params[] = $f['admin_id'];
        }
        if ($f['from'] !== '') {
            $where[]  = 'a.created_at >= ?';
            $types   .= 's';
            $params[] = $f['from'] . ' 00:00:00';
        }
        if ($f['to'] !== '') {
            $where[]  = 'a.created_at < ?';
            $types   .= 's';
            $params[] = (new DateTimeImmutable($f['to']))->modify('+1 day')->format('Y-m-d') . ' 00:00:00';
        }

        return [implode(' AND ', $where), $types, $params];
    }

    private static function fetchRows(mysqli $conn, string $sql, string $types = '', array $params = []): array
    {
        $rows = [];
        $stmt = mysqli_prepare($conn, $sql);
        if (!$stmt) {
            return $rows;
        }
        if ($types !== '') {
            mysqli_stmt_bind_param($stmt, $types, ...$params);
        }
        if (mysqli_stmt_execute($stmt)) {
            $res = mysqli_stmt_get_result($stmt);
            while ($res && ($row = mysqli_fetch_assoc($res))) {
                $rows[] = $row;
            }
        }
        mysqli_stmt_close($stmt);
        return $rows;
    }

/**

 *

    {
        $empty = [

            'activeToday' => 0, 'active7d' => 0, 'decks' => 0, 'cards' => 0,
            'reviews7d' => 0, 'reviewsPrev7d' => 0, 'retention30d' => 0.0,
        ];

        $live = "deleted_at IS NULL AND account_status = 'approved'";
        $sql = "SELECT
            (SELECT COUNT(*) FROM users WHERE $live) AS total_users,
            (SELECT COUNT(*) FROM users WHERE $live AND created_at >= NOW() - INTERVAL 7 DAY) AS new_users_7d,
            (SELECT COUNT(*) FROM users WHERE deleted_at IS NULL AND account_status = 'pending') AS pending_users,
            (SELECT COUNT(DISTINCT user_id) FROM review_sessions WHERE started_at >= CURDATE()) AS active_today,
            (SELECT COUNT(DISTINCT user_id) FROM review_sessions WHERE started_at >= NOW() - INTERVAL 7 DAY) AS active_7d,
            (SELECT COUNT(*) FROM decks d JOIN users u ON u.id = d.user_id

            (SELECT COUNT(*) FROM flashcards f JOIN decks d ON d.id = f.deck_id JOIN users u ON u.id = d.user_id

            (SELECT COALESCE(ROUND(SUM(remembered_count) / NULLIF(SUM(total_reviewed), 0) * 100, 1), 0)
               FROM review_sessions
              WHERE ended_at IS NOT NULL AND started_at >= NOW() - INTERVAL 30 DAY) AS retention_30d";

        $res = mysqli_query($conn, $sql); // static SQL, no user input

        if (!$r) {

    /**
     * Daily series with empty days filled so the chart axis is continuous.
     *
     * @param int $days One of self::RANGES (anything else falls back to 30).
     * @return array Keys: labels (Y-m-d), reviews, activeUsers, signups (parallel int arrays).
     */
    public static function getTrend(mysqli $conn, int $days = 30): array
    {
        $days = in_array($days, self::RANGES, true) ? $days : 30;
        $out = ['labels' => [], 'reviews' => [], 'activeUsers' => [], 'signups' => []];

        $today = null;
        if ($res = mysqli_query($conn, 'SELECT CURDATE()')) {
            $row = mysqli_fetch_row($res);
            $today = $row[0] ?? null;
            mysqli_free_result($res);
        }
        if (!$today) {
            return $out;
        }

        $offset = $days - 1;
        $sessions = self::fetchRows($conn,
            "SELECT DATE(started_at) AS d, SUM(total_reviewed) AS reviews, COUNT(DISTINCT user_id) AS active
             FROM review_sessions
             WHERE ended_at IS NOT NULL AND started_at >= CURDATE() - INTERVAL ? DAY
             GROUP BY d", 'i', [$offset]);
        $signups = self::fetchRows($conn,
            "SELECT DATE(created_at) AS d, COUNT(*) AS signups
             FROM users
             WHERE deleted_at IS NULL AND created_at >= CURDATE() - INTERVAL ? DAY
             GROUP BY d", 'i', [$offset]);

        $byDay = [];
        foreach ($sessions as $s) {
            $byDay[$s['d']]['reviews'] = (int) $s['reviews'];
            $byDay[$s['d']]['active']  = (int) $s['active'];
        }
        foreach ($signups as $s) {
            $byDay[$s['d']]['signups'] = (int) $s['signups'];
        }

        $end = new DateTimeImmutable($today);
        for ($i = $offset; $i >= 0; $i--) {
            $key = $end->modify("-$i days")->format('Y-m-d');
            $out['labels'][]      = $key;
            $out['reviews'][]     = $byDay[$key]['reviews'] ?? 0;
            $out['activeUsers'][] = $byDay[$key]['active'] ?? 0;
            $out['signups'][]     = $byDay[$key]['signups'] ?? 0;
        }
        return $out;
    }

    public static function getRecentSessions(mysqli $conn, int $limit = 6): array
    {
        $limit = max(1, min($limit, 50));

                    TIMESTAMPDIFF(SECOND, rs.started_at, NOW()) AS age_seconds
             FROM review_sessions rs
             JOIN users u ON u.id = rs.user_id

    /**
     * Percentage change chip data, or null when there is nothing to compare.
     *
     * @return array{text:string,tone:string}|null tone is up, down or flat.
     */
    public static function percentChip(int $current, int $previous, string $suffix): ?array
    {
        if ($previous === 0) {
            return $current > 0 ? ['text' => 'New activity', 'tone' => 'up'] : null;
        }
        $pct = (int) round((($current - $previous) / $previous) * 100);
        if ($pct === 0) {
            return ['text' => '0% ' . $suffix, 'tone' => 'flat'];
        }
        return ['text' => ($pct > 0 ? '+' : '') . $pct . '% ' . $suffix, 'tone' => $pct > 0 ? 'up' : 'down'];
    }

    /** Human-friendly relative time from a seconds-ago value computed in SQL. */
    public static function timeAgo(int $seconds): string
    {
        if ($seconds < 60)    return 'Just now';
        if ($seconds < 3600)  return floor($seconds / 60) . ' min ago';
        if ($seconds < 86400) return floor($seconds / 3600) . ' hr ago';
        $d = (int) floor($seconds / 86400);
        return $d . ($d === 1 ? ' day ago' : ' days ago');
    }

    private static function userRows(mysqli $conn, string $extraWhere, string $order, int $limit): array
    {
        $limit = max(1, min($limit, 50));
        // $extraWhere and $order are fixed strings from the two callers above, never user input.
        $rows = self::fetchRows($conn,
            "SELECT u.id, u.first_name, u.last_name, u.email, u.role, u.account_status,
                    TIMESTAMPDIFF(SECOND, u.created_at, NOW()) AS age_seconds
             FROM users u
             WHERE u.deleted_at IS NULL $extraWhere
             ORDER BY $order LIMIT ?", 'i', [$limit]);

        $out = [];
        foreach ($rows as $r) {
            $out[] = [
                'id'     => (int) $r['id'],
                'name'   => self::displayName($r),
                'email'  => (string) $r['email'],
                'role'   => (string) $r['role'],
                'status' => (string) $r['account_status'],
                'age'    => self::timeAgo((int) $r['age_seconds']),
            ];
        }
        return $out;
    }

    private static function displayName(array $r): string
    {
        $n = trim(($r['first_name'] ?? '') . ' ' . ($r['last_name'] ?? ''));
        return $n !== '' ? $n : (string) ($r['email'] ?? 'Unknown user');
    }

    private static function fetchRows(mysqli $conn, string $sql, string $types = '', array $params = []): array
    {
        $rows = [];
        $stmt = mysqli_prepare($conn, $sql);
        if (!$stmt) {
            return $rows;
        }
        if ($types !== '') {
            mysqli_stmt_bind_param($stmt, $types, ...$params);
        }
        if (mysqli_stmt_execute($stmt)) {
            $res = mysqli_stmt_get_result($stmt);
            while ($res && ($row = mysqli_fetch_assoc($res))) {
                $rows[] = $row;
            }
        }
        mysqli_stmt_close($stmt);
        return $rows;
    }

<?php
/**
 * AdminUserActionsService
 *
 * Server-enforced user management mutations:
 *  - Single & bulk actions: approve, reject, delete (soft-delete), restore, make_admin, make_user
 *  - Direct user creation by an admin
 *  - Prevents self-lockout / self-modification
 *  - Writes every action atomically to admin_audit_log
 */
class AdminUserActionsService
{
    public const ACTIONS = [
        'approve',
        'reject',
        'delete',
        'restore',
        'make_admin',
        'make_user',
    ];

    private const AUDIT_NAMES = [
        'approve'    => 'user_approved',
        'reject'     => 'user_rejected',
        'delete'     => 'user_deleted',
        'restore'    => 'user_restored',
        'make_admin' => 'role_promoted_admin',
        'make_user'  => 'role_demoted_user',
        'create'     => 'user_created_by_admin',
    ];

    /**
     * Allowed actions for a user row (shared between UI and server enforcement).
     */
    public static function allowedActions(string $role, string $status, bool$isDeleted = false): array
    {
        if ($isDeleted || $status === 'deleted') {
            return ['restore'];
        }

        $actions = [];
        switch ($status) {
            case 'pending':
                $actions = ['approve', 'reject', 'delete'];
                break;
            case 'approved':
                $actions = ['reject', 'delete'];
                break;
            case 'rejected':
                $actions = ['approve', 'delete'];
                break;
        }

        if ($status === 'approved') {
            $actions[] = ($role === 'admin') ? 'make_user' : 'make_admin';
        }

        return $actions;
    }

    /**
     * Perform an action on a single user.
     */
    public static function perform(mysqli $conn, int $actorId, int$targetId, string $action, string$ip = ''): array
    {
        if (!in_array($action, self::ACTIONS, true) || $targetId <= 0) {
            return self::fail(400, 'That request is not valid.');
        }
        if ($targetId ===$actorId) {
            return self::fail(403, 'You cannot change or delete your own account.');
        }

        try {
            mysqli_begin_transaction($conn);

            $stmt = mysqli_prepare($conn,
                'SELECT first_name, last_name, email, role, account_status, deleted_at
                   FROM users WHERE id = ? FOR UPDATE'
            );
            if (!$stmt) {
                throw new RuntimeException('prepare failed (select)');
            }
            mysqli_stmt_bind_param($stmt, 'i',$targetId);
            mysqli_stmt_execute($stmt);
            mysqli_stmt_bind_result($stmt, $first,$last, $email,$role, $status,$deletedAt);
            $found = mysqli_stmt_fetch($stmt);
            mysqli_stmt_close($stmt);

            if (!$found) {
                mysqli_rollback($conn);
                return self::fail(404, 'That account does not exist.');
            }

            $isDeleted = ($deletedAt !== null);
            $allowed   = self::allowedActions((string)$role, (string) $status,$isDeleted);

            if (!in_array($action,$allowed, true)) {
                mysqli_rollback($conn);
                return self::fail(409, 'This action is not valid for the account\'s current state. Refresh the page.');
            }

            // Prevent deleting or rejecting the last approved admin
            if ($role === 'admin' && in_array($action, ['reject', 'delete', 'make_user'], true)) {
                $admRes = mysqli_query($conn, "SELECT COUNT(*) FROM users WHERE role = 'admin' AND account_status = 'approved' AND deleted_at IS NULL");
                $admRow = $admRes ? mysqli_fetch_row($admRes) : [0];
                if ($admRes) mysqli_free_result($admRes);
                if ((int) ($admRow[0] ?? 0) <= 1) {
                    mysqli_rollback($conn);
                    return self::fail(409, 'Cannot remove the last active administrator.');
                }
            }

            $prevState = $isDeleted ? 'deleted' : (string) $status;
            $newState  =$prevState;

            if ($action === 'delete') {
                $sql      = 'UPDATE users SET deleted_at = NOW() WHERE id = ?';$newState = 'deleted';
            } elseif ($action === 'restore') {$sql      = 'UPDATE users SET deleted_at = NULL WHERE id = ?';
                $newState = (string)$status;
            } elseif ($action === 'approve' || $action === 'reject') {$newState = $action === 'approve' ? 'approved' : 'rejected';$sql      = "UPDATE users SET account_status = '$newState' WHERE id = ?";
            } elseif ($action === 'make_admin' || $action === 'make_user') {
                $newRole  =$action === 'make_admin' ? 'admin' : 'user';
                $prevState = 'role:' .$role;
                $newState  = 'role:' .$newRole;
                $sql       = "UPDATE users SET role = '$newRole' WHERE id = ?";
            }

            $up = mysqli_prepare($conn,$sql);
            if (!$up) {
                throw new RuntimeException('prepare failed (update)');
            }
            mysqli_stmt_bind_param($up, 'i',$targetId);
            mysqli_stmt_execute($up);
            mysqli_stmt_close($up);

            $name      = trim($first . ' ' .$last);
            $name      =$name !== '' ? $name :$email;
            $auditName = self::AUDIT_NAMES[$action];
            $ip        = mb_substr($ip, 0, 45);

            $log = mysqli_prepare($conn,
                'INSERT INTO admin_audit_log
                    (admin_id, action, target_user_id, target_email, target_name, previous_status, new_status, ip_address)
                 VALUES (?, ?, ?, ?, ?, ?, ?, ?)'
            );
            if (!$log) {
                throw new RuntimeException('prepare failed (audit)');
            }
            mysqli_stmt_bind_param($log, 'isisssss', $actorId,$auditName, $targetId,$email, $name,$prevState, $newState,$ip);
            mysqli_stmt_execute($log);
            mysqli_stmt_close($log);

            mysqli_commit($conn);
        } catch (Throwable $e) {
            @mysqli_rollback($conn);
            error_log('AdminUserActionsService::perform failed: ' . $e->getMessage());
            return self::fail(500, 'Something went wrong and nothing was changed. Please try again.');
        }

        $messages = [
            'approve'    => "Approved {$name}.",
            'reject'     => "Suspended {$name}.",
            'delete'     => "Archived {$name}.",
            'restore'    => "Restored {$name}.",
            'make_admin' => "Promoted {$name} to Admin.",
            'make_user'  => "Changed {$name} to regular User.",
        ];
        return ['ok' => true, 'code' => 200, 'message' => $messages[$action], 'status' =>$newState];
    }

    /**
     * Perform a bulk action on multiple user IDs.
     */
    public static function performBulk(mysqli $conn, int $actorId, array$targetIds, string $action, string$ip = ''): array
    {
        if (!in_array($action, ['approve', 'reject', 'delete', 'restore'], true)) {
            return self::fail(400, 'Invalid bulk action.');
        }

        $ids = array_values(array_unique(array_filter(array_map('intval',$targetIds), fn($id) =>$id > 0 && $id !==$actorId)));
        if (!$ids) {
            return self::fail(400, 'No valid accounts selected (you cannot modify your own account).');
        }

        $succeeded = 0;
        foreach ($ids as$id) {
            $res = self::perform($conn, $actorId,$id, $action,$ip);
            if ($res['ok']) {$succeeded++;
            }
        }

        if ($succeeded === 0) {
            return self::fail(409, 'None of the selected accounts could be updated.');
        }

        $verb = [ 'approve' => 'Approved', 'reject'  => 'Suspended', 'delete'  => 'Archived', 'restore' => 'Restored', ][$action];

        return [
            'ok'      => true,
            'code'    => 200,
            'message' => "{$verb} {$succeeded} " . ($succeeded === 1 ? 'account.' : 'accounts.'),
        ];
    }

    /**
     * Create a new user account directly from the Admin Users page.
     */
    public static function createUser(mysqli $conn, int$actorId, array $data, string$ip = ''): array
    {
        $first    = trim((string) ($data['first_name'] ?? ''));
        $last     = trim((string) ($data['last_name'] ?? ''));
        $email    = trim((string) ($data['email'] ?? ''));
        $password = (string) ($data['password'] ?? '');
        $role     = in_array($data['role'] ?? 'user', ['admin', 'user'], true) ? $data['role'] : 'user';$status   = in_array($data['status'] ?? 'approved', ['approved', 'pending'], true) ?$data['status'] : 'approved';

        if ($first === '' || $last === '' || $email === '' || $password === '') {
            return self::fail(400, 'All fields are required.');
        }
        if (!filter_var($email, FILTER_VALIDATE_EMAIL)) {
            return self::fail(400, 'Please enter a valid email address.');
        }
        if (strlen($password) < 8) {
            return self::fail(400, 'Password must be at least 8 characters long.');
        }

        try {
            mysqli_begin_transaction($conn);

            $chk = mysqli_prepare($conn, 'SELECT id FROM users WHERE email = ? LIMIT 1');
            mysqli_stmt_bind_param($chk, 's',$email);
            mysqli_stmt_execute($chk);
            mysqli_stmt_store_result($chk);
            if (mysqli_stmt_num_rows($chk) > 0) {
                mysqli_stmt_close($chk);
                mysqli_rollback($conn);
                return self::fail(409, 'An account with that email already exists.');
            }
            mysqli_stmt_close($chk);

            $hash = password_hash($password, PASSWORD_BCRYPT);
            $ins  = mysqli_prepare($conn,
                'INSERT INTO users (first_name, last_name, email, password_hash, role, account_status)
                 VALUES (?, ?, ?, ?, ?, ?)'
            );
            mysqli_stmt_bind_param($ins, 'ssssss', $first,$last, $email,$hash, $role,$status);
            mysqli_stmt_execute($ins);
            $newId = (int) mysqli_insert_id($conn);
            mysqli_stmt_close($ins);

            $set = mysqli_prepare($conn, 'INSERT IGNORE INTO user_settings (user_id) VALUES (?)');
            if ($set) {
                mysqli_stmt_bind_param($set, 'i',$newId);
                mysqli_stmt_execute($set);
                mysqli_stmt_close($set);
            }

            $name      = trim($first . ' ' .$last);
            $auditName = self::AUDIT_NAMES['create'];$ip        = mb_substr($ip, 0, 45);$prev      = null;

            $log = mysqli_prepare($conn,
                'INSERT INTO admin_audit_log
                    (admin_id, action, target_user_id, target_email, target_name, previous_status, new_status, ip_address)
                 VALUES (?, ?, ?, ?, ?, ?, ?, ?)'
            );
            mysqli_stmt_bind_param($log, 'isisssss', $actorId,$auditName, $newId,$email, $name,$prev, $status,$ip);
            mysqli_stmt_execute($log);
            mysqli_stmt_close($log);

            mysqli_commit($conn);
            return ['ok' => true, 'code' => 200, 'message' => "Created account for {$name}."];
        } catch (Throwable $e) {
            @mysqli_rollback($conn);
            error_log('AdminUserActionsService::createUser failed: ' . $e->getMessage());
            return self::fail(500, 'Unable to create the account.');
        }
    }

<?php
/**
 * AdminUsersService
 *
 * Industry-standard read & query service for the Admin Users page.
 * Supports active & soft-deleted accounts, role/status/date/search filters,
 * multi-column sorting, per-user SRS aggregate metrics, and per-user audit history.
 */
class AdminUsersService
{
    public const PER_PAGE = 15;
    public const STATUSES = ['pending', 'approved', 'rejected', 'deleted'];
    public const ROLES    = ['admin', 'user'];

    /** Sort key => SQL expression (strict allow-list). */
    private const SORTS = [
        'name'    => 'u.first_name',
        'email'   => 'u.email',
        'role'    => 'u.role',
        'status'  => 'u.account_status',
        'decks'   => 'decks',
        'reviews' => 'reviews',
        'joined'  => 'u.created_at',
    ];

    /**
     * Validate and normalise GET input.
     */
    public static function parseFilters(array $get): array
    {
        $str  = fn($k) => is_string($get[$k] ?? null) ? trim($get[$k]) : '';
        $date = function (string $v): string {
            $d = DateTimeImmutable::createFromFormat('!Y-m-d', $v);
            return ($d && $d->format('Y-m-d') === $v) ? $v : '';
        };
        $sort = $str('sort');

        return [
            'q'      => mb_substr($str('q'), 0, 100),
            'status' => in_array($str('status'), self::STATUSES, true) ? $str('status') : '',
            'role'   => in_array($str('role'), self::ROLES, true) ? $str('role') : '',
            'from'   => $date($str('from')),
            'to'     => $date($str('to')),
            'sort'   => isset(self::SORTS[$sort]) ? $sort : 'joined',
            'dir'    => $str('dir') === 'asc' ? 'asc' : 'desc',
            'page'   => max(1, (int) ($get['page'] ?? 1)),
        ];
    }

    /** Query string builder keeping only non-default parameters. */
    public static function buildQuery(array $f, array $override = []): string
    {
        $f = array_merge($f, $override);
        $q = array_filter([
            'q'      => $f['q'],
            'status' => $f['status'],
            'role'   => $f['role'],
            'from'   => $f['from'],
            'to'     => $f['to'],
            'sort'   => $f['sort'] !== 'joined' ? $f['sort'] : '',
            'dir'    => $f['dir'] !== 'desc' ? $f['dir'] : '',
            'page'   => $f['page'] > 1 ? $f['page'] : '',
        ], fn($v) => $v !== '');
        return $q ? '?' . http_build_query($q) : '';
    }

    /**
     * Account counts per status tab + headline KPIs.
     */
    public static function getStatusCounts(mysqli $conn): array
    {
        $out = [
            'all'      => 0,
            'pending'  => 0,
            'approved' => 0,
            'rejected' => 0,
            'deleted'  => 0,
            'admins'   => 0,
            'active7d' => 0,
        ];

        $rows = self::fetchRows(
            $conn,
            "SELECT
                SUM(CASE WHEN deleted_at IS NULL THEN 1 ELSE 0 END) AS all_live,
                SUM(CASE WHEN deleted_at IS NULL AND account_status = 'pending' THEN 1 ELSE 0 END) AS pending,
                SUM(CASE WHEN deleted_at IS NULL AND account_status = 'approved' THEN 1 ELSE 0 END) AS approved,
                SUM(CASE WHEN deleted_at IS NULL AND account_status = 'rejected' THEN 1 ELSE 0 END) AS rejected,
                SUM(CASE WHEN deleted_at IS NOT NULL THEN 1 ELSE 0 END) AS deleted,
                SUM(CASE WHEN deleted_at IS NULL AND role = 'admin' THEN 1 ELSE 0 END) AS admins,
                (SELECT COUNT(DISTINCT rs.user_id)
                   FROM review_sessions rs
                   JOIN users u2 ON u2.id = rs.user_id
                  WHERE u2.deleted_at IS NULL AND rs.started_at >= NOW() - INTERVAL 7 DAY) AS active_7d
             FROM users"

        if ($rows) {
            $r = $rows[0];
            $out['all']      = (int) ($r['all_live'] ?? 0);
            $out['pending']  = (int) ($r['pending'] ?? 0);
            $out['approved'] = (int) ($r['approved'] ?? 0);

            $out['deleted']  = (int) ($r['deleted'] ?? 0);
            $out['admins']   = (int) ($r['admins'] ?? 0);
            $out['active7d'] = (int) ($r['active_7d'] ?? 0);

    /**
     * Paginated (or unpaginated for CSV export) user list with per-user aggregates.
     */
    public static function getList(mysqli $conn, array $f, bool $exportAll = false): array
    {
        $empty = [
            'available' => false, 'rows' => [], 'total' => 0,
            'page' => 1, 'pages' => 1, 'first' => 0, 'last' => 0,
        ];

        [$where, $types, $params] = self::buildWhere($f);

        $c = self::fetchRows($conn, "SELECT COUNT(*) AS n FROM users u WHERE $where", $types, $params);
        if (!$c) {
            return $empty;
        }
        $total  = (int) $c[0]['n'];
        $pages  = max(1, (int) ceil($total / self::PER_PAGE));
        $page   = min($f['page'], $pages);
        $offset = ($page - 1) * self::PER_PAGE;

        $col  = self::SORTS[$f['sort']] ?? 'u.created_at';
        $dir  = $f['dir'] === 'asc' ? 'ASC' : 'DESC';
        $done = "s.user_id = u.id AND s.ended_at IS NOT NULL";

        $limitSql = $exportAll ? '' : 'LIMIT ? OFFSET ?';
        $sql = "SELECT u.id, u.first_name, u.last_name, u.email, u.role, u.account_status,
                       u.created_at, u.deleted_at,
                  (SELECT COUNT(*) FROM decks d WHERE d.user_id = u.id AND d.deleted_at IS NULL) AS decks,
                  (SELECT COUNT(*) FROM flashcards fc JOIN decks d ON d.id = fc.deck_id
                    WHERE d.user_id = u.id AND d.deleted_at IS NULL AND fc.deleted_at IS NULL) AS cards,
                  (SELECT COUNT(*) FROM review_sessions s WHERE $done) AS sessions,
                  (SELECT COALESCE(SUM(s.total_reviewed), 0) FROM review_sessions s WHERE $done) AS reviews,
                  (SELECT COALESCE(ROUND(SUM(s.remembered_count) / NULLIF(SUM(s.total_reviewed), 0) * 100, 1), 0)
                     FROM review_sessions s WHERE $done) AS accuracy,
                  (SELECT TIMESTAMPDIFF(SECOND, MAX(s.started_at), NOW())
                     FROM review_sessions s WHERE s.user_id = u.id) AS last_active_seconds
                FROM users u
                WHERE $where
                ORDER BY $col $dir, u.id DESC
                $limitSql";

        $queryTypes  = $exportAll ? $types : $types . 'ii';
        $queryParams = $exportAll ? $params : array_merge($params, [self::PER_PAGE, $offset]);
        $rows        = self::fetchRows($conn, $sql, $queryTypes, $queryParams);

        // Fetch recent audit logs for the returned users in one query
        $userIds   = array_map(fn($r) => (int) $r['id'], $rows);
        $auditByUser = self::getAuditHistoryForUsers($conn, $userIds);

        $out = [];
        foreach ($rows as $r) {
            $id        = (int) $r['id'];
            $name      = trim($r['first_name'] . ' ' . $r['last_name']);
            $isDeleted = $r['deleted_at'] !== null;
            $effStatus = $isDeleted ? 'deleted' : (string) $r['account_status'];

            $out[] = [
                'id'         => $id,
                'firstName'  => (string) $r['first_name'],
                'lastName'   => (string) $r['last_name'],
                'name'       => $name !== '' ? $name : (string) $r['email'],
                'email'      => (string) $r['email'],
                'role'       => (string) $r['role'],
                'status'     => $effStatus,
                'rawStatus'  => (string) $r['account_status'],
                'isDeleted'  => $isDeleted,
                'joined'     => date('M j, Y', strtotime($r['created_at'])),
                'joinedIso'  => date('Y-m-d', strtotime($r['created_at'])),
                'deletedAt'  => $isDeleted ? date('M j, Y', strtotime($r['deleted_at'])) : null,
                'decks'      => (int) $r['decks'],
                'cards'      => (int) $r['cards'],
                'sessions'   => (int) $r['sessions'],
                'reviews'    => (int) $r['reviews'],
                'accuracy'   => (float) $r['accuracy'],
                'lastActive' => $r['last_active_seconds'] === null
                    ? 'Never'
                    : AdminOverviewService::timeAgo((int) $r['last_active_seconds']),
                'audit'      => $auditByUser[$id] ?? [],
            ];
        }

        return [
            'available' => true,
            'rows'      => $out,
            'total'     => $total,
            'page'      => $page,
            'pages'     => $pages,
            'first'     => $total ? $offset + 1 : 0,
            'last'      => $offset + count($out),
        ];
    }

    /**
     * Fetch up to 8 recent audit events per user for the drawer timeline.
     */
    private static function getAuditHistoryForUsers(mysqli $conn, array $userIds): array
    {
        if (!$userIds) {
            return [];
        }
        $placeholders = implode(',', array_fill(0, count($userIds), '?'));
        $types        = str_repeat('i', count($userIds));
        $sql          = "SELECT a.target_user_id, a.action, a.previous_status, a.new_status, a.created_at,
                                CONCAT(COALESCE(adm.first_name,''), ' ', COALESCE(adm.last_name,'')) AS admin_name
                           FROM admin_audit_log a
                           LEFT JOIN users adm ON adm.id = a.admin_id
                          WHERE a.target_user_id IN ($placeholders)
                          ORDER BY a.created_at DESC";

        $rows = self::fetchRows($conn, $sql, $types, $userIds);
        $map  = [];
        foreach ($rows as $r) {
            $uid = (int) $r['target_user_id'];
            if (count($map[$uid] ?? []) >= 6) {
                continue;
            }
            $adminName = trim((string) $r['admin_name']);
            $map[$uid][] = [
                'action' => (string) $r['action'],
                'from'   => $r['previous_status'],
                'to'     => $r['new_status'],
                'by'     => $adminName !== '' ? $adminName : 'Admin',
                'when'   => date('M j, Y g:i A', strtotime($r['created_at'])),
            ];
        }
        return $map;
    }

    /** @return array{0:string,1:string,2:array} */
    private static function buildWhere(array $f): array
    {
        $where  = [];
        $types  = '';
        $params = [];

        if ($f['status'] === 'deleted') {
            $where[] = 'u.deleted_at IS NOT NULL';
        } else {
            $where[] = 'u.deleted_at IS NULL';
            if ($f['status'] !== '') {
                $where[]  = 'u.account_status = ?';
                $types   .= 's';
                $params[] = $f['status'];
            }
        }

        if ($f['q'] !== '') {
            $like    = '%' . addcslashes($f['q'], '%_\\') . '%';
            $where[] = "(u.first_name LIKE ? OR u.last_name LIKE ? OR u.email LIKE ? OR CONCAT(u.first_name, ' ', u.last_name) LIKE ?)";
            $types  .= 'ssss';
            array_push($params, $like, $like, $like, $like);
        }
        if ($f['role'] !== '') {
            $where[]  = 'u.role = ?';
            $types   .= 's';
            $params[] = $f['role'];
        }
        if ($f['from'] !== '') {
            $where[]  = 'u.created_at >= ?';
            $types   .= 's';
            $params[] = $f['from'] . ' 00:00:00';
        }
        if ($f['to'] !== '') {
            $where[]  = 'u.created_at < ?';
            $types   .= 's';
            $params[] = (new DateTimeImmutable($f['to']))->modify('+1 day')->format('Y-m-d') . ' 00:00:00';
        }

        return [implode(' AND ', $where), $types, $params];
    }

    private static function fetchRows(mysqli $conn, string $sql, string $types = '', array $params = []): array
    {
        $rows = [];
        $stmt = mysqli_prepare($conn, $sql);
        if (!$stmt) {
            return $rows;
        }
        if ($types !== '') {
            mysqli_stmt_bind_param($stmt, $types, ...$params);
        }
        if (mysqli_stmt_execute($stmt)) {
            $res = mysqli_stmt_get_result($stmt);
            while ($res && ($row = mysqli_fetch_assoc($res))) {
                $rows[] = $row;
            }
        }
        mysqli_stmt_close($stmt);
        return $rows;

function globalSearchRows(
    mysqli $conn,
    string $sql,
    string $types,
    array $params
): array {
    $rows = [];
    $stmt = mysqli_prepare($conn, $sql);

    if (!$stmt) {
        error_log('Global search prepare failed: ' . mysqli_error($conn));
        return [];
    }

    mysqli_stmt_bind_param($stmt, $types, ...$params);

    if (mysqli_stmt_execute($stmt)) {
        $result = mysqli_stmt_get_result($stmt);

        while ($result && ($row = mysqli_fetch_assoc($result))) {
            $rows[] = $row;
        }

        }
